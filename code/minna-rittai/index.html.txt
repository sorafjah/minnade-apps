    var t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}
function ri(a, b) { return a + Math.floor(rnd() * (b - a + 1)); }
function pick(a) { return a[Math.floor(rnd() * a.length)]; }
function shuffle(a) { for (var i = a.length - 1; i > 0; i--) { var j = Math.floor(rnd() * (i + 1)); var t = a[i]; a[i] = a[j]; a[j] = t; } return a; }

/* レベルごとの設定（先生画面の「レベルごとの内容」も、ここから作る）
   h[z][x] = 高さ。x：左→右、z：おく(0)→手前(D-1)。正面は手前側、右は +x 側。 */
var MAX_BLOCKS = 18;
var LV = {
  1: { W: 3, D: 3, maxH: 2, foot: [3, 5], tot: [4, 7],   hidden: 'none', seq: ['top', 'front', 'count'] },
  2: { W: 3, D: 3, maxH: 3, foot: [4, 7], tot: [6, 13],  hidden: 'any',  seq: ['top', 'front', 'right'] },
  3: { W: 3, D: 3, maxH: 3, foot: [4, 7], tot: [6, 13],  hidden: 'some', seq: ['count'] },
  4: { W: 4, D: 3, maxH: 3, foot: [5, 8], tot: [8, 16],  hidden: 'any',  seq: ['top', 'front', 'right'] },
  5: { W: 4, D: 3, maxH: 3, foot: [6, 9], tot: [10, 18], hidden: 'some', seq: ['count'] }
};
var TYPENAME = { top: '上から見た形', front: '正面から見た形', right: '右から見た形', count: '積み木の数' };
function lvDesc(lv) {
  var c = LV[lv];
  var kinds = c.seq.map(function (t) { return TYPENAME[t]; }).join('・');
  var only = c.seq.length === 1 ? '（数えるだけ）' : (c.seq.indexOf('count') < 0 ? '（数える問題は出ない）' : '');
  var hid = c.hidden === 'none' ? 'かくれた積み木はない' : (c.hidden === 'some' ? 'かくれた積み木がある' : 'かくれた積み木があることも');
  return kinds + only + '／' + c.W + '×' + c.D + 'マス、' + c.maxH + '段まで、積み木は' + c.tot[0] + '〜' + c.tot[1] + '個、' + hid;
}
// 数の答え方：生徒ごとの設定 → 全体の設定 → レベルのきまり（レベル5は数字を入れる、ほかは選ぶ）
function numModeOf(lv, roundMode, stuMode) {
  if (stuMode === 'choice' || stuMode === 'input') return stuMode;
  if (roundMode === 'choice' || roundMode === 'input') return roundMode;
  return lv >= 5 ? 'input' : 'choice';
}

function occ(s, x, y, z) { return x >= 0 && x < s.W && z >= 0 && z < s.D && y >= 0 && y < s.h[z][x]; }
// はじめの向き（右ななめ前の上）から見て、上・前・右がすべてふさがっている積み木＝見えない
function countHidden(s) {
  var n = 0;
  for (var z = 0; z < s.D; z++) for (var x = 0; x < s.W; x++) for (var y = 0; y < s.h[z][x]; y++)
    if (occ(s, x, y + 1, z) && occ(s, x, y, z + 1) && occ(s, x + 1, y, z)) n++;
  return n;
}
function colMax(s) { var a = []; for (var x = 0; x < s.W; x++) { var m = 0; for (var z = 0; z < s.D; z++) m = Math.max(m, s.h[z][x]); a.push(m); } return a; }
function rowMax(s) { return s.h.map(function (r) { return Math.max.apply(null, r); }); }
function flat(h) { return [].concat.apply([], h); }
function mkShape(h) {
  var f = flat(h), s = { W: h[0].length, D: h.length, h: h };
  s.total = f.reduce(function (a, b) { return a + b; }, 0);
  s.H = Math.max.apply(null, f);
  s.foot = f.filter(function (v) { return v > 0; }).length;
  s.hidden = countHidden(s);
  return s;
}
function genShape(c) {
  var W = c.W, D = c.D, DIRS = [[1, 0], [-1, 0], [0, 1], [0, -1]];
  for (var t = 0; t < 3000; t++) {
    var foot = ri(c.foot[0], c.foot[1]);
    var mask = [], z, x;
    for (z = 0; z < D; z++) { mask.push([]); for (x = 0; x < W; x++) mask[z].push(0); }
    var cells = [[ri(0, D - 1), ri(0, W - 1)]];
    mask[cells[0][0]][cells[0][1]] = 1;
    var guard = 0;
    while (cells.length < foot && guard++ < 500) {
      var c0 = pick(cells), d = pick(DIRS), nz = c0[0] + d[0], nx = c0[1] + d[1];
      if (nz < 0 || nz >= D || nx < 0 || nx >= W || mask[nz][nx]) continue;
      mask[nz][nx] = 1; cells.push([nz, nx]);
    }
    // どの列・どの行にも積み木がある
    var cover = true;
    for (x = 0; x < W; x++) if (!mask.some(function (r) { return r[x]; })) cover = false;
    for (z = 0; z < D; z++) if (!mask[z].some(function (v) { return v; })) cover = false;
    if (!cover) continue;
    var h = mask.map(function (r) { return r.map(function (v) { return v ? ri(1, c.maxH) : 0; }); });
    if (c.maxH >= 2 && flat(h).indexOf(c.maxH) < 0) { var p = pick(cells); h[p[0]][p[1]] = c.maxH; }
    var s = mkShape(h);
    if (s.total < c.tot[0] || s.total > c.tot[1] || s.total > MAX_BLOCKS) continue;
    if (c.hidden === 'none' && s.hidden > 0) continue;
    if (c.hidden === 'some' && s.hidden < 1) continue;
    return s;
  }
  return null;
}

/* 見え方（2Dの形）　cols：横に並んだ高さ　mask：上から見た形（上がおく、下が手前） */
function viewFront(s) { return { t: 'cols', a: colMax(s) }; }
function viewRight(s) { return { t: 'cols', a: rowMax(s).slice().reverse() }; } // 左はし＝手前
function viewTop(s) { return { t: 'mask', m: s.h.map(function (r) { return r.map(function (v) { return v > 0 ? 1 : 0; }); }) }; }
function viewOf(s, dir) { return dir === 'top' ? viewTop(s) : (dir === 'front' ? viewFront(s) : viewRight(s)); }
function trimCols(a) { var i = 0, j = a.length; while (i < j && !a[i]) i++; while (j > i && !a[j - 1]) j--; return a.slice(i, j); }
function trimMask(m) {
  var rows = [], cols = [], x, i;
  m.forEach(function (r, k) { if (r.some(function (v) { return v; })) rows.push(k); });
  if (!rows.length) return [];
  for (x = 0; x < m[0].length; x++) if (m.some(function (r) { return r[x]; })) cols.push(x);
  var out = [];
  for (i = rows[0]; i <= rows[rows.length - 1]; i++) out.push(m[i].slice(cols[0], cols[cols.length - 1] + 1));
  return out;
}
function gkey(g) { return g.t === 'cols' ? 'c' + trimCols(g.a).join(',') : 'm' + trimMask(g.m).map(function (r) { return r.join(''); }).join('/'); }
function gEmpty(g) { return g.t === 'cols' ? !g.a.some(function (v) { return v > 0; }) : !g.m.some(function (r) { return r.some(function (v) { return v; }); }); }
function gDims(g) {
  if (g.t === 'cols') { var a = trimCols(g.a); return { c: a.length, r: Math.max.apply(null, a) }; }
  var m = trimMask(g.m); return { c: m[0].length, r: m.length };
}
// 選択肢を4つにまとめる
function assemble(correct, g1, g2, keyFn, ok) {
  var used = {}, out = [];
  used[keyFn(correct)] = 1;
  function take(list, n) {
    var l = shuffle(list.slice());
    for (var i = 0; i < l.length; i++) {
      if (out.length >= n) break;
      var k = keyFn(l[i].g);
      if (!used[k] && !gEmpty(l[i].g) && ok(l[i].g)) { used[k] = 1; out.push(l[i]); }
    }
  }
  take(g1, 2); take(g2, 3); take(g1, 3);
  if (out.length < 3) return null;
  var choices = shuffle([{ g: correct, k: 'correct' }].concat(out));
  var ans = -1;
  choices.forEach(function (c, i) { if (c.k === 'correct') ans = i; });
  return { choices: choices, ans: ans };
}
function sideChoices(s, dir, maxH) {
  var correct = dir === 'front' ? viewFront(s) : viewRight(s);
  var other = dir === 'front' ? viewRight(s) : viewFront(s);
  var near = [], i;
  if (dir === 'front') near = s.h[s.D - 1].slice();
  else for (i = 0; i < s.D; i++) near.push(s.h[s.D - 1 - i][s.W - 1]);
  var a = correct.a;
  var g1 = [
    { g: { t: 'cols', a: a.slice().reverse() }, k: 'mirror' },
    { g: { t: 'cols', a: near }, k: 'near' },
    { g: other, k: 'other' }
  ];
  var g2 = [];
  a.forEach(function (v, k) {
    if (v < maxH) { var p = a.slice(); p[k] = v + 1; g2.push({ g: { t: 'cols', a: p }, k: 'plus' }); }
    if (v > 1) { var m = a.slice(); m[k] = v - 1; g2.push({ g: { t: 'cols', a: m }, k: 'minus' }); }
  });
  var sum = function (x) { return x.reduce(function (p, q) { return p + q; }, 0); };
  // 正解と同じ横はばで、すき間がなく、積み木の数の差が2つまでの形だけを使う
  var ok = function (g) { return g.a.length === a.length && g.a.every(function (v) { return v >= 1; }) && Math.abs(sum(g.a) - sum(a)) <= 2; };
  return assemble(correct, g1, g2, gkey, ok);
}
function rot90(m) {
  var o = [];
  for (var x = 0; x < m[0].length; x++) { var r = []; for (var z = m.length - 1; z >= 0; z--) r.push(m[z][x]); o.push(r); }
  return o;
}
// 上から見た形は、回すと同じになる形を「同じ形」とみなす
function rotKey(g) {
  var m = trimMask(g.m), ks = [];
  for (var i = 0; i < 4; i++) { ks.push(m.map(function (r) { return r.join(''); }).join('/')); m = rot90(m); }
  return 'm' + ks.sort()[0];
}
function connected(m) {
  var cells = [], DIRS = [[1, 0], [-1, 0], [0, 1], [0, -1]];
  m.forEach(function (r, z) { r.forEach(function (v, x) { if (v) cells.push(z + ',' + x); }); });
  if (!cells.length) return false;
  var seen = {}, st = [cells[0]], n = 1;
  seen[cells[0]] = 1;
  while (st.length) {
    var p = st.pop().split(','), z = +p[0], x = +p[1];
    for (var i = 0; i < 4; i++) {
      var zz = z + DIRS[i][0], xx = x + DIRS[i][1], k = zz + ',' + xx;
      if (m[zz] && m[zz][xx] && !seen[k]) { seen[k] = 1; n++; st.push(k); }
    }
  }
  return n === cells.length;
}
function topChoices(s) {
  var m = viewTop(s).m, correct = { t: 'mask', m: m }, DIRS = [[1, 0], [-1, 0], [0, 1], [0, -1]];
  var g1 = [
    { g: { t: 'mask', m: m.map(function (r) { return r.slice().reverse(); }) }, k: 'mirrorTop' },
    { g: { t: 'mask', m: m.slice().reverse().map(function (r) { return r.slice(); }) }, k: 'flipTop' }
  ];
  var g2 = [], n = flat(m).filter(function (v) { return v; }).length;
  for (var z = 0; z < s.D; z++) for (var x = 0; x < s.W; x++) {
    var c = m.map(function (r) { return r.slice(); });
    if (!m[z][x]) {
      var nb = DIRS.some(function (d) { return m[z + d[0]] && m[z + d[0]][x + d[1]]; });
      if (nb) { c[z][x] = 1; g2.push({ g: { t: 'mask', m: c }, k: 'plusTop' }); }
    } else if (n > 2) { c[z][x] = 0; g2.push({ g: { t: 'mask', m: c }, k: 'minusTop' }); }
  }
  // 正解と同じたて・横の大きさで、つながっている形だけを使う
  var d0 = gDims(correct);
  var ok = function (g) { var d = gDims(g); return d.c === d0.c && d.r === d0.r && connected(trimMask(g.m)); };
  return assemble(correct, g1, g2, rotKey, ok);
}
// 数の選択肢：見えている数だけ、上から見たマスの数だけ、などのまちがいを入れる
function countChoices(s) {
  var T = s.total, c = [{ v: T, k: 'correct' }], used = {};
  used[T] = 1;
  function add(v, k) { if (v > 0 && !used[v] && c.length < 4) { used[v] = 1; c.push({ v: v, k: k }); } }
  if (s.hidden > 0 && s.hidden <= 3) add(T - s.hidden, 'visible');
  if (T - s.foot <= 3) add(s.foot, 'foot');
  shuffle([1, -1, 2, -2, 3, -3]).forEach(function (d) { add(T + d, 'near'); });
  c.sort(function (x, y) { return x.v - y.v; });
  var ans = -1;
  c.forEach(function (x, i) { if (x.k === 'correct') ans = i; });
  return { choices: c, ans: ans };
}
function makeQuestion(level, type) {
  var cfg = LV[level];
  for (var t = 0; t < 60; t++) {
    var s = genShape(cfg), c;
    if (!s) continue;
    if (type === 'top') { c = topChoices(s); if (c) return { type: 'view', dir: 'top', s: s, choices: c.choices, ans: c.ans, level: level }; }
    else if (type === 'front' || type === 'right') { c = sideChoices(s, type, cfg.maxH); if (c) return { type: 'view', dir: type, s: s, choices: c.choices, ans: c.ans, level: level }; }
    else return { type: 'count', s: s, ans: s.total, nc: countChoices(s), level: level };
  }
  return null;
}
// 前の問題と同じ形にならないように、1問作る
function genOne(level, type, prevKey) {
  var q = null;
  for (var t = 0; t < 20; t++) { q = makeQuestion(level, type); if (q && JSON.stringify(q.s.h) !== prevKey) break; }
  return q;
}
// 「みんなで」の問題：同じ種（seed）から、どの端末でも同じ問題ができる
function buildSet(seed, level, n) {
  var out = [], used = {}, prev = '';
  rnd = seeded(seed);
  try {
    for (var i = 0; i < n; i++) {
      var lv = level === 'mix' ? 2 + (i % 4) : level;
      var seq = LV[lv].seq, k = used[lv] || 0;
      used[lv] = k + 1;
      var q = genOne(lv, seq[k % seq.length], prev);
      if (!q) continue;
      prev = JSON.stringify(q.s.h); out.push(q);
    }
  } finally { rnd = Math.random; }
  return out;
}

var FILL = '#d3e3ec', STROKE = '#2b3a46';
function svgOpen(vb, extra) { return '<svg viewBox="' + vb + '" ' + (extra || '') + ' xmlns="http://www.w3.org/2000/svg">'; }
function cellRect(x, y, cs) { return '<rect x="' + x + '" y="' + y + '" width="' + cs + '" height="' + cs + '" fill="' + FILL + '" stroke="' + STROKE + '" stroke-width="2"/>'; }
// 選択肢用：すべての選択肢を同じ大きさのわくで描く
function choiceSvg(g, dims) {
  var cs = 40, W = dims.c * cs, H = dims.r * cs, inner = '', ox, oy;
  if (g.t === 'cols') {
    var a = trimCols(g.a); ox = (dims.c - a.length) / 2 * cs;
    a.forEach(function (v, i) { for (var y = 0; y < v; y++) inner += cellRect(ox + i * cs, H - (y + 1) * cs, cs); });
    inner += '<line x1="' + (ox - 8) + '" y1="' + (H + 1) + '" x2="' + (ox + a.length * cs + 8) + '" y2="' + (H + 1) + '" stroke="#8a99a6" stroke-width="3"/>';
  } else {
    var m = trimMask(g.m); ox = (dims.c - m[0].length) / 2 * cs; oy = (dims.r - m.length) / 2 * cs;
    m.forEach(function (r, z) { r.forEach(function (v, x) { if (v) inner += cellRect(ox + x * cs, oy + z * cs, cs); }); });
  }
  return svgOpen('-8 -8 ' + (W + 16) + ' ' + (H + 16), 'preserveAspectRatio="xMidYMid meet"') + inner + '</svg>';
}
function fixedSvg(W, H, inner) { return svgOpen('-8 -8 ' + (W + 16) + ' ' + (H + 16), 'width="' + (W + 16) + '" height="' + (H + 16) + '"') + inner + '</svg>'; }

// ななめ上から見た絵（使い方の説明と、立体を表示できない端末で使う）
function isoSvg(h, labels) {
  var D = h.length, W = h[0].length, A = 0.866, pts = [], out = '';
  function P(x, y, z) { return [(x - z) * A, (x + z) * 0.5 - y]; }
  function f(a) { return a.map(function (p) { return p[0].toFixed(3) + ',' + p[1].toFixed(3); }).join(' '); }
  function poly(ps, fill, stroke, sw) {
    ps.forEach(function (p) { pts.push(p); });
    out += '<polygon points="' + f(ps) + '" fill="' + fill + '" stroke="' + stroke + '" stroke-width="' + sw + '" stroke-linejoin="round"/>';
  }
  poly([P(0, 0, 0), P(W, 0, 0), P(W, 0, D), P(0, 0, D)], '#eef4f8', '#b5c2ce', 0.03);
  for (var z = 0; z < D; z++) for (var x = 0; x < W; x++) for (var y = 0; y < h[z][x]; y++) {
    poly([P(x, y + 1, z), P(x + 1, y + 1, z), P(x + 1, y + 1, z + 1), P(x, y + 1, z + 1)], '#a8d3ee', '#2b3a46', 0.04);
    poly([P(x, y, z + 1), P(x + 1, y, z + 1), P(x + 1, y + 1, z + 1), P(x, y + 1, z + 1)], '#7fb6da', '#2b3a46', 0.04);
    poly([P(x + 1, y, z), P(x + 1, y, z + 1), P(x + 1, y + 1, z + 1), P(x + 1, y + 1, z)], '#689fc4', '#2b3a46', 0.04);
  }
  if (labels) {
    var lab = function (pos, kana, kanji) {
      var p = P(pos[0], pos[1], pos[2]), X = p[0], Y = p[1];
      pts.push([X - 1.1, Y - 0.5], [X + 1.1, Y + 0.75]);
      var font = '" text-anchor="middle" font-weight="700" font-family="BIZ UDPGothic, sans-serif" ';
      if (KANA_ONLY) out += '<text x="' + X.toFixed(3) + '" y="' + (Y + 0.35).toFixed(3) + font + 'font-size="0.42" fill="#e8892b">' + kana + '</text>';
      else out += '<text x="' + X.toFixed(3) + '" y="' + (Y - 0.1).toFixed(3) + font + 'font-size="0.3" fill="#56707e">' + kana + '</text>' +
        '<text x="' + X.toFixed(3) + '" y="' + (Y + 0.5).toFixed(3) + font + 'font-size="0.62" fill="#e8892b">' + kanji + '</text>';
    };
    lab([W / 2, 0, D + 0.9], 'しょうめん', '正面');
    lab([W + 0.9, 0, D / 2], 'みぎ', '右');
  }
  var xs = pts.map(function (p) { return p[0]; }), ys = pts.map(function (p) { return p[1]; });
  var x0 = Math.min.apply(null, xs) - 0.15, y0 = Math.min.apply(null, ys) - 0.15;
  var w = Math.max.apply(null, xs) - Math.min.apply(null, xs) + 0.3, hh = Math.max.apply(null, ys) - Math.min.apply(null, ys) + 0.3;
  return svgOpen(x0.toFixed(3) + ' ' + y0.toFixed(3) + ' ' + w.toFixed(3) + ' ' + hh.toFixed(3), 'width="' + Math.round(w * 40) + '" height="' + Math.round(hh * 40) + '"') + out + '</svg>';
}
function rotGrid(h) {
  var D = h.length, W = h[0].length, n = [], z, x;
  for (x = 0; x < W; x++) { n.push([]); for (z = 0; z < D; z++) n[x].push(0); }
  for (z = 0; z < D; z++) for (x = 0; x < W; x++) n[x][D - 1 - z] = h[z][x];
  return n;
}

var HOME = { theta: 0.62, phi: 0.98 };
function stageHtml(p, student) {
  var ans = student ? R('{正解|せいかい}') : '正解';
  return '<div class="stage"><div class="glWrap" id="' + p + 'Gl"></div>' +

function makeStage(p) {
  var T3 = window.THREE, wrap = $(p + 'Gl');
  var S = { ok: false, cur: null, drag: false, dead: false };
  var renderer = null, scene, camera, world, blocks = [], labels = {}, needs = true;
  var cam = { theta: HOME.theta, phi: HOME.phi, tween: null }, countAnim = null, hl = null;
  var boxGeo, edgeGeo, edgeMat, blockMat;
  if (T3) { try { renderer = new T3.WebGLRenderer({ antialias: true }); } catch (e) { renderer = null; } }

  function setBadge(n) {
    var html = n == null ? '' : R(n + ' {個|こ}');
    var b = swap(p + 'Badge', html);
    if (b) b.classList.toggle('hidden', n == null);
  }
  function labelTex(kana, kanji, color) {
    var c = document.createElement('canvas'); c.width = KANA_ONLY ? 384 : 256; c.height = 200;
    var g = c.getContext('2d');
    g.fillStyle = color; g.textAlign = 'center'; g.textBaseline = 'alphabetic';
    if (KANA_ONLY) { g.font = '700 72px "BIZ UDPGothic", sans-serif'; g.fillText(kana, 192, 125); }
    else {
      g.font = '700 34px "BIZ UDPGothic", sans-serif'; g.fillText(kana, 128, 44);
      g.font = '700 100px "BIZ UDPGothic", sans-serif'; g.fillText(kanji, 128, 160);
    }
    var t = new T3.CanvasTexture(c); t.anisotropy = 4;
    return t;
  }
  function makeLabel(kana, kanji) {
    var texN = labelTex(kana, kanji, '#6b7b88'), texH = labelTex(kana, kanji, '#e8892b');
    var mesh = new T3.Mesh(new T3.PlaneGeometry(KANA_ONLY ? 2.25 : 1.5, 1.17), new T3.MeshBasicMaterial({ map: texN, transparent: true, depthWrite: false }));
    mesh.rotation.x = -Math.PI / 2;
    return { mesh: mesh, texN: texN, texH: texH };
  }
  function makeArrow() {
    var sh = new T3.Shape(); sh.moveTo(0, 0.32); sh.lineTo(-0.3, -0.18); sh.lineTo(0.3, -0.18); sh.lineTo(0, 0.32);
    var m = new T3.Mesh(new T3.ShapeGeometry(sh), new T3.MeshBasicMaterial({ color: 0x9aa8b4, side: T3.DoubleSide }));
    m.rotation.order = 'YXZ';
    return m;
  }
  function clearWorld() {
    while (world.children.length) {
      var c = world.children.pop();
      c.traverse(function (n) {
        if (n.geometry && n.geometry !== boxGeo && n.geometry !== edgeGeo) n.geometry.dispose();
        if (n.material && n.material !== edgeMat && n.material !== blockMat) n.material.dispose();
      });
    }
    for (var k in labels) { labels[k].texN.dispose(); labels[k].texH.dispose(); }
    labels = {}; blocks = [];
  }
  function build(s) {
    clearWorld();
    var ox = -s.W / 2, oz = -s.D / 2, x, y, z;
    var floor = new T3.Mesh(new T3.PlaneGeometry(s.W + 0.5, s.D + 0.5), new T3.MeshBasicMaterial({ color: 0xeef4f8 }));
    floor.rotation.x = -Math.PI / 2; floor.position.y = -0.003; world.add(floor);
    var pts = [];
    for (x = 0; x <= s.W; x++) pts.push(ox + x, 0, oz, ox + x, 0, oz + s.D);
    for (z = 0; z <= s.D; z++) pts.push(ox, 0, oz + z, ox + s.W, 0, oz + z);
    var gg = new T3.BufferGeometry(); gg.setAttribute('position', new T3.Float32BufferAttribute(pts, 3));
    world.add(new T3.LineSegments(gg, new T3.LineBasicMaterial({ color: 0xb5c2ce })));
    var list = [];
    for (z = 0; z < s.D; z++) for (x = 0; x < s.W; x++) for (y = 0; y < s.h[z][x]; y++) list.push([x, y, z]);
    list.sort(function (a, b) { return a[1] - b[1] || a[2] - b[2] || a[0] - b[0]; }); // 下の段から数える
    list.forEach(function (c) {
      var g = new T3.Group();
      g.add(new T3.Mesh(boxGeo, blockMat), new T3.LineSegments(edgeGeo, edgeMat));
      g.position.set(ox + c[0] + 0.5, c[1] + 0.5, oz + c[2] + 0.5);
      world.add(g); blocks.push(g);
    });
    labels.front = makeLabel('しょうめん', '正面');
    labels.front.mesh.position.set(0, 0.002, oz + s.D + 1.35); world.add(labels.front.mesh);
    labels.right = makeLabel('みぎ', '右');
    labels.right.mesh.position.set(ox + s.W + 1.35, 0.002, 0); world.add(labels.right.mesh);
    var af = makeArrow(); af.rotation.set(-Math.PI / 2, 0, 0); af.position.set(0, 0.002, oz + s.D + 0.5); world.add(af);
    var ar = makeArrow(); ar.rotation.set(-Math.PI / 2, Math.PI / 2, 0); ar.position.set(ox + s.W + 0.5, 0.002, 0); world.add(ar);
  }
  function frame() {
    if (!renderer || !S.cur) return;
    var w = wrap.clientWidth, h = wrap.clientHeight;
    if (!w || !h) return;
    renderer.setSize(w, h, false);
    var asp = w / h, s = S.cur;
    var r = 0.5 * Math.sqrt(Math.pow(s.W + 3.4, 2) + Math.pow(s.D + 3.4, 2) + Math.pow(s.H + 1, 2)) * 0.8;
    var hh = asp >= 1 ? r : r / asp;
    camera.left = -hh * asp; camera.right = hh * asp; camera.top = hh; camera.bottom = -hh;
    camera.updateProjectionMatrix(); needs = true;
  }
  function applyCam() {
    if (!renderer || !S.cur) return;
    var r = 30, ty = S.cur.H / 2;
    camera.position.set(r * Math.sin(cam.phi) * Math.sin(cam.theta), ty + r * Math.cos(cam.phi), r * Math.sin(cam.phi) * Math.cos(cam.theta));
    camera.up.set(0, 1, 0); camera.lookAt(0, ty, 0); needs = true;
  }
  function applyHl() {
    ['front', 'right'].forEach(function (k) { if (labels[k]) labels[k].mesh.material.map = (k === hl ? labels[k].texH : labels[k].texN); });
    needs = true;
  }
  function update(dt) {
    var tw = cam.tween;
    if (tw) {
      tw.time += dt;
      var k = Math.min(1, tw.time / tw.dur), e = k < 0.5 ? 2 * k * k : 1 - Math.pow(-2 * k + 2, 2) / 2;
      cam.theta = tw.t0 + tw.dt * e; cam.phi = tw.p0 + tw.dp * e; applyCam();
      if (k >= 1) cam.tween = null;
    }
    if (countAnim) {
      countAnim.t -= dt;
      if (countAnim.t <= 0) {
        if (countAnim.i < blocks.length) {
          blocks[countAnim.i].visible = true; countAnim.i++;
          setBadge(countAnim.i); sndPop(); countAnim.t = countAnim.step; needs = true;
        } else countAnim = null;
      }
    }
  }
  function showAll() { countAnim = null; blocks.forEach(function (b) { b.visible = true; }); setBadge(null); needs = true; }
  function goView(v, dur) {
    var dt = v.theta - cam.theta;
    while (dt > Math.PI) dt -= Math.PI * 2;
    while (dt < -Math.PI) dt += Math.PI * 2;
    cam.tween = { t0: cam.theta, p0: cam.phi, dt: dt, dp: v.phi - cam.phi, time: 0, dur: dur || 1.6 };
  }
  var VIEWS = { top: { theta: 0, phi: 0.0001 }, front: { theta: 0, phi: Math.PI / 2 }, right: { theta: Math.PI / 2, phi: Math.PI / 2 }, home: HOME };

  if (renderer) {
    S.ok = true;
    renderer.setPixelRatio(Math.min(2, window.devicePixelRatio || 1));
    renderer.setClearColor(0xffffff, 1);
    wrap.appendChild(renderer.domElement);
    scene = new T3.Scene();
    camera = new T3.OrthographicCamera(-1, 1, 1, -1, 0.1, 100);
    scene.add(new T3.AmbientLight(0xffffff, 0.45));
    var dl = new T3.DirectionalLight(0xffffff, 0.7); dl.position.set(3, 8, 5); scene.add(dl);
    var dl2 = new T3.DirectionalLight(0xffffff, 0.15); dl2.position.set(-5, 2, -3); scene.add(dl2);
    world = new T3.Group(); scene.add(world);
    boxGeo = new T3.BoxGeometry(1, 1, 1);
    edgeGeo = new T3.EdgesGeometry(boxGeo);
    edgeMat = new T3.LineBasicMaterial({ color: 0x2b3a46 });
    blockMat = new T3.MeshLambertMaterial({ color: 0x8cc4e6, polygonOffset: true, polygonOffsetFactor: 1, polygonOffsetUnits: 1 });
    // 指でなぞって回す（答えたあとだけ）
    (function (el) {
      var id = null, lx = 0, ly = 0;
      el.addEventListener('pointerdown', function (e) {
        if (!S.drag) return;
        id = e.pointerId; lx = e.clientX; ly = e.clientY; cam.tween = null;
        try { el.setPointerCapture(id); } catch (err) {}
      });
      el.addEventListener('pointermove', function (e) {
        if (e.pointerId !== id) return;
        var dx = e.clientX - lx, dy = e.clientY - ly; lx = e.clientX; ly = e.clientY;
        cam.theta -= dx * 0.01;
        cam.phi = Math.min(Math.PI / 2, Math.max(0.0001, cam.phi - dy * 0.01));
        applyCam();
      });
      var up = function (e) { if (e.pointerId === id) id = null; };
      el.addEventListener('pointerup', up); el.addEventListener('pointercancel', up);
    })(renderer.domElement);
    if (window.ResizeObserver) new ResizeObserver(frame).observe(wrap);
    window.addEventListener('resize', frame);
    // 1/30秒より長い時間は切りつめる（速い画面でも同じ速さ）
    var last = performance.now();
    var loop = function (now) {
      if (S.dead) return;
      var dt = Math.min(1 / 30, Math.max(0, (now - last) / 1000)); last = now;
      update(dt);
      if (needs) { renderer.render(scene, camera); needs = false; }
      requestAnimationFrame(loop);
    };
    requestAnimationFrame(loop);
  }

  var fbGrid = null;
  function drawFallback() { wrap.innerHTML = isoSvg(fbGrid, true); }
  S.setShape = function (s) {
    S.cur = s; hl = null;
    if (!renderer) { fbGrid = s.h; drawFallback(); setBadge(null); return; }
    build(s);
    cam.theta = HOME.theta; cam.phi = HOME.phi; cam.tween = null;
    showAll(); frame(); applyCam(); applyHl();
  };
  S.rebuild = function () {
    if (!S.cur) return;
    if (!renderer) { drawFallback(); return; }
    var vis = blocks.map(function (b) { return b.visible; });
    build(S.cur);
    blocks.forEach(function (b, i) { b.visible = vis[i] !== false; });
    applyHl(); applyCam();
  };
  S.highlight = function (dir) { hl = dir; if (renderer) applyHl(); };
  S.spin = function () {
    if (!S.cur) return;
    if (!renderer) return;
    var step = Math.PI / 2;
    var base = cam.tween ? cam.tween.t0 + cam.tween.dt : cam.theta;
    var tt = HOME.theta + (Math.round((base - HOME.theta) / step) + 1) * step;
    cam.tween = { t0: cam.theta, p0: cam.phi, dt: tt - cam.theta, dp: HOME.phi - cam.phi, time: 0, dur: 1.4 };
  };
  S.view = function (name) { if (renderer && S.cur) { showAll(); goView(VIEWS[name]); } };
  S.home = function () { if (renderer && S.cur) { showAll(); goView(HOME, 1.0); } };
  S.count = function () {
    if (!S.cur) return;
    if (!renderer) { setBadge(S.cur.total); return; }
    cam.tween = null; goView(HOME, 0.6);
    blocks.forEach(function (b) { b.visible = false; });
    countAnim = { i: 0, t: 0.8, step: blocks.length > 10 ? 0.6 : 1.0 }; setBadge(0); needs = true;
  };
  S.dispose = function () {
    S.dead = true;
    if (renderer) { try { clearWorld(); renderer.dispose(); } catch (e) {} }
  };
  return S;
}

var STAGES = [];

}

function markChoices(root, ans, picked) {

  ST.qi = SF.no;
  showQ(q, SF.no);
}
function selfAfter(q, ok) {
  var r = ST.r, c = ST.conf;
  SF.no++; if (ok) SF.okN++;
  SF.results.push({ k: qKey(q), ok: ok });
  if (ok) {
    SF.up++; SF.down = 0;
    if (q.level === 1) SF.low = 0;
    if (q.level === TOP_LV) SF.top++;
    if (q.level === TOP_LV && c.hi >= TOP_LV && !c.master && SF.top >= r.masterN) SF.master = true;
  } else {
    SF.down++; SF.up = 0; SF.top = 0;
    if (q.level === 1) { SF.low++; if (SF.low >= r.markN) SF.mark = true; }
  }
  // レベルの自動上下（生徒の画面には出さない）
  if (r.auto && c.auto) {
    if (SF.up >= r.up && SF.lv < c.hi) { SF.lv++; SF.up = 0; SF.down = 0; }
    else if (SF.down >= r.down && SF.lv > c.lo) { SF.lv--; SF.up = 0; SF.down = 0; }
  }
  saveSF();
}

  T.stage.highlight(q.type === 'view' ? q.dir : null);
  var mode = numModeOf(q.level, r.numMode, '');
  var save = KANA_ONLY; KANA_ONLY = false;
  var body = answerHtml(q, mode);
  KANA_ONLY = save;
  var p = swap('tPanel', '<div class="qtext">' + (r.qi + 1) + ' / ' + r.count + '問目（レベル' + q.level + '）<br>' + esc(noRuby(QTEXT[qKey(q)])) + '</div>' +
    '<div id="tAns">' + noRubyHtml(body) + '</div><div class="feedback hidden" id="tCorrect"></div>');
  var els = p.querySelectorAll('#tAns button, #tAns input');
  for (var i = 0; i < els.length; i++) els[i].disabled = true;
}
function noRubyHtml(h) { return h.replace(/<rt>[^<]*<\/rt>/g, '').replace(/<\/?ruby>/g, ''); }
function showTAns() {
  var q = T.set && T.set[T.r.qi];
  if (!q || !T.stage) return;
  var root = $('tAns'), tx;
  if (q.type === 'view') { markChoices(root, q.ans, -1); tx = '正解は「' + KANA4[q.ans] + '」（' + noRuby(RESNAME[q.dir]) + '）'; T.stage.view(q.dir); }
  else { if (root.querySelector('.choice')) markChoices(root, q.nc.ans, -1); tx = '正解は ' + q.ans + '個'; T.stage.count(); }
  var marks = root.querySelectorAll('.is-correct .cmark');
  for (var i = 0; i < marks.length; i++) marks[i].innerHTML = '正解';
  swap('tCorrect', '<div class="fb-title ok">' + esc(tx) + '</div>').classList.remove('hidden');
}
function showQrBig(on) {
  var r = T.r, box = $('qrBig');
  if (!on || !r) { box.classList.add('hidden'); box.innerHTML = ''; return; }
  box.innerHTML = '<div class="qrL">' + qrSvg(T.url) + '</div>' +
    (r.codeOn ? '<div>参加コード <span class="codeBig" style="font-size:56px">' + esc(r.code) + '</span></div>' : '') +
    '<div class="url">' + esc(T.url) + '</div><button class="primary" data-act="qrSmall">ふつうの表示にもどす</button>';
  box.classList.remove('hidden');
}

function closeModal() { T.modal = ''; $('modal').classList.add('hidden'); swap('modalBox', ''); }

function mMsg(s) { if ($('mMsg')) swap('mMsg', esc(s)); }
function loadStuConf(roster) {
  T.croster = roster; T.crows = [];
  api('api_tGetConf', roster).then(function (d) { T.crows = d.rows || []; if (T.modal === 'stuConf') renderModal(); })
    .catch(function (e) { mMsg(errText(e, '読みこめませんでした')); });
}
function rerender() { if (T.modal) renderModal(); else if (T.view === 'setup') renderSetup(); }
function fixRow(c) {
  c.lo = Math.max(1, Math.min(5, c.lo)); c.hi = Math.max(c.lo, Math.min(5, c.hi)); c.lv = Math.max(c.lo, Math.min(c.hi, c.lv));
}
function setNum(key, row, v) {
  var lim = LIM[key] || [0, 99], obj = row != null ? T.crows[+row] : T.form;
  if (!obj) return;
  if (isNaN(v)) v = obj[key];
  obj[key] = Math.max(lim[0], Math.min(lim[1], v));
  if (row != null) fixRow(obj);
  rerender();
}
function onTeacherChange(e) {
  var t = e.target;
  if (t.getAttribute('data-in')) {
    var v = parseInt(String(t.value).replace(/[０-９]/g, function (d) { return String.fromCharCode(d.charCodeAt(0) - 0xFEE0); }), 10);
    setNum(t.getAttribute('data-in'), t.getAttribute('data-row'), v);
  } else if (t.getAttribute('data-sel') === 'roster') { T.form.roster = t.value; }
  else if (t.getAttribute('data-sel') === 'croster') { loadStuConf(t.value); renderModal(); }
}
function runAct(p, after) {
  tMsg('');
  p.then(function (d) {
    if (d && d.err) { tMsg(d.err); return; }
    if (d && 'round' in d) { T.r = d.round; T.students = d.students || []; }
    if (after) after(d);
  }).catch(function (e) { tMsg(errText(e, 'うまくいきませんでした')); });
}
function onTeacherClick(e) {
  var t = e.target, b;
  if (!t.closest) return;
  if ((b = t.closest('#tBar [data-view]'))) {
    if (!T.stage) return;
    var v = b.getAttribute('data-view');
    if (v === 'spin') T.stage.spin(); else if (v === 'ans') showTAns(); else T.stage.home();
    return;
  }
  if ((b = t.closest('[data-seg]'))) {
    var row = b.getAttribute('data-row'), obj = row != null ? T.crows[+row] : T.form;
    if (obj) { obj[b.getAttribute('data-seg')] = parseVal(b.getAttribute('data-val')); rerender(); }
    return;
  }
  if ((b = t.closest('[data-step]'))) {
    var k = b.getAttribute('data-step'), rw = b.getAttribute('data-row'), o = rw != null ? T.crows[+rw] : T.form;
    if (o) setNum(k, rw, (+o[k] || 0) + (+b.getAttribute('data-d')));
    return;
  }
  if (!(b = t.closest('[data-act]'))) return;
  var act = b.getAttribute('data-act');
  if (act === 'guide') openGuide(teacherGuide(), true, null);
  else if (act === 'detail') openModal('detail');
  else if (act === 'stuConf') { openModal('stuConf'); loadStuConf(T.croster || (T.r && T.r.join === 'roster' ? T.r.roster : T.form.roster) || T.rosters[0] || 'ゲスト'); renderModal(); }
  else if (act === 'records') {
    T.recs = []; openModal('records');
    api('api_tRecords').then(function (d) { T.recs = d.rows || []; if (T.modal === 'records') renderModal(); }).catch(function () {});
  }
  else if (act === 'closeModal') closeModal();
  else if (act === 'saveDetail') {
    mMsg('保存しています…');
    api('api_tSaveSet', T.form).then(function (d) { T.conf = d.conf; closeModal(); }).catch(function (er) { mMsg(errText(er, '保存できませんでした')); });
  }
  else if (act === 'unmaster') { var c = T.crows[+b.getAttribute('data-row')]; if (c) { c.master = false; c.unmaster = true; renderModal(); } }
  else if (act === 'saveStu') {
    mMsg('保存しています…');
    api('api_tSaveConf', T.croster, T.crows).then(function (d) { T.crows = d.rows || []; renderModal(); mMsg('保存しました'); })
      .catch(function (er) { mMsg(errText(er, '保存できませんでした')); });
  }
  else if (act === 'start') {
    b.disabled = true; tMsg('受付を始めています…');
    api('api_tStart', T.form).then(function (d) {
      if (d.err) { b.disabled = false; tMsg(d.err); return; }
      T.conf = clone(T.form); T.r = d.round; T.students = []; renderRun();
    }).catch(function (er) { b.disabled = false; tMsg(errText(er, '受付を始められませんでした')); });
  }
  else if (act === 'next') { b.disabled = true; runAct(api('api_tSetQ', T.r.qi + 1), function () { updateRun(); tSchedule(); }); }
  else if (act === 'dist') { T.showDist = !T.showDist; updateRun(); }
  else if (act === 'end') { T.endAsk = true; updateRun(); }
  else if (act === 'endNo') { T.endAsk = false; updateRun(); }
  else if (act === 'endYes') { b.disabled = true; T.endAsk = false; runAct(api('api_tEnd'), function () { showQrBig(false); updateRun(); }); }
  else if (act === 'setup') { T.form = clone(T.conf); renderSetup(); }
  else if (act === 'qrBig') showQrBig(true);
  else if (act === 'qrSmall') showQrBig(false);
}
