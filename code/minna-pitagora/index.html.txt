function surfaceAt(p, x){

  const a=ANG(p.rot), c=Math.cos(a); if(c<0.5) return null;
  let cy=p.y, w=sp.w, th=sp.h; if(p.type==='goal'){ w=2.3; th=0.3; cy=p.y-0.15; }

}
// 玉とドミノは、すぐ下に板があればぴったり乗せる（置きやすくするためのチート）
function settle(layout){
  return layout.map(p=>{ if(p.type!=='ball'&&p.type!=='domino') return {x:p.x,y:p.y};
    const c=[{y:0,a:0}]; for(const q of layout){ const s=surfaceAt(q,p.x); if(s) c.push(s); }
    let best=null;
    for(const s of c){ const off=p.type==='ball'? CFG.BALL_R/Math.cos(s.a) : PARTS.domino.h/2+Math.abs(Math.tan(s.a))*PARTS.domino.w/2;
      const cy=s.y+off+0.003, d=p.y-cy; if(d>-0.6&&d<0.8&&(best===null||cy>best)) best=cy; }
    return {x:p.x, y:best===null?p.y:best}; });
}

      if(an>L){ b.quaternion.setFromAxisAngle(Z,L); if(b.angularVelocity.z>0) b.angularVelocity.z=0; }
      else if(an<-L){ b.quaternion.setFromAxisAngle(Z,-L); if(b.angularVelocity.z<0) b.angularVelocity.z=0; } }
    while(pending.length){ const f=pending.shift(), nx=-Math.sin(f.a), ny=Math.cos(f.a), bp=f.ball.position;

    for(const b of balls) b.prevSpeed=b.velocity.length();
    for(const b of balls) if(b.position.y<R+0.08) { b.velocity.x*=CFG.GRASS; b.angularVelocity.z*=CFG.GRASS; }   // 床は芝生：玉がいつまでも転がり続けないようにする
    let inGoal=false; if(goal) for(const b of balls){ if(goal.lockOnly&&!b.locked) continue; const p=b.position;   // opt=1 のゴールは 赤い玉（動かせない玉）だけ
      if(Math.abs(p.x-goal.x)<1.0&&p.y>goal.y&&p.y<goal.y+1.15) inGoal=true; }
    winT=inGoal?winT+H:0; if(winT>0.4){ s.status='win'; events.push({k:'win'}); return; }
    let mx=0; for(const d of movers){ const v=d.body.velocity.length()+Math.abs(d.body.angularVelocity.z)*0.5; if(v>mx) mx=v; }
    stallT=(mx<0.25&&s.t>1)?stallT+H:0;
    if(stallT>stallLim||s.t>CFG.B_TIME_LIMIT){ s.status='fail'; events.push({k:'fail'}); }
  };
  return s;
}

function boxDist(b,x,y){ const c=Math.cos(b.a), s=Math.sin(b.a), dx=x-b.x, dy=y-b.y, lx=dx*c+dy*s, ly=-dx*s+dy*c;
  return Math.hypot(Math.max(Math.abs(lx)-b.hw,0),Math.max(Math.abs(ly)-b.hh,0)); }
function boxHit(A,B,m){ // 回転した長方形どうしが m より近いか（分離軸）
  for(const r of [A,B]) for(const ax of [r.a, r.a+Math.PI/2]){ const ux=Math.cos(ax), uy=Math.sin(ax);
    const proj=b=>{ const c=Math.cos(b.a), s=Math.sin(b.a); return Math.abs(b.hw*(c*ux+s*uy))+Math.abs(b.hh*(-s*ux+c*uy)); };
    if(Math.abs((A.x-B.x)*ux+(A.y-B.y)*uy)>proj(A)+proj(B)+m) return false; }
  return true; }
// 配置を走らせて、玉・ドミノ・ふりこの玉の 位置と速さを 毎ステップ記録する（ゴールなし）
function traceSim(C,layout,act,i0,eps){ const s=createBuildSim(C,layout,eps||0), tr={}, H=CFG.H, G=CFG.G;
  const ds=s.dyn.filter(d=>d.kind==='ball'||d.kind==='domino'||d.kind==='bob');
  ds.forEach(d=>tr[d.pi]={kind:d.kind,x:[],y:[],vx:[],vy:[]});
  const ta=tr[act]; let run=0, done=false;
  while(s.status==='run'&&!done){ s.step(); for(const d of ds){ const b=d.body, t=tr[d.pi]; t.x.push(b.position.x); t.y.push(b.position.y); t.vx.push(b.velocity.x); t.vy.push(b.velocity.y); }
    // 調べたい玉が 宙をとび終わったら、そこで やめる（時間の節約）
    const n=ta?ta.x.length:0; if(n>=3&&n-2>=i0){ const ay=(ta.vy[n-1]-ta.vy[n-3])/(2*H), ax=(ta.vx[n-1]-ta.vx[n-3])/(2*H);
      if(Math.abs(ay+G)<0.12*G&&Math.abs(ax)<0.12*G) run++; else { if(run>=10) done=true; run=0; } } }
  return tr; }
// i0 より後で、玉が はじめて 宙をとぶ区間 [a,b]（重力だけで動いている＝加速度が (0,-G) に近い）
function flightAfter(t,i0){ const H=CFG.H, G=CFG.G, n=t.x.length; let a=-1;
  for(let i=Math.max(1,i0);i<n-1;i++){ const ay=(t.vy[i+1]-t.vy[i-1])/(2*H), ax=(t.vx[i+1]-t.vx[i-1])/(2*H);
    const ff=Math.abs(ay+G)<0.12*G&&Math.abs(ax)<0.12*G&&t.x[i]>0.6&&t.x[i]<BOARD.W-0.6;
    if(ff){ if(a<0) a=i; } else if(a>=0){ if(i-a>=10) return [a,i-1]; a=-1; } }
  return a>=0&&n-1-a>=10?[a,n-2]:null; }

function genOnce(C,rng){
  const R=CFG.BALL_R, L=[], ri=(a,b)=>a+Math.floor(rng()*(b-a+1)), rf=(a,b)=>a+rng()*(b-a), pick=a=>a[Math.floor(rng()*a.length)];
  const q=v=>Math.round(v*20)/20, rot2=(a,lx,ly)=>({x:lx*Math.cos(a)-ly*Math.sin(a), y:lx*Math.sin(a)+ly*Math.cos(a)});
  const nBalls=()=>L.filter(p=>p.type==='ball').length;
  let dir=rng()<0.5?1:-1, active, iLand=0, note=ri(0,4), used=[];
  const X=x=>dir>0?x:BOARD.W-x;   // 左右を反転できるように
  // ---- スタート ----
  const st=pick(['slope','slope','belt']); used.push(st);
  if(st==='slope'){ const k=ri(1,2), rot=-dir*k, a=ANG(rot), cx=q(X(rf(3.6,5.2))), cy=q(rf(13.5,15.8));
    L.push(P('plankL',cx,cy,rot)); const top=rot2(a,-dir*2.3,0.15+R+0.02);
    L.push(P('ball',q(cx+top.x),q(cy+top.y))); active=1; }
  else { const rot=dir*ri(0,1), a=ANG(rot), cx=q(X(rf(2.6,4.2))), cy=q(rf(13.5,15.5)), o=ri(0,1);   // ベルトに のった玉を はこんで、はしから おとす
    L.push(P('belt',cx,cy,rot,dir>0?o:o+2)); const top=rot2(a,-dir*1.2,PARTS.belt.h/2+R+0.02);
    L.push(P('ball',q(cx+top.x),q(cy+top.y))); active=1; }
  // ---- 置いてよいか（ほかの部品と重ならない・これまでの通り道をふさがない） ----
  function fits(news,tr,iC,clr){ clr=clr||0.12; const nb=news.flatMap(partBoxes), old=L.flatMap(partBoxes);
    for(const b of nb){ const c=Math.cos(b.a), s=Math.sin(b.a);
      for(const [sx,sy] of [[1,1],[1,-1],[-1,1],[-1,-1]]){ const x=b.x+sx*b.hw*c-sy*b.hh*s, y=b.y+sx*b.hw*s+sy*b.hh*c; if(x<0.2||x>BOARD.W-0.2||y<0.05||y>BOARD.H-0.2) return false; }
      for(const o of old) if(boxHit(b,o,0.25)) return false; }
    for(const n of news) if(n.type==='ball'||n.type==='domino'){ const rr=n.type==='ball'?R:1.0; for(const o of old) if(boxDist(o,n.x,n.y)<rr+0.2) return false; }
    for(const k in tr){ const t=tr[k], rr=t.kind==='domino'?1.0:R, lim=(+k===active)?iC-3:t.x.length;
      for(let i=0;i<lim;i+=2) for(const b of nb) if(boxDist(b,t.x[i],t.y[i])<rr+clr) return false;
      if(+k!==active) for(const n of news) if(n.type==='ball'||n.type==='domino') for(let i=0;i<t.x.length;i+=3) if(Math.hypot(t.x[i]-n.x,t.y[i]-n.y)<rr+1.2) return false; }
    return true; }
  // ---- しかけ（宙をとんでいる玉を、とちゅうで受けとめる） ----
  const MOD={
    ramp(tr,i,t,type){ const x=t.x[i], y=t.y[i], vd=Math.sign(t.vx[i])||dir; let d=rng()<0.72?vd:-vd;
      const w=PARTS[type].w, ex=dd=>x+dd*w*0.85; if(ex(d)>BOARD.W-3.5||ex(d)<3.5) d=-d; if(ex(d)>BOARD.W-3.5||ex(d)<3.5) return false;
      const rot=-d*(type==='chime'?1:ri(1,2)), a=ANG(rot), th=PARTS[type].h, f=d===vd?rf(0.12,0.3):rf(0.62,0.8);
      const sl=-d*w/2+d*f*w, c=rot2(a,sl,th/2+R), opt=type==='chime'?(note=(note+ri(1,2))%8):0;
      const p=P(type,q(x-c.x),q(y-c.y),rot,opt); if(!fits([p],tr,i)) return false; L.push(p); dir=d; return true; },
    belt(tr,i,t){ const x=t.x[i], y=t.y[i], vd=Math.sign(t.vx[i])||dir, w=PARTS.belt.w; let d=rng()<0.65?vd:-vd;
      const ex=dd=>x+dd*w*0.85; if(ex(d)>BOARD.W-3.5||ex(d)<3.5) d=-d; if(ex(d)>BOARD.W-3.5||ex(d)<3.5) return false;
      const rot=d*ri(0,2), a=ANG(rot), sl=-d*w/2+d*rf(0.15,0.35)*w, c=rot2(a,sl,PARTS.belt.h/2+R), o=ri(0,1);   // うしろの はしで 受けて、すすむ向きへ（のぼりも あり）
      const p=P('belt',q(x-c.x),q(y-c.y),rot,d>0?o:o+2); if(!fits([p],tr,i)) return false; L.push(p); dir=d; return true; },
    tramp(tr,i,t){ if(t.vy[i]>-6) return false; const x=t.x[i], y=t.y[i]; let d=Math.sign(t.vx[i])||dir; if(x+d*5>BOARD.W-1||x+d*5<1) d=-d;
      const rot=-d*ri(0,1), a=ANG(rot), c=rot2(a,rf(-0.3,0.3),0.2+R); const p=P('tramp',q(x-c.x),q(y-c.y),rot);
      if(!fits([p],tr,i)) return false; L.push(p); dir=d; return true; },
    spring(tr,i,t){ const x=t.x[i], y=t.y[i]; let d=rng()<0.5?1:-1; if(x+d*6>BOARD.W-1||x+d*6<1) d=-d;
      const rot=-d*ri(1,3), a=ANG(rot), o=ri(0,2), v=CFG.SPRING_BV[o], vy=v*Math.cos(a), c=rot2(a,rf(-0.2,0.2),0.2+R);
      const sx=x-c.x, sy=y-c.y; if(sy+vy*vy/(2*CFG.G)>BOARD.H-1) return false;
      const p=P('spring',q(sx),q(sy),rot,o); if(!fits([p],tr,i)) return false; L.push(p); dir=d; return true; },
    seesaw(tr,i,t){ if(t.vy[i]>-11||nBalls()>=3) return false; const x=t.x[i], y=t.y[i], side=Math.sign(t.vx[i])||dir;
      const rot=side*2, a=ANG(rot), c=rot2(a,side*rf(1.6,2.0),0.15+R), sx=q(x-c.x), sy=q(y-c.y), b=rot2(a,-side*2.0,0.15+R+0.01);
      if(sx+side*5>BOARD.W||sx+side*5<0) return false;
      const ps=[P('seesaw',sx,sy,rot), P('ball',q(sx+b.x),q(sy+b.y))]; if(!fits(ps,tr,i)) return false;
      L.push(...ps); active=L.length-1; dir=side; return 'swap'; },
    domino(tr,i,t){ if(nBalls()>=3) return false; const x=t.x[i], y=t.y[i], d=Math.sign(t.vx[i])||dir;
      const rot=-d*ri(1,2), a=ANG(rot), f=rf(0.15,0.3), sl=-d*1.5+d*f*3, c=rot2(a,sl,0.15+R), rx=x-c.x, ry=y-c.y;
      const e=rot2(a,d*1.5,0.15), ex=rx+e.x, ey=ry+e.y-0.02;
      const n=ri(3,4), s=rf(0.85,1.05), ps=[P('plankS',q(rx),q(ry),rot), P('plankL',q(ex+d*3.0),q(ey-0.15),0)];
      for(let k=0;k<n;k++) ps.push(P('domino',q(ex+d*(1.0+k*s)),q(ey+0.9)));
      ps.push(P('ball',q(ex+d*(1.0+(n-1)*s+1.15)),q(ey+R)));
      if(!fits(ps,tr,i)) return false; L.push(...ps); active=L.length-1; dir=d; return 'swap'; }
  };
  const target=ri(CFG.RND_MODS[0],CFG.RND_MODS[1]); let mods=0;
  const flightOf=tr=>{ const t=tr[active]; if(!t) return null; const fl=flightAfter(t,iLand); return fl?{t,fl}:null; };
  // 宙をとぶ区間のうち、次のパーツを置ける点（下に向かっていて、着地点より じゅうぶん上）
  const candOf=(c,ymin)=>{ const r=[]; if(!c) return r; const ye=c.t.y[c.fl[1]], lo=ye<1.2?ymin:Math.max(ymin,ye+2.2);
    for(let i=c.fl[0]+3;i<=c.fl[1];i++) if(c.t.vy[i]<-1.5&&c.t.y[i]>lo) r.push(i); return r; };
  // 重力を少しずらしても、玉が ほぼ同じ所を とぶか（たまたま うまくいった 置き方を はやめに はじく）
  const steady=cc=>{ if(!CFG.RND_STEADY) return true; const t0=cur2.t; cc=cc.slice(0,Math.ceil(cc.length*0.5)); for(const e of (CFG.RND_STEADY>1?[CFG.RND_EPS,-CFG.RND_EPS]:[CFG.RND_EPS])){ const t=traceSim(C,L,active,iLand,e)[active]; if(!t) return false;
      for(const i of cc){ if(i>=t.x.length) return false; if(Math.hypot(t.x[i]-t0.x[i],t.y[i]-t0.y[i])>0.4) return false; } } return true; };
  let cur2=null;
  let tr=traceSim(C,L,active,iLand), cur=flightOf(tr); if(!cur) return null;
  for(let m=0;m<target;m++){
    const cand=candOf(cur,3.2); if(!cand.length) break;
    const types=['ramp','ramp','chime','chime','belt','belt','tramp','spring','seesaw','domino','domino'].filter(k=>!(k==='domino'||k==='seesaw')||!used.includes(k));
    let ok=false;
    for(let tryN=0;tryN<12&&!ok;tryN++){ const k=pick(types), t=cur.t, i=cand[Math.min(cand.length-1,Math.floor(Math.pow(rng(),1.6)*cand.length))];
      const snap={n:L.length,active,iLand,dir,note};
      const r=k==='ramp'?MOD.ramp(tr,i,t,pick(['plankS','plankL'])):k==='chime'?MOD.ramp(tr,i,t,'chime'):MOD[k](tr,i,t);
      if(!r) continue;
      iLand=r==='swap'?0:i;
      // 置いたあと 走らせて、次に 受けとめられる「宙をとぶ区間」が できるかを たしかめる
      const tr2=traceSim(C,L,active,iLand), c2=flightOf(tr2); cur2=c2;
      const cc=c2?candOf(c2,m+1<target?3.2:2.2):[];
      if(cc.length>=4&&steady(cc)){ tr=tr2; cur=c2; used.push(k); mods++; ok=true; }
      else { L.length=snap.n; active=snap.active; iLand=snap.iLand; dir=snap.dir; note=snap.note; } }
    if(!ok) break;
  }
  if(mods<CFG.RND_MODS[0]) return null;
  // ---- ゴール ----
  const t=cur.t, cand=candOf(cur,2.2).filter(i=>t.y[i]<13);
  for(let k=0;k<12&&cand.length;k++){ const i=pick(cand), gy=Math.max(0.5,t.y[i]-1.8), g=P('goal',q(t.x[i]+Math.sign(t.vx[i])*rf(0,0.35)),q(gy));
    if(t.y[i]-gy<1.75) continue;
    if(fits([g],tr,i+3,0.06)){ L.push(g); return {layout:L, used}; } }
  return null;
}
function winsAt(C,layout,eps){ const s=createBuildSim(C,layout,eps); while(s.status==='run') s.step(); return s.status==='win'; }
function verifyLayout(C,layout){ const e=CFG.RND_EPS; return winsAt(C,layout,0)&&winsAt(C,layout,e)&&winsAt(C,layout,-e); }

// ぬいた パーツは 0.5 マス目に そろえても ゴールできるものだけ（生徒が 同じ 場所に おけるように）。ぬいたままでは ゴールしないことも たしかめる
const PUZ_TYPES=['plankL','plankS','chime','tramp','belt'];
function makePuzzle(C,layout,rng){
  const R=CFG.BALL_R, g=v=>Math.round(v/CFG.GRID)*CFG.GRID;
  const holds=p=>layout.some(o=>{ if(o.type!=='ball'&&o.type!=='domino') return false; const s=surfaceAt(p,o.x); if(!s) return false;
    const off=o.type==='ball'?R/Math.cos(s.a):PARTS.domino.h/2; return Math.abs(o.y-(s.y+off))<0.25; });
  const cand=[]; layout.forEach((p,i)=>{ if(PUZ_TYPES.includes(p.type)&&!holds(p)) cand.push(i); });
  if(!cand.length) return null;
  for(const n of (cand.length>=2&&rng()<0.5?[2,1]:[1])) for(let k=0;k<8;k++){
    const pick=cand.slice().sort(()=>rng()-0.5).slice(0,n), ans=pick.map(i=>Object.assign({},layout[i],{x:g(layout[i].x),y:g(layout[i].y),lock:false}));
    const full=layout.map((p,i)=>pick.includes(i)?ans[pick.indexOf(i)]:p);
    if(!verifyLayout(C,full)) continue;
    const puzzle=layout.filter((p,i)=>!pick.includes(i)).map(p=>Object.assign({},p,{lock:true}));
    if(winsAt(C,puzzle,0)) continue;
    return {puzzle, answer:ans}; }
  return null; }
// A：つくって、たしかめる。だめなら やりなおし。RND_TRIES 回 か RND_TIME をこえたら null（→ B を使う）
function genRandom(C,seed,now){ now=now||(()=>Date.now()); const t0=now(), rng=mulberry32(seed>>>0);
  for(let n=1;n<=CFG.RND_TRIES;n++){ const g=genOnce(C,rng);
    if(g&&verifyLayout(C,g.layout)){ const pz=makePuzzle(C,g.layout,rng); if(pz) return {layout:g.layout, puzzle:pz.puzzle, answer:pz.answer, used:g.used, tries:n, ms:now()-t0}; }
    if(now()-t0>CFG.RND_TIME) return {layout:null, tries:n, ms:now()-t0}; }
  return {layout:null, tries:CFG.RND_TRIES, ms:now()-t0}; }
