<script>
/* ===== 共通（回答者・先生の両方で使う） ===== */
const IS_GAS = !!(window.google && google.script && google.script.run);

// サーバーを呼ぶ（GASがないときはデモ用のにせサーバー）
function call(fn, ...args) {
  if (!IS_GAS) {
    return new Promise((res, rej) => setTimeout(() => {
      try {
        if (DEMO.offline && fn === 'submit') throw new Error('通信できません（デモ）');
        const r = DEMO[fn](...args);
        res(r === undefined ? r : JSON.parse(JSON.stringify(r)));
      } catch (e) { rej(e); }
    }, 180));
  }
  return new Promise((res, rej) => {
    google.script.run.withSuccessHandler(res).withFailureHandler(rej)[fn](...args);
  });
}

function esc(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
}
function R(base, yomi) { return '<ruby>' + base + '<rt>' + yomi + '</rt></ruby>'; }

// 先生が書いた「漢字《かんじ》」「｜文字《もじ》」をふりがな付きにする
function rubyText(s) {
  return esc(s)
    .replace(/｜([^｜《》]+)《([^》]+)》/g, (m, b, y) => R(b, y))
    .replace(/([一-鿿㐀-䶿々〆ヵヶ]+)《([^》]+)》/g, (m, b, y) => R(b, y))
    .replace(/\n/g, '<br>');
}
function plainText(s) { return String(s || '').replace(/｜/g, '').replace(/《[^》]*》/g, ''); }

// 7時5分 / 7時
function fmtTime(ms) {
  const d = new Date(ms); const m = d.getMinutes();
  return d.getHours() + '時' + (m ? m + '分' : '');
}
function fmtDate(ms) { const d = new Date(ms); return (d.getMonth() + 1) + '月' + d.getDate() + '日'; }

const store = {
  get(k, d) { try { const v = localStorage.getItem(k); return v == null ? d : JSON.parse(v); } catch (e) { return d; } },
  set(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); } catch (e) {} },
  del(k) { try { localStorage.removeItem(k); } catch (e) {} }
};

}

const Gauge = {

  create(host, opt) {

    const el = document.createElement('div');

    const knob = el.querySelector('.g-knob');
    const hint = el.querySelector('.g-hint');
    const state = el.querySelector('.g-state');

    function draw() {
      const on = value !== null;

        hint.innerHTML = opt.hintText || 'ここを おしてね';

      knob.style.left = value + '%';
      el.setAttribute('aria-valuenow', String(value));
      el.setAttribute('aria-valuetext', String(value));
      state.innerHTML = opt.showNumber
        ? '✓ <span class="g-num">' + value + '</span>'
        : '✓ ' + (opt.doneText || R('答', 'こた') + 'えたよ');
    }
    function setFromX(x) {
      const r = track.getBoundingClientRect();
      let v = Math.round(((x - r.left) / r.width) * 100);
      v = Math.max(0, Math.min(100, v));
      if (v !== value) { value = v; draw(); if (opt.onChange) opt.onChange(v); }
    }
    let dragging = false;

    });
    draw();
    return { el, get value() { return value; } };
  },

/* ---------- 質問 ---------- */
function isEmpty(i) {
  const a = S.answers[i]; const q = S.survey[i];
  if (q.type === 'choice') return !a || !a.length;
  if (q.type === 'gauge') return a === null || a === undefined;
  return !String(a || '').trim();
}

function showQ(i) {
  S.phase = 'q'; S.idx = i; saveProgress();
  const q = S.survey[i]; const n = S.survey.length;
  let body = '';
  if (q.type === 'choice') {
    body = '<p class="note">' + (q.multi ? J('いくつでも えらべるよ', '当てはまるものをすべて選んでください（複数選択可）') : J('1つ えらんでね', '1つ選んでください')) + '</p><div class="opts">' +
      q.options.map((o, k) => '<button class="opt' + (q.multi ? ' multi' : '') + '" data-k="' + k + '" aria-pressed="false"><span class="mark"></span><span>' + qt(o) + '</span><span class="badge"></span></button>').join('') +
      '</div>';
  } else if (q.type === 'gauge') {
    body = '<p class="note">' + J('ゲージを おしてね。ゆびで うごかしても いいよ', '当てはまる位置を押してください（指でずらして調整することもできます）') + '</p><div class="gbox"></div>';
  } else {
    body = '<p class="note">' + J('ここに ' + R('書', 'か') + 'いてね', 'ご自由にお書きください') + '</p><textarea class="free" maxlength="1000" aria-label="' + J('こたえ', '回答') + '"></textarea>';
  }
  const v = show(head() +
    '<div class="progress"><i style="width:' + Math.round((i / n) * 100) + '%"></i></div>' +
    '<div class="qno">' + J('しつもん ', '質問 ') + (i + 1) + ' / ' + n +
      (q.required ? '' : J('　（' + R('答', 'こた') + 'えなくても いいよ）', '　（任意）')) + '</div>' +
    '<div class="card" style="margin-top:8px"><div class="qtext">' + qt(q.text) + '</div>' + body +
    '<div class="gentle" style="margin-top:12px"></div>' +
    '<div class="nav">' + (i > 0 ? '<button class="back btn-lg">' + J(R('前', 'まえ') + 'へ', '戻る') + '</button>' : '') +
    '<button class="pri next btn-lg">' + (i === n - 1 ? J('かくにんへ', '確認へ') : J('つぎへ', '次へ')) + '</button></div></div>');

  if (q.type === 'choice') {
    const draw = () => v.querySelectorAll('.opt').forEach(b => {
      const on = S.answers[i].indexOf(+b.dataset.k) >= 0;
      b.setAttribute('aria-pressed', on ? 'true' : 'false');
      b.querySelector('.mark').innerHTML = on ? '✓' : '';
      b.querySelector('.badge').innerHTML = on ? J('えらんだ', '選択中') : '';
    });
    draw();
    v.querySelector('.opts').addEventListener('click', e => {
      const b = e.target.closest('.opt'); if (!b) return;
      const k = +b.dataset.k; let a = S.answers[i].slice();
      if (q.multi) a = a.indexOf(k) >= 0 ? a.filter(x => x !== k) : a.concat(k).sort((x, y) => x - y);
      else a = [k];
      S.answers[i] = a; saveProgress(); draw();
      v.querySelector('.gentle').innerHTML = '';
    });
  } else if (q.type === 'gauge') {
    Gauge.create(v.querySelector('.gbox'), {
      left: q.left, right: q.right, value: S.answers[i], showNumber: S.showNumber, fmt: qt,

    const ta = v.querySelector('textarea');
    ta.value = S.answers[i] || '';

  }
  const back = v.querySelector('.back');
  if (back) back.addEventListener('click', () => showQ(i - 1));

      return;
    }
    if (i < n - 1) showQ(i + 1); else showConfirm();
  });
}

function answerText(i) {
  const q = S.survey[i], a = S.answers[i];
  if (isEmpty(i)) return '';
  if (q.type === 'choice') return a.map(k => plainText(q.options[k])).join(' / ');
  if (q.type === 'gauge') return String(a);
  return String(a).trim();
}

/* ---------- 送る（送れなかったら、ためておいて あとで送る） ---------- */
function getQueue() { return store.get('svy_queue', []); }

async function flushQueue() {
  if (S.sendBusy) return getQueue().length === 0;
  S.sendBusy = true;
  try {
    const q = getQueue(); const rest = [];
    for (const item of q) {
      try { await call('submit', item); }
      catch (e) { rest.push(item); }
    }
    // 送っているあいだに足されたものも残す
    const now = getQueue().filter(x => !q.some(y => y.sendId === x.sendId));
    store.set('svy_queue', rest.concat(now));
    return rest.length + now.length === 0;
  } finally { S.sendBusy = false; }
}

async function send() {
  const payload = {
    sendId: S.roundId + '-' + deviceId.slice(-6) + '-' + uid(),
    roundId: S.roundId, title: S.title, name: S.name, device: deviceKind(), statusKey: statusKey(),
    answers: S.survey.map((q, i) => ({
      no: i + 1, type: q.type, text: q.text, answer: answerText(i),
      value: q.type === 'gauge' ? S.answers[i] : null
    }))
  };
  store.set('svy_queue', getQueue().concat([payload]));
  store.del('svy_progress');
  if (!S.shared) store.set('svy_lastDone', S.roundId);
  show(head() + '<div class="card center msg">' + J(R('送', 'おく') + 'っているよ…', '送信中です…') + '</div>');
  await flushQueue();
  showDone(false);
}

async function startRound() {
  const c = T.cfg;
  const m = main.querySelector('#runMsg');
  const btn = main.querySelector('#startBtn');
  const say = (t, err) => { if (m) { m.className = 'msgline ' + (err ? 'err' : 'ok'); m.textContent = t; } };
  if (c.source === 'app' && !T.ed.questions.length) { say('「質問をつくる」タブで質問を作ってください。', true); return; }
  if (btn) btn.disabled = true;
  say('始めています…');
  try {
    const r = await call('tStart', T.pass, Object.assign({}, c, { questions: T.ed.questions, title: T.ed.title }));
    T.st = r.state; T.people = r.people; T.count = r.count;
    renderStatus(); renderRun();
    const m2 = main.querySelector('#runMsg'); if (m2) { m2.className = 'msgline ok'; m2.textContent = '受付を始めました。'; }
    T.resRound = T.st.roundId;
  } catch (e) {
    say('始められませんでした：' + errText(e), true);
    if (btn) btn.disabled = false;
  }
}

async function stopRound() {
  if (!await confirmBox('受付を終えますか？（終えたあとに届いた回答も記録されます）', '受付を終える')) return;
  try {
    const r = await call('tStop', T.pass);
    T.st = r.state; T.people = r.people; T.count = r.count;
    renderStatus(); if (T.tab === 'run') renderRun();
  } catch (e) { alert('終えられませんでした：' + errText(e)); }
}

function drawQList(box) {
  const qs = T.ed.questions;
  if (!qs.length) { box.innerHTML = '<p class="help">まだ質問がありません。下のボタンで足してください。</p>'; return; }
  box.innerHTML = qs.map((q, i) => {
    let extra = '';
    if (q.type === 'choice') {
      extra = '<label class="blk">選択肢（1行に1つ）</label><textarea class="t" data-f="options">' + esc((q.options || []).join('\n')) + '</textarea>' +
        '<div class="chips" style="margin-top:6px"><label><input type="checkbox" data-f="multi"' + (q.multi ? ' checked' : '') + '>いくつでも選べる</label></div>';
    } else if (q.type === 'gauge') {
      extra = '<div class="two"><div><label class="blk">左はしの言葉</label><input type="text" data-f="left" value="' + esc(q.left || '') + '" style="width:100%"></div>' +
        '<div><label class="blk">右はしの言葉</label><input type="text" data-f="right" value="' + esc(q.right || '') + '" style="width:100%"></div></div>';
    }
    return '<div class="qcard" data-i="' + i + '"><div class="qhead"><b>質問 ' + (i + 1) + '</b>' +
      '<select data-f="type">' + Object.keys(TYPE_LABEL).map(t => '<option value="' + t + '"' + (t === q.type ? ' selected' : '') + '>' + TYPE_LABEL[t] + '</option>').join('') + '</select>' +
      '<div class="chips"><label><input type="checkbox" data-f="required"' + (q.required ? ' checked' : '') + '>答えないと次へ進めない</label></div>' +
      '<div class="tools"><button data-act="up" aria-label="上へ"' + (i === 0 ? ' disabled' : '') + '>▲</button><button data-act="down" aria-label="下へ"' + (i === qs.length - 1 ? ' disabled' : '') + '>▼</button><button data-act="del">削除</button></div></div>' +
      '<label class="blk">質問文</label><textarea class="t" data-f="text">' + esc(q.text || '') + '</textarea>' +
      '<div class="rubyprev">' + (q.text ? rubyText(q.text) : '') + '</div>' + extra + '</div>';
  }).join('');
  box.oninput = e => {
    const card = e.target.closest('.qcard'); if (!card) return;
    const q = qs[+card.dataset.i]; const f = e.target.dataset.f;
    if (f === 'text') { q.text = e.target.value; card.querySelector('.rubyprev').innerHTML = q.text ? rubyText(q.text) : ''; }
    else if (f === 'options') q.options = e.target.value.split('\n').map(x => x.trim()).filter(x => x);
    else if (f === 'left' || f === 'right') q[f] = e.target.value;
    saveEd();
  };
  box.onchange = e => {
    const card = e.target.closest('.qcard'); if (!card) return;
    const i = +card.dataset.i; const q = qs[i]; const f = e.target.dataset.f;
    if (f === 'multi') q.multi = e.target.checked;
    else if (f === 'required') q.required = e.target.checked;
    else if (f === 'type') {
      const n = newQ(e.target.value); n.text = q.text; qs[i] = n; drawQList(box);
    }
    else if (f === 'options') q.options = e.target.value.split('\n').map(x => x.trim()).filter(x => x);
    saveEd();
  };
  box.onclick = async e => {
    const b = e.target.closest('button[data-act]'); if (!b) return;
    const i = +b.closest('.qcard').dataset.i;
    if (b.dataset.act === 'up' && i > 0) [qs[i - 1], qs[i]] = [qs[i], qs[i - 1]];
    else if (b.dataset.act === 'down' && i < qs.length - 1) [qs[i + 1], qs[i]] = [qs[i], qs[i + 1]];
    else if (b.dataset.act === 'del') { if (!await confirmBox('質問 ' + (i + 1) + ' を削除しますか？', '削除する')) return; qs.splice(i, 1); }
    saveEd(); drawQList(box);
  };
}

function stats(vals) {
  if (!vals.length) return null;
  const s = vals.slice().sort((a, b) => a - b);
  const sum = s.reduce((a, b) => a + b, 0);
  const mid = s.length % 2 ? s[(s.length - 1) / 2] : (s[s.length / 2 - 1] + s[s.length / 2]) / 2;
  return { n: s.length, avg: sum / s.length, min: s[0], max: s[s.length - 1], med: mid };
}

function resultsHtml(r) {
  if (!r.roundId) return '<p class="help">まだ記録がありません。受付を始めると、ここに結果が出ます。</p>';
  // 回答者ごとにまとめる
  const people = {}; const order = [];
  r.answers.forEach(a => {
    if (!people[a.sendId]) { people[a.sendId] = { name: a.name, time: a.time, by: {} }; order.push(a.sendId); }
    people[a.sendId].by[a.no] = a;
  });
  let qs = r.questions && r.questions.length ? r.questions.map((q, i) => Object.assign({ no: i + 1 }, q)) : null;
  if (!qs) {  // 質問の内容がないときは、回答から作る
    const m = {}; r.answers.forEach(a => { if (!m[a.no]) m[a.no] = { no: a.no, type: a.type, text: a.q, options: [] }; });
    qs = Object.keys(m).sort((a, b) => a - b).map(k => m[k]);
  }
  const n = order.length;
  let html = '<div class="kpi"><div><span class="n">' + n + '</span> 人が回答</div>' +
    '<div class="sub">' + esc(plainText(r.title)) + (r.started ? '　' + fmtDate(r.started) + ' ' + fmtTime(r.started) + 'に開始' : '') + (r.open ? '　<span class="pill on">受付中・自動で更新</span>' : '') + '</div></div>';
  if (!n) return html + '<p class="help">まだ回答がありません。</p>';

  qs.forEach(q => {
    const list = order.map(id => ({ p: people[id], a: people[id].by[q.no] })).filter(x => x.a);
    html += '<div class="rq"><div class="qt"><span class="tag">' + q.no + '. ' + TYPE_LABEL[q.type] + '</span>' + esc(plainText(q.text)) + '</div>';
    if (q.type === 'gauge') {
      const vals = list.filter(x => x.a.value !== null && x.a.value !== '' && !isNaN(x.a.value)).map(x => ({ v: Number(x.a.value), name: x.p.name }));
      const s = stats(vals.map(x => x.v));
      if (!s) { html += '<p class="help">数値の回答がありません。</p></div>'; return; }
      const bands = [0, 0, 0, 0, 0]; vals.forEach(x => bands[Math.min(4, Math.floor(x.v / 20))]++);
      html += '<div class="avg"><div>平均 <span class="big">' + s.avg.toFixed(1) + '</span></div>' +
        '<div>中央値 <b>' + (Math.round(s.med * 10) / 10) + '</b></div><div>いちばん低い <b>' + s.min + '</b></div><div>いちばん高い <b>' + s.max + '</b></div><div>' + s.n + '人</div></div>' +
        '<div class="plot"><div class="line"></div>' +
        vals.map((x, k) => '<span class="dot" title="' + esc(x.name) + '：' + x.v + '" style="left:' + x.v + '%;top:' + (14 + (k % 3) * 6) + 'px"></span>').join('') +
        '<span class="am" style="left:' + s.avg + '%"></span><span class="al" style="left:' + s.avg + '%">平均 ' + s.avg.toFixed(1) + '</span></div>' +
        '<div class="plot-ends"><span>0：' + esc(plainText(q.left || '')) + '</span><span>' + esc(plainText(q.right || '')) + '：100</span></div>' +
        '<div class="bands">' + ['0〜19', '20〜39', '40〜59', '60〜79', '80〜100'].map((l, k) => '<div>' + l + '<br><b>' + bands[k] + '人</b></div>').join('') + '</div>' +
        '<div class="vals">' + vals.map(x => '<span>' + esc(x.name) + '：<b>' + x.v + '</b></span>').join('') + '</div>';
    } else if (q.type === 'choice') {
      const opts = (q.options || []).map(plainText);
      const cnt = {}; opts.forEach(o => cnt[o] = 0);
      let answered = 0;
      list.forEach(x => {
        const parts = String(x.a.answer || '').split(' / ').map(s => s.trim()).filter(s => s);
        if (parts.length) answered++;
        parts.forEach(p => { cnt[p] = (cnt[p] || 0) + 1; });
      });
      const keys = Object.keys(cnt);
      const max = Math.max(1, ...keys.map(k => cnt[k]));
      html += '<div class="bars">' + keys.map(k => '<div class="bar"><span>' + esc(k) + '</span><div class="b"><i style="width:' + (cnt[k] / max * 100) + '%"></i></div><span><b>' + cnt[k] + '</b>人' + (answered ? '（' + Math.round(cnt[k] / answered * 100) + '%）' : '') + '</span></div>').join('') + '</div>' +
        (q.multi ? '<p class="help">いくつでも選べる質問です（%は答えた人の数をもとにしています）。</p>' : '');
    } else {
      const fr = list.filter(x => String(x.a.answer || '').trim());
      html += fr.length ? '<div class="frees">' + fr.map(x => '<div><b>' + esc(x.p.name) + '</b>' + esc(x.a.answer) + '</div>').join('') + '</div>' : '<p class="help">書かれた回答はありません。</p>';
    }
    html += '</div>';
  });

  // 回答者ごとの一覧
  const gaugeNos = qs.filter(q => q.type === 'gauge').map(q => q.no);
  html += '<div class="rq"><div class="qt">回答者ごとの一覧</div><div class="tbl-wrap"><table class="tbl"><thead><tr><th>名前</th><th>時刻</th>' +
    qs.map(q => '<th>' + q.no + '. ' + esc(plainText(q.text).slice(0, 14)) + '</th>').join('') +
    (gaugeNos.length > 1 ? '<th>ゲージの平均</th>' : '') + '</tr></thead><tbody>' +
    order.map(id => {
      const p = people[id];
      const gv = gaugeNos.map(no => p.by[no] && p.by[no].value).filter(v => v !== null && v !== undefined && v !== '');
      return '<tr><td>' + esc(p.name) + '</td><td>' + fmtTime(p.time) + '</td>' +
        qs.map(q => { const a = p.by[q.no]; if (!a) return '<td></td>';
          return q.type === 'gauge' ? '<td class="num">' + (a.value === null ? '' : a.value) + '</td>' : '<td class="wrap">' + esc(a.answer) + '</td>'; }).join('') +
        (gaugeNos.length > 1 ? '<td class="num">' + (gv.length ? (gv.reduce((x, y) => x + Number(y), 0) / gv.length).toFixed(1) : '') + '</td>' : '') + '</tr>';
    }).join('') + '</tbody></table></div></div>';
  return html;
}
