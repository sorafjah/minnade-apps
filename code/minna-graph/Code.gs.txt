/************************************************************
 * みんなでグラフ（棒グラフ・折れ線グラフを、みんなで1つ／ひとりずつ作る）
 *   使い方：はじめに一度だけ setup を実行し、ウェブアプリとしてデプロイ。生徒用はURL、先生用はURLの最後に ?page=teacher（URLを知っている人だけが開ける）
 *   利用条件：授業や校内での利用・改造は自由です。販売や、そのままの再配布はご遠慮ください。
 ************************************************************/

const TZ = 'Asia/Tokyo';
const SHEET_ROSTER = '名簿';
const SHEET_ODAI = 'お題';
const SHEET_LOG = '操作の記録';
const SHEET_GRAPH = 'できたグラフ';
const HEAD_ODAI = ['お題名', '項目', '単位', 'いちばん上の数', '1目盛り', 'グラフ'];
const HEAD_LOG = ['日時', '回ID', '名前', '進め方', 'お題', '項目', '前の数', 'あとの数', '操作'];
const HEAD_GRAPH = ['日時', '回ID', '進め方', '名前', 'お題', 'グラフ', '単位', 'データ', 'データ（アプリ用）'];
const CACHE_SEC = 21600; // 6時間
const MAX_ITEMS = 12;    // 「同じ表」「ひとりずつ」の項目の数の上限
const STEPS = [0.1, 0.2, 0.5, 1, 2, 5, 10, 20, 50, 100];

// setup で「お題」タブに入れる例（{漢字|かな} と書くと、生徒画面でふりがなになる）
const ODAI_EXAMPLES = [
  ['{好|す}きなくだもの', 'りんご\nバナナ\nいちご\nぶどう\nみかん', '{人|にん}', 10, 1, '棒'],
  ['1{週間|しゅうかん}の{気温|きおん}', '{月|げつ}\n{火|か}\n{水|すい}\n{木|もく}\n{金|きん}', '{度|ど}', 30, 1, '折れ線'],
  ['{植物|しょくぶつ}の{高|たか}さ', '1{週目|しゅうめ}\n2{週目|しゅうめ}\n3{週目|しゅうめ}\n4{週目|しゅうめ}\n5{週目|しゅうめ}', 'cm', 50, 1, '折れ線'],
  ['{教室|きょうしつ}にあるもの', 'つくえ\nいす\nまど\nドア\nごみばこ', 'こ', 30, 1, '棒'],
  ['ねた{時間|じかん}', '{月|げつ}\n{火|か}\n{水|すい}\n{木|もく}\n{金|きん}\n{土|ど}\n{日|にち}', '{時間|じかん}', 12, 0.5, '折れ線'],
  ['{天気|てんき}しらべ', '{晴|は}れ\nくもり\n{雨|あめ}', '{日|にち}', 20, 1, '棒']
];

const DEFAULT_SETTINGS = {
  mode: 'all', share: 'one', title: '{好|す}きなくだもの', items: ['りんご', 'バナナ', 'いちご', 'ぶどう', 'みかん'],
  unit: '{人|にん}', max: 10, step: 1, type: 'bar', canSwitch: true, free: false, showNum: true,
  kana: 'ruby', guide: 'first', join: 'roster', roster: '', useCode: false
};

/* ========== 画面 ========== */
function doGet(e) {
  const p = (e && e.parameter) || {};
  const wantTeacher = p.page === 'teacher' || p.p === 'teacher';
  const t = HtmlService.createTemplateFromFile('index');
  t.page = wantTeacher ? 'teacher' : 'student';
  return t.evaluate().setTitle(wantTeacher ? 'みんなでグラフ（先生用）' : 'みんなでグラフ')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no, viewport-fit=cover')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/* ========== はじめに一度だけ実行 ========== */
function setup() {
  const ss = ss_();
  if (!ss.getSheetByName(SHEET_ROSTER)) {
    const sh = ss.insertSheet(SHEET_ROSTER);
    sh.getRange(1, 1, 4, 1).setValues([['1組'], ['れい1'], ['れい2'], ['れい3']]);
    sh.getRange(1, 1).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  const od = sheet_(SHEET_ODAI, HEAD_ODAI);
  if (od.getLastRow() < 2) od.getRange(2, 1, ODAI_EXAMPLES.length, HEAD_ODAI.length).setValues(ODAI_EXAMPLES);
  sheet_(SHEET_LOG, HEAD_LOG);
  sheet_(SHEET_GRAPH, HEAD_GRAPH);
}

/* ========== スプレッドシート ========== */
function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }
// タブがなければ作り、足りない見出しは右に足す（前の版のシートでも動くように）
function sheet_(name, heads) {
  const ss = ss_();
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  const lastCol = sh.getLastColumn();
  const cur = lastCol ? sh.getRange(1, 1, 1, lastCol).getValues()[0].map(v => String(v).trim()) : [];
  const missing = heads.filter(h => cur.indexOf(h) < 0);
  if (missing.length) {
    const start = cur.some(v => v !== '') ? lastCol + 1 : 1;
    sh.getRange(1, start, 1, missing.length).setValues([missing]).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  return sh;
}
function heads_(sh) { return sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(v => String(v).trim()); }
function appendRows_(sh, objs) {
  if (!objs.length) return;
  const heads = heads_(sh);
  const rows = objs.map(o => heads.map(h => (Object.prototype.hasOwnProperty.call(o, h) ? o[h] : '')));
  sh.getRange(sh.getLastRow() + 1, 1, rows.length, heads.length).setValues(rows);
}
// 名簿：1行目が名簿の名前、その下に生徒の名前（列ごとに別の名簿）
function rosters_() {
  const sh = ss_().getSheetByName(SHEET_ROSTER);
  if (!sh || sh.getLastRow() < 1 || sh.getLastColumn() < 1) return [];
  const v = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getDisplayValues();
  const out = [];
  for (let c = 0; c < v[0].length; c++) {
    const title = String(v[0][c]).trim();
    if (!title) continue;
    const names = [];
    for (let r = 1; r < v.length; r++) { const n = String(v[r][c]).trim(); if (n && names.indexOf(n) < 0) names.push(n); }
    out.push({ title: title, names: names });
  }
  return out;
}
function rosterNames_(title) {
  const list = rosters_();
  const hit = list.filter(r => r.title === title)[0] || list[0];
  return hit ? hit.names : [];
}
function num_(v) {
  const s = String(v == null ? '' : v).trim().replace(/[０-９．]/g, d => String.fromCharCode(d.charCodeAt(0) - 0xFEE0));
  const n = parseFloat(s);
  return isNaN(n) ? null : n;
}
// お題タブ（なければ例）
function odaiList_() {
  const sh = sheet_(SHEET_ODAI, HEAD_ODAI);
  const last = sh.getLastRow();
  const rows = last > 1 ? sh.getRange(2, 1, last - 1, sh.getLastColumn()).getDisplayValues() : ODAI_EXAMPLES.map(r => r.map(String));
  const h = last > 1 ? heads_(sh) : HEAD_ODAI, c = x => h.indexOf(x);
  return rows.filter(r => String(r[c('お題名')]).trim()).map(r => ({
    title: String(r[c('お題名')]).trim(),
    items: String(r[c('項目')]).split(/\r?\n|、|,/).map(s => s.trim()).filter(s => s),
    unit: String(r[c('単位')]).trim(),
    max: num_(r[c('いちばん上の数')]) || 10,
    step: num_(r[c('1目盛り')]) || 1,
    type: /折/.test(String(r[c('グラフ')])) ? 'line' : 'bar'
  }));
}

/* ========== 保存場所（プロパティ・キャッシュ） ========== */
function props_() { return PropertiesService.getScriptProperties(); }
function getJSON_(k, d) { try { const v = props_().getProperty(k); return v ? JSON.parse(v) : d; } catch (err) { return d; } }
function setJSON_(k, v) { props_().setProperty(k, JSON.stringify(v)); }
function cache_() { return CacheService.getScriptCache(); }
// キャッシュのキー：日本語の名前は encodeURIComponent で（ちがう名前が同じキーにならないように）
function stKey_(rid, name) { return ('st_' + rid + '_' + encodeURIComponent(name)).slice(0, 240); }
function seenKey_(rid, name) { return ('sn_' + rid + '_' + encodeURIComponent(name)).slice(0, 240); }
function finKey_(rid, name) { return ('fn_' + rid + '_' + encodeURIComponent(name)).slice(0, 240); }
function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try { return fn(); } finally { lock.releaseLock(); }
}

/* ========== 回 ========== */
function clampN_(v, lo, hi, d) { const n = num_(v); return n === null ? d : Math.max(lo, Math.min(hi, n)); }
function sanitize_(s) {
  const d = Object.assign({}, DEFAULT_SETTINGS, s || {});
  d.mode = d.mode === 'self' ? 'self' : 'all';
  d.share = d.share === 'table' ? 'table' : 'one';
  d.title = String(d.title || '').slice(0, 60);
  d.items = (Array.isArray(d.items) ? d.items : []).map(x => String(x).trim().slice(0, 30)).filter(x => x).slice(0, MAX_ITEMS);
  d.unit = String(d.unit || '').slice(0, 20);
  d.step = STEPS.indexOf(Number(d.step)) >= 0 ? Number(d.step) : 1;
  d.max = clampN_(d.max, d.step, 100000, 10);
  d.type = d.type === 'line' ? 'line' : 'bar';
  d.canSwitch = d.canSwitch !== false;
  d.free = d.free === true;
  d.showNum = d.showNum !== false;
  d.kana = d.kana === 'hira' ? 'hira' : 'ruby';
  d.guide = d.guide === 'every' ? 'every' : 'first';
  d.join = ['roster', 'chara', 'guest'].indexOf(d.join) >= 0 ? d.join : 'roster';
  d.roster = String(d.roster || '');
  d.useCode = d.useCode === true;
  return d;
}
// 生徒に送る情報（参加コードそのもの・名簿の名前は送らない）
function pub_(r) {
  if (!r) return null;
  const s = r.settings, o = { id: r.id, prevId: r.prevId || '', status: r.status, locked: !!r.locked, code: !!r.code, startedAt: r.startedAt };
  ['mode', 'share', 'title', 'items', 'unit', 'max', 'step', 'type', 'canSwitch', 'free', 'showNum', 'kana', 'guide', 'join'].forEach(k => { o[k] = s[k]; });
  return o;
}
function modeName_(m) { return m === 'self' ? 'ひとりずつ' : 'みんなで'; }
function plain_(s) { return String(s || '').replace(/\{([^|{}]+)\|[^{}]+\}/g, '$1'); }
function fmtV_(v) { return v === null || v === undefined || v === '' ? '' : v; }
function dataText_(g) {
  return (g.items || []).map(it => plain_(it.label) + '：' + (it.v === null || it.v === undefined ? '－' : it.v)).join('、');
}
function graphRow_(r, name, g) {
  return {
    '日時': new Date(), '回ID': r.id, '進め方': modeName_(r.settings.mode), '名前': name, 'お題': plain_(g.title),
    'グラフ': g.type === 'line' ? '折れ線' : '棒', '単位': plain_(g.unit), 'データ': dataText_(g),
    'データ（アプリ用）': JSON.stringify({ title: g.title, unit: g.unit, max: g.max, step: g.step, type: g.type, items: (g.items || []).map(it => ({ label: it.label, v: it.v })) })
  };
}

/* ---------- みんなで作るグラフ（SHARED） ---------- */
function newShared_(r) {
  const s = r.settings;
  return {
    rid: r.id, ver: 1, items: s.share === 'table' ? s.items.map((lb, i) => ({ id: 'i' + i, label: lb, v: null, owner: '', by: '' })) : []
  };
}
function roundVal_(v, step) {
  if (v === null || v === undefined || v === '') return null;
  const n = Number(v);
  if (isNaN(n)) return null;
  const dec = String(step).indexOf('.') >= 0 ? String(step).split('.')[1].length : 0;
  return Math.max(0, Math.min(100000, Number(n.toFixed(dec))));
}
// 1つの操作を SHARED に入れる。記録の1行（入れられないときは null）を返す
function applyOp_(sh, op, who, teacher, s) {
  const items = sh.items, it = items.filter(x => x.id === op.itemId)[0];
  const free = s.free, one = s.share === 'one';
  const log = (label, prev, next, how) => ({ label: plain_(label), prev: fmtV_(prev), next: fmtV_(next), how: how });
  if (op.op === 'set') {
    if (!it) return null;
    if (one && !teacher && it.owner !== who) return null;
    const prev = it.v;
    it.v = roundVal_(op.v, s.step); it.by = teacher ? '先生' : who;
    return log(it.label, prev, it.v, op.how || '入力');
  }
  if (op.op === 'label') {
    if (!it) return null;
    if (!teacher && !(free && (one ? it.owner === who : true))) return null;
    const prev = it.label;
    it.label = String(op.label || '').trim().slice(0, 30) || prev;
    return log(prev, '', '', '項目の名前を「' + plain_(it.label) + '」にした');
  }
  if (op.op === 'add') {
    if (!teacher && !(free && !one)) return null;
    if (items.length >= MAX_ITEMS || !op.itemId || it) return null;
    const label = String(op.label || '').trim().slice(0, 30) || ('こうもく' + (items.length + 1));
    items.push({ id: String(op.itemId).slice(0, 40), label: label, v: null, owner: teacher ? '' : who, by: teacher ? '先生' : who });
    return log(label, '', '', '項目を足した');
  }
  if (op.op === 'del') {
    if (!it) return null;
    if (!teacher && !(free && !one && it.owner === who)) return null;
    sh.items = items.filter(x => x !== it);
    return log(it.label, it.v, '', '項目を消した');
  }
  return null;
}
// 「1人1本」で、参加した生徒の項目を作る
function ensureOwnItem_(sh, name) {
  if (sh.items.some(x => x.owner === name)) return false;
  sh.items.push({ id: 'u' + Utilities.getUuid().slice(0, 8), label: name, v: null, owner: name, by: '' });
  return true;
}
function sharedOut_(r) {
  if (!r || r.settings.mode !== 'all') return null;
  const sh = getJSON_('SHARED', null);
  return sh && sh.rid === r.id ? sh : null;
}

// 回を終える：できたグラフをシートに残す
function endRound_(r) {
  r.status = 'ended';
  r.endedAt = Date.now();
  setJSON_('ROUND', r);
  const s = r.settings, rows = [];
  if (s.mode === 'all') {
    const sh = sharedOut_(r);
    if (sh && !r.saved) {
      rows.push(graphRow_(r, 'みんな', { title: s.title, unit: s.unit, max: s.max, step: s.step, type: s.type, items: sh.items }));
      r.saved = true; setJSON_('ROUND', r);
    }
  } else {
    const mem = getJSON_('MEM', []), c = cache_();
    const sk = mem.map(m => stKey_(r.id, m.name)), fk = mem.map(m => finKey_(r.id, m.name));
    const got = mem.length ? c.getAll(sk.concat(fk)) : {}, put = {};
    mem.forEach((m, i) => {
      if (got[fk[i]] || !got[sk[i]]) return;
      let g = null;
      try { g = JSON.parse(got[sk[i]]); } catch (err) { g = null; }
      if (!g) return;
      rows.push(graphRow_(r, m.name, g));
      put[fk[i]] = '1';
    });
    if (Object.keys(put).length) c.putAll(put, CACHE_SEC);
  }
  if (rows.length) appendRows_(sheet_(SHEET_GRAPH, HEAD_GRAPH), rows);
}

/* ========== 生徒用 API ========== */
function api_poll(roundId, name) {
  const r = getJSON_('ROUND', null);
  if (r && name && roundId === r.id) cache_().put(seenKey_(r.id, name), String(Date.now()), CACHE_SEC);
  const joined = r && name && roundId === r.id;
  return { round: pub_(r), shared: joined ? sharedOut_(r) : null };
}

function api_checkCode(code) {
  const r = getJSON_('ROUND', null);
  if (!r || r.status !== 'open') return false;
  return !r.code || String(code) === r.code;
}

function api_joinInfo(tok) {
  const r = getJSON_('ROUND', null);
  const out = { round: pub_(r), names: [], taken: [] };
  if (r && r.status === 'open') {
    if (r.settings.join === 'roster') out.names = rosterNames_(r.settings.roster);
    out.taken = getJSON_('MEM', []).filter(m => m.kind === 'chara' && m.tok !== tok).map(m => m.name);
  }
  return out;
}

// 名前を入れない参加：端末ごとに「ゲスト1」「ゲスト2」…（同じ端末なら同じ番号）
function guestName_(tok) {
  const g = getJSON_('GUESTS', {});
  if (!g[tok]) {
    let max = 0;
    Object.keys(g).forEach(k => { max = Math.max(max, g[k]); });
    g[tok] = max + 1;
    setJSON_('GUESTS', g);
  }
  return 'ゲスト' + g[tok];
}

function api_join(o) {
  return withLock_(() => {
    const r = getJSON_('ROUND', null);
    if (!r || r.status !== 'open') return { ok: false, err: 'closed', round: pub_(r) };
    const tok = String((o && o.tok) || '').slice(0, 60);
    const kind = r.settings.join;
    let name = String((o && o.name) || '').trim().slice(0, 30);
    if (kind === 'guest') name = guestName_(tok);
    if (!name) return { ok: false, err: 'name' };
    const mem = getJSON_('MEM', []);
    const me = mem.filter(m => m.name === name)[0];
    const prevOk = !!(o.prevId && o.prevId === r.prevId && getJSON_('PREV_MEM', []).indexOf(name) >= 0);
    if (r.code && !me && !prevOk && String(o.code) !== r.code) return { ok: false, err: 'code' };
    if (me && me.kind === 'chara' && me.tok !== tok) return { ok: false, err: 'taken' };
    if (me) me.tok = tok;
    else mem.push({ name: name, kind: kind, tok: tok });
    setJSON_('MEM', mem);
    const out = { ok: true, name: name, round: pub_(r) };
    if (r.settings.mode === 'all') {
      const sh = sharedOut_(r) || newShared_(r);
      if (r.settings.share === 'one' && ensureOwnItem_(sh, name)) { sh.ver++; setJSON_('SHARED', sh); }
      out.shared = sh;
    } else {
      const g = cache_().get(stKey_(r.id, name));
      try { out.mine = g ? JSON.parse(g) : null; } catch (err) { out.mine = null; }
    }
    return out;
  });
}

// 生徒から届いたもの（送れなかった分もまとめて届く。同じものの2回目は使わない）
//   op …「みんなで」のグラフを変える操作　log …「ひとりずつ」の操作の記録　snap …「ひとりずつ」のいまのグラフ
function api_submit(items) {
  if (!items || !items.length) return { done: [] };
  return withLock_(() => {
    const c = cache_(), r = getJSON_('ROUND', null);
    const idKeys = items.map(it => 'it_' + String(it.id).slice(0, 60));
    const seen = c.getAll(idKeys), newIds = {};
    let sh = null, changed = false;
    const logs = [], graphRows = [], put = {};
    items.forEach((it, i) => {
      if (seen[idKeys[i]]) return;
      newIds[idKeys[i]] = '1';
      const name = String(it.name || '');
      if (it.kind === 'op') {
        if (!r || r.id !== it.roundId || r.status !== 'open' || r.locked || r.settings.mode !== 'all') return;
        sh = sh || sharedOut_(r) || newShared_(r);
        const lg = applyOp_(sh, it, name, false, r.settings);
        if (!lg) return;
        changed = true;
        logs.push(logRow_(r, name, lg, it.at));
      } else if (it.kind === 'log') {
        if (!r || r.id !== it.roundId) return;
        logs.push(logRow_(r, name, { label: plain_(it.label), prev: fmtV_(it.prev), next: fmtV_(it.next), how: it.how || '' }, it.at));
      } else if (it.kind === 'snap') {
        if (!it.snap) return;
        const k = stKey_(it.roundId || '', name);
        put[k] = JSON.stringify(it.snap);
        // 回が終わったあとに届いたグラフは、まだ残していなければ残す
        if (r && r.id === it.roundId && r.status !== 'open' && r.settings.mode === 'self' && !c.get(finKey_(r.id, name))) {
          graphRows.push(graphRow_(r, name, it.snap));
          put[finKey_(r.id, name)] = '1';
        }
      }
    });
    if (changed) { sh.ver++; setJSON_('SHARED', sh); }
    if (logs.length) appendRows_(sheet_(SHEET_LOG, HEAD_LOG), logs);
    if (graphRows.length) appendRows_(sheet_(SHEET_GRAPH, HEAD_GRAPH), graphRows);
    Object.assign(put, newIds);
    if (Object.keys(put).length) c.putAll(put, CACHE_SEC);
    return { done: items.map(it => it.id), shared: sharedOut_(r) };
  });
}
function logRow_(r, name, lg, at) {
  return {
    '日時': new Date(at || Date.now()), '回ID': r.id, '名前': name, '進め方': modeName_(r.settings.mode), 'お題': plain_(r.settings.title),
    '項目': lg.label, '前の数': lg.prev, 'あとの数': lg.next, '操作': lg.how
  };
}

/* ========== 先生用 API（先生用のURLを知っている人が呼ぶ。アカウントの確かめはしない） ========== */
function api_tInit() {
  const r = getJSON_('ROUND', null);
  return {
    settings: sanitize_(getJSON_('LAST_SETTINGS', {})),
    rosters: rosters_(),
    odai: odaiList_(),
    url: ScriptApp.getService().getUrl(),
    round: r, shared: sharedOut_(r)
  };
}

function api_tStart(settings) {
  return withLock_(() => {
    const old = getJSON_('ROUND', null);
    if (old && old.status === 'open') endRound_(old);
    const s = sanitize_(settings);
    if (s.mode === 'all' && s.share === 'table' && !s.items.length) s.items = ['こうもく1', 'こうもく2', 'こうもく3'];
    if (s.mode === 'self' && !s.items.length && !s.free) s.items = ['こうもく1', 'こうもく2', 'こうもく3'];
    const r = {
      id: Utilities.formatDate(new Date(), TZ, 'MMdd-HHmmss'),
      prevId: old ? old.id : '',
      status: 'open', locked: false,
      settings: s,
      code: s.useCode ? String(1000 + Math.floor(Math.random() * 9000)) : '',
      startedAt: Date.now()
    };
    setJSON_('PREV_MEM', getJSON_('MEM', []).map(m => m.name));
    setJSON_('MEM', []);
    setJSON_('ROUND', r);
    if (s.mode === 'all') setJSON_('SHARED', newShared_(r));
    setJSON_('LAST_SETTINGS', s);
    return { round: r, shared: sharedOut_(r) };
  });
}

function api_tPoll() {
  const r = getJSON_('ROUND', null);
  if (!r) return { round: null, members: [], shared: null };
  const mem = getJSON_('MEM', []);
  const sk = mem.map(m => stKey_(r.id, m.name)), nk = mem.map(m => seenKey_(r.id, m.name));
  const got = mem.length ? cache_().getAll(r.settings.mode === 'self' ? sk.concat(nk) : nk) : {};
  const now = Date.now();
  const members = mem.map((m, i) => {
    let g = null;
    if (r.settings.mode === 'self') { try { g = got[sk[i]] ? JSON.parse(got[sk[i]]) : null; } catch (err) { g = null; } }
    const seen = parseInt(got[nk[i]] || '0', 10);
    return { name: m.name, kind: m.kind, graph: g, ago: seen ? Math.round((now - seen) / 1000) : -1 };
  });
  return { round: r, members: members, shared: sharedOut_(r) };
}

// 入力をしめきる・再開する
function api_tLock(roundId, on) {
  return withLock_(() => {
    const r = getJSON_('ROUND', null);
    if (r && r.id === roundId && r.status === 'open') { r.locked = on === true; setJSON_('ROUND', r); }
    return r;
  });
}

function api_tEnd(roundId) {
  return withLock_(() => {
    const r = getJSON_('ROUND', null);
    if (r && r.status === 'open' && (!roundId || r.id === roundId)) endRound_(r);
    return getJSON_('ROUND', null);
  });
}

// 先生が「みんなで」のグラフを直す（数・項目の名前・項目を足す・消す）
function api_tOp(roundId, op) {
  return withLock_(() => {
    const r = getJSON_('ROUND', null);
    if (!r || r.id !== roundId || r.settings.mode !== 'all') return sharedOut_(r);
    const sh = sharedOut_(r) || newShared_(r);
    const lg = applyOp_(sh, op || {}, '先生', true, r.settings);
    if (lg) {
      sh.ver++; setJSON_('SHARED', sh);
      appendRows_(sheet_(SHEET_LOG, HEAD_LOG), [logRow_(r, '先生', lg)]);
    }
    return sh;
  });
}

// いまのお題を「お題」タブに足す
function api_tSaveOdai(o) {
  const s = sanitize_(o);
  return withLock_(() => {
    const sh = sheet_(SHEET_ODAI, HEAD_ODAI);
    appendRows_(sh, [{ 'お題名': s.title, '項目': s.items.join('\n'), '単位': s.unit, 'いちばん上の数': s.max, '1目盛り': s.step, 'グラフ': s.type === 'line' ? '折れ線' : '棒' }]);
    return odaiList_();
  });
}

function fmtJ_(v) {
  const d = v instanceof Date ? v : new Date(v);
  if (isNaN(d.getTime())) return String(v);
  const m = parseInt(Utilities.formatDate(d, TZ, 'm'), 10);
  return Utilities.formatDate(d, TZ, 'M月d日 H時') + (m ? m + '分' : '');
}

// できたグラフ（新しい順に60まで）
function api_tGraphs() {
  const sh = sheet_(SHEET_GRAPH, HEAD_GRAPH), last = sh.getLastRow(), h = heads_(sh), c = x => h.indexOf(x);
  if (last < 2) return [];
  const n = Math.min(60, last - 1);
  return sh.getRange(last - n + 1, 1, n, h.length).getValues().reverse().map(r => {
    let g = null;
    try { g = JSON.parse(r[c('データ（アプリ用）')]); } catch (err) { g = null; }
    return { at: fmtJ_(r[c('日時')]), rid: String(r[c('回ID')]), mode: String(r[c('進め方')]), name: String(r[c('名前')]), title: String(r[c('お題')]), data: String(r[c('データ')]), graph: g };
  }).filter(x => x.graph);
}
