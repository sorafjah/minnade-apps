function bindDrag(host, svg) {
  var st = host._ch;
  function pos(e) { var b = svg.getBoundingClientRect(); return { x: e.clientX - b.left, y: e.clientY - b.top }; }
  function valAt(y) {
    var g = st.g, v = (g.y1 - y) / g.ph * g.ax.am;
    v = Math.round(v / g.ax.step) * g.ax.step;
    return Number(Math.max(0, Math.min(g.ax.am, v)).toFixed(decOf(g.ax.step)));
  }
  function redraw() { svg.innerHTML = chartInner(st); }
  svg.addEventListener('pointerdown', function (e) {
    var g = st.g, p = pos(e), i = Math.floor((p.x - g.x0) / g.colW);
    if (p.x < g.x0 - 4 || i < 0 || i >= (st.m.items || []).length || !st.o.edit || !st.o.edit(i)) return;
    e.preventDefault();
    try { svg.setPointerCapture(e.pointerId); } catch (err) {}
    st.drag = { i: i, pid: e.pointerId, v0: st.m.items[i].v, v: valAt(p.y), sy: p.y, moved: false };
    redraw();
  });
  svg.addEventListener('pointermove', function (e) {
    var d = st.drag;
    if (!d || d.pid !== e.pointerId) return;
    e.preventDefault();
    var p = pos(e);
    if (Math.abs(p.y - d.sy) > 6) d.moved = true;
    var v = valAt(p.y);
    if (v !== d.v) { d.v = v; redraw(); }
  });
  function end(e, ok) {
    var d = st.drag;
    if (!d || d.pid !== e.pointerId) return;
    st.drag = null;
    var changed = ok && d.v !== d.v0;
    if (changed) st.m.items[d.i].v = d.v;
    if (changed && st.o.onSet) st.o.onSet(d.i, d.v, d.moved ? 'ドラッグ' : 'タップ');
    if (st.dirty) { st.dirty = false; chartRender(host); } else redraw();
  }
  svg.addEventListener('pointerup', function (e) { end(e, true); });
  svg.addEventListener('pointercancel', function (e) { end(e, false); });
}

function findItem(id) { return model().items.filter(function (x) { return x.id === id; })[0]; }
function doSet(id, v, how, prevOpt, noUndo) {
  var rd = ST.round, it = findItem(id);
  if (!it || !canEdit(it)) { render(); return; }
  var prev = prevOpt !== undefined ? prevOpt : it.v;
  if (v === prev) { render(); return; }
  if (!noUndo) { ST.undo.push({ id: id, prev: prev }); if (ST.undo.length > 30) ST.undo.shift(); }
  if (rd.mode === 'all') pushOp({ op: 'set', itemId: id, v: v, how: how });
  else {
    var mi = ST.my.items.filter(function (x) { return x.id === id; })[0];
    mi.v = v; saveMy();
    pushLog(mi.label, prev, v, how);
    scheduleSnap();
  }
  render();
}
// △▽は、続けて押したあと、少し待ってから1つの操作として送る
function stepTap(id, d) {
  var it = findItem(id), step = (ST.round.mode === 'self' ? ST.my.step : ST.round.step) || 1;
  if (!it || !canEdit(it)) return;
  if (!(id in ST.stepBase)) ST.stepBase[id] = it.v;
  var nv = roundVal(Math.max(0, (it.v || 0) + d * step), step);
  if (it.v == null && d < 0) nv = 0;
  ST.stepOv[id] = nv;
  render();
  clearTimeout(ST.stepT[id]);
  ST.stepT[id] = setTimeout(function () {
    var b = ST.stepBase[id], v = ST.stepOv[id];
    delete ST.stepBase[id]; delete ST.stepOv[id];
    doSet(id, v, '△▽', b);
  }, 700);
}
function undo() {
  var u = ST.undo.pop();
  if (!u) return;
  doSet(u.id, u.prev, 'もどす', undefined, true);
}
function setLabel(id, val) {
  val = String(val).trim().slice(0, 30);
  if (!val) { ST.tableSig = ''; render(); return; }
  if (ST.round.mode === 'all') pushOp({ op: 'label', itemId: id, label: val });
  else {
    var mi = ST.my.items.filter(function (x) { return x.id === id; })[0];
    if (!mi || mi.label === val) return;
    pushLog(mi.label || '（なし）', '', '', '項目の名前を「' + val + '」にした');
    mi.label = val; saveMy(); scheduleSnap();
  }
  render();
}

  if (ST.round.mode === 'all') pushOp({ op: 'del', itemId: id });

  }
  ST.undo = ST.undo.filter(function (u) { return u.id !== id; });
  render();
}

      var inp = box.querySelector('input[data-tv="' + it.id + '"]');
      if (inp && inp !== document.activeElement) inp.value = fmtN(it.v);
      var by = box.querySelector('[data-by="' + it.id + '"]');
      if (by) by.textContent = it.by || '';
    });
    return;
  }
  var a = document.activeElement;
  if (a && box.contains(a)) return;
  T.tSig = sig;
  var h = m.items.map(function (it) {
    return '<tr><td>' + (open ? '<input data-tl="' + esc(it.id) + '" value="' + esc(noRuby(it.label)) + '" style="width:130px">' : P(it.label)) + '</td>' +
      '<td>' + (open ? '<input class="v" data-tv="' + esc(it.id) + '" inputmode="decimal" value="' + fmtN(it.v) + '">' : fmtN(it.v)) + ' ' + P(s.unit) + '</td>' +
      '<td data-by="' + esc(it.id) + '">' + esc(it.by || '') + '</td>' +
      '<td>' + (open ? '<button class="ghost" style="font-size:14px;min-height:40px" data-tdel="' + esc(it.id) + '">消す</button>' : '') + '</td></tr>';
  }).join('');
  box = swap('tTable', h || '<tr><td colspan="4" class="subc">まだ項目がありません。</td></tr>');
  box.querySelectorAll('input[data-tv]').forEach(function (inp) {
    inp.addEventListener('change', function () { var v = parseNum(inp.value); if (v === undefined) { inp.value = ''; return; } tOp({ op: 'set', itemId: inp.dataset.tv, v: v, how: '先生が入力' }); });
  });
  box.querySelectorAll('input[data-tl]').forEach(function (inp) {
    inp.addEventListener('change', function () {
      var orig = (T.shared.items.filter(function (x) { return x.id === inp.dataset.tl; })[0] || {}).label;
      if (inp.value.trim() && inp.value.trim() !== noRuby(orig)) tOp({ op: 'label', itemId: inp.dataset.tl, label: inp.value });
    });
  });
  box.querySelectorAll('[data-tdel]').forEach(function (b) {
    b.addEventListener('click', function () {
      if (b.dataset.sure !== '1') { b.dataset.sure = '1'; b.textContent = 'もう一度押すと消します'; return; }
      T.tSig = ''; tOp({ op: 'del', itemId: b.dataset.tdel });
    });
  });
}

/* --- 大きく映す --- */
function openBig(b) { T.big = b; T.bvType = ''; $('bigView').classList.remove('hidden'); renderBig(); }
function closeBig() { T.big = null; $('bigView').classList.add('hidden'); }
function bigList() { return T.big.kind === 'rec' ? T.recs.map(function (x, i) { return { key: i, g: x.graph }; }) : T.members.filter(function (m) { return m.graph; }).map(function (m) { return { key: m.name, g: m.graph }; }); }
function renderBig() {
  var b = T.big;
  if (!b) return;
  var s = T.round ? T.round.settings : DEF_T, g = null, title = '', sub = '', series = null;
  KANA_ONLY = s.kana === 'hira';   // 生徒に見せる画面なので、生徒画面と同じ文字にする
  var nav = b.kind === 'stu' || b.kind === 'rec';
  $('bvPrev').classList.toggle('hidden', !nav); $('bvNext').classList.toggle('hidden', !nav);
  $('bvTypeSeg').classList.toggle('hidden', b.kind === 'cmp');
  swap('bvChips', ''); swap('bvLegend', '');
  if (b.kind === 'all') {
    if (!T.round || !T.shared) { closeBig(); return; }
    g = sharedModel(T.vType); title = s.title; sub = T.round.locked ? '入力をしめきっています' : '';
  } else if (b.kind === 'stu') {
    var m = T.members.filter(function (x) { return x.name === b.name; })[0];
    if (!m || !m.graph) { closeBig(); return; }
    g = clone(m.graph); title = g.title || ''; sub = m.name;
  } else if (b.kind === 'rec') {
    var rc = T.recs[b.idx];
    if (!rc) { closeBig(); return; }
    g = clone(rc.graph); title = g.title || rc.title; sub = rc.name + '・' + rc.at;
  } else if (b.kind === 'cmp') {
    var withG = T.members.filter(function (x) { return x.graph; });
    if (!b.sel) b.sel = withG.slice(0, 8).map(function (x) { return x.name; });
    b.sel.forEach(function (n) { colorFor(n); });
    title = s.title; sub = '重ねて比べる（8人まで）';
    series = withG.filter(function (x) { return b.sel.indexOf(x.name) >= 0; }).map(function (x) {
      return { name: x.name, color: T.cmpColor[x.name], values: s.items.map(function (lb, k) { var it = x.graph.items[k]; return it ? it.v : null; }) };
    });
    g = { title: s.title, unit: s.unit, max: s.max, step: s.step, type: 'line', items: s.items.map(function (lb, k) { return { id: 'i' + k, label: lb, v: null }; }), series: series };
    var chips = swap('bvChips', withG.map(function (x) { return '<button class="gchip' + (b.sel.indexOf(x.name) >= 0 ? ' on' : '') + '" data-n="' + esc(x.name) + '">' + esc(x.name) + '</button>'; }).join(''));
    chips.querySelectorAll('[data-n]').forEach(function (c) {
      c.addEventListener('click', function () {
        var n = c.dataset.n, i = b.sel.indexOf(n);
        if (i >= 0) { b.sel.splice(i, 1); delete T.cmpColor[n]; } else if (b.sel.length < 8) b.sel.push(n);
        renderBig();
      });
    });
    swap('bvLegend', series.map(function (x) { return '<span><i style="background:' + x.color + '"></i>' + esc(x.name) + '</span>'; }).join(''));
  }
  if (T.bvType && !series) g.type = T.bvType;
  $('bvTypeSeg').querySelectorAll('button').forEach(function (x) { x.classList.toggle('on', x.dataset.v === g.type); });
  swap('bvTitle', RE(title));
  swap('bvSub', esc(sub));
  drawChart($('bvChart'), g, { showNum: b.kind === 'cmp' ? false : s.showNum !== false || b.kind === 'rec' });
  KANA_ONLY = false;
}
// 色は人についていく（選び直しても、残った人の色は変えない）
function colorFor(n) {
  if (T.cmpColor[n]) return T.cmpColor[n];
  var used = Object.keys(T.cmpColor).map(function (k) { return T.cmpColor[k]; });
  T.cmpColor[n] = SER_COLORS.filter(function (c) { return used.indexOf(c) < 0; })[0] || SER_COLORS[0];
  return T.cmpColor[n];
}
function bigMove(d) {
  var b = T.big, list = bigList();
  if (!list.length) return;
  var cur = b.kind === 'rec' ? b.idx : b.name;
  var i = list.map(function (x) { return x.key; }).indexOf(cur);
  i = (i + d + list.length) % list.length;
  if (b.kind === 'rec') b.idx = list[i].key; else b.name = list[i].key;
  renderBig();
}
