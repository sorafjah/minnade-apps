<script>
/************************************************************
 * ジャンル：時計の読み取り
 * どのジャンルも同じ約束で作る
 *   make(level, prev)   … 問題をつくる
 *   render(q, el, opt)  … 問題の絵をかく（opt.explain で解説用の色）
 *   choices(q, n)       … 選択肢 n こ（正解1つ＋まちがいの型つきの誤答）
 *   judge(q, given)     … 正誤とまちがいの型
 *   hint(type)          … まちがえたときのヒント（ふりがなつき）
 ************************************************************/
window.GENRES = window.GENRES || {};
(function () {
  const STEP = [60, 30, 15, 5, 1];

  function fmt(h, m) { return m === 0 ? `${h}時` : `${h}時${m}分`; }
  function wrapH(h) { return ((h - 1) % 12 + 12) % 12 + 1; }
  function rint(a, b) { return a + Math.floor(Math.random() * (b - a + 1)); }

  function make(level, prev) {
    const step = STEP[level - 1];
    let h, m, guard = 0;
    do {
      h = rint(1, 12);
      m = step === 60 ? 0 : rint(0, 60 / step - 1) * step;
      // レベル4・5は「ちょうど」が出すぎないように
      if (level >= 4 && m === 0) continue;
      guard++;
    } while (guard < 30 && prev && prev.h === h && prev.m === m);
    if (level >= 4 && m === 0) m = step;
    return { h, m, level, text: fmt(h, m) + 'の時計', answerText: fmt(h, m) };
  }

  // まちがえやすい答え（よくある読みまちがい）を、出しやすい順にならべる
  function candidates(q) {
    const { h, m, level } = q;
    const list = [];
    const add = (hh, mm, type) => {
      hh = wrapH(hh); mm = ((mm % 60) + 60) % 60;
      if (hh === h && mm === m) return;
      if (list.some(c => c.h === hh && c.m === mm)) return;
      list.push({ h: hh, m: mm, type });
    };
    // 長いはりと短いはりを取りちがえる
    const swapH = m === 0 ? 12 : (Math.floor(m / 5) || 12);
    const swapM = ((h % 12) * 5 + Math.floor(m / 12)) % 60;
    add(swapH, swapM, 'はりの取りちがえ');
    // 短いはりが次の数字に近いので、次の時と読む
    if (m >= 25) add(h + 1, m, '次の時と読む');
    // 長いはりがさす数字をそのまま分と読む（20分→4分）
    if (m > 0 && m % 5 === 0) add(h, m / 5, '分の読みかた');
    add(h - 1, m, '前の時と読む');
    if (m < 25) add(h + 1, m, '次の時と読む');
    // 分が少しずれる
    if (level >= 2) {
      const s = STEP[level - 1];
      add(h, m + s, '分のずれ');
      add(h, m - s, '分のずれ');
      if (level >= 4) { add(h, m + 5 * (level === 5 ? 1 : 2), '分のずれ'); }
    }
    return list;
  }

  function choices(q, n) {
    const cands = candidates(q);
    // 上位から少しだけ入れかえて、毎回同じ並びにならないようにする
    const top = cands.slice(0, Math.min(cands.length, n + 1));
    for (let i = top.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [top[i], top[j]] = [top[j], top[i]];
    }
    const pick = top.slice(0, n - 1);
    // 足りないときは、そのレベルのきざみでずらした時刻でうめる
    const st = q.level === 1 ? 30 : STEP[q.level - 1];
    let k = 2;
    while (pick.length < n - 1 && k < 120) {
      const mm = (q.m + k * st) % 60;
      if (!pick.some(c => c.h === q.h && c.m === mm) && mm !== q.m) pick.push({ h: q.h, m: mm, type: 'そのほか' });
      k++;
    }
    const all = pick.map(c => ({ h: c.h, m: c.m, type: c.type, correct: false, text: fmt(c.h, c.m) }));
    all.push({ h: q.h, m: q.m, type: '', correct: true, text: fmt(q.h, q.m) });
    for (let i = all.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [all[i], all[j]] = [all[j], all[i]];
    }
    return all;
  }

  function judge(q, given) {
    const ok = given.h === q.h && given.m === q.m;
    if (ok) return { ok: true, type: '' };
    if (given.type) return { ok: false, type: given.type };
    const c = candidates(q).find(c => c.h === given.h && c.m === given.m);
    return { ok: false, type: c ? c.type : 'そのほか' };
  }

  // まちがえた問題に「にた問題」：分はそのまま、時だけかえる
  function similar(q) {
    let h; do { h = rint(1, 12); } while (h === q.h);
    return { h, m: q.m, level: q.level, text: fmt(h, q.m) + 'の時計', answerText: fmt(h, q.m) };
  }

  // その問題が、そのまちがいの型を練習できる問題かどうか
  function fits(q, type) {
    const m = q.m;
    switch (type) {
      case 'はりの取りちがえ': return m > 0 && Math.floor(m / 5) !== q.h % 12;
      case '次の時と読む': return m >= 30;
      case '前の時と読む': return m > 0 && m <= 20;
      case '分の読みかた': return m >= 10 && m % 5 === 0;
      case '分のずれ': return m % 15 !== 0;
      default: return false;
    }
  }

  const HINTS = {
    'はりの取りちがえ': '<ruby>短<rt>みじか</rt></ruby>い はりが「<ruby>何時<rt>なんじ</rt></ruby>」、<ruby>長<rt>なが</rt></ruby>い はりが「<ruby>何分<rt>なんぷん</rt></ruby>」を おしえているよ。',
    '次の時と読む': '<ruby>短<rt>みじか</rt></ruby>い はりは、まだ <ruby>次<rt>つぎ</rt></ruby>の <ruby>数字<rt>すうじ</rt></ruby>の <ruby>手前<rt>てまえ</rt></ruby>に あるよ。',
    '前の時と読む': '<ruby>短<rt>みじか</rt></ruby>い はりが <ruby>通<rt>とお</rt></ruby>りすぎた <ruby>数字<rt>すうじ</rt></ruby>を <ruby>読<rt>よ</rt></ruby>むよ。',
    '分の読みかた': '<ruby>長<rt>なが</rt></ruby>い はりは、<ruby>数字<rt>すうじ</rt></ruby>が 1つ すすむと 5分 だよ。',
    '分のずれ': '<ruby>長<rt>なが</rt></ruby>い はりの <ruby>目<rt>め</rt></ruby>もりを、ゆっくり <ruby>数<rt>かぞ</rt></ruby>えてみてね。',
    'そのほか': '<ruby>短<rt>みじか</rt></ruby>い はりと <ruby>長<rt>なが</rt></ruby>い はりを、ひとつずつ <ruby>見<rt>み</rt></ruby>てみてね。',
  };
  function hint(type) { return HINTS[type] || HINTS['そのほか']; }

  // 時計の絵（SVG）。explain=true のとき、短いはり＝オレンジ、長いはり＝青で色分けして名前をつける
  function svg(h, m, explain) {
    const cx = 100, cy = 100;
    let s = `<svg viewBox="0 0 200 200" xmlns="http://www.w3.org/2000/svg" role="img" aria-label="時計">`;
    s += `<circle cx="${cx}" cy="${cy}" r="94" fill="#fff" stroke="#2b3a4a" stroke-width="3"/>`;
    for (let i = 0; i < 60; i++) {
      const a = i * 6 * Math.PI / 180, five = i % 5 === 0;
      const r1 = five ? 82 : 87, r2 = 92;
      s += `<line x1="${cx + r1 * Math.sin(a)}" y1="${cy - r1 * Math.cos(a)}" x2="${cx + r2 * Math.sin(a)}" y2="${cy - r2 * Math.cos(a)}" stroke="#2b3a4a" stroke-width="${five ? 2.4 : 1}"/>`;
    }
    for (let i = 1; i <= 12; i++) {
      const a = i * 30 * Math.PI / 180;
      s += `<text x="${cx + 69 * Math.sin(a)}" y="${cy - 69 * Math.cos(a) + 7}" font-size="20" font-weight="700" text-anchor="middle" fill="#2b3a4a" font-family="BIZ UDPGothic, sans-serif">${i}</text>`;
    }
    const ha = ((h % 12) + m / 60) * 30 * Math.PI / 180;
    const ma = m * 6 * Math.PI / 180;
    const hc = explain ? '#e8913a' : '#2b3a4a', mc = explain ? '#3f8cd6' : '#2b3a4a';
    s += `<line x1="${cx}" y1="${cy}" x2="${cx + 46 * Math.sin(ha)}" y2="${cy - 46 * Math.cos(ha)}" stroke="${hc}" stroke-width="8" stroke-linecap="round"/>`;
    s += `<line x1="${cx}" y1="${cy}" x2="${cx + 74 * Math.sin(ma)}" y2="${cy - 74 * Math.cos(ma)}" stroke="${mc}" stroke-width="5" stroke-linecap="round"/>`;
    s += `<circle cx="${cx}" cy="${cy}" r="5" fill="#2b3a4a"/>`;
    s += `</svg>`;
    return s;
  }

  function render(q, el, opt) {
    const explain = opt && opt.explain;
    let html = `<div class="clock-box">${svg(q.h, q.m, explain)}</div>`;
    if (explain) {
      html += `<div class="clock-legend"><span class="lg lg-h">■ <ruby>短<rt>みじか</rt></ruby>い はり＝<ruby>何時<rt>なんじ</rt></ruby></span>`
        + `<span class="lg lg-m">■ <ruby>長<rt>なが</rt></ruby>い はり＝<ruby>何分<rt>なんぷん</rt></ruby></span></div>`;
    }
    el.innerHTML = html;
  }

  window.GENRES.clock = {
    key: 'clock',
    name: '時計の読み取り',
    nameRuby: '<ruby>時計<rt>とけい</rt></ruby>',
    prompt: '<ruby>何時何分<rt>なんじなんぷん</rt></ruby>かな？',
    answerKind: 'time',
    levels: [
      { label: 'ちょうど', ex: '3時' },
      { label: '30分きざみ', ex: '3時30分' },
      { label: '15分きざみ', ex: '3時15分' },
      { label: '5分きざみ', ex: '3時25分', mode: 'input' },
      { label: '1分きざみ', ex: '3時27分', mode: 'input' },
    ],
    fmt, make, render, choices, judge, hint, similar, fits,
  };
})();

/************************************************************
 * ジャンル：何分まえ・何分あと
 * 時計（はじめの時刻）と「○分あと／まえ」を見て、その時刻を答える
 * どのジャンルも同じ約束で作る（genre_clock.html を参照）

window.GENRES = window.GENRES || {};
(function () {
  // step：時刻と分のきざみ、diffs：何分（最小・最大）、both：まえも出す、cross：時をこえるか
  const CONF = [
    { step: 10, min: 10, max: 30, both: false, cross: 'no' },
    { step: 10, min: 10, max: 30, both: true,  cross: 'no' },
    { step: 5,  min: 5,  max: 30, both: true,  cross: 'no' },
    { step: 5,  min: 5,  max: 45, both: true,  cross: 'yes' },
    { step: 1,  min: 1,  max: 59, both: true,  cross: 'any' },
  ];

  function fmt(h, m) { return m === 0 ? `${h}時` : `${h}時${m}分`; }
  function rint(a, b) { return a + Math.floor(Math.random() * (b - a + 1)); }
  function toMin(h, m) { return (h % 12) * 60 + m; }
  function fromMin(t) {
    t = ((t % 720) + 720) % 720;
    const h = Math.floor(t / 60);
    return { h: h === 0 ? 12 : h, m: t % 60 };
  }
  function wrapH(h) { return ((h - 1) % 12 + 12) % 12 + 1; }
  const AFTER = 'あと', BEFORE = 'まえ';

  function build(h, m, diff, after, level) {
    const e = fromMin(toMin(h, m) + (after ? diff : -diff));
    const end = m + (after ? diff : -diff);
    return {
      h0: h, m0: m, diff, after, level,
      h: e.h, m: e.m, cross: end < 0 || end >= 60,
      text: `${fmt(h, m)}の${diff}分${after ? AFTER : BEFORE}`,
      answerText: fmt(e.h, e.m),
    };
  }

  function make(level, prev) {
    const c = CONF[level - 1];
    for (let guard = 0; guard < 200; guard++) {
      const diff = rint(c.min / c.step, c.max / c.step) * c.step;
      const after = c.both ? Math.random() < 0.5 : true;
      const h = rint(1, 12);
      const m = rint(0, 60 / c.step - 1) * c.step;
      const end = m + (after ? diff : -diff);
      const cross = end < 0 || end >= 60;
      if (c.cross === 'no' && cross) continue;
      if (c.cross === 'yes' && !cross) continue;
      if (prev && prev.h0 === h && prev.m0 === m && prev.diff === diff) continue;
      return build(h, m, diff, after, level);
    }
    return build(3, 10, 20, true, level);
  }

  // まちがえやすい答え（出しやすい順）

      hh = wrapH(hh); mm = ((mm % 60) + 60) % 60;
      if (hh === q.h && mm === q.m) return;
      if (list.some(c => c.h === hh && c.m === mm)) return;
      list.push({ h: hh, m: mm, type });
    };
    // まえ と あと を反対にした
    const o = fromMin(toMin(q.h0, q.m0) + (q.after ? -q.diff : q.diff));
    add(o.h, o.m, 'まえとあとの取りちがえ');
    if (q.cross) {
      // 分は合っているが、時をかえわすれた
      add(q.h0, q.m, '時がかわっていない');
      add(q.h + (q.after ? 1 : -1), q.m, '時のかえすぎ');
    } else {
      add(q.h + 1, q.m, '時のまちがい');
      add(q.h - 1, q.m, '時のまちがい');
    }
    const s = CONF[q.level - 1].step;
    add(q.h, q.m + s, '分の計算');
    add(q.h, q.m - s, '分の計算');
    if (s === 1) { add(q.h, q.m + 10, '分の計算'); add(q.h, q.m - 10, '分の計算'); }
    // はじめの時刻のまま
    add(q.h0, q.m0, 'そのほか');
    return list;
  }

  function choices(q, n) {
    const cands = candidates(q);
    const top = cands.slice(0, Math.min(cands.length, n + 1));
    for (let i = top.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [top[i], top[j]] = [top[j], top[i]];
    }
    const pick = top.slice(0, n - 1);
    const st = CONF[q.level - 1].step;
    let k = 2;
    while (pick.length < n - 1 && k < 120) {
      const t = fromMin(toMin(q.h, q.m) + (k % 2 ? 1 : -1) * Math.ceil(k / 2) * st);
      if (!(t.h === q.h && t.m === q.m) && !pick.some(c => c.h === t.h && c.m === t.m)) pick.push({ h: t.h, m: t.m, type: 'そのほか' });
      k++;
    }
    const all = pick.map(c => ({ h: c.h, m: c.m, type: c.type, correct: false, text: fmt(c.h, c.m) }));
    all.push({ h: q.h, m: q.m, type: '', correct: true, text: fmt(q.h, q.m) });
    for (let i = all.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [all[i], all[j]] = [all[j], all[i]];
    }
    return all;
  }

      const q = op === '+' ? build(10 - b, b, '+', level) : build(10, b, '-', level);
      q.ten = true;
      if (!(prev && prev.a === q.a && prev.b === q.b && prev.op === q.op)) return q;
    }
    return Object.assign(build(4, 6, '+', level), { ten: true });
  }

  // want：にた問題を作るときの しゅるい（op・くり上がり・くり下がり・10になる）
  function make(level, prev, want) {
    const c = CONF[level - 1];
    if (c.ten && (want ? want.ten : Math.random() < TEN_RATE)) {
      const op = want ? want.op : (c.ten === 'both' ? (Math.random() < 0.5 ? '+' : '-') : c.ten);
      return tenQ(op, level, prev);
    }
    // 2けた：くり上がり・くり下がりのある問題が半分くらい
    const regroup = c.big ? (want ? (want.carry || want.borrow) : Math.random() < 0.5) : null;
    const bigOp = c.big ? (want ? want.op : c.big[rint(0, c.big.length - 1)]) : null;
    for (let g = 0; g < 400; g++) {
      let q;
      if (c.big) {
        const a = rint(10, 99), b = rint(10, 99);
        if (bigOp === '+' && a + b > 99) continue;
        if (bigOp === '-' && a - b < 1) continue;
        q = build(a, b, bigOp, level);
        if ((q.carry || q.borrow) !== regroup) continue;
      } else {
        const sc = SMALL[c.small[rint(0, c.small.length - 1)]];
        let a = rint(sc.a[0], sc.a[1]), b = rint(sc.b[0], sc.b[1]);
        if (!sc.ok(a, b)) continue;
        if (sc.op === '+' && Math.random() < 0.5) [a, b] = [b, a];
        q = build(a, b, sc.op, level);
        if (want && (want.op !== q.op || want.carry !== q.carry || want.borrow !== q.borrow)) continue;
      }
      if (prev && prev.a === q.a && prev.b === q.b && prev.op === q.op) continue;
      return q;
    }
    return build(3, 2, '+', level);
  }

  function candidates(q) {
    const { a, b, op, v } = q, list = [];
    const add = (x, type) => {
      if (x < 0 || x === v || list.some(c => c.v === x)) return;
      list.push({ v: x, type });
    };
    if (q.carry) add(v - 10, 'くり上がりわすれ');
    if (q.borrow) {
      // 一のくらいを「大きいほうから小さいほう」をひいてしまう（13−8＝15）
      const tens = Math.floor(a / 10) - Math.floor(b / 10);
      add(tens * 10 + Math.abs((a % 10) - (b % 10)), 'くり下がりわすれ');
      add(v + 10, 'くり下がりわすれ');
    }
    add(op === '+' ? a - b : a + b, 'たし算とひき算の取りちがえ');
    add(v + 1, '1ちがい');
    add(v - 1, '1ちがい');
    if (q.level >= 6) { add(v + 10, '十のくらいのまちがい'); add(v - 10, '十のくらいのまちがい'); }
    return list;
  }

  function choices(q, n) {
    const cands = candidates(q);
    // 一番よくあるまちがい（はじめの1つ）は入れて、あとはまぜる
    const first = cands.slice(0, 1), rest = cands.slice(1);
    for (let i = rest.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [rest[i], rest[j]] = [rest[j], rest[i]];
    }
    const pick = first.concat(rest).slice(0, n - 1);
    let k = 2;
    while (pick.length < n - 1 && k < 40) {
      const x = q.v + (k % 2 ? 1 : -1) * Math.ceil(k / 2);
      if (x >= 0 && x !== q.v && !pick.some(c => c.v === x)) pick.push({ v: x, type: 'そのほか' });
      k++;
    }
    const all = pick.map(c => ({ v: c.v, type: c.type, correct: false, text: String(c.v) }));
    all.push({ v: q.v, type: '', correct: true, text: String(q.v) });
    for (let i = all.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [all[i], all[j]] = [all[j], all[i]];
    }
    return all;
  }

  function judge(q, given) {
    if (given.v === q.v) return { ok: true, type: '' };
    if (given.type) return { ok: false, type: given.type };
    const c = candidates(q).find(c => c.v === given.v);
    return { ok: false, type: c ? c.type : 'そのほか' };
  }

      case 'くり上がりわすれ': return q.carry;
      case 'くり下がりわすれ': return q.borrow;
      case 'たし算とひき算の取りちがえ': return true;
      case '1ちがい': return true;
      case '十のくらいのまちがい': return q.level >= 6;
      default: return false;

  const IE = R('家', 'いえ'), EKI = R('駅', 'えき'), ARU = R('歩', 'ある'), DEN = R('電車', 'でんしゃ');
  const NO = R('乗', 'の'), DE = R('出', 'で'), NANJI = R('何時何分', 'なんじなんぷん');

  // walk/ride：かかる分のはんい、back：まえに もどして考える、cross：時がかわるか（'no'／'half'／'any'）
  const CONF = [
    { segs: ['walk'], walk: [5, 15], back: false, cross: 'no' },
    { segs: ['ride'], ride: [10, 35], back: false, cross: 'half' },
    { segs: ['walk'], walk: [5, 20], back: true, cross: 'no' },
    { segs: ['ride'], ride: [10, 35], back: true, cross: 'half' },
    { segs: ['walk', 'ride'], walk: [5, 20], ride: [10, 35], back: 'both', cross: 'any' },
  ];
  const STEP = 5;

  function rint(a, b) { return a + Math.floor(Math.random() * (b - a + 1)); }
  function rstep(r) { return rint(r[0] / STEP, r[1] / STEP) * STEP; }
  function hm(t) { t = ((t % 1440) + 1440) % 1440; return { h: Math.floor(t / 60), m: t % 60 }; }
  function fmtT(t) { const x = hm(t); return fmt(x.h, x.m); }
  function fmt(h, m) { return m === 0 ? `${h}時` : `${h}時${m}分`; }

  function build(level, durs, back, start) {
    const total = durs.reduce((a, b) => a + b.dur, 0);
    const end = start + total;
    const ans = back ? start : end;
    const known = back ? end : start;
    const a = hm(ans);
    const segTxt = durs.map(d => (d.kind === 'walk' ? '歩いて' : '電車で') + d.dur + '分').join('＋');
    return {
      level, durs, back, start, end, total, known,
      h: a.h, m: a.m,
      cross: Math.floor(start / 60) !== Math.floor(end / 60),
      text: back ? `${fmtT(end)}につくには（${segTxt}）` : `${fmtT(start)}に出て（${segTxt}）`,
      answerText: fmt(a.h, a.m),
    };
  }

  function make(level, prev, want) {
    const c = CONF[level - 1];
    const back = want ? want.back : (c.back === 'both' ? Math.random() < 0.5 : c.back);
    const wantCross = want ? want.cross : (c.cross === 'half' ? Math.random() < 0.5 : null);
    for (let g = 0; g < 400; g++) {
      const durs = c.segs.map(k => ({ kind: k, dur: rstep(k === 'walk' ? c.walk : c.ride) }));
      const start = rint(7 * 12, 17 * 12) * STEP;   // 7時〜17時ごろ、5分きざみ
      const q = build(level, durs, back, start);
      if (c.cross === 'no' && q.cross) continue;
      if (wantCross !== null && wantCross !== undefined && q.cross !== wantCross) continue;
      if (prev && prev.start === q.start && prev.total === q.total) continue;
      return q;
    }
    return build(level, [{ kind: c.segs[0], dur: 10 }], back, 8 * 60);
  }

  function candidates(q) {
    const ansT = q.back ? q.start : q.end;
    const list = [];
    const add = (t, type) => {
      const x = hm(t);
      if (x.h === q.h && x.m === q.m) return;
      if (list.some(c => c.h === x.h && c.m === x.m)) return;
      list.push({ h: x.h, m: x.m, type });
    };
    // 時を かえわすれる（分だけ 計算して、時は わかっている時刻のまま）
    if (q.cross) {
      const k = hm(q.known);
      add(k.h * 60 + hm(ansT).m, '時がかわるのをわすれた');
    }
    // たす と ひく を 反対にした
    add(q.back ? q.known + q.total : q.known - q.total, 'まえとあとの取りちがえ');
    // 2つのうち 1つしか 計算していない
    if (q.durs.length === 2) q.durs.forEach(d => add(q.back ? q.known - d.dur : q.known + d.dur, 'かかる時間のたしわすれ'));
    add(ansT + 60, '時のまちがい');
    add(ansT - 60, '時のまちがい');
    add(ansT + 5, '分の計算');
    add(ansT - 5, '分の計算');
    add(ansT + 10, '分の計算');
    add(ansT - 10, '分の計算');
    return list;
  }

  function choices(q, n) {
    const cands = candidates(q);
    // はじめの1つ（いちばん よくある まちがい）は入れて、のこりは まぜる
    // 歩きと電車の問題では「たしわすれ」も1つ入れる
    const first = cands.slice(0, 1);
    const forgot = cands.find(c => c.type === 'かかる時間のたしわすれ' && c !== first[0]);
    if (forgot && n >= 4) first.push(forgot);
    const rest = cands.filter(c => first.indexOf(c) < 0);
    for (let i = rest.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [rest[i], rest[j]] = [rest[j], rest[i]];
    }
    const pick = first.concat(rest).slice(0, n - 1);
    const all = pick.map(c => ({ h: c.h, m: c.m, type: c.type, correct: false, text: fmt(c.h, c.m) }));
    all.push({ h: q.h, m: q.m, type: '', correct: true, text: fmt(q.h, q.m) });
    for (let i = all.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [all[i], all[j]] = [all[j], all[i]];
    }
    return all;
  }

  function judge(q, given) {
    if (given.h === q.h && given.m === q.m) return { ok: true, type: '' };
    if (given.type) return { ok: false, type: given.type };
    const c = candidates(q).find(c => c.h === given.h && c.m === given.m);
    return { ok: false, type: c ? c.type : 'そのほか' };
  }

  // にた問題：同じレベル・同じ むき（まえ／あと）・時がかわるかどうかも同じで、数をかえる
  function similar(q) { return make(q.level, q, { back: q.back, cross: q.cross }); }

  function fits(q, type) {
    switch (type) {
      case '時がかわるのをわすれた': return q.cross;
      case 'まえとあとの取りちがえ': return true;
      case 'かかる時間のたしわすれ': return q.durs.length === 2;
      case '時のまちがい': return !q.cross;
      case '分の計算': return true;
      default: return false;
    }
  }

  function hint(type, q) {
    const back = q && q.back;
    switch (type) {
      case '時がかわるのをわすれた':
        return back
          ? `<ruby>分<rt>ふん</rt></ruby>が たりないときは、<ruby>時<rt>じ</rt></ruby>を 1つ もどして 60<ruby>分<rt>ぷん</rt></ruby>を たして <ruby>考<rt>かんが</rt></ruby>えるよ。`
          : `<ruby>分<rt>ふん</rt></ruby>が 60を こえたら、<ruby>時<rt>じ</rt></ruby>が 1つ すすむよ。`;
      case 'まえとあとの取りちがえ':
        return back
          ? `つく <ruby>時刻<rt>じこく</rt></ruby>から、かかる <ruby>時間<rt>じかん</rt></ruby>の ぶん もどして（ひいて）<ruby>考<rt>かんが</rt></ruby>えるよ。`
          : `${DE}た <ruby>時刻<rt>じこく</rt></ruby>に、かかる <ruby>時間<rt>じかん</rt></ruby>を たして <ruby>考<rt>かんが</rt></ruby>えるよ。`;
      case 'かかる時間のたしわすれ':
        return `${ARU}く <ruby>時間<rt>じかん</rt></ruby>と ${DEN}の <ruby>時間<rt>じかん</rt></ruby>を、どちらも <ruby>計算<rt>けいさん</rt></ruby>するよ。`;
      case '時のまちがい':
        return `<ruby>分<rt>ふん</rt></ruby>が 60を こえないときは、<ruby>時<rt>じ</rt></ruby>は そのままだよ。`;
      case '分の計算':
        return `<ruby>分<rt>ふん</rt></ruby>の <ruby>計算<rt>けいさん</rt></ruby>を、もう<ruby>一<rt>いち</rt></ruby>ど たしかめてね。`;
      default:
        return `<ruby>下<rt>した</rt></ruby>の じゅんばんを <ruby>見<rt>み</rt></ruby>て、<ruby>時刻<rt>じこく</rt></ruby>を たどってみてね。`;
    }
  }

  function sentence(q) {
    const T = t => `<span class="t">${fmtT(t)}</span>`;
    const D = d => `<span class="d">${d}分</span>`;
    const w = q.durs.find(d => d.kind === 'walk'), r = q.durs.find(d => d.kind === 'ride');
    if (q.durs.length === 2) {
      const how = `${IE}から A${EKI}まで ${ARU}いて ${D(w.dur)}、<br>A${EKI}から B${EKI}まで ${DEN}で ${D(r.dur)} かかるよ。`;
      return q.back
        ? `${T(q.end)}に B${EKI}に つきたいよ。<br>${how}<br>${IE}を ${NANJI}に ${DE}たら いいかな？`
        : `${T(q.start)}に ${IE}を ${DE}たよ。<br>${how}<br>B${EKI}に つくのは ${NANJI}かな？`;
    }
    if (w) {
      return q.back
        ? `${T(q.end)}に ${EKI}に つきたいよ。<br>${IE}から ${EKI}まで ${ARU}いて ${D(w.dur)} かかるよ。<br>${IE}を ${NANJI}に ${DE}たら いいかな？`
        : `${T(q.start)}に ${IE}を ${DE}たよ。<br>${IE}から ${EKI}まで ${ARU}いて ${D(w.dur)} かかるよ。<br>${EKI}に つくのは ${NANJI}かな？`;
    }
    return q.back
      ? `${T(q.end)}に つきたいよ。<br>${DEN}は ${D(r.dur)} かかるよ。<br>${NANJI}の ${DEN}に ${NO}れば いいかな？`
      : `${T(q.start)}の ${DEN}に ${NO}ったよ。<br>${DEN}は ${D(r.dur)} かかるよ。<br>つくのは ${NANJI}かな？`;
  }

  // 解説：時刻の じゅんばんを 左から右へ ならべる（答えのところに色）
  function line(q) {
    const two = q.durs.length === 2, walk = q.durs[0].kind === 'walk';
    const caps = two ? [`${IE}を ${DE}る`, `A${EKI}に つく`, `B${EKI}に つく`]
      : walk ? [`${IE}を ${DE}る`, `${EKI}に つく`] : [`${DEN}に ${NO}る`, 'つく'];
    const times = [q.start];
    q.durs.forEach(d => times.push(times[times.length - 1] + d.dur));
    const ansIdx = q.back ? 0 : times.length - 1;
    let h = `<div class="dr-line">`;
    times.forEach((t, i) => {
      h += `<div class="dr-box ${i === ansIdx ? 'ans' : ''}"><div class="cap">${caps[i]}</div><div class="tm">${fmtT(t)}</div></div>`;
      if (i < q.durs.length) {
        const d = q.durs[i];
        h += `<div class="dr-arrow">${d.kind === 'walk' ? ARU + 'いて' : DEN + 'で'}<br>${d.dur}分<br>→</div>`;
      }
    });
    return h + `</div>`;
  }

  function render(q, el, opt) {
    let h = `<div class="dr-text">${sentence(q)}</div>`;
    if (opt && opt.explain) h += line(q);
    el.innerHTML = h;
  }

  window.GENRES.deru = {
    key: 'deru',
    name: '何時に出るかな',
    nameRuby: `「${R('何時', 'なんじ')}に ${DE}るかな」`,
    prompt: '',
    answerKind: 'time',
    levels: [
      { label: `${ARU}いて つく ${R('時刻', 'じこく')}`, ex: `8時10分に ${DE}て 15分` },
      { label: `${DEN}で つく ${R('時刻', 'じこく')}`, ex: `8時50分の ${DEN}で 20分` },
      { label: `${IE}を ${DE}る ${R('時刻', 'じこく')}`, ex: '9時に つくには 15分まえ' },
      { label: `${DEN}に ${NO}る ${R('時刻', 'じこく')}`, ex: '10時5分に つくには 25分まえ' },
      { label: `${ARU}きと ${DEN}`, ex: `${ARU}いて 10分＋${DEN}で 20分`, mode: 'input' },
    ],
    fmt, make, render, choices, judge, hint, similar, fits,
  };
})();

function fmtRemain(sec){

/* ============================================================
   3. レベルの決めかた
   ・レベルに数字がある → 先生が指定（前回のつづきが「する」なら前回から）
   ・下限・上限がある   → その中から自分で選ぶ
   ・ぜんぶ空           → 自由に選ぶ
   複数ジャンルのときは、生徒は選ばない（ジャンルごとのレベル設定で始める）
     指定なし → 範囲があれば下限、なければレベル1（前回のつづきなら前回のレベル）
============================================================ */
function planLevel(G, c){
  const max = G.levels.length;
  c = c || {};
  const n = v => (v === "" || v === null || v === undefined) ? null : parseInt(v);
  const clamp = (v, a, b)=> Math.min(b, Math.max(a, v));
  const lv = n(c.level), lo = n(c.lower), up = n(c.upper), last = n(c.last);
  if(lv){
    let start = clamp(lv, 1, max);
    if(S.sess.continueLevel && last) start = clamp(last, 1, max);
    return { fixed:true, level:start, minLv:1, maxLv:max, auto:!!c.auto, last };
  }
  let a = lo || 1, b = up || max;
  if(a > b) [a,b] = [b,a];
  a = clamp(a, 1, max); b = clamp(b, 1, max);
  const ranged = !!(lo || up);
  let start = a;
  if(S.sess.continueLevel && last) start = clamp(last, a, b);
  return { fixed:false, level:start, a, b, minLv: ranged ? a : 1, maxLv: ranged ? b : max, auto:!!c.auto, last };
}

}

}

function nextQuestion(){

/* つぎの問題のジャンル
   じゅんばん：時間をジャンルの数で分けて、選んだ順に
   まぜる　　：1問ごとに ばらばら（同じジャンルが3回つづかないように） */
function chooseGenre(){
  const keys = S.gkeys;
  if(keys.length === 1) return keys[0];
  if(S.sess.order === "mix"){
    const h = S.genreHist || [];
    let cand = keys;
    if(h.length >= 2 && h[0] === h[1]) cand = keys.filter(k=> k !== h[0]);
    const k = cand[Math.floor(Math.random() * cand.length)];
    S.genreHist = [k].concat(h).slice(0,2);
    return k;
  }
  const total = S.sess.minutes * 60000;
  const used = total - (S.endAt - Date.now());
  const i = Math.floor(used / (total / keys.length));
  return keys[Math.max(0, Math.min(keys.length - 1, i))];
}

/* ============================================================
   問題のえらびかた（「まちがえた問題を多めに」のとき）
   ・まちがえた問題は、2問あとに「にた問題」でもう一度（同じ問題で2回まで）
   ・それ以外のときも、半分くらいは、まちがいの多い型を練習できる問題にする
============================================================ */
function pickQuestion(){
  const G = S.G;
  const prev = (S.q && S.q._g === S.gcur) ? S.q : S.lastQ;
  S.qSrc = "ふつう"; S.qTries = 0;
  if(S.sess.weakFocus !== false){
    const i = S.review.findIndex(r=> r.due <= S.count);
    if(i >= 0){
      const r = S.review.splice(i,1)[0];
      S.qSrc = "やり直し"; S.qTries = r.tries;
      return G.similar ? G.similar(r.q) : r.q;
    }
    const keys = Object.keys(S.types);
    if(keys.length && G.fits && Math.random() < 0.5){
      // まちがえた回数が多い型ほど えらばれやすい
      let total = 0; keys.forEach(k=> total += S.types[k]);
      let x = Math.random() * total, type = keys[0];
      for(const k of keys){ x -= S.types[k]; if(x < 0){ type = k; break; } }
      for(let k=0;k<12;k++){
        const q = G.make(S.level, prev);
        if(G.fits(q, type)){ S.qSrc = "にがて"; return q; }
      }
    }
  }
  return G.make(S.level, prev);
}

function buildNumInput(){
  const box = $("q-answer");

function numKey(k){
  if(S.answered) return;
  if(k === "けす") S.input.n = S.input.n.slice(0,-1);
  else if(k === "こたえる"){
    if(S.input.n === "") return;
    const v = parseInt(S.input.n);
    answer({ v }, S.G.fmt(v));
    return;
  }
  else if(S.input.n === "0") S.input.n = k;          // 0 のあとに数字を押したら入れかえる（05 にしない）
  else if(S.input.n.length < (S.G.maxDigits || 5)) S.input.n += k;
  $("slot-n").textContent = S.input.n;
}

function drawInput(){
  const sn = $("slot-n"); if(sn) sn.classList.toggle("cur", !S.answered);
  const sh = $("slot-h"), sm = $("slot-m");
  if(!sh) return;
  sh.textContent = S.input.h; sm.textContent = S.input.m;
  sh.classList.toggle("cur", S.input.focus === "h" && !S.answered);
  sm.classList.toggle("cur", S.input.focus === "m" && !S.answered);
}
function inputKey(k){
  if(S.answered) return;
  const f = S.input.focus;
  if(k === "けす"){
    if(S.input[f] === "" && f === "m"){ S.input.focus = "h"; }
    else S.input[f] = S.input[f].slice(0,-1);
  }else if(k === "こたえる"){
    if(S.input.h === "") { S.input.focus = "h"; drawInput(); return; }
    const h = parseInt(S.input.h), m = S.input.m === "" ? 0 : parseInt(S.input.m);
    answer({ h, m }, S.G.fmt(h, m));
    return;
  }else{
    if(S.input[f].length < 2) S.input[f] += k;
    // 時を入れ終わったら、自動で分へ
    if(f === "h" && (S.input.h.length === 2 || parseInt(S.input.h) >= 2)) S.input.focus = "m";
  }
  drawInput();
}

function answer(given, givenText){
  S.answered = true;
  drawInput();
  const res = S.G.judge(S.q, given);
  const sec = Math.round((Date.now() - S.qStart)/100)/10;
  S.count++; S.gCount++;
  S.lastQ = S.q;
  const lvAtAnswer = S.level;
  const fb = $("feedback");
  if(res.ok){
    S.correct++; S.gCorrect++; S.streakOk++; S.streakNg = 0;
    fb.className = "feedback ok";
    fb.innerHTML = "せいかい！";
    soundOk();
  }else{
    S.streakNg++; S.streakOk = 0;
    S.types[res.type] = (S.types[res.type] || 0) + 1;
    S.recent.unshift({ q:S.q.answerText, a:givenText, type:res.type, g: isMulti() ? S.G.name : "" });
    S.recent = S.recent.slice(0,5);
    if(S.sess.weakFocus !== false){
      const tries = S.qSrc === "やり直し" ? S.qTries + 1 : 1;
      if(tries <= 2) S.review.push({ q:S.q, due:S.count + 2, tries });
    }
    fb.className = "feedback ng";
    fb.innerHTML = `こたえは ${S.q.answerHtml || S.q.answerText} だよ<div class="hint">${S.G.hint(res.type, S.q)}</div>`;
    S.G.render(S.q, $("q-visual"), { explain:true });
    soundNg();
  }
  drawScore();
  S.queue.push({
    ts: Date.now(), genre:S.G.name, level:lvAtAnswer, q:S.q.text, ans:S.q.answerText,
    given:givenText, ok:res.ok, sec, type:res.type, mode:modeName(), src:S.qSrc,
  });
  savePending();
  autoAdjust();
  flush();

  const stop = S.timeUp || S.teacherEnded;
  const row = $("next-row");
  row.innerHTML = `<button class="big-btn" id="btn-next">${stop ? "おわる" : "つぎへ"}</button>`;
  $("btn-next").addEventListener("click", nextQuestion);
  // せいかいのときは、少しまって自動で次へ
  if(res.ok && !stop){
    const q = S.q;
    setTimeout(()=>{ if(S.q === q && S.answered && !S.finished) nextQuestion(); }, 1100);
  }
}
/* 答え方。「レベルどおり」のときは、ジャンルがレベルごとに決めた答え方（決めていなければ4択） */
function curMode(){
  const m = S.sess.answerMode;
  if(m !== "level") return m;
  const L = S.G.levels[(S.q ? S.q.level : S.level) - 1];
  return (L && L.mode) || "4";
}
function modeName(){ const m = curMode(); return m === "input" ? "入力" : (m === "2" ? "2択" : "4択"); }

/* レベルの自動調整（その生徒が「する」のときだけ） */
function autoAdjust(){
  if(!S.auto) return;
  const up = S.sess.upN || 3, down = S.sess.downN || 2;
  if(S.streakOk >= up && S.level < S.maxLv){
    S.level++; S.streakOk = 0;
    if(S.sess.levelUpNotice){
      const el = $("levelup"); el.classList.add("show");
      setTimeout(()=> el.classList.remove("show"), 1400);
    }
  }else if(S.streakNg >= down && S.level > S.minLv){
    S.level--; S.streakNg = 0;   // 下がったときは、なにも出さない
  }
}

/* ============================================================
   送る（うらで順番に。失敗したらためておいて、あとで送り直す）

function pendKey(){ return "jibundrill_q_" + (S.sess ? S.sess.id : "") + "_" + S.name; }
function savePending(){ try{ localStorage.setItem(pendKey(), JSON.stringify(S.queue)); }catch(e){} }
function loadPending(){
  try{
    const v = localStorage.getItem(pendKey());
    if(v){ const arr = JSON.parse(v); if(Array.isArray(arr)) S.queue = arr.concat(S.queue); }
  }catch(e){}
}
function topOf(types){
  let top = "", topN = 0;
  Object.keys(types || {}).forEach(k=>{ if(types[k] > topN){ top = k; topN = types[k]; } });
  return top;
}
function snapshot(state){
  saveGenre();
  // 複数ジャンルのときは「ジャンル名：まちがいの型」でまとめる
  let types = S.types;
  if(isMulti()){
    types = {};
    S.gkeys.forEach(k=>{ const g = S.gs[k]; if(g) Object.keys(g.types).forEach(t=> types[g.G.name + "：" + t] = g.types[t]); });
  }
  const per = S.gkeys.filter(k=> S.gs[k]).map(k=>{ const g = S.gs[k];
    return { genre:k, name:g.G.name, level:g.level, startLevel:g.startLevel, count:g.gCount, correct:g.gCorrect }; });
  return {
    state: state || (S.finished ? "done" : "playing"),
    level:S.level, startLevel:S.startLevel, count:S.count, correct:S.correct,
    types, top:topOf(types), recent:S.recent, remain: S.endAt ? remainSec() : null, auto:S.auto, dev:S.dev,
    genre: S.G ? S.G.name : "", per: isMulti() ? per : null,
  };
}
function flush(){
  if(S.sending) return;
  S.sending = true;
  const items = S.queue.slice();
  api("studentLog", S.sess.id, S.name, items, snapshot()).then(r=>{
    S.queue.splice(0, items.length);
    savePending();
    if(r && r.open === false && !S.teacherEnded && !S.finished){
      S.teacherEnded = true;
      onStopSignal("<ruby>先生<rt>せんせい</rt></ruby>が おわりに したよ。");
    }
  }).catch(()=>{}).finally(()=>{
    S.sending = false;
    if(S.queue.length && S.finishing) setTimeout(flush, 1500);
  });
}
function sendStatus(state){
  api("studentLog", S.sess.id, S.name, [], snapshot(state)).catch(()=>{});
}

function finish(){
  if(S.finished) return;
  S.finished = true; S.finishing = true;
  clearInterval(timerId); clearInterval(hbId);
  $("res-count").textContent = `${S.count}もん といたよ`;
  $("res-score").innerHTML = S.sess.showScore ? `せいかいは ${S.correct}もん だったよ` : "よく がんばったね";
  saveGenre();
  $("res-per").innerHTML = isMulti() ? S.gkeys.map(k=>{ const g = S.gs[k];
    return `${g.G.nameRuby}：${g.gCount}もん` + (S.sess.showScore ? `（せいかい ${g.gCorrect}）` : ""); }).join("<br>") : "";
  show("s-result");
  sendAll();
}
function sendAll(){
  $("send-state").innerHTML = "<ruby>先生<rt>せんせい</rt></ruby>に おくっているよ…";
  saveGenre();
  const share = (isMulti() && S.sess.order !== "mix") ? Math.round(S.sess.minutes / S.gkeys.length * 10) / 10 : S.sess.minutes;
  const per = S.gkeys.map(k=>{ const g = S.gs[k];
    return { genre:k, genreName:g.G.name, startLevel:g.startLevel, endLevel:g.level,
             count:g.gCount, correct:g.gCorrect, topType:topOf(g.types), minutes:share }; });
  const sum = Object.assign({}, per[0], { per });
  const tryOnce = ()=>{
    const items = S.queue.slice();
    api("studentLog", S.sess.id, S.name, items, snapshot("done")).then(()=>{
      S.queue.splice(0, items.length); savePending();
      return api("studentFinish", S.sess.id, S.name, sum);
    }).then(()=>{
      $("send-state").innerHTML = "<ruby>先生<rt>せんせい</rt></ruby>に おくったよ";
      startWaitNext();
      try{ localStorage.removeItem(pendKey()); }catch(e){}
    }).catch(()=>{
      $("send-state").innerHTML = "つながるのを まっているよ…";
      setTimeout(tryOnce, 4000);
    });
  };
  // 送っている途中のものが終わるのを少しまつ
  setTimeout(tryOnce, S.sending ? 1500 : 0);
}

let nextTimer = null, nextStart = 0;
function startWaitNext(){
  $("wait-next").innerHTML = "つぎの もんだいが はじまったら、<br>じどうで かわるよ。このまま まってね";
  clearInterval(nextTimer);
  nextStart = Date.now();
  nextTimer = setInterval(checkNext, POLL_MS);
}
function stopWaitNext(){ clearInterval(nextTimer); nextTimer = null; }
function checkNext(){
  if(Date.now() - nextStart > 40*60000){ stopWaitNext(); $("wait-next").innerHTML = ""; return; }
  api("studentWait", S.sess.id, S.dev).then(r=>{
    if(!r || (!r.ok && !r.needCode)) return;
    stopWaitNext();
    const prev = S.name;
    resetForNext();
    S.prevName = prev;
    $("wait-next").innerHTML = "";
    if(r.ok){
      S.code = r.session.code;
      enter(r);
    }else{
      S.code = ""; drawCode();
      freshEl("code-err").innerHTML = "あたらしい もんだいが はじまったよ。<br><ruby>先生<rt>せんせい</rt></ruby>の <ruby>画面<rt>がめん</rt></ruby>の <ruby>数字<rt>すうじ</rt></ruby>を <ruby>入<rt>い</rt></ruby>れてね";
      show("s-code");
    }
  }).catch(()=>{});
}
/* つぎの回のために、前の回の じょうたいを もどす */

    gs:{}, gkeys:[], gcur:"", confs:null, lastQ:null, gCount:0, gCorrect:0, genreHist:[],
  });
}

let lastTouch = 0;
document.addEventListener("touchend", e=>{
  const now = Date.now();
  const t = e.target;
  const inter = t && t.closest && t.closest("button,input,select,textarea,a,label,.slot");
  if(!inter && now - lastTouch <= 300) e.preventDefault();
  lastTouch = now;
}, { passive:false });
