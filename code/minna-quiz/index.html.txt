function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));
}

/** 漢字《かんじ》、｜漢字《かんじ》 を ruby にする */
function rubyHtml(s) {
  let h = escapeHtml(s || '');
  h = h.replace(/[｜|]([^｜|《》]+?)《([^》]+?)》/g, '<ruby>$1<rt>$2</rt></ruby>');
  h = h.replace(/([一-鿿々〆ヶ]+)《([^》]+?)》/g, '<ruby>$1<rt>$2</rt></ruby>');
  return h.replace(/\n/g, '<br>');
}

function setHtml(el, h) {
  if (el._h !== h) {
    el._h = h;
    el.innerHTML = h;
    if (h.indexOf('data-img') >= 0) hydrateImgs(el);
  }

/* ===== 画像の よみこみ =====
 * スプレッドシートの 画像は「img:ファイルID」の しるしで とどく。
 * アプリ（サーバー）から 画像の データを もらって 表示する。 */
const IMG = { cache: {}, state: {}, queue: [], running: 0 };

function loadImg(t) {
  if (IMG.cache[t] || IMG.state[t] === 'q') return;
  if (typeof IMG.state[t] === 'number' && IMG.state[t] > Date.now()) return; // しっぱい したら すこし まつ
  IMG.state[t] = 'q';
  IMG.queue.push(t);
  pumpImg();
}

function pumpImg() {
  while (IMG.running < 3 && IMG.queue.length) {
    const t = IMG.queue.shift();
    IMG.running++;
    // 生徒が いっせいに もらわないように、すこし ずらす
    const delay = active === 'student' ? Math.random() * 800 : 0;
    setTimeout(() => {
      call('getImage', t.slice(4), CID).then(d => {
        IMG.cache[t] = d;
        delete IMG.state[t];
        document.querySelectorAll('img[data-img]').forEach(el => { if (el.dataset.img === t) el.src = d; });
      }).catch(() => {
        IMG.state[t] = Date.now() + 4000;
      }).then(() => {
        IMG.running--;
        pumpImg();
      });
    }, delay);
  }
}

/** まだ 出ていない 画像を よみこむ */
function hydrateImgs(root) {
  root.querySelectorAll('img[data-img]').forEach(el => {
    const t = el.dataset.img, d = IMG.cache[t];
    if (d) { if (el.getAttribute('src') !== d) el.src = d; }
    else loadImg(t);
  });
}

/**
 * 画像の タグ
 * p が 't'（先生）の ときだけ、見つからない 画像の おしらせを 出す
 */
function imgTag(src, cls, p) {
  if (!src) return '';
  if (src.indexOf('nf:') === 0) return p === 't' ? `<div class="nfimg">${escapeHtml(src.slice(3))}</div>` : '';
  if (src.indexOf('img:') === 0) return `<img class="${cls}" data-img="${escapeHtml(src)}" alt="">`;
  return `<img class="${cls}" src="${escapeHtml(src)}" alt="">`;
}

/** のこり時間（ミリびょう）。時間が ないときは null */
function remainMs(st, offset) {
  if (!st || st.phase !== 'open' || !st.limit) return null;
  return Math.max(0, st.openedAt + st.limit * 1000 - (Date.now() + offset));
}

function remainSec(st, offset) {
  const ms = remainMs(st, offset);
  return ms === null ? null : Math.ceil(ms / 1000);
}

/**
 * のこり時間の 表示（バー・びょうの 文字）
 * p: 's'（生徒）か 't'（先生）。みせかたは せっていで えらぶ（はじめは バー）
 */
function paintTimer(p, st, offset) {
  const ms = remainMs(st, offset);
  const view = (st && st.timeView) || 'bar';
  const txt = $(p + 'Time'), bar = $(p + 'Bar');
  if (ms === null) {
    if (txt.textContent !== '') txt.textContent = '';
    bar.classList.add('hidden');
    return null;
  }
  const sec = Math.ceil(ms / 1000);
  const label = view === 'bar' ? '' : 'のこり ' + sec + 'びょう';
  if (txt.textContent !== label) txt.textContent = label;
  txt.classList.toggle('warn', sec <= 5);
  const showBar = view !== 'text';
  bar.classList.toggle('hidden', !showBar);
  if (showBar) {
    bar.classList.toggle('warn', sec <= 5);
    const w = (ms / (st.limit * 1000) * 100).toFixed(1) + '%';
    const fill = bar.firstElementChild;
    if (fill.style.width !== w) fill.style.width = w;
  }
  return sec;
}

function isLive(st) {
  return !!st && st.round > 0 && ['open', 'closed', 'reveal'].includes(st.phase);
}

/** ヒント・解説の 1こぶん（ことば か 画像） */
function toItem(v) {
  if (v && typeof v === 'object') return { text: String(v.text || ''), img: String(v.img || '') };
  const s = String(v || '').trim();
  if (/^(https?:|data:image|img:|nf:)/.test(s)) return { text: '', img: s };
  return { text: s, img: '' };
}

function itemHtml(v, p) {
  const it = toItem(v);
  return (it.text ? rubyHtml(it.text) : '') + imgTag(it.img, 'subimg', p);
}

/** 問題文・がぞう・ヒント（先生と 生徒で おなじ 見た目） */
function renderQ(p, st, show) {
  const has = show && !!st.text;
  $(p + 'Q').classList.toggle('hidden', !has);
  setHtml($(p + 'Q'), has ? rubyHtml(st.text) : '');
  const ih = show ? imgTag(st.img, 'qimg', p) : '';
  $(p + 'Img').classList.toggle('hidden', !ih);
  setHtml($(p + 'Img'), ih);
  setHtml($(p + 'Hints'), show ? st.hints.map(h => `<div class="hint">ヒント：${itemHtml(h, p)}</div>`).join('') : '');
  hydrateImgs($(p + 'Stage')); // よみこみに しっぱいした 画像を もういちど
}

/** かいせつ（こたえあわせの ときだけ） */
function renderExplain(p, st, show) {
  const el = $(p + 'Explain');
  const list = (show && st.phase === 'reveal' && st.explain) ? st.explain : [];
  const h = list.length ? `<div class="lab">かいせつ</div>` + list.map(v => itemHtml(v, p)).join('<br>') : '';
  el.classList.toggle('hidden', !h);
  const changed = el._h !== h;
  setHtml(el, h);
  if (h && changed && el.scrollIntoView) {
    try { el.scrollIntoView({ block: 'nearest', behavior: 'smooth' }); } catch (e) { /* なにもしない */ }
  }
}

/**
 * せんたくし（先生と 生徒で おなじ 見た目）
 * o.tap: おせる / o.mine: じぶんの こたえ / o.counts: 人数 / o.showText: ことばを 出す
 */
function optsHtml(st, o) {
  const rev = st.phase === 'reveal';
  const max = o.counts ? Math.max(1, ...st.choices.map((_, i) => o.counts[LETTERS[i]] || 0)) : 1;
  return st.choices.map((t, i) => {
    const L = LETTERS[i], l = L.toLowerCase();
    let cls = 'opt ' + l;
    if (rev) cls += st.correct === L ? ' win' : (o.mine === L ? ' mine' : ' dim');
    else if (o.mine) cls += o.mine === L ? ' mine' : (o.ask === L ? ' ask' : (o.nodim ? '' : ' dim'));
    const text = o.showText && t ? rubyHtml(t) : '';
    let extra = '';
    if (o.counts) {
      const n = o.counts[L] || 0;
      extra += `<div class="track"><div class="fill ${l}" style="width:${n / max * 100}%"></div></div><span class="n">${n}<ruby>人<rt>にん</rt></ruby></span>`;
    }
    if (rev && st.correct === L) extra += '<span class="tag ok">せいかい</span>';
    else if (o.mine === L) extra += '<span class="tag me">えらんだよ</span>';
    else if (o.ask === L) extra += '<span class="tag me">これに かえる？</span>';
    const tag = o.tap ? 'button' : 'div';
    return `<${tag} class="${cls}" data-c="${L}">${chip(L)}<span class="otxt">${text}</span>${extra}</${tag}>`;
  }).join('');
}

function applyStudent(r) {
  if (r.needCode) {
    // コードが ひつよう（はじめて・コードが かわった とき）
    S.needCode = true; S.codeBad = !!r.bad; S.codeWait = !!r.wait; S.code = '';
    saveCode('');

    return;

  S.needCode = false;
  if (r.now) S.offset = r.now - Date.now();
  if (typeof r.nick === 'string') S.nick = r.nick;
  if (r.taken) S.taken = r.taken;
  const st = r.state;
  if (st.round !== S.myRound) { S.myRound = st.round; S.mine = ''; S.chg = false; S.ask = ''; S.pending = ''; }
  S.state = st;
  renderStudent();
}

function studentMsg(st) {

async function studentStart() {
  clearTimeout(S.timer);
  try {
    const r = await callS('studentHello', S.code || loadCode());

      applyStudent(r);
      $('sNet').textContent = '';

    S.myRound = r.state.round; S.mine = r.mine; S.chg = !!r.chg; S.ask = ''; S.pending = '';
    if (S.code) saveCode(S.code);
    applyStudent(r);
    $('sNet').textContent = '';
  } catch (e) {
    $('sNet').textContent = 'つながりにくいよ。すこし まってね';
  }
  S.timer = setTimeout(studentPoll, 2000);
}

async function studentPoll() {
  if (active !== 'student') return;

    try {
      const r = await callS('getState');
      $('sNet').textContent = '';
      applyStudent(r);
      if (S.needCode) return; // コードが かわった。いれなおすまで とめる
    } catch (e) {
      $('sNet').textContent = 'つながりにくいよ。すこし まってね';
    }
  }
  if (active === 'student') S.timer = setTimeout(studentPoll, 2000);
}

/** 画像を ちいさく する（長い ほうを 1280pxまで） */
function shrinkImage(file) {
  return new Promise((resolve, reject) => {
    const fr = new FileReader();
    fr.onerror = () => reject(new Error('よめない ファイルだよ'));
    fr.onload = () => {
      const img = new Image();
      img.onerror = () => reject(new Error('よめない がぞうだよ'));
      img.onload = () => {
        const MAX = 1280;
        let w = img.naturalWidth || 800, h = img.naturalHeight || 600;
        const keepPng = /png|gif|svg/.test(file.type);
        if (w <= MAX && h <= MAX && file.size < 400 * 1024 && /^image\/(png|jpeg|gif|webp)$/.test(file.type)) {
          resolve(fr.result); // ちいさい ときは そのまま
          return;
        }
        const r = Math.min(1, MAX / Math.max(w, h));
        w = Math.max(1, Math.round(w * r)); h = Math.max(1, Math.round(h * r));
        const c = document.createElement('canvas');
        c.width = w; c.height = h;
        const g = c.getContext('2d');
        if (!keepPng) { g.fillStyle = '#fff'; g.fillRect(0, 0, w, h); }
        g.drawImage(img, 0, 0, w, h);
        let url = c.toDataURL(keepPng ? 'image/png' : 'image/jpeg', 0.85);
        if (keepPng && url.length > 900 * 1024) {
          // 大きすぎる PNG は しろい はいけいの JPEG にする
          g.globalCompositeOperation = 'destination-over';
          g.fillStyle = '#fff'; g.fillRect(0, 0, w, h);
          url = c.toDataURL('image/jpeg', 0.85);
        }
        resolve(url);
      };
      img.src = fr.result;
    };
    fr.readAsDataURL(file);
  });
}

/** 文字を コピーする */
function copyText(t) {
  let ok = false;
  const ta = document.createElement('textarea');
  ta.value = t;
  ta.setAttribute('readonly', '');
  ta.style.position = 'fixed'; ta.style.top = '0'; ta.style.opacity = '0';
  document.body.appendChild(ta);
  ta.select();
  try { ta.setSelectionRange(0, t.length); } catch (e) { /* なにもしない */ }
  try { ok = document.execCommand('copy'); } catch (e) { ok = false; }
  document.body.removeChild(ta);
  if (!ok && navigator.clipboard) {
    try { navigator.clipboard.writeText(t); ok = true; } catch (e) { ok = false; }
  }
  return ok;
}

/* ===== 時間の表示 ===== */
function tickAll() {
  if (active === 'student') tickStudent();
  if (active === 'teacher') tickTeacher();
}
// 画面が かくれているときも しめきりが うごくように、タイマーも のこす
setInterval(tickAll, 500);
// バーを なめらかに へらす（時刻から 計算するので、120Hzでも 速さは かわらない）
(function frame() { tickAll(); requestAnimationFrame(frame); })();

/* ===== 画面の切りかえ ===== */
function show(page) {
  active = page;
  clearTimeout(S.timer); clearTimeout(T.timer);
  $('student').classList.toggle('hidden', page !== 'student');
  $('teacher').classList.toggle('hidden', page !== 'teacher');
  if (page === 'student') studentStart();
  else { T.footKey = ''; teacherPoll(); }
}

if (RAW_PAGE === 'teacher' || RAW_PAGE === 'student') {
  show(RAW_PAGE);
} else {
  $('pv').classList.remove('hidden');
  $('pvT').addEventListener('click', () => { $('pvT').classList.add('on'); $('pvS').classList.remove('on'); show('teacher'); });
  $('pvS').addEventListener('click', () => { $('pvS').classList.add('on'); $('pvT').classList.remove('on'); show('student'); });
  show('teacher');
}
