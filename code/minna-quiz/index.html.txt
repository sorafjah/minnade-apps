function esc(s) {
  return String(s === null || s === undefined ? '' : s).replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));

/** 「漢字《かんじ》」「｜漢字《かんじ》」を ruby にする */
function rubyHtml(s) {
  let h = esc(s);
  h = h.replace(/[｜|]([^｜|《》]+?)《([^》]+?)》/g, '<ruby>$1<rt>$2</rt></ruby>');
  h = h.replace(/([一-鿿々〆ヶ]+)《([^》]+?)》/g, '<ruby>$1<rt>$2</rt></ruby>');
  return h.replace(/\n/g, '<br>');
}

}

/* ===== 画像のよみこみ（「img:ファイルID」のしるしは、アプリから画像のデータをもらう） ===== */
const IMG = { cache: {}, state: {}, queue: [], running: 0 };
function loadImg(t) {
  if (IMG.cache[t] || IMG.state[t] === 'q') return;
  if (typeof IMG.state[t] === 'number' && IMG.state[t] > Date.now()) return;
  IMG.state[t] = 'q';
  IMG.queue.push(t);
  pumpImg();
}
function pumpImg() {
  while (IMG.running < 3 && IMG.queue.length) {
    const t = IMG.queue.shift();
    IMG.running++;
    const delay = active === 'student' ? Math.random() * 800 / SPEED : 0;
    setTimeout(() => {
      call('getImage', t.slice(4), CID, RAW_PAGE === 'teacher' ? 'teacher' : '').then(d => {
        IMG.cache[t] = d;
        delete IMG.state[t];
        document.querySelectorAll('img[data-img]').forEach(el => { if (el.dataset.img === t) el.src = d; });
      }).catch(() => {
        IMG.state[t] = Date.now() + 4000;
      }).then(() => {
        IMG.running--;
        pumpImg();
      });
    }, delay);
  }
}
function hydrateImgs(root) {
  root.querySelectorAll('img[data-img]').forEach(el => {
    const t = el.dataset.img, d = IMG.cache[t];
    if (d) { if (el.getAttribute('src') !== d) el.src = d; }
    else loadImg(t);
  });
}

/** のこり時間の表示（バー・秒の文字）。p は 's'（生徒）か 't'（先生） */
function paintTimer(p, st, offset) {
  const ms = remainMs(st, offset);
  const view = (st && st.timeView) || 'bar';
  const bar = $(p + 'Bar');
  if (ms === null) {
    setText(p + 'Time', '');
    bar.classList.add('hidden');
    return null;
  }
  const sec = Math.ceil(ms / 1000);
  const label = view === 'bar' ? '' : (p === 's' ? 'のこり ' : 'のこり ') + sec + '秒';
  setText(p + 'Time', label);
  $(p + 'Time').classList.toggle('warn', sec <= 5);
  const showBar = view !== 'text';
  bar.classList.toggle('hidden', !showBar);
  if (showBar) {
    bar.classList.toggle('warn', sec <= 5);
    const w = (ms / (st.limit * 1000) * 100).toFixed(1) + '%';
    const fill = bar.firstElementChild;
    if (fill.style.width !== w) fill.style.width = w;
  }
  return sec;

function applyStudent(r) {
  if (r.needCode) {
    S.needCode = true; S.codeBad = !!r.bad; S.codeWait = !!r.wait; S.code = '';
    saveCode('');
    renderStudent();
    return;
  }
  S.needCode = false;
  if (r.now) S.offset = r.now - Date.now();
  if (typeof r.nick === 'string') S.nick = r.nick;
  if (r.taken) S.taken = r.taken;
  const st = r.state;
  if (st.sid !== S.curSid) { S.curSid = st.sid; S.res = null; S.soloEnd = false; S.joined = false; }
  if (st.round !== S.myRound) { S.myRound = st.round; S.mine = ''; S.chg = false; S.ask = ''; S.pending = ''; S.typed = ''; S.judge = null; resetTyped(); }
  if (typeof r.joined === 'boolean') S.joined = r.joined;
  if (typeof r.n === 'number') S.n = r.n;
  if (r.res) S.res = r.res;
  if (typeof r.judge === 'boolean') S.judge = r.judge;
  KANA = st.kana === 'hira' ? 'hira' : 'ruby';
  Snd.on = st.sound !== false && S.soundPref;
  S.state = st;
  renderStudent();
}

/** 入るまえに開いていた生徒・結果画面で待っていた生徒が、つぎの回に自動で入る */
async function ensureJoined() {
  const st = S.state;
  if (!st || !S.nick || S.joined || S.needCode) return;
  if (['lobby', 'open', 'closed', 'reveal', 'play'].indexOf(st.phase) < 0) return;
  if (S.joinTry === st.sid && Date.now() - S.joinAt < 5000 / SPEED) return;
  S.joinTry = st.sid; S.joinAt = Date.now();
  try {
    const r = await callS('joinSession');
    if (r && r.ok) { S.joined = true; S.n = r.n || S.n; }
    else if (r && r.needCode) applyStudent({ needCode: true });
  } catch (e) { /* つぎの更新でもう一度 */ }
  renderStudent();
}

  renderQ('s', { show: st.showQ, text: st.text, img: st.img, hints: st.hints });
  const canChange = !!S.mine && !S.chg && !st.input;
  const canTap = st.phase === 'open' && !S.sending && (!S.mine || canChange) && !st.input;
  const rev = st.phase === 'reveal';
  $('sOpts').classList.toggle('only', !withText);
  $('sOpts').classList.toggle('hidden', !!st.input);
  setHtml($('sOpts'), st.input ? '' : optsHtml(st.choices, { p: 's', tap: canTap, mine: S.pending || S.mine, showText: withText,
    nodim: canChange && st.phase === 'open', ask: S.ask, correct: rev ? st.correct : '' }));
  // 入力で答えるとき
  const showIn = !!st.input && st.phase === 'open' && !S.mine;
  $('sInputRow').classList.toggle('hidden', !showIn);
  const typedBox = $('sTypedShow');
  const showTyped = !!st.input && !!S.mine;
  typedBox.classList.toggle('hidden', !showTyped);
  typedBox.classList.toggle('win', rev && S.judge === true);
  if (showTyped) {
    let t = ks('あなたの 答《こた》え：') + esc(S.mine);
    if (rev) t += '<br>' + ks('正解《せいかい》：') + ks(st.correctText);
    setHtml(typedBox, t);
  } else if (!!st.input && rev) {
    setHtml(typedBox, ks('正解《せいかい》：') + ks(st.correctText));
    typedBox.classList.remove('hidden');
  }
  renderExplain('s', rev ? st.explain : []);
  const m = liveMsg(st);
  setMsg(m[0], m[1]);
  if (rev && S.soundRound !== st.round) {
    S.soundRound = st.round;
    if (m[1] === 'ok') Snd.play('ok'); else if (S.mine || S.judge !== null) Snd.play('ng');
  }
}

/* ----- ひとりずつ ----- */
function normAnsC(s) {
  s = String(s || '');
  try { s = s.normalize('NFKC'); } catch (e) { /* なにもしない */ }
  return s.toLowerCase().replace(/[\s　]+/g, '').replace(/[ァ-ヶ]/g, c => String.fromCharCode(c.charCodeAt(0) - 0x60)).replace(/[、。,.!！?？・「」『』（）()]/g, '');
}
function judgeInputC(q, text) {
  const n = normAnsC(text);
  return n !== '' && (q.accept || []).some(a => normAnsC(a) === n);
}
/** まちがいの型（記録に残すことば） */
function whyOfC(q, given, timeUp) {
  if (timeUp || given === '' || given === null || given === undefined) return '時間切れ（答えなし）';
  if (q.input) {
    const g = normAnsC(given);
    for (let i = 0; i < q.choices.length; i++) {
      if (q.map[i] === q.answer) continue;
      if (normAnsC(plainText(q.choices[i])) === g || normAnsC(readingText(q.choices[i])) === g) {
        const t = q.types[q.map[i]];
        if (t) return plainText(t);
      }
    }
    return '入力ちがい';
  }
  const t = q.types[given];
  return t ? plainText(t) : '選びまちがい';
}

function curSolo() { return S.solo && S.state && S.solo.sid === S.state.sid ? S.solo : null; }
function curQ(s) { return s.pack.qs.find(q => q.id === s.cur) || null; }
function soloSave(s) { try { lsSet(SOLO_KEY, JSON.stringify(s || S.solo)); } catch (e) { /* 保存できなくても続ける */ } }
function soloLoad(sid) {
  try {
    const o = JSON.parse(lsGet(SOLO_KEY) || 'null');
    if (o && o.sid === sid && o.nick === S.nick && o.pack) return o;
  } catch (e) { /* 読めないときは新しく始める */ }
  return null;
}
function soloInit(pack) {
  const qs = pack.qs.filter(q => q.level >= pack.lo && q.level <= pack.hi && (q.input ? (q.accept || []).length > 0 : !!q.answer));
  const s = {
    sid: pack.sid, nick: S.nick, pack: Object.assign({}, pack, { qs: qs }), total: Math.min(SOLO_MAX, qs.length),
    lv: pack.lv0, lv0: pack.lv0, upRun: 0, dnRun: 0, asked: [], n: 0, ok: 0, hist: [pack.lv0], types: {}, marks: [], seq: 0, outbox: [],
    cur: 0, curStart: 0, hint: 0, fb: null, done: false
  };
  soloNext(s);
  return s;
}
/** つぎの問題を選ぶ（いまのレベルの問題をまず出す。なければ、いちばん近いレベルから） */
function soloNext(s) {
  s.fb = null; s.hint = 0;
  if (s.n >= s.total) { s.done = true; s.cur = 0; soloSave(s); return; }
  const left = s.pack.qs.filter(q => s.asked.indexOf(q.id) < 0);
  if (!left.length) { s.done = true; s.cur = 0; soloSave(s); return; }
  let best = 99;
  left.forEach(q => { const d = Math.abs(q.level - s.lv); if (d < best) best = d; });
  const c = left.filter(q => Math.abs(q.level - s.lv) === best);
  s.cur = c[Math.floor(Math.random() * c.length)].id;
  s.curStart = Date.now();
  soloSave(s);
}
function soloStatus() {
  const s = S.solo;
  return { st: s.done ? 'done' : 'play', n: s.n, ok: s.ok, lv: s.lv, lv0: s.lv0, total: s.total, hist: s.hist.slice(-80), types: s.types };
}

async function ensureSolo() {
  const st = S.state;
  if (!st || st.mode !== 'solo' || st.phase !== 'play' || !S.nick || S.needCode) return;
  if (curSolo() || S.soloBusy) return;
  S.soloBusy = true;
  try {
    const saved = soloLoad(st.sid);
    if (saved) S.solo = saved;
    else {
      const r = await callS('getSoloPack');
      if (r && r.ok) { S.solo = soloInit(r.pack); soloFlush(); }
      else if (r && r.needCode) applyStudent({ needCode: true });
    }
  } catch (e) { setText('sNet', 'つながりにくいよ。すこし まってね'); }
  S.soloBusy = false;
  renderStudent();
}

/** 記録を先生に送る。送れなかった記録はlocalStorageにたまっていて、つながったときに送りなおす */
async function soloFlush() {
  const s = S.solo;
  if (!s || S.needCode) return;
  if (S.flushing) { S.flushAgain = true; return; }
  S.flushing = true;
  try {
    for (let guard = 0; guard < 12; guard++) {
      S.flushAgain = false;
      const items = s.outbox.slice(0, 50);
      const r = await callS('studentLog', s.sid, items, soloStatus());
      if (!r) break;
      if (r.needCode) { applyStudent({ needCode: true }); break; }
      if (r.gone) { s.outbox = []; soloSave(s); break; }
      if (!r.ok) break;
      s.outbox.splice(0, items.length);
      soloSave(s);
      if (!s.outbox.length && !S.flushAgain) break;
    }
  } catch (e) { /* つながったときに送りなおす */ }
  S.flushing = false;
}

function soloAnswer(given, timeUp) {
  const s = S.solo;
  if (!s || s.fb || !s.cur) return;
  const q = curQ(s); if (!q) return;
  let ok = false;
  if (!timeUp) ok = q.input ? judgeInputC(q, given) : given === q.answer;
  const sec = Math.round((Date.now() - s.curStart) / 100) / 10;
  const why = ok ? '' : whyOfC(q, timeUp ? '' : given, timeUp);
  const lvNow = s.lv;
  s.asked.push(q.id); s.n++; if (ok) s.ok++;
  s.marks.push(timeUp ? '-' : (ok ? 'o' : 'x'));
  s.seq++;
  s.outbox.push({ seq: s.seq, ts: Date.now(), no: s.n, qid: q.id,
    given: timeUp ? '' : (q.input ? String(given).slice(0, 40) : (q.map[LETTERS.indexOf(given)] || given)),
    ok: ok, text: q.text, why: why, level: lvNow, qlevel: q.level, sec: sec, hint: s.hint > 0 });
  if (!ok && why) s.types[why] = (s.types[why] || 0) + 1;
  let lvMsg = '';
  if (s.pack.auto) {
    if (ok) {
      s.upRun++; s.dnRun = 0;
      if (s.upRun >= s.pack.upN) { s.upRun = 0; if (s.lv < s.pack.hi) { s.lv++; s.hist.push(s.lv); lvMsg = 'up'; } }
    } else {
      s.dnRun++; s.upRun = 0;
      if (s.dnRun >= s.pack.downN) { s.dnRun = 0; if (s.lv > s.pack.lo) { s.lv--; s.hist.push(s.lv); lvMsg = 'down'; } }
    }
  }
  s.fb = { ok: ok, timeUp: !!timeUp, given: timeUp ? '' : given, lvMsg: lvMsg };
  soloSave();
  Snd.play(lvMsg === 'up' ? 'up' : ok ? 'ok' : 'ng');
  renderStudent();
  soloFlush();
}

function tickStudent() {
  const st = S.state;
  if (!st || S.needCode) { paintTimer('s', null, 0); return; }
  const s = curSolo();
  if (st.mode === 'solo') {
    if (st.phase === 'play' && s && s.cur && !s.fb && !s.done) {

      if (q) {

function studentDelay() {
  const st = S.state;
  if (!st) return 3000;
  if (st.mode === 'solo' && st.phase === 'play') return 8000;
  if (st.phase === 'wait') return 5000;
  if (st.phase === 'end') return 6000;
  return 3000;
}

async function studentStart() {
  clearTimeout(S.timer);
  try {
    const r = await callS('studentHello', S.code || loadCode());
    if (r.needCode) {
      applyStudent(r);
      setText('sNet', '');
      return;
    }
    S.myRound = r.state.round; S.mine = r.mine || ''; S.chg = !!r.chg; S.ask = ''; S.pending = '';
    if (S.code) saveCode(S.code);
    applyStudent(r);
    setText('sNet', '');
    await ensureJoined();
    await ensureSolo();
  } catch (e) {
    setText('sNet', 'つながりにくいよ。すこし まってね');
  }
  S.timer = setTimeout(studentPoll, jitter(3000));
}

async function studentPoll() {
  if (active !== 'student') return;
  if (!S.sending) {
    try {
      const r = await callS('getState');
      setText('sNet', '');
      applyStudent(r);
      if (S.needCode) return;
      await ensureJoined();
      await ensureSolo();
      if (curSolo() && S.state.mode === 'solo' && S.state.phase === 'play') await soloFlush();
    } catch (e) {
      setText('sNet', 'つながりにくいよ。すこし まってね');
    }
  }
  if (active === 'student') S.timer = setTimeout(studentPoll, jitter(studentDelay()));
}

/* ----- 操作 ----- */
async function onOpt(btn) {
  Snd.unlock();   // 音の解錠は、通信より前（タップの中）で行う
  const st = S.state; if (!st) return;
  const c = btn.dataset.c;
  if (st.mode === 'solo') {
    if (curSolo() && S.solo.cur && !S.solo.fb) soloAnswer(c, false);
    return;
  }
  if (S.sending || st.phase !== 'open') return;
  if (S.mine) {
    // かえられるのは1回だけ。まちがえて押さないように、2回押してきめる
    if (S.chg) return;
    if (c === S.mine) { S.ask = ''; renderStudent(); return; }
    if (S.ask !== c) { S.ask = c; renderStudent(); return; }
  }
  S.ask = '';
  S.sending = true; S.pending = c; renderStudent();
  try {
    const r = await callS('submitAnswer', st.round, c);
    S.sending = false; S.pending = '';
    if (r.needNick) S.nick = '';
    else if (!r.ok) setText('sNet', 'しめきったあとだったよ');
    applyStudent(r);
    if (r.ok) { S.mine = r.choice; S.chg = !!r.chg; renderStudent(); }
  } catch (err) {
    S.sending = false; S.pending = '';
    setText('sNet', 'おくれなかったよ。もういちど おしてね');
    renderStudent();
  }
}

async function sendTyped() {
  Snd.unlock();
  const st = S.state; if (!st) return;
  const v = ($('sTyped').value || '').trim();
  if (!v) return;
  if (st.mode === 'solo') {
    if (curSolo() && S.solo.cur && !S.solo.fb) { resetTyped(); soloAnswer(v, false); }
    return;
  }
  if (S.sending || st.phase !== 'open' || S.mine) return;
  S.sending = true; renderStudent();
  try {
    const r = await callS('submitAnswer', st.round, v);
    S.sending = false;
    if (r.needNick) S.nick = '';
    else if (!r.ok) setText('sNet', 'しめきったあとだったよ');
    applyStudent(r);
    if (r.ok) { S.mine = r.choice; S.chg = false; resetTyped(); renderStudent(); }
  } catch (err) {
    S.sending = false;
    setText('sNet', 'おくれなかったよ。もういちど おしてね');
    renderStudent();
  }
}

document.addEventListener('click', e => {
  if (active !== 'student') return;
  let b;
  if ((b = e.target.closest('#sOpts button.opt'))) { onOpt(b); return; }
  if (e.target.closest('#sSend')) { sendTyped(); return; }
  if ((b = e.target.closest('#sAct button'))) {
    Snd.unlock();
    const s = curSolo();
    if (b.dataset.a === 'hint' && s && !s.fb) { s.hint++; soloSave(); renderStudent(); }
    else if (b.dataset.a === 'next' && s && s.fb) { soloNext(s); renderStudent(); soloFlush(); }
    return;

/* ----- QRコード ----- */
function loadQrLib() {
  if (T.qrState === 'loading' || T.qrState === 'fail' || typeof qrcode === 'function') return;
  T.qrState = 'loading';
  const s = document.createElement('script');
  s.src = QR_LIB;
  s.onload = function () { T.qrState = 'ok'; if (T.last) renderTeacher(T.last); };
  s.onerror = function () { T.qrState = 'fail'; if (T.last) renderTeacher(T.last); };
  document.head.appendChild(s);
}
function qrSvg(url) {
  if (!url) return '';
  if (T.qrCache.url === url && T.qrCache.svg) return T.qrCache.svg;
  if (typeof qrcode !== 'function') { loadQrLib(); return ''; }
  try {
    const q = qrcode(0, 'M');
    q.addData(url);
    q.make();
    const n = q.getModuleCount(), m = 4, size = n + m * 2;
    let d = '';
    for (let r = 0; r < n; r++) for (let c = 0; c < n; c++) if (q.isDark(r, c)) d += 'M' + (c + m) + ' ' + (r + m) + 'h1v1h-1z';
    const svg = '<svg xmlns="' + SVG_NS + '" viewBox="0 0 ' + size + ' ' + size + '" shape-rendering="crispEdges" role="img" aria-label="参加用のQRコード"><rect width="' + size + '" height="' + size + '" fill="#fff"/><path d="' + d + '" fill="#000"/></svg>';
    T.qrCache = { url: url, svg: svg };
    return svg;
  } catch (e) { return ''; }
}
function qrBlock(r) {
  const svg = qrSvg(r.url);
  if (svg) return svg;
  if (!r.url) return H('div', 'note', 'このアプリのアドレスが取れませんでした。デプロイしたURLを、生徒に伝えてください。');
  return H('div', 'note', T.qrState === 'fail' ? 'QRコードを読みこめませんでした。下のアドレスを生徒に伝えてください。' : 'QRコードを作っています…');
}

function renderTeacher(r) {
  T.last = r;
  T.offset = r.now - Date.now();
  const st = r.state, se = r.settings, phase = st.phase;
  const on = phase !== 'wait';
  const mode = on ? st.mode : se.mode;
  const style = on ? st.style : se.style;
  $('tSet').textContent = (r.setName || '') + (on && st.sid ? '　回 ' + st.sid : '');
  $('tPhase').textContent = modeText(mode, style) + '　' + PHASE_TEXT[phase];
  const codeOn = !!(r.codeOn && r.code);
  setHtml($('tCode'), codeOn ? '参加コード <b>' + r.code + '</b>' : '');
  $('tCode').classList.toggle('hidden', !codeOn || !on);
  $('tFormBtn').classList.toggle('hidden', !(on && mode === 'all' && st.source === 'manual' && phase !== 'end'));

  const vLive = mode === 'all' && isLive(st);
  const vBoard = phase === 'play' || phase === 'end';
  $('tvStart').classList.toggle('hidden', phase !== 'wait');
  $('tvLobby').classList.toggle('hidden', phase !== 'lobby');
  $('tvLive').classList.toggle('hidden', !vLive);
  $('tvBoard').classList.toggle('hidden', !vBoard);

  if (phase === 'wait') syncStart(r);

  if (vBoard) setHtml($('tvBoard'), boardHtml(r));
  if (!$('setOv').classList.contains('hidden')) syncSettings(r);
  renderFoot(r);

}

function renderFoot(r) {
  const st = r.state, se = r.settings, phase = st.phase;
  const key = [phase, st.round, st.correct, st.hints.length, st.source, r.hasKey, st.index, st.total, st.mode, st.input, T.busy].join('|');
  if (key === T.footKey) return;
  T.footKey = key;
  const btn = (act, label, cls, arg, confirm) => H('button', cls || 'main-btn', esc(label), 'type="button" data-act="' + act + '" data-arg="' + esc(arg || '') + '"' + (confirm ? ' data-confirm="' + esc(confirm) + '"' : ''));
  const sheet = st.source === 'sheet';
  let h = '';
  if (phase === 'wait') h = btn('teacherStart', 'この設定で受付を始める');
  else if (phase === 'lobby') {
    if (st.mode === 'solo') h = btn('teacherBegin', 'この回をはじめる（生徒が自分のペースで進めます）');
    else if (sheet) h = btn('teacherBegin', 'この回をはじめる（1問目を出す）');
    else h = btn('form', '問題を入力して始める');
    h += btn('teacherRestart', '受付をやめる', 'sub-btn');
  } else if (phase === 'play') {
    h = btn('teacherEnd', 'この回を終わる', 'main-btn', '', 'もう一度押すと、この回が終わります');
  } else if (phase === 'open') {
    if (st.hints.length < st.hintTotal) h += btn('teacherHint', 'ヒントを出す（' + (st.hints.length + 1) + '）', 'sub-btn');
    h += btn('teacherClose', 'しめきる');
  } else if (phase === 'closed') {
    if (r.hasKey) h = btn('teacherReveal', '答え合わせ');
    else h = H('span', 'lab', '正解を選んでください') + st.choices.map((_, i) => btn('teacherReveal', LETTERS[i], 'pickbtn', LETTERS[i])).join('');
  } else if (phase === 'reveal') {
    if (sheet) h = st.index < st.total ? btn('teacherNext', 'つぎの問題へ') : btn('teacherEnd', 'この回を終わる');
    else h = btn('form', 'つぎの問題を入力する') + btn('teacherEnd', 'この回を終わる', 'sub-btn', '', 'もう一度押すと、この回が終わります');

  }
  $('tFoot').innerHTML = h;
}

function tickTeacher() {
  const r = T.last;
  if (!r) return;
  const sec = paintTimer('t', r.state, T.offset);
  if (sec === 0 && r.state.phase === 'open' && T.autoClosed !== r.state.round && !T.busy) {
    T.autoClosed = r.state.round;
    teacherAct('teacherClose');
  }
}

async function teacherPoll() {
  if (active !== 'teacher') return;
  if (!T.busy) {
    try {
      const r = await call('teacherStatus');
      setText('tNet', '');
      renderTeacher(r);
    } catch (e) {
      setText('tNet', (e && e.message) ? e.message : 'つながりにくいです');
    }
  }
  if (active === 'teacher') {
    const ph = T.last ? T.last.state.phase : 'wait';
    T.timer = setTimeout(teacherPoll, jitter(ph === 'play' || ph === 'wait' || ph === 'end' ? 5000 : 3000));
  }
}

async function teacherAct(fn, arg) {
  if (T.busy) return;
  T.busy = true;
  $('tFoot').querySelectorAll('button').forEach(b => { b.disabled = true; });
  try {
    if (arg === undefined) await call(fn); else await call(fn, arg);
    const r = await call('teacherStatus');
    T.footKey = '';
    if (fn === 'teacherSettings' || fn === 'teacherStart' || fn === 'teacherRestart') T.setsStale = true;
    setText('tNet', '');
    T.busy = false;
    renderTeacher(r);
  } catch (e) {
    setText('tNet', (e && e.message) ? e.message : 'うまくいきませんでした');
    T.footKey = '';
    T.busy = false;
  }
  T.busy = false;
  $('tFoot').querySelectorAll('button').forEach(b => { b.disabled = false; });
}

document.addEventListener('click', e => {
  if (active !== 'teacher') return;
  let b = e.target.closest('#tFoot button[data-act]');
  if (b) {
    const confirmText = b.dataset.confirm;
    if (confirmText) {
      if (!b._arm) {
        b._arm = true; b.classList.add('arm'); b._label = b.textContent; b.textContent = confirmText;
        setTimeout(() => { b._arm = false; b.classList.remove('arm'); if (b._label) b.textContent = b._label; }, 3500);
        return;
      }
    }
    if (b.dataset.act === 'form') openForm();
    else teacherAct(b.dataset.act, b.dataset.arg || undefined);
    return;
  }
  b = e.target.closest('[data-a="bigqr"]');
  if (b) { T.bigQr = true; if (T.last) renderTeacher(T.last); return; }
  b = e.target.closest('[data-a="normalqr"]');
  if (b) { T.bigQr = false; if (T.last) renderTeacher(T.last); }
});

const newImgs = {};
function shrinkImage(file) {
  return new Promise((resolve, reject) => {
    const fr = new FileReader();
    fr.onerror = () => reject(new Error('読めないファイルです'));
    fr.onload = () => {
      const img = new Image();
      img.onerror = () => reject(new Error('読めない画像です'));
      img.onload = () => {
        const MAX = 1280;
        let w = img.naturalWidth || 800, h = img.naturalHeight || 600;
        const keepPng = /png|gif|svg/.test(file.type);
        if (w <= MAX && h <= MAX && file.size < 400 * 1024 && /^image\/(png|jpeg|gif|webp)$/.test(file.type)) { resolve(fr.result); return; }
        const r = Math.min(1, MAX / Math.max(w, h));
        w = Math.max(1, Math.round(w * r)); h = Math.max(1, Math.round(h * r));
        const c = document.createElement('canvas');
        c.width = w; c.height = h;
        const g = c.getContext('2d');
        if (!keepPng) { g.fillStyle = '#fff'; g.fillRect(0, 0, w, h); }
        g.drawImage(img, 0, 0, w, h);
        let url = c.toDataURL(keepPng ? 'image/png' : 'image/jpeg', 0.85);
        if (keepPng && url.length > 900 * 1024) {
          g.globalCompositeOperation = 'destination-over';
          g.fillStyle = '#fff'; g.fillRect(0, 0, w, h);
          url = c.toDataURL('image/jpeg', 0.85);
        }
        resolve(url);
      };
      img.src = fr.result;
    };
    fr.readAsDataURL(file);
  });
}
function copyText(t) {
  let ok = false;
  const ta = document.createElement('textarea');
  ta.value = t; ta.setAttribute('readonly', '');
  ta.style.position = 'fixed'; ta.style.top = '0'; ta.style.opacity = '0';
  document.body.appendChild(ta);
  ta.select();
  try { ta.setSelectionRange(0, t.length); } catch (e) { /* なにもしない */ }
  try { ok = document.execCommand('copy'); } catch (e) { ok = false; }
  document.body.removeChild(ta);
  if (!ok && navigator.clipboard) { try { navigator.clipboard.writeText(t); ok = true; } catch (e) { ok = false; } }
  return ok;
}
