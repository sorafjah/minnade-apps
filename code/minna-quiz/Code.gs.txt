function doGet(e) {
  const p = (e && e.parameter) || {};
  const want = p.page || p.p || '';   // ?page=teacher でも ?p=teacher でも先生画面
  const t = HtmlService.createTemplateFromFile('index');
  t.page = want === 'teacher' ? 'teacher' : 'student';
  t.code = String(p.code || '').replace(/\D/g, '').slice(0, 4);
  return t.evaluate()
    .setTitle('みんなでクイズ')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/** 記録・まとめの中身を消す（確かめのときの記録を消して、ほかの先生に渡すときに使う） */
function clearRecords_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let n = 0;
  ss.getSheets().forEach(sh => {
    const name = sh.getName();
    const isLog = name === SH.log || name === SH.logOld || name.indexOf(SH.log + '：') === 0 || name.indexOf(SH.logOld + '：') === 0;
    if (!isLog && name !== SH.sum) return;
    const last = sh.getLastRow();
    if (last > 1) { sh.getRange(2, 1, last - 1, Math.max(sh.getLastColumn(), 1)).clearContent(); n += last - 1; }
  });
  return n;
}

function menuClearRecords() {
  const ui = SpreadsheetApp.getUi();
  const r = ui.alert('テストの記録を消す',
    '「記録」「まとめ」のタブの中身をすべて消します。問題・名簿・生徒設定は消えません。\nもとにもどせませんが、消してよろしいですか。', ui.ButtonSet.YES_NO);
  if (r !== ui.Button.YES) return;
  ui.alert('消しました（' + clearRecords_() + '行）');
}

function prop_(key, def) {
  const v = PropertiesService.getScriptProperties().getProperty(key);
  const base = def ? JSON.parse(JSON.stringify(def)) : null;
  if (!v) return base;
  const o = JSON.parse(v);
  return (base && o && typeof o === 'object') ? Object.assign(base, o) : o;
}

function setProp_(key, val) {
  PropertiesService.getScriptProperties().setProperty(key, JSON.stringify(val));
}

function cget_(k) {
  const c = CacheService.getScriptCache().get(k);
  return c ? JSON.parse(c) : null;
}

function cput_(k, v, ttl) {
  CacheService.getScriptCache().put(k, JSON.stringify(v), ttl || TTL);
}

function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try { return fn(); } finally { lock.releaseLock(); }
}

/** 日本語の名前を キャッシュの キーに する（encodeURIComponent。長すぎる ときは みじかくする） */
function ckey_(prefix, name) {
  let k = prefix + encodeURIComponent(String(name));
  if (k.length > 200) {
    let h = 5381;
    for (let i = 0; i < k.length; i++) h = ((h * 33) ^ k.charCodeAt(i)) >>> 0;
    k = prefix + 'h' + h.toString(36) + '_' + k.length;
  }
  return k;
}

/** 大きい もの（問題の セットなど）を 分けて キャッシュに 入れる */
function bigPut_(key, obj, ttl) {
  const s = JSON.stringify(obj), CH = 90000, parts = {};
  let n = 0;
  for (let i = 0; i < s.length; i += CH) parts[key + '_' + (n++)] = s.slice(i, i + CH);
  const c = CacheService.getScriptCache();
  try { c.putAll(parts, ttl || TTL); c.put(key, String(n), ttl || TTL); } catch (e) { /* きおくできなくても つづける */ }
}

function bigGet_(key) {
  const c = CacheService.getScriptCache();
  const head = c.get(key);
  if (!head) return null;
  const keys = [];
  for (let i = 0; i < Number(head); i++) keys.push(key + '_' + i);
  const got = c.getAll(keys);
  if (!keys.every(k => got[k])) return null;
  try { return JSON.parse(keys.map(k => got[k]).join('')); } catch (e) { return null; }
}

/** 「漢字《かんじ》」の ふりがなを とった 文字 */
function plain_(s) { return String(s || '').replace(/[｜|]/g, '').replace(/《[^》]*》/g, ''); }
/** 「漢字《かんじ》」を よみ（かんじ）に した 文字 */
function reading_(s) {
  return String(s || '')
    .replace(/[｜|]([^｜|《》]+)《([^》]+)》/g, '$2')
    .replace(/[一-鿿々〆ヶ]+《([^》]+)》/g, '$1')
    .replace(/《[^》]*》/g, '');
}

/** 入力の こたえを くらべる ために そろえる（全角半角・大文字小文字・かたかな・空白・かっこ） */
function normAns_(s) {
  s = String(s || '');
  try { s = s.normalize('NFKC'); } catch (e) { /* なにもしない */ }
  return s.toLowerCase().replace(/[\s　]+/g, '')
    .replace(/[ァ-ヶ]/g, c => String.fromCharCode(c.charCodeAt(0) - 0x60))
    .replace(/[、。,.!！?？・「」『』（）()]/g, '');
}

function judgeInput_(q, text) {
  const n = normAns_(text);
  return n !== '' && (q.accept || []).some(a => normAns_(a) === n);
}

/** 文字から 数の ならびを つくる（おなじ もじなら おなじ ならび。2択の えらびに つかう） */
function rng_(seed) {
  const s = String(seed);
  let h = 1779033703 ^ s.length;
  for (let i = 0; i < s.length; i++) { h = Math.imul(h ^ s.charCodeAt(i), 3432918353); h = (h << 13) | (h >>> 19); }
  let a = h >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const CID_RE = /^[A-Za-z0-9_-]{8,48}$/;
const CODE_FAIL_MAX = 30; // 1分に まちがえて よい 回数（ぜんぶの iPadで）

function newCodeStr_(old) {
  let c;
  do { c = String(Math.floor(Math.random() * 10000)).padStart(4, '0'); } while (c === old);
  return c;
}

/** いまの コード（まだ なければ つくる） */
function joinCode_() {
  const set = prop_('settings', DEF_SET);
  if (/^\d{4}$/.test(set.code || '')) return set.code;
  return withLock_(() => {
    const s = prop_('settings', DEF_SET);
    if (!/^\d{4}$/.test(s.code || '')) { s.code = newCodeStr_(''); setProp_('settings', s); }
    return s.code;
  });
}

function okCids_() {
  let a = cget_('okcids');
  if (!a) {
    a = prop_('okcids', null);
    if (!Array.isArray(a)) a = [];
    cput_('okcids', a);
  }
  return a;
}

function passed_(cid) { return CID_RE.test(String(cid || '')) && okCids_().indexOf(String(cid)) >= 0; }

/** 生徒の iPadを とおして よいか（コードを つかわない ときは だれでも よい） */
function guard_(cid, pub) { return !pub.codeOn || passed_(cid); }

/** 'ok' / 'bad' / 'wait'（まちがいが おおいので すこし まつ） */
function tryJoin_(cid, code) {
  cid = String(cid || ''); code = String(code || '');
  if (!CID_RE.test(cid)) return 'bad';
  if (passed_(cid)) return 'ok';
  if (!/^\d{4}$/.test(code)) return 'bad';
  const cache = CacheService.getScriptCache();
  const fails = Number(cache.get('codefail') || 0);
  if (fails >= CODE_FAIL_MAX) return 'wait';
  if (code !== joinCode_()) { cache.put('codefail', String(fails + 1), 60); return 'bad'; }
  withLock_(() => {
    const a = okCids_().filter(x => x !== cid);
    a.push(cid);
    while (a.length > 800) a.shift();
    setProp_('okcids', a);
    cput_('okcids', a);
  });
  return 'ok';
}

/** 生徒にも見せてよい情報だけをまとめる（正解はこたえあわせまで入れない） */
function buildPublic_() {
  const st = getState_();
  const set = prop_('settings', DEF_SET);
  const q = prop_('cur', null);
  const on = !!st.sid && st.phase !== 'wait';
  const pub = {
    v: PUB_VER, sid: st.sid || '', mode: on ? st.mode : set.mode, style: on ? st.style : set.style,
    round: st.round, phase: st.phase, index: st.index, total: st.total, logTab: st.logTab || '',
    correct: st.phase === 'reveal' ? st.correct : '', correctText: '',
    timeView: set.timeView || 'bar', showQ: set.showQ, kana: set.kana === 'hira' ? 'hira' : 'ruby', sound: set.sound !== false,
    source: on ? st.source : set.source, openedAt: st.openedAt, setName: st.setName, defLimit: effLimit_(set),
    nameMode: 'chara', roster: [], codeOn: set.codeOn === true,
    text: '', choices: [], input: false, img: '', hints: [], hintTotal: 0, limit: 0, explain: []
  };
  if (q && st.round > 0 && LIVE.indexOf(st.phase) >= 0) {
    pub.text = q.text;
    pub.input = !!q.input;
    pub.choices = q.input ? [] : q.choices;
    pub.img = q.img;
    pub.limit = q.limit;
    pub.hints = q.hints.slice(0, st.hintShown);
    pub.hintTotal = q.hints.length;
    if (st.phase === 'reveal') {
      pub.explain = q.explain || [];
      if (q.input) pub.correctText = q.correctText;
    }
  }
  if (set.nameMode !== 'chara') {
    let names = prop_('rosterNames', null);
    if (!Array.isArray(names)) names = refreshRoster_(set).names;
    if (names.length) { pub.nameMode = 'roster'; pub.roster = names; }
  }
  return pub;
}

function getPublic_() {
  let pub = cget_('pub');
  if (!pub || pub.v !== PUB_VER) { pub = buildPublic_(); cput_('pub', pub); }
  return pub;
}

function save_(st, q) {
  setProp_('state', st);
  if (q !== undefined) setProp_('cur', q);
  const pub = buildPublic_();
  cput_('pub', pub);
  return pub;
}

function normLetter_(s) {
  s = String(s || '').trim().toUpperCase()
    .replace(/[Ａ-Ｄ]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0))
    .replace(/[１-４]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0));
  if (/^[1-4]$/.test(s)) s = LETTERS[Number(s) - 1];
  return (s.length === 1 && LETTERS.indexOf(s) >= 0) ? s : '';
}

/**
 * 画像の しるしに かえる
 *  画像の なまえ・ドライブの リンク → 「img:ファイルID」（アプリが 中継して わたす）
 *  ほかの URL → そのまま
 *  見つからない → 「nf:メッセージ」（先生の 画面にだけ 出す）
 */
function imgUrl_(s) {
  s = String(s || '').trim();
  if (!s) return '';
  if (/^(img|nf):/.test(s)) return s;
  if (/^https?:\/\//.test(s)) {
    if (/(drive|docs)\.google\.com/.test(s)) {
      const m = s.match(/\/d\/([\w-]{20,})/) || s.match(/[?&]id=([\w-]{20,})/);
      if (m) return 'img:' + m[1];
    }
    return s;
  }
  const map = imgMapSafe_();
  if (map[s]) return 'img:' + map[s];
  if (/^[\w-]{25,}$/.test(s)) return 'img:' + s;
  return 'nf:画像「' + s + '」が見つかりません';
}

function item_(v) {
  if (v && typeof v === 'object') return { text: String(v.text || '').trim(), img: imgUrl_(v.img) };
  const s = String(v || '').trim();
  if (/^https?:\/\//.test(s)) return { text: '', img: imgUrl_(s) };
  return { text: s, img: '' };
}

function items_(list, max) {
  return (list || []).map(item_).filter(x => x.text || x.img).slice(0, max);
}

/** セルの中身（セルの中に 入れた 画像は 生徒に 見せられないので、先生に しらせる） */
function cellItem_(disp, vals, i, col) {
  if (col < 0 || col >= disp[i].length) return { text: '', img: '' };
  const raw = vals[i][col];
  if (raw && typeof raw === 'object' && typeof raw.getContentUrl === 'function') {
    return { text: '', img: 'nf:セルの中の画像は使えません。「画像」ボタンから入れて、名前を書いてください' };
  }
  return item_(disp[i][col]);
}

/** レベル：1〜9（からっぽ・数字でないときは 1） */
function normLevel_(v) {
  const n = parseInt(zen2han_(String(v === null || v === undefined ? '' : v)).replace(/[^0-9]/g, ''), 10);
  return isNaN(n) ? 1 : Math.max(1, Math.min(9, n));
}

function normQ_(o) {
  const raw = (o.choices || []).slice(0, 4).map(s => String(s || '').trim());
  let n = 0;
  raw.forEach((s, i) => { if (s) n = i + 1; });
  const choices = n === 0 ? ['', '', '', ''] : raw.slice(0, Math.max(n, 2));
  while (choices.length < 2) choices.push('');
  let answer = normLetter_(o.answer);
  if (answer && LETTERS.indexOf(answer) >= choices.length) answer = '';
  const types = {};
  LETTERS.split('').forEach((L, i) => { const t = str_((o.types || [])[i]); if (t && i < choices.length) types[L] = t; });
  const alt = String(o.alt || '').split(/[／\/,、，;；\n]/).map(s => s.trim()).filter(Boolean).slice(0, 10);
  return {
    text: String(o.text || '').trim(),
    choices: choices,
    answer: answer,
    hints: items_(o.hints, 5),
    explain: items_(o.explain, 4),
    img: imgUrl_(o.img),
    limit: normLimit_(o.limit),
    level: normLevel_(o.level),
    types: types,
    alt: alt
  };
}

/** 制限時間：からっぽなら null（せっていの 時間を つかう） */

  if (v === null || v === undefined || String(v).trim() === '') return null;
  const n = parseInt(zen2han_(String(v)), 10);
  return isNaN(n) ? null : Math.max(0, Math.min(600, n));
}

/**
 * 問題を 生徒に 出す 形に する（答え方に 合わせる）
 *  2択：正解と、まちがいを 1つ（seed で 決まる）。ならびも seed で 決まる
 *  入力：せんたくしを 見せず、正解の ことばを 打ってもらう（正解が ないときは 4択のまま）
 *  map：出した 選択肢の 文字（A〜D）が、シートの どの 選択肢か
 */
function present_(q, style, seed) {
  const rnd = rng_(seed);
  const n = q.choices.length;
  const o = {
    text: q.text, img: q.img, hints: q.hints, explain: q.explain, limit: q.limit, level: q.level || 1,
    choices: q.choices.slice(), map: LETTERS.slice(0, n).split(''), answer: q.answer,
    input: false, accept: [], correctText: '', types: {}
  };
  LETTERS.slice(0, n).split('').forEach(L => { if (q.types && q.types[L]) o.types[L] = q.types[L]; });
  const ai = q.answer ? LETTERS.indexOf(q.answer) : -1;
  if (style === 'two' && ai >= 0 && n > 2) {
    const wrong = [];
    for (let i = 0; i < n; i++) if (i !== ai && q.choices[i]) wrong.push(i);
    if (wrong.length) {
      const w = wrong[Math.floor(rnd() * wrong.length)];
      const pair = rnd() < 0.5 ? [ai, w] : [w, ai];
      o.choices = pair.map(i => q.choices[i]);
      o.map = pair.map(i => LETTERS[i]);
      o.answer = LETTERS[pair.indexOf(ai)];
      o.types = {};
      pair.forEach((i, k) => { const t = q.types && q.types[LETTERS[i]]; if (t) o.types[LETTERS[k]] = t; });
    }
  } else if (style === 'input' && ai >= 0 && q.choices[ai]) {
    o.input = true;
    o.correctText = q.choices[ai];
    o.accept = uniq_([plain_(q.choices[ai]), reading_(q.choices[ai])].concat(q.alt || []).filter(Boolean));
  }
  return o;
}

/** まちがいの型（記録に 残す 言葉） */
function whyOf_(q, given, changedFromRight) {
  if (given === '' || given === null || given === undefined) return '時間切れ（答えなし）';
  if (changedFromRight) return '答えを変えてまちがえた';
  if (q.input) {
    const g = normAns_(given);
    for (let i = 0; i < q.choices.length; i++) {
      if (q.map[i] === q.answer) continue;
      if (normAns_(plain_(q.choices[i])) === g || normAns_(reading_(q.choices[i])) === g) {
        const t = q.types[q.map[i]];
        if (t) return plain_(t);
      }
    }
    return '入力ちがい';
  }
  const t = q.types[given];
  return t ? plain_(t) : '選びまちがい';
}

/** 見出しから、どの列に何があるかをさがす（列の順番や数がちがっても読める） */
function findCols_(head) {
  const h = head.map(s => String(s).replace(/\s/g, '')
    .replace(/[Ａ-Ｚ０-９]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0)));
  const isImg = s => /(画像|がぞう|写真|しゃしん)/.test(s);
  const isHint = s => /^(ヒント|ひんと)/.test(s);
  const isEx = s => /^(解説|かいせつ|説明|せつめい|こたえの説明|答えの説明)/.test(s);
  const isType = s => /^[A-D](の)?(型|まちがい)/.test(s);
  const find = (fn, def) => { const i = h.findIndex(fn); return i >= 0 ? i : def; };

  const hintMap = {}, hintOrder = [];
  let lastKey = null, auto = 0;
  h.forEach((s, i) => {
    if (!isHint(s)) return;
    const m = s.match(/\d+/);
    let key;
    if (m) key = 'n' + m[0];
    else if (isImg(s) && lastKey) key = lastKey;
    else key = 'a' + (auto++);
    if (!hintMap[key]) { hintMap[key] = { t: -1, i: -1 }; hintOrder.push(key); }
    if (isImg(s)) hintMap[key].i = i; else hintMap[key].t = i;
    lastKey = key;
  });
  hintOrder.sort((a, b) => {
    const na = a[0] === 'n' ? Number(a.slice(1)) : 1000, nb = b[0] === 'n' ? Number(b.slice(1)) : 1000;
    return na - nb;
  });

  const exText = [], exImg = [];
  h.forEach((s, i) => { if (isEx(s)) (isImg(s) ? exImg : exText).push(i); });

  return {
    text: find(s => /^(問題|もんだい)/.test(s) && !isImg(s), 0),
    img: find(s => isImg(s) && !isHint(s) && !isEx(s), -1),
    choices: LETTERS.split('').map((L, k) => find(s => s === L, 1 + k)),
    answer: find(s => (/^(正解|せいかい)/.test(s) && !/別/.test(s)) || s === '答え' || s === 'こたえ', 5),
    hints: hintOrder.map(k => hintMap[k]),
    exText: exText,
    exImg: exImg,
    limit: find(s => /^(制限|せいげん|時間|じかん)/.test(s), -1),
    level: find(s => /^(レベル|れべる)/.test(s), -1),
    types: LETTERS.split('').map(L => find(s => isType(s) && s[0] === L, -1)),
    alt: find(s => /^(正解の別|別の書き方|別の答え|ほかの答え|こたえの別)/.test(s), -1),
    set: find(s => /^(セット|せっと)/.test(s), -1)
  };
}

function pairItem_(disp, vals, i, tCol, iCol) {
  const t = cellItem_(disp, vals, i, tCol);
  const g = cellItem_(disp, vals, i, iCol);
  return { text: t.text, img: g.img || imgUrl_(g.text) || t.img };
}

/** タブを読んで [{set, q}] にする */
function readSheet_(sh) {
  const range = sh.getDataRange();
  const disp = range.getDisplayValues();
  const vals = range.getValues();
  const c = findCols_(disp[0] || []);
  const cell = (r, i) => (i >= 0 && i < r.length) ? r[i] : '';
  const out = [];
  let curSet = '';
  for (let i = 1; i < disp.length; i++) {
    const r = disp[i];
    if (c.set >= 0) {
      const v = String(cell(r, c.set)).trim();
      if (v) curSet = v;
    }
    const qItem = pairItem_(disp, vals, i, c.text, c.img);
    const choices = c.choices.map(k => cell(r, k));
    if (!qItem.text && !qItem.img && !choices.join('').trim()) continue;
    const q = normQ_({
      text: qItem.text, choices: choices, answer: cell(r, c.answer),
      hints: c.hints.map(p => pairItem_(disp, vals, i, p.t, p.i)),
      explain: c.exText.map(k => cellItem_(disp, vals, i, k)).concat(c.exImg.map(k => {
        const g = cellItem_(disp, vals, i, k);
        return { text: '', img: g.img || imgUrl_(g.text) };
      })),
      img: qItem.img, limit: cell(r, c.limit), level: cell(r, c.level),
      types: c.types.map(k => cell(r, k)), alt: cell(r, c.alt)
    });
    out.push({ set: c.set >= 0 ? curSet : '', q: q });
  }
  return out;
}

function setLabel_(tab, grp) { return grp ? tab + '：' + grp : tab; }

function listSets_() {
  const out = [];
  qSheets_().forEach(sh => {
    const tab = sh.getName();
    const groups = [], cnt = {};
    readSheet_(sh).forEach(r => {
      if (!(r.set in cnt)) { cnt[r.set] = 0; groups.push(r.set); }
      cnt[r.set]++;
    });
    groups.forEach(g => out.push({ id: tab + SEP + g, name: setLabel_(tab, g), count: cnt[g] }));
  });
  return out;
}

function loadSet_(id) {
  if (id) {
    const k = id.indexOf(SEP);
    const tab = k >= 0 ? id.slice(0, k) : id;
    const grp = k >= 0 ? id.slice(k + 1) : '';
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tab);
    if (sh && isQSheet_(sh)) {
      const rows = readSheet_(sh).filter(r => r.set === grp).map(r => r.q);
      if (rows.length) return { id: tab + SEP + grp, name: setLabel_(tab, grp), tab: tab, rows: rows };
    }
  }
  const sets = listSets_();
  if (!sets.length || sets[0].id === id) return { id: '', name: '', tab: '', rows: [] };
  return loadSet_(sets[0].id);
}

/** 前の版で作ったタブ：ひらがなの見出しを漢字にし、ない見出しを右に足す */
function fixHead_(sh, head, legacy) {
  const lc = Math.max(sh.getLastColumn(), 1);
  const have = sh.getRange(1, 1, 1, lc).getValues()[0].map(str_);
  let changed = false;
  if (legacy) {
    have.forEach((h, i) => {
      if (legacy[h] && have.indexOf(legacy[h]) < 0) { have[i] = legacy[h]; changed = true; }
    });
    if (changed) sh.getRange(1, 1, 1, lc).setValues([have]);
  }
  const add = head.filter(h => have.indexOf(h) < 0);
  if (add.length) {
    const empty = have.every(h => !h);
    sh.getRange(1, empty ? 1 : lc + 1, 1, add.length).setValues([add]).setFontWeight('bold');
  }
}

  return { sh: sh, head: head, idx: idx };
}

/** みんなで：1問ぶんの 回答を 記録に 書く（正解発表の ときに まとめて 1回で 書く）。correct が null の ときは 正誤を 書かない */
function commitRound_(st, cur, correct) {
  if (!st.sid || st.committed === st.round || !cur) return;
  const list = getAnswers_(st.round);
  const who = getJoined_(st.sid).slice();
  list.forEach(a => { if (!who.some(j => j.w === a.e)) who.push({ w: a.e, n: a.n }); });
  const tl = cget_('tl_' + st.sid) || {};
  const ty = cget_('ty_' + st.sid) || {};
  const rows = [];
  const pos = Math.max(0, (st.qn || st.index || 1) - 1);
  who.forEach(j => {
    const a = list.find(x => x.e === j.w);
    let ok = null, why = '', given = '';
    if (a) {
      given = cur.input ? a.c : (cur.map[LETTERS.indexOf(a.c)] || a.c);
      if (correct !== null) {
        ok = cur.input ? judgeInput_(cur, a.c) : a.c === correct;
        if (!ok) why = whyOf_(cur, a.c, !!(a.x && a.p && !cur.input && a.p === correct));
      }
    } else if (correct !== null) {
      ok = false; why = whyOf_(cur, '', false);
    }
    rows.push({
      '日時': new Date(), 'セット': st.setName, '何問目': st.index || '', '番号': st.round, '端末': j.w, '名前': plain_(j.n),
      '答え': cellSafe_(given), '正誤': ok === null ? '' : (ok ? '○' : '×'), '問題文': plain_(cur.text).slice(0, 60),
      'かえた': a && a.x ? 'かえた' : '', '回ID': st.sid, 'まちがいの型': why, '進め方': MODE_LABEL.all,
      '答え方': styleLabel_(st.style), '秒数': a && a.t ? Math.round(a.t / 100) / 10 : ''
    });
    const t = tl[j.w] = tl[j.w] || { n: j.n, marks: [], types: {} };
    t.n = j.n;
    if (correct !== null) t.marks[pos] = a ? (ok ? 'o' : 'x') : '-';
    if (why) { t.types[why] = (t.types[why] || 0) + 1; ty[why] = (ty[why] || 0) + 1; }
  });
  withLock_(() => appendRows_(logSheet_(st.logTab), rows));
  cput_('tl_' + st.sid, tl);
  cput_('ty_' + st.sid, ty);
  st.committed = st.round;
}

/** 回の まとめを 1回だけ 書く */
function finishSession_(st) {
  if (!st.sid || st.sumDone) return st;
  const cur = prop_('cur', null);
  if (st.mode === 'all' && (st.phase === 'open' || st.phase === 'closed') && st.committed !== st.round) commitRound_(st, cur, null);
  writeSummary_(st);
  st.sumDone = true;
  return st;
}

function writeSummary_(st) {
  const joined = getJoined_(st.sid);
  const rows = [];
  const base = { '日時': new Date(), '回ID': st.sid, 'セット': st.setName || MANUAL_NAME, '進め方': MODE_LABEL[st.mode],
    '答え方': styleLabel_(st.style) };
  if (st.mode === 'solo') {
    const sts = soloStatuses_(st.sid, joined);
    joined.forEach((j, i) => {
      const s = sts[i];
      if (!s) return;
      const hist = s.hist || [];
      rows.push(Object.assign({}, base, {
        '名前': plain_(j.n), '問題数': s.total || '', '答えた数': s.n || 0, '正解数': s.ok || 0,
        '正答率': s.n ? Math.round(s.ok / s.n * 100) + '%' : '',
        '最初のレベル': s.lv0 || '', '最後のレベル': s.lv || '', 'レベルの変化': hist.join('→'),
        'まちがいの型': '', '多かったまちがい': topKeys_(s.types, 2).join('、')
      }));
    });
  } else {
    const tl = cget_('tl_' + st.sid) || {};
    const total = st.qn || st.total || 0;
    joined.forEach(j => {
      const t = tl[j.w] || { marks: [], types: {} };
      let a = 0, ok = 0;
      for (let i = 0; i < total; i++) { const m = t.marks[i]; if (m === 'o') { a++; ok++; } else if (m === 'x') a++; }
      rows.push(Object.assign({}, base, {
        '名前': plain_(j.n), '問題数': total, '答えた数': a, '正解数': ok, '正答率': a ? Math.round(ok / a * 100) + '%' : '',
        '多かったまちがい': topKeys_(t.types, 2).join('、')
      }));
    });
  }
  if (rows.length) withLock_(() => appendRows_(sumSheet_(), rows));
}

function getAnswers_(round) {
  if (!round || round <= 0) return [];
  return cget_('ans_' + round) || [];
}

function autoCode_(v) {
  const s = str_(v);
  if (!s) return '';
  if (/^(あり|オン|on|○|はい)/i.test(s)) return 'on';
  if (/^(なし|オフ|off|×|いいえ)/i.test(s)) return 'off';
  return '';
}

function readStudentSet_() {
  const t = readTable_(SH.stu);
  const map = {};
  t.rows.forEach(r => {
    const name = str_(cell_(t, r, '名前'));
    if (!name) return;
    map[name] = { lv: num_(cell_(t, r, '開始レベル')) || 0, lo: num_(cell_(t, r, '最低レベル')) || 0,
      hi: num_(cell_(t, r, '最高レベル')) || 0, auto: autoCode_(cell_(t, r, '自動上下')) };
  });
  return map;
}

/** 名簿にいて、生徒設定にいない生徒の行を足す */
function syncStudentSet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ensureSheet_(ss, SH.stu, STU_HEAD);
  const t = readTable_(SH.stu);
  const have = t.rows.map(r => str_(cell_(t, r, '名前')));
  const names = [];
  readRoster_().groups.forEach(g => g.names.forEach(n => { if (names.indexOf(n) < 0) names.push(n); }));
  const add = names.filter(n => have.indexOf(n) < 0);
  if (add.length) appendRows_(sh, add.map(n => ({ '名前': n })));
}

function getStudentTable(group) {
  checkTeacher_();
  syncStudentSet_();
  const map = readStudentSet_();
  const r = readRoster_();
  const g = r.groups.find(x => x.name === group) || r.groups[0];
  const names = g ? g.names : [];
  const set = prop_('settings', DEF_SET);
  const st = getState_();
  const rows = (st.sid && st.source === 'sheet') ? getSnap_(st) : currentSet_().rows;
  const lvs = rows.map(q => q.level || 1);
  return {
    rows: names.map(n => { const s = map[n] || {}; return { name: n, lv: s.lv || 0, lo: s.lo || 0, hi: s.hi || 0, auto: s.auto || '' }; }),
    minL: lvs.length ? Math.min.apply(null, lvs) : 1, maxL: lvs.length ? Math.max.apply(null, lvs) : 1, auto: set.auto !== false
  };
}

function saveStudentTable(rows) {
  checkTeacher_();
  return withLock_(() => {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    ensureSheet_(ss, SH.stu, STU_HEAD);
    const t = readTable_(SH.stu);
    const idx = {};
    t.rows.forEach((r, i) => { idx[str_(cell_(t, r, '名前'))] = i; });
    (rows || []).forEach(x => {
      let r;
      const name = str_(x.name);
      if (!name) return;
      if (idx[name] !== undefined) r = t.rows[idx[name]];
      else { r = rowFrom_(t, { '名前': name }); idx[name] = t.rows.length; t.rows.push(r); }
      const lv = Math.max(0, Math.min(9, parseInt(x.lv, 10) || 0));
      const lo = Math.max(0, Math.min(9, parseInt(x.lo, 10) || 0));
      const hi = Math.max(0, Math.min(9, parseInt(x.hi, 10) || 0));
      setCell_(t, r, '開始レベル', lv || '');
      setCell_(t, r, '最低レベル', lo || '');
      setCell_(t, r, '最高レベル', hi || '');
      setCell_(t, r, '自動上下', x.auto === 'on' ? 'あり' : x.auto === 'off' ? 'なし' : '');
    });
    if (t.rows.length) t.sh.getRange(2, 1, t.rows.length, t.head.length).setValues(t.rows);
    return true;
  });
}

function resultFor_(sid, who, total) {
  const tl = cget_('tl_' + sid) || {};
  const t = tl[who] || { marks: [] };
  const marks = [];
  let ok = 0, ans = 0;
  for (let i = 0; i < total; i++) {
    const m = t.marks[i] || '-';
    marks.push(m);
    if (m === 'o') { ok++; ans++; } else if (m === 'x') ans++;
  }
  return { total: total, ok: ok, answered: ans, marks: marks };
}

function stateReply_(cid, pub, hello) {
  const who = who_(cid);
  const m = getMembers_();
  const jl = getJoined_(pub.sid);
  const r = { state: pub, now: Date.now(), nick: myNick_(m, who, pub), taken: takenBy_(m, who),
    joined: jl.some(j => j.w === who), n: jl.length };
  if (hello && pub.mode === 'all' && LIVE.indexOf(pub.phase) >= 0) {
    const had = getAnswers_(pub.round).find(a => a.e === who);
    r.mine = had ? had.c : '';
    r.chg = !!(had && had.x);
  }
  if (pub.mode === 'all' && pub.phase === 'reveal' && pub.input) {
    const cur = prop_('cur', null);
    const had = getAnswers_(pub.round).find(x => x.e === who);
    r.judge = !!(had && cur && judgeInput_(cur, had.c));
  }
  if (pub.phase === 'end' && pub.mode === 'all' && pub.sid) r.res = resultFor_(pub.sid, who, pub.total);
  return r;
}

function studentHello(cid, code) {

    if (j !== 'ok') return { needCode: true, wait: j === 'wait', bad: j === 'bad' && /^\d{4}$/.test(String(code || '')) };
  }
  return stateReply_(cid, pub, true);
}

function getState(cid) {

}

/** 名前を えらぶ（回が 始まって いれば、そのまま 回に 入る） */
function pickNick(cid, nick) {
  return withLock_(() => {
    const pub = getPublic_();
    if (!guard_(cid, pub)) return { ok: false, needCode: true };
    const who = who_(cid);
    const m = getMembers_();
    if (choices_(pub).indexOf(nick) < 0) return { ok: false, again: true, nick: myNick_(m, who, pub), taken: takenBy_(m, who) };
    // クイズの とちゅうは かえられない（まだ えらんでいない人は えらべる）
    const live = pub.round > 0 && (LIVE.indexOf(pub.phase) >= 0 || pub.phase === 'play');
    const now = myNick_(m, who, pub);
    if (live && now) return { ok: false, locked: true, nick: now, taken: takenBy_(m, who) };
    if (takenBy_(m, who).indexOf(nick) >= 0) return { ok: false, nick: now, taken: takenBy_(m, who) };
    m[who] = nick;
    setMembers_(m);
    let n = 0, joined = false;
    if (pub.sid && ACTIVE.indexOf(pub.phase) >= 0) { n = joinWho_(pub.sid, who, nick); joined = true; }
    return { ok: true, nick: nick, taken: takenBy_(m, who), joined: joined, n: n };
  });
}

/** 回に 入る（受付前に 開いていた 生徒や、前の 回の 結果画面の 生徒が 自動で よぶ） */
function joinSession(cid) {
  return withLock_(() => {
    const pub = getPublic_();
    if (!guard_(cid, pub)) return { ok: false, needCode: true };
    if (!pub.sid || ACTIVE.indexOf(pub.phase) < 0) return { ok: false, notActive: true };
    const who = who_(cid);
    const nick = myNick_(getMembers_(), who, pub);
    if (!nick) return { ok: false, needNick: true };
    return { ok: true, joined: true, n: joinWho_(pub.sid, who, nick) };
  });
}

/** みんなで：答えを 送る（えらび＝A〜D、入力＝ことば）。ロックの 中では キャッシュだけ さわる */
function submitAnswer(cid, round, choice) {
  return withLock_(() => {
    const pub = getPublic_();
    if (!guard_(cid, pub)) return { ok: false, needCode: true };
    const now = Date.now();
    let valid = pub.phase === 'open' && pub.round === round && pub.mode === 'all';
    if (valid) {
      if (pub.input) {
        choice = String(choice === null || choice === undefined ? '' : choice).trim().slice(0, 40);
        valid = choice.length > 0;
      } else {
        valid = LETTERS.indexOf(choice) >= 0 && LETTERS.indexOf(choice) < pub.choices.length;
      }
    }
    const late = pub.limit > 0 && now > pub.openedAt + pub.limit * 1000 + 2000;
    if (!valid || late) return { ok: false, state: pub, now: now };
    const who = who_(cid);
    const nick = myNick_(getMembers_(), who, pub);
    if (!nick) return { ok: false, needNick: true, state: pub, now: now };
    joinWho_(pub.sid, who, nick);
    const list = getAnswers_(round);
    const had = list.find(a => a.e === who);
    if (had) {
      if (had.c === choice || had.x || pub.input) return { ok: true, choice: had.c, chg: !!had.x, state: pub, now: now };
      // こたえを かえられるのは 1かいだけ
      had.c = choice; had.x = 1;
      cput_('ans_' + round, list);
      return { ok: true, choice: choice, chg: true, state: pub, now: now };
    }
    list.push({ e: who, n: nick, c: choice, p: choice, x: 0, t: now - pub.openedAt });
    cput_('ans_' + round, list);
    return { ok: true, choice: choice, chg: false, state: pub, now: now };
  });
}

/** ひとりずつ：生徒が 自分で すすめる ための 問題と 設定を わたす（答えも 入っているので、通信が 切れても すすめられる） */
function getSoloPack(cid) {
  const pub = getPublic_();
  if (!guard_(cid, pub)) return { ok: false, needCode: true };
  const st = getState_();
  if (st.mode !== 'solo' || st.phase !== 'play') return { ok: false, notPlay: true };
  const who = who_(cid);
  const nick = myNick_(getMembers_(), who, pub);
  if (!nick) return { ok: false, needNick: true };
  const qs = getPresented_(st);
  const lvs = qs.map(q => q.level);
  const minL = Math.min.apply(null, lvs), maxL = Math.max.apply(null, lvs);
  const set = prop_('settings', DEF_SET);
  const cfg = readStudentSet_()[nick] || {};
  const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
  let lo = clamp(cfg.lo || minL, minL, maxL), hi = clamp(cfg.hi || maxL, minL, maxL);
  if (lo > hi) hi = lo;
  const lv0 = clamp(cfg.lv || lo, lo, hi);
  const auto = cfg.auto === 'off' ? false : (cfg.auto === 'on' ? true : set.auto !== false);
  return { ok: true, pack: {
    sid: st.sid, style: st.style, auto: auto, upN: set.upN, downN: set.downN, lv0: lv0, lo: lo, hi: hi,
    minL: minL, maxL: maxL, levels: uniq_(lvs).sort((a, b) => a - b), qs: qs, defLimit: effLimit_(set)
  } };
}

/** ひとりずつ：答えの 記録を まとめて 送る（seq で 二重に 書かない）と、いまの ようす（先生画面用） */
function studentLog(cid, sid, items, status) {
  const pub = getPublic_();
  if (!guard_(cid, pub)) return { ok: false, needCode: true };
  const st = getState_();
  if (!st.sid || st.sid !== sid) return { ok: false, gone: true };
  const who = who_(cid);
  const nick = myNick_(getMembers_(), who, pub);
  if (!nick) return { ok: false, needNick: true };
  const key = ckey_('lg_' + sid + '_', who);
  const cache = CacheService.getScriptCache();
  const last = Number(cache.get(key) || 0);
  const fresh = (items || []).filter(it => Number(it.seq) > last).slice(0, 50);
  if (fresh.length) {
    withLock_(() => {
      const rows = fresh.map(it => ({
        '日時': new Date(Number(it.ts) || Date.now()), 'セット': st.setName, '何問目': Number(it.no) || '', '番号': Number(it.qid) || '',
        '端末': who, '名前': plain_(nick), '答え': cellSafe_(it.given), '正誤': it.ok ? '○' : '×',
        '問題文': cellSafe_(plain_(it.text).slice(0, 60)), '回ID': sid, 'まちがいの型': it.ok ? '' : cellSafe_(it.why),
        'レベル': Number(it.level) || '', '問題のレベル': Number(it.qlevel) || '', '進め方': MODE_LABEL.solo,
        '答え方': styleLabel_(st.style), '秒数': Number(it.sec) || '', 'ヒント': it.hint ? '使った' : ''
      }));
      appendRows_(logSheet_(st.logTab), rows);
      const ty = cget_('ty_' + sid) || {};
      fresh.forEach(it => { if (!it.ok && it.why) { const w = plain_(String(it.why)).slice(0, 40); ty[w] = (ty[w] || 0) + 1; } });
      cput_('ty_' + sid, ty);
      cache.put(key, String(Math.max.apply(null, fresh.map(it => Number(it.seq)))), TTL);
    });
  }
  if (status && typeof status === 'object') {
    cput_(ckey_('st_' + sid + '_', who), {
      st: status.st === 'done' ? 'done' : 'play', n: Number(status.n) || 0, ok: Number(status.ok) || 0,
      lv: Number(status.lv) || 0, lv0: Number(status.lv0) || 0, total: Number(status.total) || 0,
      hist: (status.hist || []).slice(-80).map(Number), types: status.types || {}, t: Date.now()
    });
  }
  return { ok: true, play: st.phase === 'play', end: st.phase === 'end' };
}

function soloStatuses_(sid, joined) {
  const keys = joined.map(j => ckey_('st_' + sid + '_', j.w));
  const got = keys.length ? CacheService.getScriptCache().getAll(keys) : {};
  return joined.map((j, i) => { try { return got[keys[i]] ? JSON.parse(got[keys[i]]) : null; } catch (e) { return null; } });
}

function qrUrl_(set) {
  let u = set.qrUrl || '';
  if (!u) { try { u = ScriptApp.getService().getUrl() || ''; } catch (e) { u = ''; } }
  if (u && set.codeOn === true) u += (u.indexOf('?') >= 0 ? '&' : '?') + 'code=' + joinCode_();
  return u;
}

function publicSettings_(set) {
  return { mode: set.mode, style: set.style, source: set.source, showQ: set.showQ, timeView: set.timeView, kana: set.kana,
    sound: set.sound !== false, limit: effLimit_(set), auto: set.auto !== false, upN: set.upN, downN: set.downN,

}

function teacherStatus() {
  checkTeacher_();
  const st = getState_();
  const pub = getPublic_();
  const set = prop_('settings', DEF_SET);
  const cur = prop_('cur', null);
  const on = !!st.sid && st.phase !== 'wait';
  const joined = on ? getJoined_(st.sid) : [];
  const out = {
    state: pub, now: Date.now(), settings: publicSettings_(set), url: qrUrl_(set),
    codeOn: pub.codeOn, code: pub.codeOn ? joinCode_() : '',
    joined: joined.map(j => plain_(j.n)), list: [], hasKey: false, qTypes: {}, levels: [], board: [], rows: [], types: []
  };
  if (on) { out.qCount = st.total; out.setName = st.setName; }
  else if (set.source === 'sheet' || set.mode === 'solo') { const s = currentSet_(); out.qCount = s.rows.length; out.setName = s.name; }
  else { out.qCount = 0; out.setName = MANUAL_NAME; }
  if (on && st.source === 'manual') out.setName = MANUAL_NAME;
  if (on && st.mode === 'all' && LIVE.indexOf(st.phase) >= 0) {
    out.list = getAnswers_(st.round).map(a => ({ name: plain_(a.n), choice: a.c }));
    out.hasKey = !!(cur && cur.answer);
    if (cur) {
      Object.keys(cur.types || {}).forEach(L => { out.qTypes[L] = plain_(cur.types[L]); });
      if (cur.input) out.correctText = cur.correctText;
    }
  }
  const solo = on ? st.mode === 'solo' : set.mode === 'solo';
  if (solo) {
    const rows = on ? getSnap_(st) : currentSet_().rows;
    out.levels = levelsOf_(rows);
  }
  if (on && st.mode === 'solo' && (st.phase === 'play' || st.phase === 'end')) {
    const sts = soloStatuses_(st.sid, joined);
    out.board = joined.map((j, i) => {
      const s = sts[i] || {};
      return { name: plain_(j.n), st: s.st || 'ready', n: s.n || 0, ok: s.ok || 0, lv: s.lv || 0, lv0: s.lv0 || 0, hist: s.hist || [], total: s.total || 0 };
    });
  }
  if (on && st.mode === 'all' && st.phase === 'end') {
    const tl = cget_('tl_' + st.sid) || {};
    out.rows = joined.map(j => { const r = resultFor_(st.sid, j.w, st.total); return { name: plain_(j.n), ok: r.ok, answered: r.answered, total: r.total }; });
  }
  if (on) out.types = topKeys_(cget_('ty_' + st.sid) || {}, 6).map(k => ({ type: k, n: (cget_('ty_' + st.sid) || {})[k] }));
  return out;
}

/** 「この設定で受付を始める」：新しい回（回ID＝月日-時分秒）を始める */
function teacherStart() {
  checkTeacher_();
  return withLock_(() => {
    const old = finishSession_(getState_());
    if (old.sid) setProp_('state', old);
    const set = prop_('settings', DEF_SET);
    if (set.nameMode !== 'chara') refreshRoster_(set);
    dropCurSet_();
    const mode = set.mode === 'solo' ? 'solo' : 'all';
    const source = mode === 'solo' ? 'sheet' : set.source;
    let s = { id: '', name: MANUAL_NAME, tab: '', rows: [] };
    if (source === 'sheet') {
      s = currentSet_();
      if (!s.rows.length) throw new Error('問題が見つかりません。「問題」タブに問題を書いてください');
    }
    let sid = Utilities.formatDate(new Date(), TZ, 'MMdd-HHmmss');
    if (sid === old.sid) sid += 'b';
    const st = Object.assign({}, DEF_STATE, {
      round: old.round, phase: 'lobby', sid: sid, mode: mode, style: set.style, source: source, setId: s.id, setName: s.name,
      logTab: s.tab || '', total: s.rows.length, qn: 0, committed: -1, sumDone: false, startedAt: Date.now()
    });
    if (source === 'sheet') bigPut_('snap_' + sid, s.rows);
    cput_('jn_' + sid, []);
    return save_(st, null);
  });
}

/** 「この回をはじめる」（みんなで＝1問めを出す／ひとりずつ＝生徒が自分で進みはじめる） */
function teacherBegin() {
  checkTeacher_();
  const st = getState_();
  if (st.phase !== 'lobby') return getPublic_();
  if (st.mode === 'solo') {
    st.phase = 'play';
    st.openedAt = Date.now();
    warmImgs_(getSnap_(st));
    return save_(st);
  }
  if (st.source === 'sheet') return teacherNext();
  return getPublic_();
}

/** セットの 次の問題を出す */
function teacherNext() {
  checkTeacher_();
  const st = getState_();
  if (st.mode !== 'all' || st.source !== 'sheet') return getPublic_();
  const rows = getSnap_(st);
  const idx = st.index + 1;
  if (idx > rows.length) {
    st.phase = 'end';
    st.total = rows.length;
    finishSession_(st);
    return save_(st);
  }
  const q = present_(withLimit_(rows[idx - 1]), st.style, st.sid + '#' + idx);
  warmImgs_(q);
  return save_(Object.assign(st, { round: st.round + 1, phase: 'open', index: idx, total: rows.length, hintShown: 0,
    openedAt: Date.now(), correct: '', qn: idx }), q);
}

/** 先生画面で入れた1問を出す */
function teacherPush(o) {
  checkTeacher_();
  const st = getState_();
  if (st.mode !== 'all' || st.source !== 'manual' || !st.sid) throw new Error('受付を始めてから、問題を入力してください');
  const q = present_(withLimit_(normQ_(o || {})), st.style, st.sid + '#m' + (st.round + 1));
  warmImgs_(q);
  return save_(Object.assign(st, { round: st.round + 1, phase: 'open', index: 0, total: 0, hintShown: 0,
    openedAt: Date.now(), correct: '', qn: (st.qn || 0) + 1 }), q);
}

function teacherReveal(choice) {
  checkTeacher_();
  const st = getState_();
  const q = prop_('cur', null);
  if (!q) throw new Error('問題が出ていません');
  let c = '';
  if (!q.input) {
    c = normLetter_(choice) || q.answer || '';
    if (!c || LETTERS.indexOf(c) >= q.choices.length) throw new Error('正解を選んでください');
  }
  st.phase = 'reveal';
  st.correct = c;
  const pub = save_(st);      // 先に 状態を かえて、これ以上 答えが 入らないように する
  commitRound_(st, q, c);     // 記録に 正誤つきで まとめて 書く
  setProp_('state', st);
  return pub;
}

function teacherSettings(o) {
  checkTeacher_();
  o = o || {};
  const set = prop_('settings', DEF_SET);
  const pick = (v, list) => list.indexOf(v) >= 0;
  if (pick(o.mode, ['all', 'solo'])) set.mode = o.mode;
  if (pick(o.style, ['four', 'two', 'input'])) set.style = o.style;
  if (pick(o.source, ['sheet', 'manual'])) set.source = o.source;
  if (typeof o.showQ === 'boolean') set.showQ = o.showQ;
  if (typeof o.setId === 'string') set.setId = o.setId;
  if (o.limit !== undefined) { set.limit = normLimit_(o.limit) || 0; set.limitSet = true; }
  if (pick(o.timeView, ['bar', 'both', 'text'])) set.timeView = o.timeView;
  if (pick(o.kana, ['ruby', 'hira'])) set.kana = o.kana;
  if (typeof o.sound === 'boolean') set.sound = o.sound;
  if (typeof o.codeOn === 'boolean') set.codeOn = o.codeOn;
  if (typeof o.auto === 'boolean') set.auto = o.auto;
  if (o.upN !== undefined) set.upN = Math.max(1, Math.min(20, parseInt(o.upN, 10) || DEF_SET.upN));
  if (o.downN !== undefined) set.downN = Math.max(1, Math.min(20, parseInt(o.downN, 10) || DEF_SET.downN));
  if (typeof o.qrUrl === 'string') set.qrUrl = o.qrUrl.trim().slice(0, 300);
  let names = false;
  if (pick(o.nameMode, ['roster', 'chara']) && o.nameMode !== set.nameMode) { set.nameMode = o.nameMode; names = true; }
  if (typeof o.rosterGroup === 'string' && o.rosterGroup !== set.rosterGroup) { set.rosterGroup = o.rosterGroup; names = true; }
  setProp_('settings', set);
  dropCurSet_();
  if (names) {
    // えらびかたが かわったら みんなに えらびなおしてもらう
    refreshRoster_(set);
    withLock_(() => {
      setMembers_({});
      const st = getState_();
      if (st.sid) cput_('jn_' + st.sid, []);
    });
  }
  return save_(getState_());
}

/* ---------- 画像 ---------- */

const IMG_FOLDER = 'みんなでクイズの画像';
const IMG_OLD_FOLDER = 'みんなでクイズの がぞう';
const IMG_MAX = 6 * 1024 * 1024;
let IMGMAP_ = null;

/** 画像のフォルダを作る（setup の中で呼ぶ。これだけ実行してもよい。ドライブを使う許可もここでとる） */
function setupImages() {
  const f = imgFolder_();
  Logger.log('画像のフォルダ：' + f.getName() + '  ' + f.getUrl());
  return f.getUrl();
}

function imgFolder_() {
  const p = PropertiesService.getScriptProperties();
  const id = p.getProperty('imgFolder');
  if (id) {
    try {
      const f = DriveApp.getFolderById(id);
      if (!f.isTrashed()) return f;
    } catch (e) { /* なくなっていたら つくりなおす */ }
  }
  let parent;
  try {
    const it = DriveApp.getFileById(SpreadsheetApp.getActiveSpreadsheet().getId()).getParents();
    parent = it.hasNext() ? it.next() : DriveApp.getRootFolder();
  } catch (e) {
    parent = DriveApp.getRootFolder();
  }
  const old = parent.getFoldersByName(IMG_OLD_FOLDER);   // 前の版で作ったフォルダがあれば、それを使い続ける
  const same = parent.getFoldersByName(IMG_FOLDER);
  const f = old.hasNext() ? old.next() : (same.hasNext() ? same.next() : parent.createFolder(IMG_FOLDER));
  p.setProperty('imgFolder', f.getId());
  return f;
}

function imgMap_(fresh) {
  if (IMGMAP_ && !fresh) return IMGMAP_;
  if (!fresh) {
    const c = cget_('imgmap');
    if (c) return (IMGMAP_ = c);
  }
  const map = {}, base = {};
  const it = imgFolder_().getFiles();
  while (it.hasNext()) {
    const f = it.next();
    if (!/^image\//.test(f.getMimeType())) continue;
    const n = f.getName();
    map[n] = f.getId();
    const b = n.replace(/\.[^.]+$/, '');
    base[b] = (b in base) ? '' : f.getId();
  }
  Object.keys(base).forEach(b => { if (base[b] && !(b in map)) map[b] = base[b]; });
  IMGMAP_ = map;
  try { cput_('imgmap', map, 120); } catch (e) { /* 大きすぎるときは きおくしない */ }
  return map;
}

function imgMapSafe_() {
  try { return imgMap_(); } catch (e) { return {}; }
}

function uploadImage(dataUrl, baseName) {
  checkTeacher_();
  const m = String(dataUrl || '').match(/^data:(image\/[\w.+-]+);base64,(.+)$/);
  if (!m) throw new Error('画像ではありません');
  const mime = m[1];
  const ext = { 'image/jpeg': 'jpg', 'image/png': 'png', 'image/gif': 'gif', 'image/webp': 'webp', 'image/svg+xml': 'svg' }[mime] || 'png';
  const base = String(baseName || '').replace(/\.[^.]*$/, '')
    .replace(/[\\\/:*?"<>|#%]/g, '').replace(/\s+/g, '_').trim().slice(0, 40) || '画像';
  return withLock_(() => {
    const map = imgMap_(true);
    let name = base + '.' + ext, n = 1;
    while (map[name] || map[name.replace(/\.[^.]+$/, '')]) name = base + '_' + (++n) + '.' + ext;
    const file = imgFolder_().createFile(Utilities.newBlob(Utilities.base64Decode(m[2]), mime, name));
    CacheService.getScriptCache().remove('imgmap');
    IMGMAP_ = null;
    return { name: name, id: file.getId() };
  });
}

function listImages() {
  checkTeacher_();
  const folder = imgFolder_();
  const out = [];
  const it = folder.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    if (!/^image\//.test(f.getMimeType())) continue;
    out.push({ id: f.getId(), name: f.getName(), t: f.getLastUpdated().getTime(), size: f.getSize() });
  }
  out.sort((a, b) => b.t - a.t);
  return { folderUrl: folder.getUrl(), items: out.slice(0, 80), total: out.length };
}

/**
 * 画像を わたす（data URL）
 * 生徒は いま 画面に 出ている 画像だけ もらえる（ひとりずつ では この回の 問題の 画像）
 */
function getImage(id, cid, who) {
  id = String(id || '');
  if (!/^[\w-]{10,}$/.test(id)) throw new Error('画像がありません');
  if (who !== 'teacher') {
    const pub = getPublic_();
    if (!guard_(cid, pub)) throw new Error('コードを入れてください');
    let ok = JSON.stringify(pub).indexOf('img:' + id) >= 0;
    if (!ok && pub.mode === 'solo' && (pub.phase === 'play' || pub.phase === 'end')) {
      ok = JSON.stringify(getSnap_(getState_())).indexOf('img:' + id) >= 0;
    }
    if (!ok) throw new Error('いまは見られません');
  }
  return imgData_(id);
}

function imgData_(id) {
  const cache = CacheService.getScriptCache();
  const head = cache.get('im_' + id);
  if (head) {
    const keys = [];
    for (let i = 0; i < Number(head); i++) keys.push('im_' + id + '_' + i);
    const got = cache.getAll(keys);
    if (keys.every(k => got[k])) return keys.map(k => got[k]).join('');
  }
  const f = DriveApp.getFileById(id);
  const mime = f.getMimeType();
  if (!/^image\//.test(mime)) throw new Error('画像ではありません');
  if (f.getSize() > IMG_MAX) throw new Error('画像が大きすぎます');
  const data = 'data:' + mime + ';base64,' + Utilities.base64Encode(f.getBlob().getBytes());
  const CH = 90000, parts = {};
  let n = 0;
  for (let i = 0; i < data.length; i += CH) parts['im_' + id + '_' + (n++)] = data.slice(i, i + CH);
  try {
    cache.putAll(parts, TTL);
    cache.put('im_' + id, String(n), TTL);
  } catch (e) { /* きおくできなくても わたせる */ }
  return data;
}

/** 問題を 出す ときに 画像を 先に 読んでおく（生徒が いっせいに もらっても はやい） */
function warmImgs_(q) {
  const ids = (JSON.stringify(q || {}).match(/img:[\w-]+/g) || []).map(s => s.slice(4));
  uniq_(ids).forEach(id => {
    try { imgData_(id); } catch (e) { /* 読めない 画像は とばす */ }
  });
}
