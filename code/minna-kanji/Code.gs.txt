// レベル：1＝小1・やさしい、2＝小1・むずかしい、3＝小2・やさしい … 13＝中学・やさしい、14＝中学・むずかしい
const MAX_LV = 14;
const GRADES = ['小1', '小2', '小3', '小4', '小5', '小6', '中学'];
function lvName_(lv) { lv = parseInt(lv, 10); return (lv >= 1 && lv <= MAX_LV) ? GRADES[Math.ceil(lv / 2) - 1] + '・' + (lv % 2 ? 'やさしい' : 'むずかしい') : ''; }

/* ========== はじめに一度だけ実行（何回実行してもよい） ========== */
function setup() {
  const ss = ss_();
  if (!ss.getSheetByName(SHEET_ROSTER)) {
    const sh = ss.insertSheet(SHEET_ROSTER);
    sh.getRange(1, 1, 4, 1).setValues([['1組'], ['れい1'], ['れい2'], ['れい3']]);
    sh.getRange(1, 1).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  const q = sheet_(SHEET_Q, HEAD_Q);
  if (q.getLastRow() < 2) writeWords_(q);   // 「問題」タブが空のときだけ、ことばを入れる
  sheet_(SHEET_CONF, HEAD_CONF);
  sheet_(SHEET_LOG, HEAD_LOG);
  sheet_(SHEET_SUM, HEAD_SUM);
  cache_().remove('W_ok');
}

// 「問題」タブに、もとのことば（約6,800語）を書く
function writeWords_(sh) {
  const heads = heads_(sh), rows = [];
  WORD_DATA.forEach((line, i) => {
    const lv = i + 1, grade = GRADES[Math.ceil(lv / 2) - 1], diff = lv % 2 ? 'やさしい' : 'むずかしい';
    line.split(',').forEach(item => {
      if (!item) return;
      const p = item.split(':'), rs = (p[1] || '').split('/');
      const o = { '学年': grade, '難易度': diff, 'ことば': p[0], '読み': rs[0], 'ほかに正解とする読み': rs.slice(1).join('・'), '出さない': '' };
      rows.push(heads.map(h => (Object.prototype.hasOwnProperty.call(o, h) ? o[h] : '')));
    });
  });
  if (rows.length) sh.getRange(2, 1, rows.length, heads.length).setValues(rows);
}

/* ========== スプレッドシート ========== */
function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }

// タブがなければ作り、足りない見出しは右に足す（前の版のシートでも動くように）
function sheet_(name, heads) {
  const ss = ss_();
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  const lastCol = sh.getLastColumn();
  const cur = lastCol ? sh.getRange(1, 1, 1, lastCol).getValues()[0].map(v => String(v).trim()) : [];
  const missing = heads.filter(h => cur.indexOf(h) < 0);
  if (missing.length) {
    const start = cur.some(v => v !== '') ? lastCol + 1 : 1;
    sh.getRange(1, start, 1, missing.length).setValues([missing]).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  return sh;
}
function heads_(sh) { return sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(v => String(v).trim()); }
function appendRows_(sh, objs) {
  if (!objs.length) return;
  const heads = heads_(sh);
  const rows = objs.map(o => heads.map(h => (Object.prototype.hasOwnProperty.call(o, h) ? o[h] : '')));
  sh.getRange(sh.getLastRow() + 1, 1, rows.length, heads.length).setValues(rows);
}

// 名簿：1行目が名簿の名前、その下に生徒の名前（列ごとに別の名簿）
function rosters_() {
  const sh = ss_().getSheetByName(SHEET_ROSTER);
  if (!sh || sh.getLastRow() < 1 || sh.getLastColumn() < 1) return [];
  const v = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getDisplayValues();
  const out = [];
  for (let c = 0; c < v[0].length; c++) {
    const title = String(v[0][c]).trim();
    if (!title) continue;
    const names = [];
    for (let r = 1; r < v.length; r++) { const n = String(v[r][c]).trim(); if (n && names.indexOf(n) < 0) names.push(n); }
    out.push({ title: title, names: names });
  }
  return out;
}
function rosterNames_(title) {
  const list = rosters_();
  const hit = list.filter(r => r.title === title)[0] || list[0];
  return hit ? hit.names : [];
}

/* ---------- 問題（ことば） ---------- */
function hira_(s) {
  return String(s == null ? '' : s).replace(/[\s　]/g, '').replace(/[ァ-ヶ]/g, c => String.fromCharCode(c.charCodeAt(0) - 0x60));
}
// 「問題」タブを読んで、レベルごとに分ける（「出さない」に何か入っている行は出さない）。1語は「ことば\t読み\tほかの読み」
function readWords_() {
  const sh = sheet_(SHEET_Q, HEAD_Q), last = sh.getLastRow(), heads = heads_(sh);
  const out = [];
  for (let i = 0; i <= MAX_LV; i++) out.push([]);
  if (last < 2) return out;
  const col = h => heads.indexOf(h);
  const gi = col('学年'), di = col('難易度'), wi = col('ことば'), ri = col('読み'), ai = col('ほかに正解とする読み'), ni = col('出さない');
  const rows = sh.getRange(2, 1, last - 1, heads.length).getDisplayValues();
  rows.forEach(r => {
    if (String(r[ni]).trim() !== '') return;
    const w = String(r[wi]).trim(), rd = hira_(r[ri]);
    if (!w || !rd) return;
    let g = GRADES.indexOf(String(r[gi]).trim());
    if (g < 0) { const n = parseInt(String(r[gi]).replace(/[^0-9]/g, ''), 10); g = /中/.test(r[gi]) ? 6 : (n >= 1 && n <= 6 ? n - 1 : -1); }
    if (g < 0) return;
    const hard = /むずか|難/.test(String(r[di]));
    const alts = String(r[ai] || '').split(/[・、,\/\s]+/).map(hira_).filter(x => x && x !== rd);
    out[g * 2 + (hard ? 2 : 1)].push(w + '\t' + rd + '\t' + alts.join('/'));
  });
  return out;
}
function cacheWords_(all) {
  const c = cache_();
  for (let lv = 1; lv <= MAX_LV; lv++) { try { c.put('W_' + lv, all[lv].join('\n'), CACHE_SEC); } catch (err) { /* 大きすぎるときは、毎回シートから読む */ } }
  c.put('W_ok', '1', CACHE_SEC);
}
function parseWord_(line, lv) { const p = line.split('\t'); return [p[0], p[1], p[2] ? p[2].split('/') : [], lv]; }
// あるレベルのことば（そのレベルに1つもないときは、近いレベルから）
function levelWords_(lv, all) {
  lv = clampN_(lv, 1, MAX_LV, 1);
  const c = cache_();
  const get = k => {
    if (all) return all[k];
    const s = c.get('W_' + k);
    return s == null ? null : (s ? s.split('\n') : []);
  };
  if (!all && !c.get('W_ok')) { all = readWords_(); cacheWords_(all); }
  for (let d = 0; d < MAX_LV; d++) {
    const ks = d ? [lv - d, lv + d] : [lv];
    for (let i = 0; i < ks.length; i++) {
      const k = ks[i];
      if (k < 1 || k > MAX_LV) continue;
      let list = get(k);
      if (list == null) { all = readWords_(); cacheWords_(all); list = all[k]; }
      if (list.length) return list.map(x => parseWord_(x, k));
    }
  }
  return [];
}
function sample_(list, n) {
  const a = list.slice();
  for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); const t = a[i]; a[i] = a[j]; a[j] = t; }
  return a.slice(0, n);
}

/* ---------- 生徒設定（生徒ごとに1行） ---------- */
function num_(v) {
  const s = String(v == null ? '' : v).trim().replace(/[０-９]/g, d => String.fromCharCode(d.charCodeAt(0) - 0xFEE0));
  const n = parseInt(s, 10);
  return isNaN(n) ? '' : n;
}
function yes_(v, def) {
  const s = String(v == null ? '' : v).trim();
  if (!s) return def;
  return !/^(×|x|ｘ|no|しない|なし|false|0|いいえ)$/i.test(s);
}
function howOf_(v) { const s = String(v == null ? '' : v); return /えら|tile/.test(s) ? 'tile' : /書|入力|type/.test(s) ? 'type' : ''; }
function howName_(h) { return h === 'tile' ? '文字をえらぶ' : h === 'type' ? '字を書く' : ''; }
function lvClamp_(v) { const n = num_(v); return n === '' ? '' : Math.max(1, Math.min(MAX_LV, n)); }
function readConf_() {
  const sh = sheet_(SHEET_CONF, HEAD_CONF);
  const last = sh.getLastRow(), heads = heads_(sh);
  const col = h => heads.indexOf(h);
  const rows = last > 1 ? sh.getRange(2, 1, last - 1, heads.length).getValues() : [];
  return { sh: sh, heads: heads, col: col, rows: rows };
}
function confRow_(d, r) {
  const c = d.col;
  return { level: lvClamp_(r[c('レベル')]), lower: lvClamp_(r[c('下限')]), upper: lvClamp_(r[c('上限')]), auto: yes_(r[c('自動調整')], true), how: howOf_(r[c('答え方')]), last: lvClamp_(r[c('前回のレベル')]) };
}
function blankConf_() { return { level: '', lower: '', upper: '', auto: true, how: '', last: '' }; }
function confFor_(name) {
  const d = readConf_();
  const row = d.rows.filter(r => String(r[d.col('名前')]).trim() === name)[0];
  return row ? confRow_(d, row) : blankConf_();
}
function patchConf_(d, name, patch) {
  const ni = d.col('名前');
  let row = d.rows.filter(r => String(r[ni]).trim() === name)[0];
  if (!row) { row = d.heads.map(() => ''); row[ni] = name; d.rows.push(row); }
  Object.keys(patch).forEach(h => { const i = d.col(h); if (i >= 0) row[i] = patch[h]; });
}
function writeConf_(d) { if (d.rows.length) d.sh.getRange(2, 1, d.rows.length, d.heads.length).setValues(d.rows); }

/* ========== 保存場所（プロパティ・キャッシュ） ========== */
function props_() { return PropertiesService.getScriptProperties(); }
function getJSON_(k, d) { try { const v = props_().getProperty(k); return v ? JSON.parse(v) : d; } catch (err) { return d; } }
function setJSON_(k, v) { props_().setProperty(k, JSON.stringify(v)); }
// 長い文字列は分けて保存する（プロパティ1つは9KBまで）
function putBig_(prefix, str) {
  const size = 2500, n = Math.ceil(str.length / size), obj = {};
  for (let i = 0; i < n; i++) obj[prefix + '_' + i] = str.substr(i * size, size);
  obj[prefix + '_n'] = String(n);
  props_().setProperties(obj);
}
function getBig_(prefix) {
  const p = props_(), n = parseInt(p.getProperty(prefix + '_n') || '0', 10);
  let s = '';
  for (let i = 0; i < n; i++) s += p.getProperty(prefix + '_' + i) || '';
  return s;
}
// キャッシュのキー：日本語の名前は encodeURIComponent で（ちがう名前が同じキーにならないように）
function stKey_(rid, name) { return ('st_' + rid + '_' + encodeURIComponent(name)).slice(0, 240); }
function seenKey_(rid, name) { return ('sn_' + rid + '_' + encodeURIComponent(name)).slice(0, 240); }
function cache_() { return CacheService.getScriptCache(); }
function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try { return fn(); } finally { lock.releaseLock(); }
}

/* ========== 回 ========== */
function clampN_(v, lo, hi, d) { const n = parseInt(v, 10); return isNaN(n) ? d : Math.max(lo, Math.min(hi, n)); }
function sanitize_(s) {
  const d = Object.assign({}, DEFAULT_SETTINGS, s || {});
  d.mode = d.mode === 'self' ? 'self' : 'all';
  d.level = clampN_(d.level, 1, MAX_LV, 1);
  d.levelMode = d.levelMode === 'each' ? 'each' : 'all';
  d.count = clampN_(d.count, 1, 30, 10);
  d.minutes = clampN_(d.minutes, 1, 60, 10);
  d.how = d.how === 'type' ? 'type' : 'tile';
  d.autoLv = d.autoLv !== false;
  d.up = clampN_(d.up, 1, 100, 30); d.down = clampN_(d.down, 1, 10, 2);
  d.flagN = clampN_(d.flagN, 2, 10, 3);
  d.sound = d.sound !== false;
  d.kana = d.kana === 'hira' ? 'hira' : 'ruby';
  d.guide = d.guide === 'every' ? 'every' : 'first';
  d.join = ['roster', 'chara', 'guest'].indexOf(d.join) >= 0 ? d.join : 'roster';
  d.roster = String(d.roster || '');
  d.useCode = d.useCode === true;
  return d;
}
// 生徒に送る情報（参加コードそのもの・名簿は送らない）
function pub_(r) {
  if (!r) return null;
  const s = r.settings, o = { id: r.id, prevId: r.prevId || '', status: r.status, curQ: r.curQ, code: !!r.code, startedAt: r.startedAt, showRes: !!r.showRes };
  ['mode', 'level', 'count', 'minutes', 'how', 'autoLv', 'up', 'down', 'flagN', 'sound', 'kana', 'guide', 'join'].forEach(k => { o[k] = s[k]; });
  return o;
}
function modeName_(m) { return m === 'self' ? 'ひとりずつ' : 'みんなで'; }

// 回を終える：まだまとめを書いていない「みんなで」の生徒の分を、キャッシュから書く
function endRound_(r) {
  r.status = 'ended';
  r.endedAt = Date.now();
  setJSON_('ROUND', r);
  if (r.settings.mode !== 'all') return;
  const mem = getJSON_('MEM', []);
  if (!mem.length) return;
  const c = cache_(), keys = mem.map(m => stKey_(r.id, m.name)), got = c.getAll(keys);
  const rows = [], put = {};
  mem.forEach((m, i) => {
    let s = null;
    try { s = got[keys[i]] ? JSON.parse(got[keys[i]]) : null; } catch (err) { s = null; }
    if (!s || s.sumd) return;
    s.sumd = true; put[keys[i]] = JSON.stringify(s);
    let correct = 0; const mist = {};
    Object.keys(s.ans || {}).forEach(q => { const a = s.ans[q]; if (a[1]) correct++; else if (a[2]) mist[a[2]] = (mist[a[2]] || 0) + 1; });
    rows.push(sumRow_(r.id, 'all', m.name, m.lv, m.lv, r.settings.count, correct, mist));
  });
  if (rows.length) appendRows_(sheet_(SHEET_SUM, HEAD_SUM), rows);
  if (Object.keys(put).length) c.putAll(put, CACHE_SEC);
}
function topOf_(mist) {
  let top = '', n = 0;
  Object.keys(mist || {}).forEach(k => { if (mist[k] > n) { top = k; n = mist[k]; } });
  return top ? top + '（' + n + '回）' : '';
}
function sumRow_(rid, mode, name, lv0, lv1, n, correct, mist) {
  return {
    '日時': new Date(), '回ID': rid, '名前': name, '進め方': modeName_(mode),
    'はじめのレベル': lv0, 'おわりのレベル': lv1, '解いた数': n, '正解数': correct,
    '正答率': n ? Math.round(correct / n * 100) + '%' : '', '多かったまちがい': typeof mist === 'string' ? mist : topOf_(mist)
  };
}

/* ========== 生徒用 API ========== */
function api_poll(roundId, name) {
  const r = getJSON_('ROUND', null);
  if (r && name && roundId === r.id) cache_().put(seenKey_(r.id, name), String(Date.now()), CACHE_SEC);
  return { round: pub_(r) };
}

function api_checkCode(code) {
  const r = getJSON_('ROUND', null);
  if (!r || r.status !== 'open') return false;
  return !r.code || String(code) === r.code;
}

function api_joinInfo(tok) {
  const r = getJSON_('ROUND', null);
  const out = { round: pub_(r), names: [], taken: [] };
  if (r && r.status === 'open') {
    if (r.settings.join === 'roster') out.names = rosterNames_(r.settings.roster);
    out.taken = getJSON_('MEM', []).filter(m => m.kind === 'chara' && m.tok !== tok).map(m => m.name);
  }
  return out;
}

// 名前を入れない参加：端末ごとに「ゲスト1」「ゲスト2」…（同じ端末なら同じ番号）
function guestName_(tok) {
  const g = getJSON_('GUESTS', {});
  if (!g[tok]) {
    let max = 0;
    Object.keys(g).forEach(k => { max = Math.max(max, g[k]); });
    g[tok] = max + 1;
    setJSON_('GUESTS', g);
  }
  return 'ゲスト' + g[tok];
}
function guestList_() {
  const g = getJSON_('GUESTS', {});
  return Object.keys(g).map(k => g[k]).sort((a, b) => a - b).map(n => 'ゲスト' + n);
}

function api_join(o) {
  return withLock_(() => {
    const r = getJSON_('ROUND', null);
    if (!r || r.status !== 'open') return { ok: false, err: 'closed', round: pub_(r) };
    const tok = String((o && o.tok) || '').slice(0, 60);
    const kind = r.settings.join;
    let name = String((o && o.name) || '').trim().slice(0, 30);
    if (kind === 'guest') name = guestName_(tok);
    if (!name) return { ok: false, err: 'name' };
    const mem = getJSON_('MEM', []);
    const me = mem.filter(m => m.name === name)[0];
    const prevOk = !!(o.prevId && o.prevId === r.prevId && getJSON_('PREV_MEM', []).indexOf(name) >= 0);
    if (r.code && !me && !prevOk && String(o.code) !== r.code) return { ok: false, err: 'code' };
    if (me && me.kind === 'chara' && me.tok !== tok) return { ok: false, err: 'taken' };
    let lv = String(r.settings.level);
    if (r.settings.mode === 'all' && r.settings.levelMode === 'each' && r.levelMap && r.levelMap[name]) lv = String(r.levelMap[name]);
    if (r.sets && r.sets.length && r.sets.indexOf(lv) < 0) lv = r.sets[0];
    if (me) me.tok = tok;
    else mem.push({ name: name, kind: kind, tok: tok, lv: lv });
    setJSON_('MEM', mem);
    return { ok: true, name: name, setKey: me ? me.lv : lv, conf: confFor_(name), round: pub_(r) };
  });
}

// 「みんなで」の問題（その回・そのレベルの分）
function api_questions(roundId, key) {
  const r = getJSON_('ROUND', null);
  if (!r || r.id !== roundId) return null;
  const c = cache_(), ck = 'qs_' + roundId + '_' + key;
  let s = c.get(ck);
  if (!s) {
    s = getBig_('QS_' + key);
    if (s && s.length < 30000) c.put(ck, s, CACHE_SEC);
  }
  return s || null;
}

// 「ひとりずつ」の問題：そのレベルのことばを n こ（ばらばらに選ぶ）
function api_words(roundId, lv, n) {
  const r = getJSON_('ROUND', null);
  if (!r || r.id !== roundId || r.status !== 'open') return [];
  return sample_(levelWords_(lv), clampN_(n, 1, 60, 40));
}

// 記録を受け取る（送れなかった分もまとめて届く。同じものの2回目は書かない）
function api_submit(items) {
  if (!items || !items.length) return { done: [] };
  return withLock_(() => {
    const c = cache_();
    const idKeys = items.map(it => 'it_' + String(it.id).slice(0, 60));
    const seen = c.getAll(idKeys);
    const stKeys = [];
    items.forEach(it => { const k = stKey_(it.roundId || '', it.name || ''); if (stKeys.indexOf(k) < 0) stKeys.push(k); });
    const got = c.getAll(stKeys), st = {};
    stKeys.forEach(k => { try { st[k] = got[k] ? JSON.parse(got[k]) : null; } catch (err) { st[k] = null; } });
    const logRows = [], sumRows = [];
    let conf = null;
    const cf = () => conf || (conf = readConf_());
    const newIds = {};
    items.forEach((it, i) => {
      if (seen[idKeys[i]]) return;
      newIds[idKeys[i]] = '1';
      const k = stKey_(it.roundId || '', it.name || '');
      const s = st[k] || (st[k] = { ans: {}, snap: null, sumd: false, done: false });
      if (it.kind === 'answer') {
        if (it.mode === 'all') {
          if (s.ans[it.qi]) return;
          s.ans[it.qi] = [-1, it.ok ? 1 : 0, it.ml || '', String(it.pick || '').slice(0, 20)];
        }
        logRows.push({
          '日時': new Date(it.at || Date.now()), '回ID': it.roundId, '名前': it.name, '進め方': modeName_(it.mode),
          'レベル': it.level, '内容': lvName_(it.level), '問題': it.q || '', '正しい答え': it.correct || '', '答え': it.pick || '',
          '正誤': it.na ? '－' : (it.ok ? '○' : '×'), 'まちがいの型': it.ml || '', '答え方': howName_(it.how), 'かかった秒数': it.sec
        });
      } else if (it.kind === 'snap') {
        s.snap = it.snap || null;
      } else if (it.kind === 'finish') {
        s.done = true;
        if (s.sumd) return;
        s.sumd = true;
        sumRows.push(sumRow_(it.roundId, it.mode, it.name, it.start, it.end, it.n, it.correct, it.top || ''));
        if (it.mode === 'self' && it.end) patchConf_(cf(), it.name, { '前回のレベル': it.end });
      }
    });
    if (logRows.length) appendRows_(sheet_(SHEET_LOG, HEAD_LOG), logRows);
    if (sumRows.length) appendRows_(sheet_(SHEET_SUM, HEAD_SUM), sumRows);
    if (conf) writeConf_(conf);
    const put = Object.assign({}, newIds);
    Object.keys(st).forEach(k => { if (st[k]) put[k] = JSON.stringify(st[k]); });
    if (Object.keys(put).length) c.putAll(put, CACHE_SEC);
    return { done: items.map(it => it.id) };
  });
}

function api_tStart(settings) {
  return withLock_(() => {
    const old = getJSON_('ROUND', null);
    if (old && old.status === 'open') endRound_(old);
    const s = sanitize_(settings);
    const all = readWords_();
    cacheWords_(all);
    // 「みんなで」で生徒ごとのレベルを使うとき：生徒設定のレベル（なければ前回のレベル）
    const levelMap = {};
    if (s.mode === 'all' && s.levelMode === 'each') {
      const d = readConf_();
      d.rows.forEach(r => {
        const n = String(r[d.col('名前')]).trim(), c = confRow_(d, r), lv = c.level || c.last;
        if (n && lv) levelMap[n] = lv;
      });
    }
    const sets = {};
    if (s.mode === 'all') {
      const keys = [String(s.level)];
      Object.keys(levelMap).forEach(n => { const k = String(levelMap[n]); if (keys.indexOf(k) < 0) keys.push(k); });
      keys.forEach(k => { sets[k] = sample_(levelWords_(+k, all), s.count); });
      let min = s.count;
      keys.forEach(k => { min = Math.min(min, sets[k].length); });
      if (min < 1) throw new Error('「問題」タブに、出せることばがありません。');
      s.count = min;
      keys.forEach(k => { sets[k] = sets[k].slice(0, min); });
    }
    const p = props_();
    p.getKeys().forEach(k => { if (k.indexOf('QS_') === 0) p.deleteProperty(k); });
    Object.keys(sets).forEach(k => putBig_('QS_' + k, JSON.stringify(sets[k])));
    const r = {
      id: Utilities.formatDate(new Date(), TZ, 'MMdd-HHmmss'),
      prevId: old ? old.id : '',
      status: 'open',
      curQ: -1,
      settings: s,
      code: s.useCode ? String(1000 + Math.floor(Math.random() * 9000)) : '',
      sets: Object.keys(sets),
      levelMap: levelMap,
      startedAt: Date.now()
    };
    setJSON_('PREV_MEM', getJSON_('MEM', []).map(m => m.name));
    setJSON_('MEM', []);
    setJSON_('ROUND', r);
    setJSON_('LAST_SETTINGS', s);
    return r;
  });
}

function api_tPoll() {
  const r = getJSON_('ROUND', null);
  if (!r) return { round: null, members: [] };
  const mem = getJSON_('MEM', []);
  const sk = mem.map(m => stKey_(r.id, m.name)), nk = mem.map(m => seenKey_(r.id, m.name));
  const got = mem.length ? cache_().getAll(sk.concat(nk)) : {};
  const now = Date.now();
  const members = mem.map((m, i) => {
    let s = null;
    try { s = got[sk[i]] ? JSON.parse(got[sk[i]]) : null; } catch (err) { s = null; }
    const seen = parseInt(got[nk[i]] || '0', 10);
    return { name: m.name, kind: m.kind, lv: m.lv, ans: s ? s.ans : {}, snap: s ? s.snap : null, done: !!(s && s.done), ago: seen ? Math.round((now - seen) / 1000) : -1 };
  });
  return { round: r, members: members };
}

function api_tSetQ(roundId, idx) {

    if (!r || r.id !== roundId || r.status !== 'open') return r;
    r.curQ = Math.max(-1, Math.min(r.settings.count - 1, parseInt(idx, 10)));
    setJSON_('ROUND', r);
    return r;
  });
}

function fmtJ_(v) {
  const d = v instanceof Date ? v : new Date(v);
  if (isNaN(d.getTime())) return String(v);
  const m = parseInt(Utilities.formatDate(d, TZ, 'm'), 10);
  return Utilities.formatDate(d, TZ, 'M月d日 H時') + (m ? m + '分' : '');
}

// 記録を見る：回ごとの記録（新しい順）と、生徒ごとのまちがいの型・まちがえたことば
function api_tRecords() {
  const readLast = (sh, max) => {
    const last = sh.getLastRow(), heads = heads_(sh);
    if (last < 2) return { heads: heads, rows: [] };
    const n = Math.min(max, last - 1);
    return { heads: heads, rows: sh.getRange(last - n + 1, 1, n, heads.length).getValues() };
  };
  const s = readLast(sheet_(SHEET_SUM, HEAD_SUM), 200), h = s.heads, col = x => h.indexOf(x);
  const sums = s.rows.reverse().map(r => ({
    at: fmtJ_(r[col('日時')]), name: String(r[col('名前')]), mode: String(r[col('進め方')]),
    lv0: String(r[col('はじめのレベル')]), lv1: String(r[col('おわりのレベル')]), n: r[col('解いた数')], correct: r[col('正解数')],
    rate: String(r[col('正答率')]), mist: String(r[col('多かったまちがい')])
  }));
  const l = readLast(sheet_(SHEET_LOG, HEAD_LOG), 3000), lh = l.heads;
  const ni = lh.indexOf('名前'), mi = lh.indexOf('まちがいの型'), qi = lh.indexOf('問題'), oi = lh.indexOf('正誤');
  const by = {}, words = {};
  l.rows.forEach(r => {
    const n = String(r[ni]).trim(), m = String(r[mi]).trim();
    if (!n || String(r[oi]) !== '×') return;
    by[n] = by[n] || {};
    by[n][m || 'そのほか'] = (by[n][m || 'そのほか'] || 0) + 1;
    words[n] = words[n] || [];
    const w = String(r[qi]);
    if (w && words[n].indexOf(w) < 0) words[n].push(w);
  });
  const mist = Object.keys(by).map(n => {
    const items = Object.keys(by[n]).map(t => [t, by[n][t]]).sort((a, b) => b[1] - a[1]);
    return { name: n, total: items.reduce((a, b) => a + b[1], 0), items: items, words: words[n].slice(-20).reverse() };
  }).sort((a, b) => b.total - a.total);
  return { sums: sums, mist: mist };
}
