const T = { init: null, set: null, screen: 'home', view: null, sig: '', poll: null, busy: false, saveT: null,
  makeTab: '', listTab: '', pend: { a: null, b: null }, recent: [], pairs: null, delArm: -1 };

async function tBoot() {
  app.innerHTML = '<div class="screen"><div class="center"><div class="msg">読み込んでいます…</div></div></div>';
  try {
    T.init = await call('tInit');
    T.set = T.init.set;
    if (!T.init.tabs.some(t => t.name === T.set.tab) && T.init.tabs.length) T.set.tab = T.init.tabs[0].name;
    soundOn = T.set.sound;
    T.view = T.init.view;
    T.screen = (T.view.st.phase === 'idle') ? 'home' : 'play';
    tRender();
  } catch (e) {
    app.innerHTML = `<div class="screen"><div class="center"><div class="msg">${esc(e.message)}</div></div></div>`;
  }
}
function tabOf(name) { return (T.init.tabs || []).find(t => t.name === name) || null; }

function tRender() {
  clearTimeout(T.poll);
  if (T.screen === 'home') tHome();
  else if (T.screen === 'make') tMake();
  else if (T.screen === 'list') tList();
  else if (T.screen === 'tabs') tTabs();
  else if (T.screen === 'bulk') tBulk();
  else if (T.screen === 'play') { T.sig = ''; tPlay(); tPollLoop(); }
}

/* ---- ホーム（せってい） ---- */
function seg(key, opts) {
  return `<div class="seg">${opts.map(o => `<button class="${T.set[key] === o[0] ? 'on' : ''}" ${o[2] ? 'disabled' : ''}
    onclick='setOpt(${JSON.stringify(key)},${JSON.stringify(o[0])})'>${o[1]}</button>`).join('')}</div>`;
}
function numBox(key, min, max, step, unit) {
  return `<div class="num"><input inputmode="numeric" value="${T.set[key]}" onchange="numSet('${key}',this.value,${min},${max})">
    <button onclick="numStep('${key}',${step},${min},${max})">△</button><button onclick="numStep('${key}',${-step},${min},${max})">▽</button>
    <span>${unit}</span></div>`;
}
function setOpt(k, v) { T.set[k] = v; if (k === 'sound') soundOn = v; saveSoon(); tHome(); }
function numSet(k, v, min, max) { const n = parseInt(String(v).replace(/[^\d]/g, ''), 10); T.set[k] = isNaN(n) ? min : Math.max(min, Math.min(max, n)); saveSoon(); tHome(); }
function numStep(k, d, min, max) { T.set[k] = Math.max(min, Math.min(max, (parseInt(T.set[k], 10) || 0) + d)); saveSoon(); tHome(); }
function saveSoon() {
  clearTimeout(T.saveT);
  T.saveT = setTimeout(() => call('tSaveSettings', T.set).catch(e => toast(e.message)), 600);
}

/* ---- すすめる画面 ---- */
function tPollLoop(wait) {
  clearTimeout(T.poll);
  T.poll = setTimeout(async () => {
    if (T.screen !== 'play' || PAGE !== 'teacher') return;
    const t0 = Date.now();
    try { const v = await call('tView'); net(false); if (PAGE !== 'teacher' || T.screen !== 'play') return; if (!T.busy) { T.view = v; tPlay(); } }
    catch (e) { net(true); }
    tPollLoop(Math.max(300, 1500 - (Date.now() - t0)));
  }, wait == null ? 1500 : wait);
}
async function tAct(fn) {
  if (T.busy) return;
  T.busy = true;
  try { T.view = await call(fn); tPlay(); } catch (e) { toast(e.message); }
  T.busy = false;
}

function tPlay() {
  const v = T.view, st = v.st;
  clockOffset = v.now - Date.now();
  if (st.phase === 'idle') { T.screen = 'home'; tRender(); return; }
  const sig = [st.round, st.phase, st.index].join('|');
  if (sig !== T.sig) {
    const first = T.sig !== '';
    T.sig = sig;
    if (st.phase === 'wait') app.innerHTML = lobbyHTML(v);
    else if (st.phase === 'q' || st.phase === 'reveal') {
      app.innerHTML = tQHTML(v);
      timerInfo = st.phase === 'q' ? { limit: st.limit, openedAt: st.openedAt } : null;
      if (st.phase === 'reveal' && first) sfx('pair');
      ensureImages(itemNames(v));
      fillImgs();
    }
    else if (st.phase === 'end') app.innerHTML = tEndHTML(v);
    if (st.phase === 'wait') ensureImages(st.imgs || []);
  }
  // 人数などだけを書きかえる
  const pc = document.getElementById('people');
  if (pc) {
    pc.innerHTML = v.people.length ? v.people.map(n => `<span class="chip">${esc(n)}</span>`).join('') : '<span class="note">まだ だれも いないよ</span>';
    document.getElementById('pcount').textContent = v.people.length + 'にん';
  }
  const an = document.getElementById('ansNames');
  if (an && v.answered) {
    an.innerHTML = v.people.map(n => `<span class="chip ${v.answered.indexOf(n) >= 0 ? 'done' : ''}">${esc(n)}</span>`).join('');
    document.getElementById('ansn').textContent = `こたえた ひと ${v.answered.length} / ${v.people.length}にん`;
  }
}

function tQHTML(v) {
  const st = v.st, q = v.q, rev = st.phase === 'reveal';
  const n = q.choices.length;
  const ch = q.choices.map((c, i) => {
    let cls = 'choice', tag = '';
    if (rev) { if (i === v.correct) { cls += ' correct'; tag = '<span class="tag">○ せいかい</span>'; } else cls += ' dim'; }
    const cnt = rev ? `<span class="cnt">${v.counts[i]}にん</span>` : '';
    return `<div class="${cls}"><span class="no">${i + 1}</span>${tag}${itemHTML(c)}${cnt}</div>`;
  }).join('');
  const left = `<div class="qbox">${itemHTML(q.item)}</div>`;
  const last = st.index + 1 >= st.total;
  const btns = rev
    ? `<button class="btn main big" onclick="tAct('tNext')">${last ? 'けっかを みる' : 'つぎへ'}</button><button class="btn" onclick="tAct('tFinish')">おわる</button>`
    : `<button class="btn main big" onclick="tAct('tReveal')">こたえ</button><button class="btn" onclick="tAct('tFinish')">おわる</button>`;
  return `<div class="play">
    <div class="topbar"><span>${st.index + 1} / ${st.total}もんめ</span><span id="ansn"></span>
      ${T.set.codeOn ? `<span class="code">さんかコード ${esc(v.code)}</span>` : ''}</div>
    ${rev ? '' : timerHTML({ limit: st.limit })}
    <div class="prompt">${promptText(st.split, st.la, st.lb, q.side)}</div>
    <div class="stage"><div class="qside">${left}</div><div class="choices n${n}">${ch}</div></div>
    <div class="chips" id="ansNames"></div>
    <div class="btns">${btns}</div>
  </div>`;
}

function resizeFile(file) {
  return new Promise((res, rej) => {
    const fr = new FileReader();
    fr.onload = () => {
      const im = new Image();
      im.onload = () => {
        const M = 720, k = Math.min(1, M / Math.max(im.width, im.height));
        const w = Math.max(1, Math.round(im.width * k)), h = Math.max(1, Math.round(im.height * k));
        const c = document.createElement('canvas'); c.width = w; c.height = h;
        const x = c.getContext('2d'); x.fillStyle = '#fff'; x.fillRect(0, 0, w, h); x.drawImage(im, 0, 0, w, h);
        res(c.toDataURL('image/jpeg', 0.85));
      };
      im.onerror = () => rej(new Error('絵を読み込めませんでした'));
      im.src = fr.result;
    };
    fr.onerror = () => rej(new Error('絵を読み込めませんでした'));
    fr.readAsDataURL(file);
  });
}
async function pickImg(side, input) {
  const f = input.files && input.files[0];
  if (!f) return;
  try {
    const data = await resizeFile(f);
    const text = document.getElementById('t_' + side).value;
    T.pend[side] = { data, text };
    tMake();
  } catch (e) { toast(e.message); }
}
async function addPair() {
  if (T.busy) return;
  const a = T.pend.a || {}, b = T.pend.b || {};
  a.text = document.getElementById('t_a').value.trim();
  b.text = document.getElementById('t_b').value.trim();
  if (!a.data && !a.text && !b.data && !b.text) return toast('絵か文字を入れてください');
  T.busy = true;
  const btn = document.getElementById('addBtn');
  btn.disabled = true; btn.textContent = '登録しています…';
  try {
    const side = async p => {
      let img = '';
      if (p.data) { img = await call('tUpload', p.data); IMG[img] = p.data; }
      return { img, text: p.text || '' };
    };
    const ra = await side(a), rb2 = await side(b);
    await call('tAddPair', T.makeTab, ra, rb2);
    T.recent.unshift({ a: ra, b: rb2 });
    const tab = tabOf(T.makeTab);
    if (tab) { if ((ra.img || ra.text) && (rb2.img || rb2.text)) tab.pairs++; else tab.dummy++; }
    T.pend = { a: null, b: null };
    toast('登録しました');
  } catch (e) { toast(e.message); }
  T.busy = false;
  tMake();
}

}

let delTimer;
async function delPair(i) {
  if (T.delArm !== i) {
    T.delArm = i; tList();
    clearTimeout(delTimer); delTimer = setTimeout(() => { T.delArm = -1; if (T.screen === 'list') tList(); }, 3000);
    return;
  }
  T.delArm = -1;
  const p = T.pairs[i];
  const check = [p.a.img, p.a.text, p.b.img, p.b.text].join('|');
  try {
    T.pairs = await call('tDelPair', T.listTab, p.row, check);
    const tab = tabOf(T.listTab);
    if (tab) { tab.pairs = T.pairs.filter(x => (x.a.img || x.a.text) && (x.b.img || x.b.text)).length; tab.dummy = T.pairs.length - tab.pairs; }
    toast('消しました');
  } catch (e) { toast(e.message); T.pairs = null; }
  tList();
}

// ファイルの なまえから もじを つくる（カメラの なまえは つかわない）
function baseName(n) { return String(n || '').replace(/\.[^.]+$/, '').trim(); }
function textFromName(n) {
  const b = baseName(n);
  if (!b || /^(img|image|dsc|pxl|photo|screenshot|スクリーンショット|写真)/i.test(b) || /^[\d_\-\s.]+$/.test(b)) return '';
  return b;
}
const PAIR_RULE = /^(.+?)[_\-－＿ ]?([12１２abAB])$/;
function suffixSide(s) { return /[1１aA]/.test(s) ? 'a' : 'b'; }

function tBulk() {
  const tab = tabOf(T.bulkTab);
  const la = tab && tab.split ? tab.la : 'A', lb = tab && tab.split ? tab.lb : 'B';

function setBulkTab(v) { T.bulkTab = v; tBulk(); }
function bText(i, side, v) { T.bulk[i][side] = Object.assign(T.bulk[i][side] || { data: '' }, { text: v }); }
function bSwap(i) { const p = T.bulk[i]; [p.a, p.b] = [p.b, p.a]; T.bsel = null; tBulk(); }
function bDel(i) { T.bulk.splice(i, 1); T.bsel = null; tBulk(); }
function bClear() { T.bulk = []; T.bsel = null; tBulk(); }
function bTap(i, side) {
  if (T.bulkBusy) return;
  if (!T.bsel) { T.bsel = { i, side }; }
  else if (T.bsel.i === i && T.bsel.side === side) T.bsel = null;
  else {
    const s = T.bsel, x = T.bulk[s.i][s.side];
    T.bulk[s.i][s.side] = T.bulk[i][side];
    T.bulk[i][side] = x;
    T.bsel = null;
  }
  tBulk();
}
async function bulkPick(input) {
  const files = Array.from(input.files || []);
  input.value = '';
  if (!files.length) return;
  const prog = document.getElementById('bprog');
  const items = [];
  for (let k = 0; k < files.length; k++) {
    if (prog) prog.textContent = `読み込んでいます ${k + 1} / ${files.length}`;
    try { items.push({ name: files[k].name, data: await resizeFile(files[k]) }); }
    catch (e) { toast(files[k].name + ' を読み込めませんでした'); }
  }
  // なまえの きまり（○○_1・○○_2）で ぜんぶ ペアに なるか しらべる
  const groups = {};
  let ruleOK = items.length >= 2;
  items.forEach(it => {
    const m = baseName(it.name).match(PAIR_RULE);
    if (!m) { ruleOK = false; return; }
    const g = groups[m[1]] = groups[m[1]] || {};
    const side = suffixSide(m[2]);
    if (g[side]) ruleOK = false;
    g[side] = { data: it.data, text: textFromName(m[1]) };
  });
  if (ruleOK) {
    Object.keys(groups).sort().forEach(k => T.bulk.push({ a: groups[k].a || null, b: groups[k].b || null }));
    toast('ファイル名でペアにしました');
  } else {
    items.forEach(it => {
      const v = { data: it.data, text: textFromName(it.name) };
      const last = T.bulk[T.bulk.length - 1];
      if (last && last.a && !last.b) last.b = v;
      else T.bulk.push({ a: v, b: null });
    });
    toast('選んだ順に並べました');
  }
  T.bsel = null;
  tBulk();
}
async function bAddAll() {
  if (T.bulkBusy) return;
  const has = it => it && (it.data || (it.text || '').trim());
  const list = T.bulk.filter(p => has(p.a) || has(p.b));
  if (!list.length) return toast('絵か文字を入れてください');
  T.bulkBusy = true;
  const btn = document.getElementById('bAdd'), prog = document.getElementById('bprog');
  btn.disabled = true;
  let done = 0, err = '';
  const total = list.length;
  try {
    while (T.bulk.length) {
      const chunk = T.bulk.slice(0, 5).filter(p => has(p.a) || has(p.b));
      const out = [];
      for (const p of chunk) {
        const datas = [p.a && p.a.data || '', p.b && p.b.data || ''];
        const names = await call('tUploadMany', datas);
        datas.forEach((d, k) => { if (names[k]) IMG[names[k]] = d; });
        out.push({ a: { img: names[0], text: ((p.a && p.a.text) || '').trim() }, b: { img: names[1], text: ((p.b && p.b.text) || '').trim() } });
        done++;
        btn.textContent = `登録しています ${done} / ${total}`;
      }
      if (out.length) await call('tAddPairs', T.bulkTab, out);
      T.bulk.splice(0, 5);
      const tab = tabOf(T.bulkTab);
      if (tab) out.forEach(r => { if ((r.a.img || r.a.text) && (r.b.img || r.b.text)) tab.pairs++; else tab.dummy++; });
    }
  } catch (e) { err = e.message; }
  T.bulkBusy = false;
  T.bsel = null;
  tBulk();
  if (err) toast('途中で止まりました（' + err + '）。残りをもう一度登録してください');
  else toast(total + '組 登録しました');
}

const S = { o: null, sig: '', poll: null, typed: '', busy: false, pre: -1, last: null };

function sBoot() { S.sig = ''; S.o = null; app.innerHTML = '<div class="screen"><div class="center"><div class="msg">つないでいるよ…</div></div></div>'; sTick(); }
async function sTick() {
  clearTimeout(S.poll);
  const t0 = Date.now();
  try {
    const o = await call('sPoll', CID, CODE);
    net(false);
    if (PAGE !== 'student') return;
    if (!S.busy) sApply(o);
  } catch (e) { net(true); }
  if (PAGE === 'student') S.poll = setTimeout(sTick, Math.max(300, 1500 - (Date.now() - t0)));
}
function sApply(o) {
  clockOffset = o.now - Date.now();
  soundOn = o.sound;
  const prev = S.o;
  S.o = o;
  if (o.imgs && S.pre !== o.round) { S.pre = o.round; ensureImages(o.imgs); }
  if (o.phase === 'reveal' && o.q && prev && prev.phase === 'q' && prev.index === o.index) {
    if (o.mine >= 0) sfx(o.mine === o.correct ? 'ok' : 'ng'); else sfx('pair');
  }
  sRender();
}
function sRender(force) {
  const o = S.o;
  if (!o) return;
  let sig, html;
  if (o.needCode) { sig = 'code|' + S.typed; html = codeHTML(); }
  else if (!o.myName) {
    sig = 'name|' + JSON.stringify(o.names);
    html = nameHTML(o);
  } else if ((o.phase === 'q' || o.phase === 'reveal') && o.q) {
    const up = timeLeft(o.q) <= 0;
    sig = ['q', o.round, o.phase, o.index, o.mine, o.changed, up].join('|');
    html = sPlayHTML(o, up);
  } else if (o.phase === 'end') {
    sig = 'end|' + o.round + '|' + o.score; html = sEndHTML(o);
  } else {
    sig = 'wait|' + o.myName + '|' + o.phase; html = sWaitHTML(o);
  }
  if (sig === S.sig && !force) return;
  S.sig = sig;
  app.innerHTML = html;
  timerInfo = (o.phase === 'q' && o.q && o.myName) ? { limit: o.q.limit, openedAt: o.q.openedAt, fired: timeLeft(o.q) <= 0 } : null;
  if (o.q) ensureImages(itemNames(o));
  fillImgs();
}

    ${rev ? '' : timerHTML(q)}
    <div class="prompt">${promptText(q.split, q.la, q.lb, q.side)}</div>
    <div class="sstage">${top}<div class="choices n${n}">${ch}</div></div>
    ${status}
  </div>`;
}
async function sPick(i) {
  const o = S.o;
  if (!o || o.phase !== 'q' || S.busy) return;
  if (timeLeft(o.q) <= 0) return toast('じかんが おわったよ');
  if (o.mine === i) return;
  if (o.mine >= 0 && o.changed >= 1) return toast('もう かえられないよ');
  S.busy = true;
  sfx('tap');
  const oldMine = o.mine, oldCh = o.changed;
  o.changed = (o.mine >= 0) ? o.changed + 1 : o.changed; o.mine = i;
  sRender();
  try {
    const r = await call('sAnswer', CID, CODE, i);
    o.mine = r.mine; o.changed = r.changed;
  } catch (e) { o.mine = oldMine; o.changed = oldCh; toast(e.message); }
  S.busy = false;
  sRender();
}

    <div class="score">${o.score} / ${o.total}</div>
    <div class="msg">もん せいかい</div>
  </div></div>`;
}
