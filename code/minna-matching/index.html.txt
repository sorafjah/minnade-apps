function add(el, c) {
  if (c == null || c === false) return;
  if (Array.isArray(c)) { c.forEach(function (x) { add(el, x); }); return; }
  if (c.nodeType) el.appendChild(c); else el.appendChild(document.createTextNode(String(c)));
}
function h(tag, props) {
  var el = document.createElement(tag), val = null;
  if (props) {
    Object.keys(props).forEach(function (k) {
      var v = props[k];
      if (v == null || v === false) return;
      if (k === 'class') el.className = v;
      else if (k === 'style') el.style.cssText = v;
      else if (k.slice(0, 2) === 'on') el.addEventListener(k.slice(2), v);
      else if (k === 'value') val = v;
      else if (k === 'checked' || k === 'disabled' || k === 'selected') el[k] = true;
      else el.setAttribute(k, v === true ? '' : v);
    });
  }
  for (var i = 2; i < arguments.length; i++) add(el, arguments[i]);
  if (val !== null) el.value = val;
  return el;
}

}

    var wrap = h('div', { class: 'wrap' }, head(), offline());
    if (p.needCode) wrap.appendChild(codeScreen());
    else if (!p.myName) wrap.appendChild(nameScreen());
    else if (p.pace === 'self' && (p.phase === 'run' || p.phase === 'end') && p.me) wrap.appendChild(p.me.done ? endScreen() : playScreen());
    else if (p.pace !== 'self' && (p.phase === 'q' || p.phase === 'reveal') && p.q) wrap.appendChild(playScreen());
    else if (p.phase === 'end') wrap.appendChild(endScreen());
    else wrap.appendChild(h('div', { class: 'msgbig' }, t(p.phase === 'idle' ? 'はじまるまで^まっていてね' : 'もうすぐ^はじまるよ')));
    if (S.help) wrap.appendChild(helpOverlay());
    swap(wrap);
    if (S.timerFn) { var tf = S.timerFn; S.timerFn = null; tf(); }
  }

  function helpSlides() {
    var p = S.p || {}, self = p.pace === 'self', playing = p.phase && p.phase !== 'idle' && p.phase !== 'wait';
    var second = self ? 'えらぶと、すぐに「せいかい！」か「おしい！」が^出《で》るよ。「つぎへ」を^おすと、つぎの^問題《もんだい》に^すすむよ。'
      : playing ? 'えらんだあとも、1回《いっかい》だけ^変《か》えられるよ。先生《せんせい》が^こたえを^見《み》せるまで、まっていてね。'
      : '「みんなで」の^ときは、えらんだあとも^1回《いっかい》だけ^変《か》えられるよ。「ひとりずつ」の^ときは、えらぶと^すぐに^こたえが^出《で》るよ。';
    return [
      { title: 'ペアを^さがすよ', body: 'うえに^カードが^出《で》るよ。ペアに^なる^カードを、したから^えらんでね。', demo: 'q' },
      { title: 'えらびかた', body: second, demo: 'pick' },
      { title: 'まちがえても^だいじょうぶ', body: 'まちがえても、だいじょうぶ。「おしい！」と^こたえが^出《で》るから、つぎに^いかそうね。', demo: 'fb' },
      { title: '音《おと》と^つかいかた', body: '「おと」の^ボタンで、音《おと》を^オン・オフ できるよ。わからなく なったら、「つかいかた」の^ボタンで、また^見《み》られるよ。' }

  }
  function helpOverlay() {

  function codeScreen() {
    var inp = h('input', { type: 'text', inputmode: 'numeric', maxlength: '4', pattern: '[0-9]*', autocomplete: 'off', value: '' });
    function go() {
      unlock();
      var v = inp.value.replace(/[^0-9]/g, '');
      if (v.length < 4) { toast(t('4けたの^数字《すうじ》を^入《い》れてね')); return; }
      S.code = v; lsSet('mm_code', v); S.codeErr = false; doTick();

  }
  function nameScreen() {
    var p = S.p, box = h('div', { class: 'names' });
    (p.names || []).forEach(function (o) {
      box.appendChild(h('button', {
        type: 'button', class: 'nbtn', disabled: o.t,

          call('sName', S.cid, S.code, o.n).then(function () { return doTick(); }).catch(function (e) {
            toast(isNet(e) ? t('つうしんが^うまくいかないよ') : t(msgOf(e)));
            doTick();
          });
        }

  }

  function playScreen() {
    var p = S.p, self = p.pace === 'self', q = p.q, hira = p.hira;
    var idx = self ? p.me.i : p.index;
    var reveal = self ? p.me.ph === 'f' : p.phase === 'reveal';
    var mine = typeof p.mine === 'number' ? p.mine : -1;
    var items = [q.item].concat(q.choices);
    needImgs(items.map(function (x) { return x.img; }));
    var timeUp = false;
    if (q.limit > 0 && !reveal) { tm.total = q.limit; tm.openedAt = q.openedAt; timeUp = tmLeft() <= 0; }
    var locked = reveal || S.busy || timeUp || (self ? false : (mine >= 0 && p.changed >= 1)) || OUT.length > 0;
    var wrap = h('div', null);
    var prog = h('div', { class: 'prog' }, h('span', { class: 'chip' }, t('問題《もんだい》'), ' ', String(idx + 1), ' / ', String(p.total)));
    var fill = null, txt = null;
    if (q.limit > 0 && !reveal) {
      fill = h('div'); txt = h('span', { class: 'chip acc' }, '');
      prog.appendChild(h('div', { class: 'timer' }, fill)); prog.appendChild(txt);
    }
    wrap.appendChild(prog);
    var ask = q.split ? (q.side === 'a' ? q.la : q.lb) : '';
    wrap.appendChild(h('div', { class: 'qcard' },
      h('div', { class: 'ask' }, ask ? [t(ask), t(' の ペアに^なるのは^どれかな？')] : t('ペアに^なるのは^どれかな？')),
      cardNode(q.item, hira)));
    var grid = h('div', { class: 'choices' });
    q.choices.forEach(function (c, i) {
      var cls = 'choice', badge = null;
      if (reveal && i === p.correct) { cls += ' right'; badge = h('span', { class: 'badge ok' }, t('こたえ')); }
      else if (reveal && i === mine) { cls += ' wrongmine'; badge = h('span', { class: 'badge ng' }, t('あなたの^えらんだもの')); }
      else if (!reveal && i === mine) { cls += ' mine'; badge = h('span', { class: 'badge' }, t('えらんだよ')); }
      if (reveal && i === p.correct && i === mine) badge = h('span', { class: 'badge ok' }, t('こたえ・あなたも^これだよ'));
      grid.appendChild(h('button', {
        type: 'button', class: cls, disabled: locked || (!self && i === mine),
        onclick: function () { pick(i); }
      }, cardNode(c, hira), badge));
    });
    wrap.appendChild(grid);
    // ことば（色や音だけにたよらない）
    if (!reveal) {
      if (S.busy || OUT.length) wrap.appendChild(h('div', { class: 'fb plain' }, t('おくっているよ…')));
      else if (timeUp) wrap.appendChild(h('div', { class: 'fb plain' }, t('時間《じかん》だよ')));
      else if (!self && mine >= 0) wrap.appendChild(h('div', { class: 'fb plain' }, t(p.changed >= 1 ? 'こたえたよ！^つぎまで^まっていてね' : 'こたえたよ！^もう^1回《いっかい》だけ^変《か》えられるよ')));
    } else if (self) {
      var ok = mine >= 0 && mine === p.correct;
      wrap.appendChild(h('div', { class: 'fb ' + (ok ? 'ok' : 'ng') }, t(ok ? 'せいかい！' : (mine < 0 ? '時間《じかん》だよ。こたえは^ここだよ' : 'おしい！^こたえは^ここだよ'))));
      var last = idx + 1 >= p.total;
      wrap.appendChild(h('div', { style: 'text-align:center' }, h('button', {
        type: 'button', class: 'btn big', disabled: S.busy,
        onclick: function () {
          unlock(); S.busy = true; sRender(true);
          call('sNext', S.cid, S.code, p.round).catch(function () {}).then(function () { S.busy = false; return doTick(); }).then(function () { sRender(true); });
        }
      }, t(last ? 'おわる' : 'つぎへ'))));
    } else {
      var ok2 = mine >= 0 && mine === p.correct;
      wrap.appendChild(h('div', { class: 'fb ' + (ok2 ? 'ok' : 'ng') }, t(ok2 ? 'せいかい！' : (mine < 0 ? 'こたえは^ここだよ' : 'おしい！^こたえは^ここだよ'))));
      wrap.appendChild(h('div', { class: 'fb plain' }, t('つぎの^問題《もんだい》まで^まっていてね')));
    }
    S.timerFn = function () {
      if (fill) tmStart(q.limit, q.openedAt, fill, txt, function () {
        if (self && p.me.ph === 'a' && !S.busy) pick(-1); else sRender(true);
      });
    };
    return wrap;

    function pick(i) {
      if (S.busy) return;
      unlock(); S.busy = true; sRender(true);
      var pr;
      if (self) pr = submit('sAnswerSelf', [S.cid, S.code, p.round, idx, i]);
      else pr = submit('sAnswer', [S.cid, S.code, i, p.round, idx]);
      pr.then(function (res) {
        if (S.snd) { if (self && res) beep(res.ok ? 'ok' : 'soft'); else beep('ok'); }
      }).catch(function (e) {
        if (!e.queued) toast(t(msgOf(e)));
      }).then(function () { S.busy = false; return doTick(); }).then(function () { sRender(true); });
    }
  }

  function endScreen() {
    var p = S.p, hira = p.hira;
    var w = h('div', null, h('div', { class: 'msgbig' }, t('おつかれさま！')));
    if (p.showScore && typeof p.score === 'number') w.appendChild(h('div', { class: 'fb ok' }, String(p.score), t(' 問《もん》^せいかい')));
    w.appendChild(h('div', { class: 'fb plain' }, t('つぎが^はじまるまで^まっていてね')));
    return w;
  }

  loop();
}

function teacherApp() {
  var T = { init: null, view: null, set: null, screen: 'home', big: false, help: false, sig: '', lastRun: false, msg: null, ok: null, busy: false };
  var P = { tab: '', rows: null, a: { text: '', data: '' }, b: { text: '', data: '' }, bulk: '', newTab: '', del: null };

  IMGDONE = function () { if (T.screen === 'pairs' || runMode()) renderT(true); };

  function runMode() { return !!(T.view && T.view.st && T.view.st.phase !== 'idle'); }
  function err(e) { T.msg = msgOf(e); T.ok = null; renderT(true); }
  function note(s) { T.ok = s; T.msg = null; renderT(true); }
  function go(screen) { T.screen = screen; T.msg = null; T.ok = null; renderT(true); }
  function act(fn) {   // サーバーを呼ぶボタン用
    if (T.busy) return;
    T.busy = true;
    var args = Array.prototype.slice.call(arguments, 1);
    return call.apply(null, [fn].concat(args)).then(function (v) { T.busy = false; return v; }, function (e) { T.busy = false; throw e; });
  }
  function setView(v) { T.view = v; T.msg = null; renderT(true); }

  function load() {
    call('tInit').then(function (d) {
      T.init = d; T.set = JSON.parse(JSON.stringify(d.set)); T.view = d.view;
      if (!T.set.tab && d.tabs.length) T.set.tab = d.tabs[0].name;
      renderT(true); poll();
    }).catch(function (e) {
      swap(h('div', { class: 'wrap' }, h('div', { class: 'err' }, isNet(e) ? 'サーバーにつながっていません' : msgOf(e))));
    });
  }
  function poll() {
    setTimeout(function () {
      if (T.busy || document.hidden) { poll(); return; }
      call('tView').then(function (v) {
        var was = runMode(); T.view = v;
        if (runMode() || was) renderT(false);
      }).catch(function () {}).then(poll);
    }, runMode() ? 4000 : 10000);
  }

  var scrollY = 0;
  function renderT(force) {
    if (!T.init) return;
    var run = runMode();
    var sig = run ? JSON.stringify([T.view, T.big, T.msg, T.ok, T.help], function (k, v) { return k === 'now' ? undefined : v; }) : 'form';
    if (!force && sig === T.sig) return;
    T.sig = sig;
    scrollY = window.scrollY || 0;
    var body;
    if (run) body = runScreen();
    else if (T.screen === 'pairs') body = pairsScreen();
    else if (T.screen === 'tabs') body = tabsScreen();
    else if (T.screen === 'students') body = studentsScreen();
    else if (T.screen === 'report') body = reportScreen();
    else body = homeScreen();
    if (T.help && body.className !== 'bigover') body.appendChild(teacherHelp());
    swap(body);
    window.scrollTo(0, scrollY);
  }
  function msgs() { return [T.msg ? h('div', { class: 'err' }, T.msg) : null, T.ok ? h('div', { class: 'okmsg' }, T.ok) : null]; }

    var box = h('div', { class: 'seg' });

      box.appendChild(h('button', { type: 'button', class: 'segb' + (T.set[key] === o[0] ? ' on' : ''), onclick: function () { T.set[key] = o[0]; renderT(true); } }, o[1]));

  function stepper(key, min, max, unit) {
    var wrap = h('div', { class: 'step' });
    function build() {
      var inp = h('input', { type: 'text', inputmode: 'numeric', value: String(T.set[key]) });
      function put(v) { v = parseInt(v, 10); if (isNaN(v)) v = T.set[key]; T.set[key] = Math.max(min, Math.min(max, v)); wrap.replaceChildren.apply(wrap, build()); }
      inp.addEventListener('change', function () { put(inp.value); });
      return [
        h('button', { type: 'button', 'aria-label': '1へらす', onclick: function () { put(T.set[key] - 1); } }, '▽'),
        inp,
        h('button', { type: 'button', 'aria-label': '1ふやす', onclick: function () { put(T.set[key] + 1); } }, '△'),
        unit ? h('span', { class: 'unit' }, unit) : null
      ].filter(Boolean);
    }
    wrap.replaceChildren.apply(wrap, build());
    return wrap;

  function readImg(file) {
    return new Promise(function (res, rej) {
      var fr = new FileReader();
      fr.onload = function () {
        var im = new Image();
        im.onload = function () {
          var m = 480, s = Math.min(1, m / Math.max(im.width, im.height)), c = document.createElement('canvas');
          c.width = Math.max(1, Math.round(im.width * s)); c.height = Math.max(1, Math.round(im.height * s));
          var g = c.getContext('2d'); g.fillStyle = '#fff'; g.fillRect(0, 0, c.width, c.height); g.drawImage(im, 0, 0, c.width, c.height);
          res(c.toDataURL('image/jpeg', 0.82));
        };
        im.onerror = function () { rej(new Error('絵を読みこめませんでした')); };
        im.src = fr.result;
      };
      fr.onerror = function () { rej(new Error('絵を読みこめませんでした')); };
      fr.readAsDataURL(file);
    });
  }

    w.appendChild(h('div', { class: 'panel' }, h('div', { class: 'row' }, h('span', { class: 'chip' }, (st.index + 1) + ' / ' + st.total + '問'),
      st.limit > 0 ? h('span', { class: 'chip acc' }, '制限時間 ' + st.limit + '秒') : null),
      h('div', { class: 'qcard' }, ask ? h('div', { class: 'ask' }, ask + ' のペアになるのは？') : null, cardNode(q.item, false, true))));
    var grid = h('div', { class: 'choices' });
    q.choices.forEach(function (c, i) {
      grid.appendChild(h('div', { class: 'choice' + (reveal && i === v.correct ? ' right' : '') }, cardNode(c, false, true),
        reveal && i === v.correct ? h('span', { class: 'badge ok' }, 'こたえ') : null,
        reveal ? h('div', { class: 'cnt' }, v.counts[i] + '人') : null));
    });
    w.appendChild(grid);
    var done = v.answered.length, tot = v.people.length;
    var p2 = h('div', { class: 'panel', style: 'margin-top:12px' }, h('h2', null, done + ' / ' + tot + ' 人が答えました'));
    if (st.live === 'names') {
      v.people.forEach(function (n) { p2.appendChild(h('span', { class: 'pill' + (v.answered.indexOf(n) >= 0 ? ' done' : '') }, plain(n) + (v.answered.indexOf(n) >= 0 ? ' ✓' : ''))); });
    }
    w.appendChild(p2);
    var last = st.index + 1 >= st.total;
    w.appendChild(h('div', { class: 'row' },
      !reveal ? h('button', { type: 'button', class: 'btn big', onclick: function () { var pr = act('tReveal'); if (pr) pr.then(setView).catch(err); } }, '答えを見せる')
        : h('button', { type: 'button', class: 'btn big', onclick: function () { var pr = act('tNext'); if (pr) pr.then(setView).catch(err); } }, last ? '結果を見る' : 'つぎの問題へ'),
      h('button', { type: 'button', class: 'btn warn', onclick: function () { var pr = act('tFinish'); if (pr) pr.then(setView).catch(err); } }, 'ここで終わりにする')));
    return w;
  }
