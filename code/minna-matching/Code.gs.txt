  ss_().getSheets().forEach(function (sh) {

/* ---------- 保存（プロパティ＋キャッシュ） ---------- */

function props_() { return PropertiesService.getScriptProperties(); }
function cache_() { return CacheService.getScriptCache(); }

function load_(key, def) {
  const c = cache_().get('k_' + key);
  if (c) return JSON.parse(c);
  const p = props_().getProperty(key);
  if (p) {
    try { cache_().put('k_' + key, p, 21600); } catch (e) {}
    return JSON.parse(p);
  }
  return def;
}
function save_(key, obj) {
  const s = JSON.stringify(obj);
  props_().setProperty(key, s);
  try { cache_().put('k_' + key, s, 21600); } catch (e) {}
}
// 大きいデータ（問題の一覧）は分けて保存する
function saveBig_(key, obj) {
  const s = JSON.stringify(obj);
  const size = 2500;
  const n = Math.max(1, Math.ceil(s.length / size));
  const o = {};
  for (let i = 0; i < n; i++) o[key + '_' + i] = s.slice(i * size, (i + 1) * size);
  o[key + '_n'] = String(n);
  props_().setProperties(o);
  try { cache_().put('k_' + key, s, 21600); } catch (e) {}
}
function loadBig_(key, def) {
  const c = cache_().get('k_' + key);
  if (c) return JSON.parse(c);
  const n = parseInt(props_().getProperty(key + '_n') || '0', 10);
  if (!n) return def;
  let s = '';
  for (let i = 0; i < n; i++) s += props_().getProperty(key + '_' + i) || '';
  try { cache_().put('k_' + key, s, 21600); } catch (e) {}
  return JSON.parse(s);
}
function withLock_(fn, ms) {
  const lock = LockService.getScriptLock();
  lock.waitLock(ms || 15000);
  try { return fn(); } finally { lock.releaseLock(); }
}

function keyMap_(keyOf, cids, sure) {
  const m = {};
  let all = null;
  for (let j = 0; j < cids.length; j += 100) {
    const part = cids.slice(j, j + 100);
    const keys = part.map(function (c) { return 'k_' + keyOf(c); });
    const got = cache_().getAll(keys);
    part.forEach(function (c, k) {
      let v = got[keys[k]];
      if (!v && sure) { if (!all) all = props_().getProperties(); v = all[keyOf(c)]; }
      if (v) { try { m[c] = JSON.parse(v); } catch (e) {} }
    });
  }
  return m;
}
function ansMap_(st, people, sure) { return keyMap_(function (c) { return ansKey_(st, st.index, c); }, Object.keys(people), sure); }

// 見出しの名前で列をさがす（ない見出しは、見つからないだけ。書きかえない）
function colsOf_(sh, F) {
  const lc = Math.max(sh.getLastColumn(), 1);
  const head = sh.getRange(1, 1, 1, lc).getValues()[0].map(cell_);
  const idx = {};
  let width = lc;
  F.forEach(function (f) {
    let c = -1;
    for (let k = 0; k < f[1].length && c < 0; k++) c = head.indexOf(f[1][k]);
    if (c >= 0) { idx[f[0]] = c; width = Math.max(width, c + 1); }
  });
  return { idx: idx, width: width, head: head };
}
// ない見出しを右に足す（前の版のタブでも、そのまま動く）
function ensureCols_(sh, F) {
  const c = colsOf_(sh, F);
  const add = F.filter(function (f) { return c.idx[f[0]] === undefined; });
  if (add.length) {
    const empty = c.head.every(function (h) { return !h; });
    const start = empty ? 0 : c.head.length;
    sh.getRange(1, start + 1, 1, add.length).setValues([add.map(function (f) { return f[1][0]; })]).setFontWeight('bold');
    if (empty) sh.setFrozenRows(1);
    return colsOf_(sh, F);
  }
  return c;
}
function sheetWith_(name, F) {
  let sh = ss_().getSheetByName(name);
  if (!sh) sh = ss_().insertSheet(name);
  ensureCols_(sh, F);
  return sh;
}
function rowOf_(c, obj) {
  const r = new Array(c.width).fill('');
  Object.keys(obj).forEach(function (k) { if (c.idx[k] !== undefined) r[c.idx[k]] = obj[k]; });
  return r;
}

    const rows = objs.map(function (o) { return rowOf_(c, o); });
    sh.getRange(sh.getLastRow() + 1, 1, rows.length, c.width).setValues(rows);
  }, 20000);
}

function readPairs_(tab) {
  const sh = ss_().getSheetByName(tab);
  if (!sh) throw new Error('タブ「' + tab + '」がありません');

  const out = [];
  for (let r = 1; r < v.length; r++) {

/* ---------- 問題をつくる ---------- */

function shuffle_(a, rnd) {
  rnd = rnd || Math.random;
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rnd() * (i + 1));
    const t = a[i]; a[i] = a[j]; a[j] = t;
  }
  return a;
}
function hash_(s) {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 16777619); }
  return h >>> 0;
}
// 同じ種（たね）からは、いつも同じ並びになる乱数
function rng_(seed) {
  let a = hash_(String(seed));
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
// 絵・文字を出すかどうか。絵を出さないのに文字がないときは絵を、どちらも出さないときは文字を出す
function pub_(it, showImg, showText) {
  let img = (showImg !== false || !it.text) ? it.img : '';
  let text = (showText || !img) ? it.text : '';
  if (!img && !text) img = it.img;
  return { img: img, text: text };
}
function dkey_(it, set) { const p = pub_(it, set.imgC, set.textC); return p.img + '|' + p.text; }

// 1問ぶん：問題カード q、正解 a、まちがい w（いちばん多いレベルの数まで）。選択肢のならべは、あとで choicesFor_ で決める
function buildQs_(rows, info, set, rnd) {
  rnd = rnd || Math.random;
  const full = shuffle_(rows.filter(function (r) { return has_(r.a) && has_(r.b); }), rnd);
  const qs = [];
  const want = Math.max(1, parseInt(set.count, 10) || 10);
  const maxW = LV_N[LV_N.length - 1] - 1;
  for (let i = 0; i < full.length && qs.length < want; i++) {
    const p = full[i];
    let qSide;
    if (info.split && set.dir === 'ab') qSide = 'a';
    else if (info.split && set.dir === 'ba') qSide = 'b';
    else qSide = rnd() < 0.5 ? 'a' : 'b';
    const oSide = qSide === 'a' ? 'b' : 'a';
    const ans = p[oSide];
    let pool = [];
    rows.forEach(function (r) {
      if (r.row === p.row) return;
      const dm = !(has_(r.a) && has_(r.b));
      if (info.split) { if (has_(r[oSide])) pool.push({ img: r[oSide].img, text: r[oSide].text, dm: dm ? 1 : 0 }); }
      else {
        if (has_(r.a)) pool.push({ img: r.a.img, text: r.a.text, dm: dm ? 1 : 0 });
        if (has_(r.b)) pool.push({ img: r.b.img, text: r.b.text, dm: dm ? 1 : 0 });
      }
    });
    // 見た目が同じもの（正解・問題・ほかの選択肢と）は選択肢に入れない
    const seen = {};
    seen[dkey_(ans, set)] = 1;
    seen[dkey_(p[qSide], set)] = 1;
    pool = shuffle_(pool, rnd).filter(function (it) { const k = dkey_(it, set); if (seen[k]) return false; seen[k] = 1; return true; });
    const wrong = pool.slice(0, maxW);
    if (!wrong.length) continue;
    qs.push({
      q: { img: p[qSide].img, text: p[qSide].text }, qs: qSide,
      a: { img: ans.img, text: ans.text }, w: wrong
    });
  }
  return qs;
}
// n この選択肢をならべる（同じ seed なら、いつも同じならび）
function choicesFor_(q, n, seed) {
  const k = Math.max(2, Math.min(n, 1 + q.w.length));
  const items = [{ img: q.a.img, text: q.a.text, ok: 1, dm: 0 }]
    .concat(q.w.slice(0, k - 1).map(function (x) { return { img: x.img, text: x.text, ok: 0, dm: x.dm ? 1 : 0 }; }));
  const arr = shuffle_(items, rng_(seed));
  let correct = 0;
  arr.forEach(function (x, i) { if (x.ok) correct = i; });
  return { choices: arr, correct: correct };
}
function outChoices_(ch, st) { return ch.choices.map(function (c) { return pub_(c, st.imgC, st.textC); }); }

function whyOf_(ch, c) { return c < 0 ? WHY.none : (ch.choices[c].dm ? WHY.dummy : WHY.other); }
function topType_(tp) {
  const ks = Object.keys(tp || {}).sort(function (a, b) { return tp[b] - tp[a]; });
  return ks.length ? ks[0] : '';
}

function sPoll(cid, code) {
  const set = settings_();
  const st = state_();
  const o = {
    round: st.round || '', phase: st.phase, index: st.index, total: st.total, sound: st.sound !== false,
    hira: !!set.hira, pace: st.pace || 'together', now: Date.now()
  };
  if (!okCode_(set, code)) { o.needCode = true; return o; }
  let people = load_('people', {});
  if (!people[cid] && set.nameMode === 'guest') { guestJoin_(cid); people = load_('people', {}); }
  o.myName = people[cid] || '';
  if (!o.myName) {
    const taken = {};
    Object.keys(people).forEach(function (k) { if (k !== cid) taken[people[k]] = 1; });
    o.names = load_('names', []).map(function (n) { return { n: n, t: !!taken[n] }; });
    return o;
  }
  if (st.pace !== 'self') {
    if (st.phase === 'q' || st.phase === 'reveal') {
      const q = loadBig_('qs', [])[st.index];
      if (q) {
        const ch = choicesFor_(q, st.nCh, st.round + ':' + st.index);
        o.q = {
          item: pub_(q.q, st.imgQ, st.textQ), choices: outChoices_(ch, st),
          openedAt: st.openedAt, limit: st.limit, split: st.split, la: st.la, lb: st.lb, side: q.qs
        };
        const a = load_(ansKey_(st, st.index, cid), null);
        o.mine = a ? a.c : -1;
        o.changed = a ? a.n : 0;
        if (st.phase === 'reveal') o.correct = ch.correct;
      }
    }
    if (st.phase === 'end') {
    }
    return o;
  }
  // ひとりずつ
  if (st.phase === 'run' || st.phase === 'end') {
    let me = load_(meKey_(st, cid), null);
    if (!me && st.phase === 'run') { me = newMe_(st, o.myName); save_(meKey_(st, cid), me); }
    if (me) {
      o.me = { i: me.i, lv: me.lv, ok: me.ok, n: me.n, ph: me.ph, fb: me.fb, nextLv: me.nl, done: me.i >= st.total, auto: autoOf_(st, me), nextN: nOfLv_(me.nl), curN: nOfLv_(me.lv) };
      if (!o.me.done) {
        const q = loadBig_('qs', [])[me.i];
        if (q) {
          const nC = nOfLv_(me.lv);
          const ch = choicesFor_(q, nC, seedSelf_(cid, st, me.i, me.lv));
          o.q = {
            item: pub_(q.q, st.imgQ, st.textQ), choices: outChoices_(ch, st),
            openedAt: me.at, limit: st.limit, split: st.split, la: st.la, lb: st.lb, side: q.qs
          };
          o.index = me.i;
          o.mine = me.c;
          if (me.ph === 'f') o.correct = ch.correct;
        }
      }
    }
  }
  return o;
}

// みんなで：1回だけ変えられる
function sAnswer(cid, code, choice, round, index) {
  const set = settings_();
  if (!okCode_(set, code)) throw new Error('参加《さんか》コードが^ちがうよ');
  const st = state_();
  if (st.pace === 'self') throw new Error('いまは^答《こた》えられないよ');
  if (st.phase !== 'q' || (round != null && st.round !== round) || (index != null && st.index !== index)) {
    throw new Error('もう^次《つぎ》の^問題《もんだい》に^なったよ');
  }
  if (st.limit > 0 && Date.now() > st.openedAt + st.limit * 1000 + 1500) throw new Error('時間《じかん》が^終《お》わったよ');
  if (!load_('people', {})[cid]) throw new Error('名前《なまえ》を^選《えら》んでね');
  const key = ansKey_(st, st.index, cid);
  const old = load_(key, null);
  if (old && old.c === choice) return { mine: old.c, changed: old.n };
  if (old && old.n >= 1) throw new Error('もう^変《か》えられないよ');
  const a = { c: choice, n: old ? 1 : 0 };
  save_(key, a);
  return { mine: choice, changed: a.n };
}

// ひとりずつ：答えると、正解かどうかとレベルの動きを返す（同じ問題への2回めの送信は、前の答えを返すだけ）
function sAnswerSelf(cid, code, round, i, c) {
  const set = settings_();
  if (!okCode_(set, code)) throw new Error('参加《さんか》コードが^ちがうよ');
  const st = state_();
  if (st.pace !== 'self' || st.phase !== 'run' || st.round !== round) throw new Error('いまは^答《こた》えられないよ');
  const name = load_('people', {})[cid];
  if (!name) throw new Error('名前《なまえ》を^選《えら》んでね');
  const key = meKey_(st, cid);
  const me = load_(key, null);
  if (!me) throw new Error('もういちど^ひらいてね');
  const qs = loadBig_('qs', []);
  const q = qs[me.i];
  const ch = choicesFor_(q, nOfLv_(me.lv), seedSelf_(cid, st, me.i, me.lv));
  if (me.i !== i || me.ph !== 'a') return { ok: me.c === ch.correct, correct: ch.correct, mine: me.c, fb: me.fb, nextLv: me.nl };
  c = parseInt(c, 10);
  if (isNaN(c) || c >= ch.choices.length) throw new Error('もういちど^選《えら》んでね');
  if (st.limit > 0 && Date.now() > me.at + st.limit * 1000 + 1500) c = -1;
  if (c < 0 && !(st.limit > 0 && Date.now() >= me.at + st.limit * 1000 - 800)) throw new Error('もういちど^選《えら》んでね');
  const ok = c === ch.correct;
  me.n++;
  if (ok) me.ok++; else { const w = whyOf_(ch, c); me.tp[w] = (me.tp[w] || 0) + 1; }
  me.lg.push([me.i, me.lv, c, Date.now()]);
  me.c = c;
  me.ph = 'f';
  // レベルの自動上下（記録には、そのときのレベルを残す。生徒設定のレベルは書きかえない）
  me.fb = '';
  me.nl = me.lv;
  if (ok) { me.so++; me.sn = 0; } else { me.sn++; me.so = 0; }
  if (autoOf_(st, me)) {
    if (ok && me.so >= st.upN && me.lv < LV_N.length) { me.nl = me.lv + 1; me.so = 0; me.fb = 'up'; }
    else if (!ok && me.sn >= st.downN && me.lv > 1) { me.nl = me.lv - 1; me.sn = 0; me.fb = 'down'; }
    if (me.fb) me.ch.push([me.i + 1, me.lv, me.nl]);
  }
  save_(key, me);
  if (me.i + 1 >= st.total) {   // 最後の問題：この人の記録をまとめて書く
    try { flushMe_(st, qs, cid, name, me); } catch (e) {}
  }
  return { ok: ok, correct: ch.correct, mine: c, fb: me.fb, nextLv: me.nl };
}
function sNext(cid, code, round) {
  const set = settings_();
  if (!okCode_(set, code)) throw new Error('参加《さんか》コードが^ちがうよ');
  const st = state_();
  if (st.pace !== 'self' || st.round !== round) return false;
  const key = meKey_(st, cid);
  const me = load_(key, null);
  if (!me || me.ph !== 'f') return false;
  me.i++;
  me.lv = me.nl;
  me.ph = 'a';
  me.c = -1;
  me.fb = '';
  me.at = Date.now();
  save_(key, me);
  return true;
}

/* 記録の書きだし（シートに書くのは、ロックの外で行う） */
function recsOfMe_(st, qs, name, me) {
  return me.lg.slice(me.fl).map(function (g) {
    const q = qs[g[0]];
    const ch = choicesFor_(q, nOfLv_(g[1]), seedSelf_(me.cid, st, g[0], g[1]));
    const c = g[2];
    const ok = c === ch.correct;
    return {
      date: new Date(g[3]), round: st.round, mode: 'ひとりずつ', no: g[0] + 1, name: name, q: lab_(q.q),
      right: lab_(ch.choices[ch.correct]), given: c >= 0 ? lab_(ch.choices[c]) : '', res: c < 0 ? '−' : (ok ? '○' : '×'),
      level: g[1], why: ok ? '' : whyOf_(ch, c)
    };
  });
}
function flushMe_(st, qs, cid, name, me) {
  me.cid = cid;
  const recs = recsOfMe_(st, qs, name, me);
  delete me.cid;
  if (!recs.length) return;
  writeRec_(st.tab, recs);
  me.fl = me.lg.length;
  save_(meKey_(st, cid), me);
}

  if (st.pace !== 'self' && (st.phase === 'q' || st.phase === 'reveal')) {
    const q = loadBig_('qs', [])[st.index];
    const ch = choicesFor_(q, st.nCh, st.round + ':' + st.index);
    const ans = ansMap_(st, people, false);
    const done = {};
    Object.keys(ans).forEach(function (cid) { if (people[cid]) done[people[cid]] = 1; });
    v.q = { item: pub_(q.q, st.imgQ, st.textQ), choices: outChoices_(ch, st), side: q.qs };
    v.answered = Object.keys(done);
    v.counts = ch.choices.map(function () { return 0; });
    Object.keys(ans).forEach(function (cid) { if (people[cid] && v.counts[ans[cid].c] != null) v.counts[ans[cid].c]++; });
    if (st.phase === 'reveal') v.correct = ch.correct;
  }
  if (st.pace === 'self' && (st.phase === 'run' || st.phase === 'end')) {
    const mm = meMap_(st, people, st.phase === 'end');
    v.rows = Object.keys(people).map(function (cid) {
      const me = mm[cid];
      return {
        name: people[cid], joined: !!me, n: me ? me.n : 0, ok: me ? me.ok : 0, lv: me ? me.lv : 0, l0: me ? me.l0 : 0,
        ch: me ? me.ch : [], done: me ? finished_(st, me) : false, top: me ? topType_(me.tp) : ''
      };
    });
  }
  if (st.phase === 'end') {
    if (st.pace === 'self') v.scores = v.rows.filter(function (r) { return r.joined; }).map(function (r) { return { name: r.name, ok: r.ok, ans: r.n }; });
    else { const sc = load_('scores', {}); v.scores = Object.keys(sc).map(function (k) { return sc[k]; }); }
  }
  return v;
}
function tView() { return view_(); }

function tStart(set) {
  const s = cleanSet_(Object.assign(settings_(), set || {}));
  save_('set', s);
  const info = infoOf_(tabInfo_(), s.tab);
  const qs = buildQs_(readPairs_(s.tab), info, s);
  if (!qs.length) throw new Error('ペアが足りません（2組以上必要です）');
  save_('names', namesFor_(s));
  const lvmap = stuMap_();
  const automap = autoMap_();
  return withLock_(function () {
    const old = state_();
    const pr = props_();
    pr.getKeys().forEach(function (k) {
      if (k.indexOf('ans:') === 0 || k.indexOf('me:') === 0 || k.indexOf('ans_') === 0) { pr.deleteProperty(k); cache_().remove('k_' + k); }
    });
    saveBig_('qs', qs);
    save_('scores', {});
    save_('lvmap', lvmap);
    save_('automap', automap);
    let id = newRoundId_();
    if (id === old.round) id = id + 'b';
    save_('state', {
      round: id, phase: 'wait', index: 0, total: qs.length, openedAt: 0, pace: s.pace, nCh: s.nChoices,
      limit: s.limit, live: s.live || 'num', imgQ: s.imgQ, imgC: s.imgC, textQ: s.textQ, textC: s.textC,
      sound: s.sound, tab: s.tab, split: info.split, la: info.la, lb: info.lb,
      auto: s.auto, upN: s.upN, downN: s.downN, sumDone: false
    });
    return view_();
  });
}

function open_(st, i) {
  st.phase = 'q';
  st.index = i;
  st.openedAt = Date.now();
  save_('state', st);
}
function tOpen() {
  return withLock_(function () {
    const st = state_();
    if (st.phase === 'wait') {
      if (st.pace === 'self') { st.phase = 'run'; st.openedAt = Date.now(); save_('state', st); }
      else open_(st, 0);
    }
    return view_();
  });
}

// 終わるときに、記録とまとめを書く材料をそろえる（ロックの中で呼ぶ。シートに書くのは、そのあと）
function collectEnd_(st) {
  const people = load_('people', {});
  const qs = loadBig_('qs', []);
  const job = { tab: st.tab, recs: [], sums: [] };
  const sum = function (name, n, ok, lv, tp) {
    return {
      date: new Date(), round: st.round, mode: st.pace === 'self' ? 'ひとりずつ' : 'みんなで', tab: st.tab, name: name,
      total: st.total, n: n, ok: ok, rate: n ? Math.round(ok / n * 100) + '%' : '', lv: lv, top: topType_(tp)
    };
  };
  if (st.pace === 'self') {
    const mm = meMap_(st, people, true);
    Object.keys(people).forEach(function (cid) {
      const me = mm[cid];
      if (!me) return;
      me.cid = cid;
      job.recs = job.recs.concat(recsOfMe_(st, qs, people[cid], me));
      delete me.cid;
      me.fl = me.lg.length;
      save_(meKey_(st, cid), me);
      job.sums.push(sum(people[cid], me.n, me.ok, me.ph === 'f' ? me.nl : me.lv, me.tp));
    });
  } else {
    const sc = load_('scores', {});
    Object.keys(sc).forEach(function (cid) { job.sums.push(sum(sc[cid].name, sc[cid].ans, sc[cid].ok, '', sc[cid].tp)); });
  }
  return job;
}
function writeJob_(job) {
  if (!job) return;
  try { writeRec_(job.tab, job.recs); } catch (e) {}
  try { writeSum_(job.sums); } catch (e) {}
}
function endNow_(st) {
  let job = null;
  if (!st.sumDone && (st.phase === 'q' || st.phase === 'reveal' || st.phase === 'run')) job = collectEnd_(st);
  st.phase = 'end';
  st.sumDone = true;
  save_('state', st);
  return job;
}

function tNext() {
  let job = null;
  const v = withLock_(function () {
    const st = state_();
    if (st.phase !== 'reveal' || st.pace === 'self') return view_();
    if (st.index + 1 < st.total) open_(st, st.index + 1);
    else job = endNow_(st);
    return view_();
  });
  writeJob_(job);
  return v;
}
function tFinish() {
  let job = null;
  const v = withLock_(function () {
    const st = state_();
    if (st.phase === 'idle' || st.phase === 'end') return view_();
    job = endNow_(st);
    return view_();
  });
  writeJob_(job);
  return v;
}
function tStop() {
  return withLock_(function () { const st = state_(); st.phase = 'idle'; save_('state', st); return view_(); });
}

// みんなで：答えを見せて、記録を書く
function tReveal() {
  let rows = [], tab = '';
  const v = withLock_(function () {
    const st = state_();
    if (st.phase !== 'q' || st.pace === 'self') return view_();
    const q = loadBig_('qs', [])[st.index];
    const ch = choicesFor_(q, st.nCh, st.round + ':' + st.index);
    const people = load_('people', {});
    const ans = ansMap_(st, people, true);
    const sc = load_('scores', {});
    const now = new Date();
    tab = st.tab;
    Object.keys(people).forEach(function (cid) {
      const a = ans[cid];
      const ok = !!a && a.c === ch.correct;
      const s = sc[cid] || { name: people[cid], ok: 0, ans: 0, tp: {} };
      s.name = people[cid];
      if (a) s.ans++;
      if (ok) s.ok++;
      let why = '';
      if (!ok) { why = whyOf_(ch, a ? a.c : -1); s.tp = s.tp || {}; s.tp[why] = (s.tp[why] || 0) + 1; }
      sc[cid] = s;
      rows.push({
        date: now, round: st.round, mode: 'みんなで', no: st.index + 1, name: people[cid], q: lab_(q.q),
        right: lab_(ch.choices[ch.correct]), given: a ? lab_(ch.choices[a.c]) : '', res: a ? (ok ? '○' : '×') : '−',
        level: '', why: why
      });
    });
    save_('scores', sc);
    st.phase = 'reveal';
    save_('state', st);
    return view_();
  });
  try { writeRec_(tab, rows); } catch (e) {}   // 記録はロックの外で書く
  return v;
}

/* ---------- 先生：記録を見る ---------- */

function tReport(todayOnly) {
  const per = {}, order = [], types = {};
  const today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyyMMdd');
  ss_().getSheets().forEach(function (sh) {
    if (!isRecTab_(sh.getName())) return;
    const last = sh.getLastRow();
    if (last < 2) return;
    const c = colsOf_(sh, REC_F);
    if (c.idx.name === undefined) return;
    const start = Math.max(2, last - 2999);
    const v = sh.getRange(start, 1, last - start + 1, Math.max(sh.getLastColumn(), 1)).getValues();
    v.forEach(function (r) {
      const name = cell_(r[c.idx.name]);
      if (!name) return;
      if (todayOnly) {
        const d = c.idx.date !== undefined ? r[c.idx.date] : null;
        if (!(d instanceof Date) || Utilities.formatDate(d, 'Asia/Tokyo', 'yyyyMMdd') !== today) return;
      }
      if (!per[name]) { per[name] = { n: 0, ok: 0, tp: {} }; order.push(name); }
      const o = per[name];
      const res = cell_(c.idx.res !== undefined ? r[c.idx.res] : '');
      o.n++;
      if (res === '○') o.ok++;
      else {
        const t = (c.idx.why !== undefined ? cell_(r[c.idx.why]) : '') || (res === '−' ? WHY.none : '');
        if (t) { o.tp[t] = (o.tp[t] || 0) + 1; types[t] = (types[t] || 0) + 1; }
      }
    });
  });
  const top = function (m) { return Object.keys(m).sort(function (a, b) { return m[b] - m[a]; }).map(function (k) { return { type: k, n: m[k] }; }); };
  return {
    rows: order.map(function (n) { return { name: n, n: per[n].n, ok: per[n].ok, types: top(per[n].tp).slice(0, 3) }; }),
    types: top(types)
  };
}
