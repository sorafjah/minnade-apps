function niceScale(min,max,zero){
  if(zero&&min>0)min=0;var steps=[1,2,5,10,20,50];
  for(var k=0;k<steps.length;k++){var s=steps[k],lo=Math.floor(min/s)*s,hi=Math.ceil(max/s)*s;if(hi===lo)hi+=s;
    if((hi-lo)/s<=8){if((hi-lo)/s<4)hi+=s;return {lo:lo,hi:hi,step:s};}}
  return {lo:0,hi:Math.ceil(max/100)*100,step:100};
}

// 正しいグラフ1つと、1か所だけちがうグラフ2つ（レベル3の2本のグラフは、もようが入れかわったものもある）
function graphQ(o){
  var all=[].concat.apply([],o.series),neg=Math.min.apply(null,all)<0,type=(o.type==='bar'&&!neg)?'bar':'line';
  var sc=niceScale(Math.min.apply(null,all),Math.max.apply(null,all),type==='bar'||o.zero);
  var ks=o.L===1?[2,3]:o.L===2?[1.5,2]:[1,1.5],vs=[{series:o.series,swap:false,ok:true}],used={},guard=0;
  if(o.series.length===2&&o.L===3)vs.push({series:o.series,swap:true});
  while(vs.length<3&&guard++<300){
    var j=ri(0,o.series.length-1),i=ri(0,o.series[0].length-1);if(used[j+'_'+i])continue;
    var d=pick(ks)*sc.step*(rnd()<0.5?1:-1),v=o.series[j][i]+d;
    if(v<sc.lo||v>sc.hi)v=o.series[j][i]-d;
    if(v<sc.lo||v>sc.hi)continue;
    used[j+'_'+i]=1;var ns=o.series.map(function(a){return a.slice();});ns[j][i]=v;vs.push({series:ns,swap:false});
  }
  if(vs.length<3)return null;
  return {type:type,items:vs.map(function(x){return {ok:!!x.ok,ty:x.ok?'':(x.swap?'グラフのもようの{取|と}りちがえ':'グラフの{読|よ}みまちがい'),
    s:chart({type:type,labels:o.labels,series:x.series,swap:x.swap,unit:o.unit,xunit:o.xunit,sc:sc,vert:o.vert})};})};
}

  function fakeStudents() {
    if (S.timer) clearInterval(S.timer);
    var r = S.round, s = r.settings, sets = {};
    var names = s.join === 'roster' ? ((ROSTERS.filter(function (x) { return x.title === s.roster; })[0] || ROSTERS[0]).names.slice(0, 4)) : s.join === 'chara' ? ['くま', 'うさぎ', 'ねこ'] : ['g1', 'g2', 'g3'];
    var fakes = names.map(function (n, i) { return { name: n, joinAt: Date.now() + 1200 + i * 1100, joined: false, nextAt: 0, skill: 0.45 + Math.random() * 0.5 }; });
    S.timer = setInterval(function () {
      if (S.round !== r || r.status !== 'open') { clearInterval(S.timer); S.timer = null; return; }
      var now = Date.now();
      fakes.forEach(function (f) {
        if (!f.joined) {
          if (now >= f.joinAt) {
            var j = join({ name: f.name, tok: 'fake-' + f.name, code: r.code });
            f.joined = !!j.ok; f.name = j.name || f.name; f.nextAt = now + 2000;
            if (s.mode === 'self') { var c = j.conf || {}; f.sim = { level: c.level || c.last || s.level, start: c.level || c.last || s.level, count: 0, correct: 0, setNo: 1, qi: 0 }; }
          }
          return;
        }
        S.seen[key(r.id, f.name)] = now;
        if (now < f.nextAt) return;
        f.nextAt = now + 2500 + Math.random() * 4000;
        if (s.mode === 'self') { stepSelf(f, r); return; }
        var gi = r.curQ;
        if (gi < 0) return;
        var st = S.st[key(r.id, f.name)];
        if (st && st.ans[gi]) return;
        var si = Math.floor(gi / QPER), set = sets[si] || (sets[si] = makeSet(r.seed, s.genres, s.level, s.graph, si, s.answer)), q = set.qs[gi % QPER];
        var ok = Math.random() < f.skill, pick, ml = '';
        if (q.kind === 'num') { pick = ok ? numText(q.num.v, q.num.dec) + noRuby(q.num.u) : numText(q.num.v + 1, q.num.dec) + noRuby(q.num.u); ml = ok ? '' : '近い数とまちがえた'; }
        else {
          var idx = q.ok;
          if (!ok) { var o = q.opts.map(function (c, i) { return i; }).filter(function (i) { return i !== q.ok; }); idx = o[Math.floor(Math.random() * o.length)]; ml = q.ty[idx]; }
          pick = pickText(q, idx);
        }
        submit([{ id: uid(), kind: 'answer', roundId: r.id, name: f.name, mode: 'all', genre: set.id, level: set.level, qi: gi, ok: ok, pick: pick, ml: ml }]);
      });
    }, 1000);
  }
  function stepSelf(f, r) {
    var sim = f.sim, ok = Math.random() < f.skill;
    sim.count++; if (ok) sim.correct++;
    sim.qi++; if (sim.qi >= QPER) { sim.qi = 0; sim.setNo++; if (ok && sim.level < 3 && Math.random() < 0.5) sim.level++; }
    var remain = r.settings.timeOn ? Math.max(0, r.settings.minutes * 60 - Math.round((Date.now() - r.startedAt) / 1000)) : null;
    submit([{ id: uid(), kind: 'snap', roundId: r.id, name: f.name, snap: { state: 'playing', genre: genreAt(r.settings.genres, f.name, sim.setNo - 1), setNo: sim.setNo, qi: sim.qi, count: sim.count, correct: sim.correct, start: sim.start, level: sim.level, remain: remain, top: ok ? '' : '以上・未満のまちがい' } }]);
  }

    },

      return { sums: S.sums.slice().reverse(), mist: mist };
    }
  };

  };

      else if (ST.view === 'result' && !rd.showRes) show('done');
    }
    var open = rd && rd.status === 'open';

/* --- 参加する --- */
function doJoin(name, code, prevId) {
  showWait('loading');
  return api('api_join', { name: name, code: code, tok: me.tok, prevId: prevId || '' }).then(function (r) {
    if (!r || !r.ok) {
      var err = r && r.err;
      if (err === 'code') showCode(true);
      else if (err === 'taken') return showName('taken');
      else showWait('wait');
      return;
    }
    var rd = r.round, same = me.roundId === rd.id;
    ST.round = rd; setRoundLook(rd);
    me.name = r.name;
    if (!same) { me.roundId = rd.id; me.gi = -1; me.res = []; me.fin = false; me.selfState = null; }
    clearInterval(clockTimer);
    swap('timer', '').classList.add('hidden');
    ST.set = null; ST.sets = {}; ST.srcKey = '';
    saveMe();
    if (rd.mode === 'all') {
      D = null; ST.all = true;
      afterGuide(function () { ST.view = ''; teacherStep(ST.round); });
      return;
    }
    ST.all = false;
    setupSelf(rd, r.conf || {});
    afterGuide(function () { showSelfReady(); });
  }).catch(function () { showWait('net'); });
}

function teacherStep(rd) {
  if (!ST.all) return;

  var target = Math.min(rd.curQ, rd.count * QPER - 1);
  if (ST.view === 'quiz' && ST.gi === target) return;
  if (ST.view === 'quiz' && !ST.answered && ST.gi < target) recordAll(false, '', '', true);
  var newTable = !ST.set || Math.floor(ST.gi / QPER) !== Math.floor(target / QPER);

  if (!set) { showWait('net'); return; }
  showQuestion(set, target % QPER);
  if (newTable && target > 0 && target % QPER === 0) banner('つぎの {表|ひょう}だよ', 1600);
}

function setupSelf(rd, conf) {
  var saved = me.selfState && me.selfState.rid === rd.id ? me.selfState : null;
  var lo = conf.lower || 1, up = conf.upper || 3;
  if (lo > up) { var t = lo; lo = up; up = t; }
  var start = conf.level || conf.last || rd.level;
  start = Math.max(lo, Math.min(up, start));
  D = { min: lo, max: up, auto: rd.autoLv && conf.auto !== false, level: start, start: start, okRun: 0, ngRun: 0, setNo: 0, setLevel: start, qi: 0, answered: false,
    count: 0, correct: 0, types: {}, gen: {}, res: [], endAt: 0, begun: false, timeUp: false, ended: false, finished: false };
  if (saved) Object.keys(saved).forEach(function (k) { if (k !== 'rid') D[k] = saved[k]; });
  D.finished = false; D.timeUp = false; D.ended = false;
}
function saveSelf() {
  if (!D || !me.roundId) return;
  var s = { rid: me.roundId };
  ['min', 'max', 'auto', 'level', 'start', 'okRun', 'ngRun', 'setNo', 'setLevel', 'qi', 'answered', 'count', 'correct', 'types', 'gen', 'res', 'endAt', 'begun'].forEach(function (k) { s[k] = D[k]; });

  saveMe();
}

function startSelf() {

function startClock() { clearInterval(clockTimer); clockTimer = setInterval(tickSelf, 250); tickSelf(); }
var lastShown = '';
function tickSelf() {
  if (!D || D.finished || !D.endAt) return;
  var left = Math.max(0, Math.ceil((D.endAt - Date.now()) / 1000));
  var txt = 'のこり ' + (left >= 60 ? Math.floor(left / 60) + '分' + (left % 60) + '秒' : left + '秒');
  if (ST.view === 'quiz' && txt !== lastShown) { lastShown = txt; swap('timer', txt).classList.remove('hidden'); }
  if (left <= 0 && !D.timeUp) { D.timeUp = true; stopSignal('じかんに なったよ。'); }
}
function stopSignal(text) {
  if (!D || D.finished) return;
  if (ST.view !== 'quiz') { finishSelf(); return; }
  if (ST.answered) { var nb = $('nextBtn'); if (nb) nb.innerHTML = R('おわる'); return; }
  banner(text + '<br>この {問題|もんだい}で おわりだよ', 2200);
}
function curSelfSet() {
  if (!ST.set && D.setNo > 0) ST.set = selfSet(D.setNo - 1, D.setLevel);
  return ST.set;
}
function nextSelf() {
  if (!D || D.finished) return;
  if (D.timeUp || D.ended) { finishSelf(); return; }
  var rd = ST.round, s = curSelfSet(), fresh = false;
  if (!s || (D.answered && D.qi >= QPER - 1)) {
    if (!rd.timeOn && D.setNo >= rd.count) { finishSelf(); return; }
    D.setLevel = D.level;
    ST.set = selfSet(D.setNo, D.level);
    if (!ST.set) { finishSelf(); return; }
    D.setNo++; D.qi = 0; fresh = D.setNo > 1;
  } else if (D.answered) D.qi++;
  D.answered = false;
  saveSelf();
  showQuestion(ST.set, D.qi);
  if (fresh) banner('つぎの {表|ひょう}だよ', 1500);
}
function afterSelf(ok, ml, pick) {
  var q = ST.q, set = ST.set, rd = ST.round;
  D.count++; if (ok) D.correct++;
  if (!ok && ml) D.types[ml] = (D.types[ml] || 0) + 1;
  var g = D.gen[set.id] || (D.gen[set.id] = { n: 0, c: 0 });
  g.n++; if (ok) g.c++;
  D.res.push({ no: (set.no + 1) + '-' + (D.qi + 1), ok: !!ok });
  if (D.res.length > 80) D.res = D.res.slice(-80);
  D.answered = true;
  outPush({
    id: uid(), kind: 'answer', at: Date.now(), roundId: me.roundId, name: me.name, mode: 'self', genre: set.id, level: set.level,
    no: (set.no + 1) + '-' + (D.qi + 1), q: noRuby(q.t), correct: noRuby(q.ans), pick: pick, ok: !!ok, ml: ok ? '' : ml,
    hint: ST.hint, how: howName(q), sec: Math.round((Date.now() - ST.qStart) / 1000)
  });
  // レベルの自動上下（つぎの表から。生徒の画面には何も出さない）
  if (ok) { D.okRun++; D.ngRun = 0; } else { D.ngRun++; D.okRun = 0; }
  if (D.auto) {
    if (D.okRun >= rd.up && D.level < D.max) { D.level++; D.okRun = 0; }
    else if (D.ngRun >= rd.down && D.level > D.min) { D.level--; D.ngRun = 0; }
  }
  saveSelf();
  sendSnap();
}
function topType(types) {
  var top = '', tn = 0;
  Object.keys(types || {}).forEach(function (t) { if (types[t] > tn) { top = t; tn = types[t]; } });
  return top ? top + '（' + tn + '回）' : '';
}

  };
}

}

function bindNum(area, n) {
  var inp = area.querySelector('#numIn');
  function clamp(v) {
    v = +v.toFixed(n.dec);
    if (n.min != null && v < n.min) v = n.min;
    if (n.max != null && v > n.max) v = n.max;
    return v;
  }
  function put(v) { ST.num = v; inp.value = v == null ? '' : (v < 0 ? '-' : '') + Math.abs(v).toFixed(n.dec); swap('numMsg', ''); }
  area.querySelectorAll('.sbtn').forEach(function (b) {
    b.addEventListener('click', function () {
      if (ST.answered) return;
      var cur = parseNum(inp.value), d = +b.dataset.d;
      put(clamp((cur == null ? 0 : cur) + d));
    });
  });
  var sg = area.querySelector('#signBtn');
  if (sg) sg.addEventListener('click', function () { if (ST.answered) return; var cur = parseNum(inp.value); if (cur == null || cur === 0) return; put(clamp(-cur)); });
  inp.addEventListener('input', function () { ST.num = parseNum(inp.value); swap('numMsg', ''); });
  inp.addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); goNum(); } });
  $('goBtn').addEventListener('click', function () { unlockAudio(); goNum(); });
}
function goNum() {
  if (ST.answered) return;
  var inp = $('numIn'), v = parseNum(inp ? inp.value : '');
  if (v == null) { swap('numMsg', R('{数|かず}を {入|い}れてね')); return; }
  var n = ST.q.num, ok = numText(v, n.dec) === numText(n.v, n.dec), ml = '';
  if (!ok) {
    n.ws.forEach(function (w) { if (!ml && numText(w[0], n.dec) === numText(v, n.dec)) ml = noRuby(w[1]); });
    if (!ml && numText(-v, n.dec) === numText(n.v, n.dec)) ml = 'マイナスのまちがい';
    if (!ml) ml = 'そのほか';
  }
  if (inp) inp.blur();
  answer(ok, ml, numText(v, n.dec) + noRuby(n.u), -1);
}

function answer(ok, ml, pick, idx) {
  if (ST.answered) return;
  ST.answered = true;
  var q = ST.q, rd = ST.round;
  if (q.kind !== 'num') {
    document.querySelectorAll('#answerArea .choice').forEach(function (b, k) {
      b.disabled = true;
      if (k === q.ok) { b.classList.add('is-correct'); b.querySelector('.cmark').innerHTML = R('せいかい'); }
      else if (k === idx) { b.classList.add('is-wrong'); b.querySelector('.cmark').innerHTML = R('{選|えら}んだもの'); }
      else b.classList.add('dim');
    });
  } else document.querySelectorAll('#answerArea button, #answerArea input').forEach(function (b) { b.disabled = true; });
  ok ? sndOK() : sndNG();
  var at = noRuby(q.ans), html;
  var ansLine = q.kind === 'num' || at.length <= 14 ? '「' + q.ans + '」' : '「せいかい」と {書|か}いて ある もの';
  if (ok) html = '<div class="fb-title ok">せいかい！</div>' + (q.kind === 'num' ? '<div>こたえは ' + q.ans + ' だよ。</div>' : '');
  else html = '<div class="fb-title ng">おしい！ せいかいは ' + ansLine + ' だよ。</div>';
  if (q.why) html += '<div>' + q.why + '</div>';
  if (q.L && q.L.length) { drawSrc(q.L, true); html += '<div class="note">' + R('{表|ひょう}の {色|いろ}が ついた ところを {見|み}てね。') + '</div>'; }
  swap('hintArea', '');
  if (rd.mode === 'all') { recordAll(ok, pick, ml, false); finishFeedback(html); return; }
  afterSelf(ok, ml, pick);
  finishFeedback(html);
}
function finishFeedback(html) {

  } else {
    var stop = D.timeUp || D.ended || (!rd.timeOn && D.qi >= QPER - 1 && D.setNo >= rd.count);
    html += '<div class="fb-row"><button id="nextBtn" class="primary">' + (stop ? 'おわる' : 'つぎへ') + '</button></div>';
  }
  var fb = swap('feedback', R(html));
  fb.classList.remove('hidden');
  var nb = $('nextBtn');
  if (nb) nb.addEventListener('click', function () { unlockAudio(); nextSelf(); });
  setTimeout(function () { try { fb.scrollIntoView({ block: 'nearest', behavior: 'smooth' }); } catch (e) {} }, 50);
}

function loadCfg() {
  renderCfgTabs();
  swap('cfgBody', '<tr><td colspan="6" class="subc">読み込んでいます…</td></tr>');
  api('api_tGetConf', T.cfg.group).then(function (rows) { T.cfg.rows = rows || []; drawCfg(); })
    .catch(function (e) { swap('cfgBody', '<tr><td colspan="6">' + esc(errText(e, '読み込めませんでした。もう一度ページを開きなおしてください。')) + '</td></tr>'); });
}
function stepCell(i, k, v) {
  return '<div class="stepper"><button data-i="' + i + '" data-k="' + k + '" data-d="-1">▽</button><input data-i="' + i + '" data-k="' + k + '" inputmode="numeric" value="' + (v === '' || v == null ? '' : v) + '"><button data-i="' + i + '" data-k="' + k + '" data-d="1">△</button></div>';
}
function drawCfg() {
  var rows = T.cfg.rows;
  if (!rows.length) { swap('cfgBody', '<tr><td colspan="6" class="subc">' + (T.cfg.group === '__guest' ? 'まだゲストで参加した端末はありません。' : 'この名簿には名前がありません。') + '</td></tr>'); return; }
  var b = swap('cfgBody', rows.map(function (r, i) {
    return '<tr><td><b>' + esc(r.name) + '</b></td>' +
      '<td>' + stepCell(i, 'level', r.level) + '</td><td>' + stepCell(i, 'lower', r.lower) + '</td><td>' + stepCell(i, 'upper', r.upper) + '</td>' +
      '<td><button class="tog' + (r.auto ? ' on' : '') + '" data-i="' + i + '" data-t="auto">' + (r.auto ? 'する' : 'しない') + '</button></td>' +
      '<td>' + (r.last || '<span class="subc">—</span>') + '</td></tr>';
  }).join(''));
  var setV = function (i, k, v) {
    v = (v === '' || isNaN(v) || v < 1) ? '' : Math.min(3, v);
    T.cfg.rows[i][k] = v;
    var inp = b.querySelector('input[data-i="' + i + '"][data-k="' + k + '"]'); if (inp) inp.value = v;
    swap('cfgMsg', '');
  };
  b.querySelectorAll('.stepper button').forEach(function (x) {
    x.addEventListener('click', function () {
      var i = +x.dataset.i, k = x.dataset.k, cur = T.cfg.rows[i][k], d = +x.dataset.d;
      setV(i, k, cur === '' || cur == null ? (d > 0 ? 1 : '') : +cur + d);
    });
  });
  b.querySelectorAll('.stepper input').forEach(function (x) {
    x.addEventListener('change', function () { setV(+x.dataset.i, x.dataset.k, x.value === '' ? '' : parseInt(x.value.replace(/[^0-9]/g, ''), 10)); });
  });
  b.querySelectorAll('[data-t]').forEach(function (x) {
    x.addEventListener('click', function () { var r = T.cfg.rows[+x.dataset.i]; r.auto = !r.auto; swap('cfgMsg', ''); drawCfg(); });
  });
}
function saveCfg() {
  swap('cfgMsg', '保存しています…');
  api('api_tSaveConf', T.cfg.rows).then(function () { swap('cfgMsg', '保存しました（' + fmtHM(Date.now()) + '）'); loadCfg(); })
    .catch(function (e) { swap('cfgMsg', errText(e, '保存できませんでした。もう一度押してください。')); });
}

/* --- レベルごとの内容（問題を作るのと同じ定義から出す） --- */
function renderLevels() {
  $('lvList').innerHTML = GENRES.map(function (g) {
    return '<div class="gcol"><h3>' + esc(noRuby(g.name)) + '</h3><ol>' + g.lv.map(function (t, i) {
      var p = makeProblem(g.id, i + 1, 'any');
      var ex = p ? p.qs.filter(function (q) { return !(q.c && q.c.g); }).map(function (q) { return '・' + esc(noRuby(q.t)); }).join('<br>') : '';
      return '<li><b>レベル' + (i + 1) + '</b>' + esc(t) + '<span class="ex">例：<br>' + ex + '</span></li>';
    }).join('') + '</ol></div>';
  }).join('');
}

/* --- 記録 --- */
function loadRecords() {
  swap('recOut', '<p class="note">読み込んでいます…</p>');
  api('api_tRecords').then(function (d) {
    var mist = (d.mist || []).map(function (x) {
      return '<tr><td><b>' + esc(x.name) + '</b></td><td>' + x.total + '</td><td style="font-size:13px">' + x.items.map(function (a) { return esc(a[0]) + '×' + a[1]; }).join('<br>') + '</td></tr>';
    }).join('');
    var sums = (d.sums || []).map(function (x) {
      return '<tr><td>' + esc(x.at) + '</td><td>' + esc(x.name) + '</td><td>' + esc(x.mode) + '</td><td style="font-size:13px">' + esc(x.genre) + '</td><td>' + lvChange(x.lv0, x.lv1) + '</td><td>' + esc(x.correct) + ' / ' + esc(x.n) + '（' + esc(x.rate) + '）</td><td style="font-size:13px">' + esc(x.mist) + '</td></tr>';
    }).join('');
    swap('recOut', '<h2 style="margin-top:14px">回ごとの記録（新しい順。「回ごとの記録」タブ）</h2><p class="note">レベルの「1→2」は、その回のはじめとおわりのレベルです。1問ごとの記録（問題・答え・ヒントを使ったか）は「記録」タブにあります。</p>' +
      '<div class="tblwrap"><table class="mt"><thead><tr><th>日時</th><th>名前</th><th>進め方</th><th>ジャンル</th><th>レベル</th><th>正解</th><th>多かったまちがい</th></tr></thead><tbody>' + (sums || '<tr><td colspan="7" class="subc">まだ記録がありません。</td></tr>') + '</tbody></table></div>' +
      '<h2 style="margin-top:18px">生徒ごとのまちがいの型（「記録」タブの直近3000行）</h2>' +
      '<div class="tblwrap"><table class="mt"><thead><tr><th>名前</th><th>まちがい</th><th>まちがいの型（回数）</th></tr></thead><tbody>' + (mist || '<tr><td colspan="3" class="subc">まだ記録がありません。</td></tr>') + '</tbody></table></div>');
  }).catch(function (e) { swap('recOut', '<p class="note">' + esc(errText(e, '読み込めませんでした。もう一度押してください。')) + '</p>'); });
}

function initTeacher() {
  document.body.classList.add('t-page');
  $('teacherApp').classList.remove('hidden');
  $('setCard').addEventListener('click', function (e) {
    var g = e.target.closest('.gchip[data-g]');
    if (g) {
      var s = T.settings, k = g.dataset.g, i = s.genres.indexOf(k);
      if (i >= 0) { if (s.genres.length > 1) s.genres.splice(i, 1); }
      else s.genres = GIDS.filter(function (x) { return x === k || s.genres.indexOf(x) >= 0; });
      renderSettingsT(); return;
    }
    var rb = e.target.closest('[data-roster]');
    if (rb) { T.settings.roster = rb.dataset.roster; renderSettingsT(); return; }
    var st = e.target.closest('[data-step]');
    if (st) { setNum(st.dataset.step, T.settings[st.dataset.step] + (+st.dataset.d)); return; }
    var b = e.target.closest('.seg[data-key] button');
    if (b) {
      var key = b.parentElement.dataset.key, v = b.dataset.val;
      T.settings[key] = v === 'true' ? true : v === 'false' ? false : v;
      renderSettingsT();
    }
  });
  Object.keys(LIMITS).forEach(function (k) {
    var el = $('in-' + k);
    if (!el) return;
    el.addEventListener('change', function () { setNum(k, el.value); });
    el.addEventListener('focus', function () { el.select(); });
  });
  $('startBtn').addEventListener('click', startRound);
  $('cfgGroup').addEventListener('click', function (e) { var b = e.target.closest('button[data-v]'); if (b) { T.cfg.group = b.dataset.v; loadCfg(); } });
  $('cfgSave').addEventListener('click', saveCfg);
  $('advBtn').addEventListener('click', function () {
    var open = $('advBox').classList.contains('hidden');
    $('advBox').classList.toggle('hidden', !open);
    $('advBtn').textContent = open ? 'くわしい設定を閉じる' : 'くわしい設定を開く';
  });
  $('lvOpenBtn').addEventListener('click', function () {
    var open = $('cfgCard').classList.contains('hidden');
    $('cfgCard').classList.toggle('hidden', !open); $('lvCard').classList.toggle('hidden', !open);
    $('lvOpenBtn').textContent = open ? '生徒ごとのレベル設定を閉じる' : '生徒ごとのレベル設定を開く';
    if (open) { loadCfg(); renderLevels(); $('cfgCard').scrollIntoView({ block: 'start', behavior: 'smooth' }); }
  });
  $('lvRefresh').addEventListener('click', renderLevels);
  $('recBtn').addEventListener('click', loadRecords);
  $('whoBtn').addEventListener('click', function () { setWhoMode(whoMode() === 'count' ? 'name' : 'count'); if (T.round) renderWho(T.round); });
  $('qvClose').addEventListener('click', function () { T.qView = false; renderTQ(); });
  $('revealBtn').addEventListener('click', function () { T.reveal = !T.reveal; renderTQ(); });
  $('qrBigBtn').addEventListener('click', function () {
    $('qrBigSvg').innerHTML = qrSvg(studentUrl()) || '<p style="font-size:24px">' + esc(studentUrl()) + '</p>';
    $('qrBigCode').textContent = T.round && T.round.code ? '参加コード ' + T.round.code : '';
    $('qrBig').classList.remove('hidden');
  });
  $('qrBigClose').addEventListener('click', function () { $('qrBig').classList.add('hidden'); });
  $('tHelpBtn').addEventListener('click', teacherGuide);
  api('api_tInit').then(function (d) {
    T.settings = Object.assign({}, DEF_T, d.settings || {}); T.rosters = d.rosters || []; T.guests = d.guests || []; T.url = d.url || ''; T.round = d.round;
    setNetT(true);
  }).catch(function (e) {
    T.settings = Object.assign({}, DEF_T);
    swap('startMsg', esc(errText(e, '読み込めませんでした。もう一度ページを開きなおしてください。')));
    setNetT(false);
  }).then(function () {
    renderSettingsT();
    if (T.round) tPoll(); else renderAll();
  });
}
