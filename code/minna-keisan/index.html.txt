  function keyOf(c) { return c.v + '|' + (c.r || 0); }

  /* 選択肢を組み立てる共通の部品
     cands：まちがいの候補（はじめの1つは かならず入れる。のこりはまぜる）
     fill：足りないときの候補を順に返す関数 */
  function assemble(q, cands, n, fill) {
    var used = {}; used[keyOf(q)] = 1;
    var list = [];
    cands.forEach(function (c) {
      if (!c || c.v < 0 || (c.r || 0) < 0) return;
      var k = keyOf(c);
      if (!used[k]) { used[k] = 1; list.push(c); }
    });
    var pick = list.slice(0, 1).concat(shuffle(list.slice(1))).slice(0, n - 1);
    for (var k = 1; pick.length < n - 1 && k < 300; k++) {
      var c = fill(k);
      if (!c || c.v < 0 || (c.r || 0) < 0) continue;
      var kk = keyOf(c);
      if (used[kk]) continue;
      used[kk] = 1; pick.push(c);
    }
    var all = pick.map(function (c) { return { v: c.v, r: c.r || 0, type: c.type || 'そのほか', correct: false, text: ansText(c.v, c.r) }; });
    all.push({ v: q.v, r: q.r || 0, type: '', correct: true, text: q.answerText });
    return shuffle(all);
  }
  function judgeNum(q, given, cands) {
    var gv = given.v, gr = given.r || 0;
    if (gv === q.v && gr === (q.r || 0)) return { ok: true, type: '' };
    if (given.type) return { ok: false, type: given.type };
    var c = cands.filter(function (c) { return c.v === gv && (c.r || 0) === gr; })[0];
    return { ok: false, type: c ? c.type : 'そのほか' };
  }
  function fillNum(q) { return function (k) { return { v: q.v + (k % 2 ? 1 : -1) * Math.ceil(k / 2), type: 'そのほか' }; }; }

  function dotsSvg(N, colorOf, cross) {
    var frames = Math.max(1, Math.ceil(N / 10));
    var r = 15, gap = 36, fw = gap * 5 + 8, fh = gap * 2 + 8;
    var W = frames * fw + (frames - 1) * 16, H = fh, f, i;
    var s = '<svg viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="まるの図">';
    for (f = 0; f < frames; f++) {
      var x0 = f * (fw + 16);
      s += '<rect x="' + (x0 + 1) + '" y="1" width="' + (fw - 2) + '" height="' + (fh - 2) + '" rx="8" fill="#fff" stroke="#9fb3c8" stroke-width="2"/>';
      for (i = 0; i < 10; i++) {
        s += '<circle cx="' + (x0 + 4 + gap / 2 + (i % 5) * gap) + '" cy="' + (4 + gap / 2 + Math.floor(i / 5) * gap) + '" r="' + r + '" fill="none" stroke="#e1e9f1" stroke-width="1.5"/>';
      }
    }
    for (i = 0; i < N; i++) {
      var ff = Math.floor(i / 10), kk = i % 10;
      var cx = ff * (fw + 16) + 4 + gap / 2 + (kk % 5) * gap, cy = 4 + gap / 2 + Math.floor(kk / 5) * gap;
      s += '<circle cx="' + cx + '" cy="' + cy + '" r="' + r + '" fill="' + colorOf(i) + '"/>';
      if (cross && cross(i)) s += '<path d="M' + (cx - 9) + ' ' + (cy - 9) + 'L' + (cx + 9) + ' ' + (cy + 9) + 'M' + (cx + 9) + ' ' + (cy - 9) + 'L' + (cx - 9) + ' ' + (cy + 9) + '" stroke="#7a8896" stroke-width="3" stroke-linecap="round"/>';

  var TEN_RATE = 0.2;   // レベル1〜3で「10になる」問題（3＋7、10−3）を出すわりあい
  function buildAS(g, level, a, b, op) {
    var v = op === '+' ? a + b : a - b;
    return {
      g: g, level: level, a: a, b: b, op: op, v: v,
      carry: op === '+' && (a % 10) + (b % 10) >= 10, borrow: op === '-' && (a % 10) < (b % 10),
      text: a + (op === '+' ? '＋' : '−') + b, answerText: String(v)
    };
  }
  function makeAS(g, level, prev) {
    var plus = g === 'tashi', op = plus ? '+' : '-';
    var same = function (q) { return prev && prev.a === q.a && prev.b === q.b && prev.op === q.op; };
    if (level <= 3 && Math.random() < TEN_RATE) {
      for (var t = 0; t < 20; t++) {
        var bb = rint(1, 9), tq = plus ? buildAS(g, level, 10 - bb, bb, op) : buildAS(g, level, 10, bb, op);
        tq.ten = true;
        if (!same(tq)) return tq;
      }
    }
    for (var n = 0; n < 500; n++) {
      var a, b;
      if (plus) {
        if (level === 1) { a = rint(1, 8); b = rint(1, 9 - a); }                         // こたえが9まで
        else if (level === 2) { a = rint(11, 18); b = rint(1, 9); if ((a % 10) + b >= 10) continue; if (Math.random() < 0.5) { var x = a; a = b; b = x; } }
        else if (level === 3) { a = rint(2, 9); b = rint(2, 9); if (a + b <= 10) continue; }
        else { a = rint(10, 89); b = rint(10, 89); if (a + b > 99) continue; if (((a % 10) + (b % 10) >= 10) !== (level === 5)) continue; }
      } else {
        if (level === 1) { a = rint(2, 9); b = rint(1, a - 1); }                         // 9までの数から
        else if (level === 2) { a = rint(11, 19); b = rint(1, 9); if ((a % 10) < b) continue; }
        else if (level === 3) { a = rint(11, 18); b = rint(2, 9); if ((a % 10) >= b || a - b < 1) continue; }
        else { a = rint(20, 99); b = rint(10, 89); if (a - b < 1) continue; if (((a % 10) < (b % 10)) !== (level === 5)) continue; }
      }
      var q = buildAS(g, level, a, b, op);
      if (same(q)) continue;
      return q;
    }
    return plus ? buildAS(g, level, 3, 2, '+') : buildAS(g, level, 5, 2, '-');
  }
  function candAS(q) {
    var a = q.a, b = q.b, v = q.v, list = [];
    var add = function (x, type) { if (x >= 0 && x !== v && !list.some(function (c) { return c.v === x; })) list.push({ v: x, type: type }); };
    if (q.carry && !q.ten) add(v - 10, 'くり上がりわすれ');
    if (q.borrow) {
      add((Math.floor(a / 10) - Math.floor(b / 10)) * 10 + Math.abs((a % 10) - (b % 10)), 'くり下がりわすれ');
      add(v + 10, 'くり下がりわすれ');
    }
    if (q.op === '+') add(Math.abs(a - b), 'たし算とひき算の取りちがえ'); else add(a + b, 'たし算とひき算の取りちがえ');
    add(v + 1, '1ちがい'); add(v - 1, '1ちがい');
    if (q.level >= 4) { add(v + 10, '十のくらいのまちがい'); add(v - 10, '十のくらいのまちがい'); }
    return list;
  }

  (function () {

    if (actx) {
      if (actx.state === 'suspended') actx.resume();

    }
  } catch (e) {}

function tone(f, st, dur, vol) {
  if (!SOUND || !actx) return;
  vol = vol || 0.12;
  try {
    var t = actx.currentTime + st, o = actx.createOscillator(), g = actx.createGain();
    o.type = 'sine'; o.frequency.value = f;
    g.gain.setValueAtTime(0.0001, t);
    g.gain.exponentialRampToValueAtTime(vol, t + 0.02);
    g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
    o.connect(g); g.connect(actx.destination); o.start(t); o.stop(t + dur + 0.05);
  } catch (e) {}
}

  // 画面を読みこみ直したときは、それまでの解いた数・レベルなどをもどす
  if (saved && saved.gs) {
    D.count = saved.count || 0; D.correct = saved.correct || 0; D.taKey = saved.taKey || ''; D.cur = saved.cur || '';
    Object.keys(saved.gs).forEach(function (k) {
      if (!GENRES[k]) return;
      var g = initG(k), sv = saved.gs[k];
      ['level', 'okRun', 'ngRun', 'topRun', 'lowNg', 'rvNg', 'types', 'n', 'mixN', 'correct', 'mastered', 'newly', 'best'].forEach(function (f) { if (sv[f] !== undefined) g[f] = sv[f]; });
    });
  }
  D.main = allowed.filter(function (k) { return !D.gs[k].mastered; });
  D.review = allowed.filter(function (k) { return D.gs[k].mastered; });
  if (saved && saved.gs && saved.main) {
    D.main = saved.main.filter(function (k) { return D.gs[k]; });
    D.review = (saved.review || []).filter(function (k) { return D.gs[k]; });
  }
  if (!D.main.length && rd.next) {
    var nk = nextGenre(allowed[allowed.length - 1]);
    if (nk) { initG(nk); D.main.push(nk); }
  }
  if (!D.main.length && !rd.ta) D.main = allowed.slice();
}
function initG(k) {
  if (D.gs[k]) return D.gs[k];
  var c = D.conf[k] || {}, max = topLv(k), rd = ST.round;
  var lo = c.lower || 1, up = c.upper || max;
  if (lo > up) { var t = lo; lo = up; up = t; }
  lo = Math.max(1, Math.min(max, lo)); up = Math.max(1, Math.min(max, up));
  var start = c.level || c.last || lo;
  start = Math.max(lo, Math.min(up, start));
  return (D.gs[k] = {
    key: k, level: start, start: start, min: lo, max: up, auto: rd.autoLv && c.auto !== false,
    okRun: 0, ngRun: 0, topRun: 0, lowNg: 0, rvNg: 0, types: {}, n: 0, mixN: 0, correct: 0,
    mastered: !!c.mastered, newly: false, best: c.best || 0, prev: null
  });
}
// むずかしさの順で、k のつぎの「出す」で まだ習得していないジャンル
function nextGenre(k) {
  var i = GENRE_ORDER.indexOf(k);
  var list = GENRE_ORDER.slice(i + 1).concat(GENRE_ORDER.slice(0, i + 1));
  for (var j = 0; j < list.length; j++) {
    var g = list[j], c = D.conf[g] || {};
    if (c.show === false) continue;
    if (D.gs[g] ? D.gs[g].mastered : c.mastered) continue;
    if (D.main.indexOf(g) >= 0) continue;
    return g;
  }
  return '';
}
// まぜるジャンル：k より前（むずかしさの順）で、習得ずみで、「出す」が×でないもの
function mixPrevs(k) {
  if (!ST.round.mixPrev) return [];
  var i = GENRE_ORDER.indexOf(k);
  return GENRE_ORDER.slice(0, Math.max(0, i)).filter(function (g) {
    var c = D.conf[g] || {};
    if (c.show === false) return false;
    return D.gs[g] ? D.gs[g].mastered : !!c.mastered;
  });
}
function genreList(keys) { return keys.map(function (k) { return GENRES[k].nameS; }).join('、'); }
function showSelfReady() {
  var rd = ST.round, msg;
  if (D.main.length) msg = genreList(D.main) + 'の {問題|もんだい}を ' + rd.minutes + '分 やるよ。';
  else msg = '{習得|しゅうとく}した ジャンルで タイムアタックを するよ。' + rd.minutes + '分 やるよ。';
  swap('readyMsg', R('「' + esc(me.name) + '」で {参加|さんか}しているよ。<span class="sub">' + msg + '<br>じゅんびが できたら「はじめる」を おしてね。</span>'));
  $('readyDots').classList.add('hidden');
  $('readyGoRow').classList.remove('hidden');
  $('renameBtn').classList.toggle('hidden', rd.join === 'guest');
  show('ready');
  if (D.endAt) startSelf();   // 読みこみ直したときは、そのまま続ける
}
function startSelf() {
  if (!D.endAt) D.endAt = Date.now() + ST.round.minutes * 60000;
  saveSelf();
  startClock();
  sendSnap();
  nextSelf();
}
var clockTimer = null;
function startClock() { clearInterval(clockTimer); clockTimer = setInterval(tickSelf, 250); tickSelf(); }
var lastShown = '';
function tickSelf() {
  if (!D || D.finished) return;
  var left = Math.max(0, Math.ceil((D.endAt - Date.now()) / 1000));
  var txt = 'のこり ' + (left >= 60 ? Math.floor(left / 60) + '分' + (left % 60) + '秒' : left + '秒');
  if (D.ta && !D.ta.over) {
    var tl = Math.max(0, Math.ceil((D.ta.endAt - Date.now()) / 1000));
    txt = 'タイムアタック のこり ' + tl + '秒';
    if (tl <= 0) endTA();
  }
  if (ST.view === 'quiz' && txt !== lastShown) { lastShown = txt; var el = swap('timer', txt); el.classList.remove('hidden'); }
  if (left <= 0 && !D.timeUp) { D.timeUp = true; stopSignal('じかんに なったよ。'); }
}
function stopSignal(text) {
  if (!D || D.finished) return;
  if (ST.view !== 'quiz' || ST.answered || (D.ta && D.ta.over)) {
    var nb = $('nextBtn'); if (nb) nb.innerHTML = R('おわる');
    if (ST.view !== 'quiz') finishSelf();
    return;
  }
  banner(text + '<br>この もんだいで おわりだよ', 2200);
}
// ふつうの練習で出すジャンル
function mainGenre() {
  var rd = ST.round;
  if (!D.main.length) return null;
  if (D.main.length === 1) return { key: D.main[0], src: 'ふつう' };
  if (rd.order === 'mix') {
    var h = D.hist, cand = D.main;
    if (h.length >= 2 && h[0] === h[1]) cand = D.main.filter(function (k) { return k !== h[0]; });
    var k = cand[Math.floor(Math.random() * cand.length)];
    D.hist = [k].concat(h).slice(0, 2);
    return { key: k, src: 'ふつう' };
  }
  var total = rd.minutes * 60000, used = total - (D.endAt - Date.now());
  var i = Math.floor(used / (total / D.main.length));
  return { key: D.main[Math.max(0, Math.min(D.main.length - 1, i))], src: 'ふつう' };
}
// つぎの問題のジャンル（前のジャンルをまぜる・たまに出す も、ここで決める）
function chooseGenre() {
  var rd = ST.round, c = mainGenre();
  if (!c) return null;
  var prevs = mixPrevs(c.key);
  if (prevs.length && Math.random() * 100 < rd.mixRate) return { key: prevs[Math.floor(Math.random() * prevs.length)], src: 'まぜる', base: c.key };
  var rv = D.review.filter(function (k) { return prevs.indexOf(k) < 0; });
  if (rd.sometimes && rv.length && Math.random() * 100 < rd.rate) return { key: rv[Math.floor(Math.random() * rv.length)], src: 'たまに' };
  return c;
}
function nextSelf() {
  if (D.finished) return;
  if (D.timeUp || D.ended) { finishSelf(); return; }
  if (D.ta) {
    if (D.ta.over) return;
    askSelf(D.ta.key, topLv(D.ta.key), 'タイムアタック');
    return;
  }
  var c = chooseGenre();
  if (!c) { startTA(); return; }
  var g = initG(c.key), lv;
  if (c.src === 'たまに') lv = topLv(c.key);
  else if (c.src === 'まぜる') lv = Math.max(1, Math.min(initG(c.base).level, g.max));   // いまのジャンルと同じ番号のレベル（上限はこえない）
  else lv = g.level;
  if (c.src === 'ふつう' && D.cur && D.cur !== c.key && D.count > 0 && ST.round.order !== 'mix') banner('ここから<br>' + GENRES[c.key].nameS + 'の もんだいだよ', 1800);
  if (c.src === 'ふつう') D.cur = c.key;
  askSelf(c.key, lv, c.src);
}
function askSelf(key, lv, src) {
  var g = initG(key), G = GENRES[key];
  var q = G.make(lv, g.prev);
  g.prev = q;
  var k = choiceCount(q, ST.round.answer);
  D.src = src;
  showQuestion(q, k ? G.choices(q, k) : null);
}

/* --- タイムアタック --- */
function startTA() {
  var list = D.review.length ? D.review : Object.keys(D.gs);
  if (!list.length) { finishSelf(); return; }
  var i = list.indexOf(D.taKey);
  var key = list[(i + 1) % list.length];
  D.taKey = key;
  D.ta = { key: key, endAt: Date.now() + ST.round.taSec * 1000, correct: 0, n: 0, over: false, best: initG(key).best || 0 };
  banner('タイムアタック！<br>' + GENRES[key].nameS + ' ' + ST.round.taSec + '秒', 1600);
  nextSelf();
}
function endTA() {
  var ta = D.ta;
  if (!ta || ta.over) return;
  ta.over = true;
  clearTimeout(ST.autoNext);
  var g = initG(ta.key), rec = ta.correct > (g.best || 0);
  if (rec) {
    g.best = ta.correct;
    outPush({ id: uid(), kind: 'best', at: Date.now(), roundId: me.roundId, name: me.name, genre: ta.key, best: ta.correct });
    flushOutbox();
    sndBig();
  }
  var h = '<div class="fb-title ' + (rec ? 'ok' : 'info') + '">タイムアタック おわり！</div>' +
    '<p>せいかいは <b style="font-size:30px">' + ta.correct + '</b>もん だったよ。' + (rec ? '<br>あたらしい さいこうきろく！' : '<br>さいこうきろくは ' + ta.best + 'もん だよ。') + '</p>';
  var more = !D.timeUp && !D.ended;
  h += '<div class="fb-row"><button id="nextBtn" class="primary">' + (more ? 'もういちど' : 'おわる') + '</button></div>';
  $('answerArea').querySelectorAll('button').forEach(function (b) { b.disabled = true; });
  var fb = swap('feedback', R(h)); fb.classList.remove('hidden');
  swap('hintArea', '');
  $('nextBtn').addEventListener('click', function () {
    unlockAudio();
    D.ta = null;
    if (D.timeUp || D.ended) finishSelf(); else startTA();
  });
  sendSnap();
}

/* --- 1問の答えのあと（ひとりずつ） --- */
function afterSelf(ok, type, pick) {
  afterSelf0(ok, type, pick);
  saveSelf();
}
// 途中経過を端末にとっておく（iPadの画面が読みこみ直されても、解いた数が消えないように）
function saveSelf() {
  if (!D || !me.roundId) return;
  var gs = {};
  Object.keys(D.gs).forEach(function (k) {
    var g = D.gs[k];
    gs[k] = { level: g.level, okRun: g.okRun, ngRun: g.ngRun, topRun: g.topRun, lowNg: g.lowNg, rvNg: g.rvNg, types: g.types, n: g.n, mixN: g.mixN, correct: g.correct, mastered: g.mastered, newly: g.newly, best: g.best };
  });
  me.selfState = { rid: me.roundId, endAt: D.endAt, count: D.count, correct: D.correct, taKey: D.taKey, cur: D.cur, main: D.main, review: D.review, gs: gs };
  saveMe();
}
function afterSelf0(ok, type, pick) {
  var q = ST.q, src = D.src, g = initG(q.g), rd = ST.round;
  D.count++; g.n++;
  if (ok) { D.correct++; g.correct++; }
  outPush({
    id: uid(), kind: 'answer', at: Date.now(), roundId: me.roundId, name: me.name, mode: 'self', genre: q.g, level: q.level,
    q: q.text, correct: q.answerText, pick: pick, ok: ok, ml: ok ? '' : type, how: howName(ST.ch ? ST.ch.length : 0), src: src,
    sec: Math.round((Date.now() - ST.qStart) / 1000)
  });
  if (src === 'タイムアタック') { if (ok) D.ta.correct++; D.ta.n++; sendSnap(); return; }
  if (!ok) g.types[type] = (g.types[type] || 0) + 1;
  // まぜた問題：記録だけ（レベルも習得も変えない）
  if (src === 'まぜる') { g.mixN++; sendSnap(); return; }
  if (src === 'たまに') {
    g.rvNg = ok ? 0 : g.rvNg + 1;
    if (g.rvNg >= 2) {     // 2問つづけてまちがえたら、習得を外して、ふつうにもどす
      g.mastered = false; g.rvNg = 0; g.topRun = 0; g.level = g.max;
      D.review = D.review.filter(function (k) { return k !== q.g; });
      if (D.main.indexOf(q.g) < 0) D.main.push(q.g);
      outPush({ id: uid(), kind: 'master', at: Date.now(), roundId: me.roundId, name: me.name, genre: q.g, on: false });
    }
    sendSnap(); return;
  }
  // 習得：いちばん上のレベルで、つづけて N問正解（上限を低くしている生徒は習得にならない）
  if (q.level === topLv(q.g) && g.max === topLv(q.g)) {
    g.topRun = ok ? g.topRun + 1 : 0;
    if (g.topRun >= rd.master) masterGenre(g);
  }
  // いちばんやさしいレベルで、まちがいがつづいている
  if (q.level === 1) g.lowNg = ok ? 0 : g.lowNg + 1; else g.lowNg = 0;
  // レベルの自動上下（変わっても何も表示しない）
  if (ok) { g.okRun++; g.ngRun = 0; } else { g.ngRun++; g.okRun = 0; }
  if (g.auto && !g.mastered) {
    if (g.okRun >= rd.up && g.level < g.max) { g.level++; g.okRun = 0; g.topRun = 0; }
    else if (g.ngRun >= rd.down && g.level > g.min) { g.level--; g.ngRun = 0; g.topRun = 0; }
  }
  sendSnap();
}
function masterGenre(g) {
  var rd = ST.round;
  g.mastered = true; g.newly = true; g.topRun = 0;
  outPush({ id: uid(), kind: 'master', at: Date.now(), roundId: me.roundId, name: me.name, genre: g.key, on: true });
  D.main = D.main.filter(function (k) { return k !== g.key; });
  if (D.review.indexOf(g.key) < 0) D.review.push(g.key);
  if (!D.main.length && rd.next) {
    var nk = nextGenre(g.key);
    if (nk) { initG(nk); D.main.push(nk); }
  }
  if (!D.main.length && !rd.ta) D.main = [g.key];
}
function topType(types) {
  var top = '', tn = 0;
  Object.keys(types).forEach(function (t) { if (types[t] > tn) { top = t; tn = types[t]; } });
  return { t: top, n: tn };
}
function snapshot() {
  var per = Object.keys(D.gs).map(function (k) {
    var g = D.gs[k], tp = topType(g.types);
    return { genre: k, start: g.start, level: g.level, n: g.n, mixN: g.mixN, correct: g.correct, top: tp.t, mastered: g.mastered, newly: g.newly, flag: g.lowNg >= ST.round.flagN, best: g.best };
  });
  return {
    state: D.finished ? 'done' : (ST.view === 'quiz' ? 'playing' : 'ready'), cur: D.ta ? D.ta.key : (ST.q ? ST.q.g : D.cur), src: D.ta ? 'タイムアタック' : D.src,
    count: D.count, correct: D.correct, remain: D.endAt ? Math.max(0, Math.round((D.endAt - Date.now()) / 1000)) : null,
    ta: D.ta ? { key: D.ta.key, correct: D.ta.correct } : null, per: per
  };
}
function sendSnap() {
  if (!D || !me.roundId) return;
  outPush({ id: uid(), kind: 'snap', roundId: me.roundId, name: me.name, snap: snapshot() });
}
function finishSelf() {
  if (!D || D.finished) return;
  D.finished = true;
  clearInterval(clockTimer); clearTimeout(ST.autoNext);
  if (!me.fin) {
    var per = Object.keys(D.gs).filter(function (k) { return D.gs[k].n > 0 || D.gs[k].newly; }).map(function (k) {
      var g = D.gs[k], tp = topType(g.types);
      return { genre: k, start: g.start, end: g.level, n: g.n, correct: g.correct, top: tp.t ? tp.t + '（' + tp.n + '回）' : '', mastered: g.newly, mixOnly: g.n > 0 && g.n === g.mixN };
    });
    sendSnap();
    outPush({ id: uid(), kind: 'finish', at: Date.now(), roundId: me.roundId, name: me.name, mode: 'self', per: per });
    me.fin = true; saveMe();
    flushOutbox();
  }
  showSelfResult();
}
function showResultOrWait() {

}

}

/* ==================== 問題の画面（両方で使う） ==================== */
function showQuestion(q, ch) {
  var G = GENRES[q.g], rd = ST.round;
  ST.q = q; ST.ch = ch; ST.answered = false; ST.qStart = Date.now(); ST.helpShown = false;
  var slots = G.slots(q);
  ST.input = { slots: slots, vals: {}, f: slots[0].k };
  clearTimeout(ST.autoNext);
  show('quiz');
  if (rd.mode === 'all') { swap('prog', (ST.qi + 1) + ' / ' + ST.Q.length); $('timer').classList.add('hidden'); }
  else swap('prog', R(D.ta ? 'せいかい ' + D.ta.correct : (D.count + 1) + '{問|もん}め'));
  var tag = swap('gTag', R(G.nameS + '　レベル' + q.level + (D && D.src === 'たまに' ? '　たしかめ' : '')));
  tag.classList.toggle('ta', !!(D && D.ta));
  swap('meChip', esc(me.name || ''));
  swap('qPrompt', R(promptOf(G, q)));
  swap('qVisual', R(G.render(q, {})));
  var fb = swap('feedback', ''); fb.classList.add('hidden');
  var easy = G.levels[q.level - 1].easy;
  swap('hintArea', easy ? R('<div class="hintrow"><button id="hintBtn" class="hintbtn">ヒント</button></div>') : '');
  if ($('hintBtn')) $('hintBtn').addEventListener('click', function () { unlockAudio(); showHelp(); });
  var html;
  if (ch) {
    html = '<div class="choices">' + ch.map(function (c, i) {
      return '<button class="choice" data-i="' + i + '"><span class="chip">' + 'ABCD'.charAt(i) + '</span><span class="ctext">' + c.text + '</span><span class="cmark"></span></button>';
    }).join('') + '</div>';
  } else {
    html = '<div class="numin">' + slots.map(function (s) {
      return (s.pre ? '<span class="slot-lab">' + s.pre + '</span>' : '') + '<span class="slot' + (s.len === 1 ? ' w1' : '') + '" data-slot="' + s.k + '"></span>';
    }).join('') + '</div>' +
      '<div class="pad small" id="inPad">' + ['1', '2', '3', '4', '5', '6', '7', '8', '9', 'del', '0', 'go'].map(function (k) {
        return '<button data-k="' + k + '" class="' + (k === 'del' ? 'del' : k === 'go' ? 'go' : '') + '">' + (k === 'del' ? 'けす' : k === 'go' ? 'こたえる' : k) + '</button>';
      }).join('') + '</div>';
  }
  var area = swap('answerArea', html);
  area.querySelectorAll('.choice').forEach(function (b) {
    b.addEventListener('click', function () { unlockAudio(); pickChoice(+b.dataset.i); });
  });
  if (!ch) {
    area.querySelectorAll('#inPad button').forEach(function (b) { b.addEventListener('click', function () { unlockAudio(); inputKey(b.dataset.k); }); });
    area.querySelectorAll('[data-slot]').forEach(function (s) { s.addEventListener('click', function () { if (ST.answered) return; ST.input.f = s.dataset.slot; drawInput(); }); });
    drawInput();
  }
  // もう答えた問題（みんなで・読みこみ直したとき）
  var prev = rd.mode === 'all' && me.res && me.res[ST.qi];
  if (prev) {
    ST.answered = true;
    area.querySelectorAll('button').forEach(function (b) { b.disabled = true; });
    finishFeedback('<div class="fb-title info">この {問題|もんだい}は、もう こたえたよ。</div>', false);
  }
}
function showHelp() {
  if (ST.answered || ST.helpShown) return;
  ST.helpShown = true;
  var G = GENRES[ST.q.g];
  swap('qVisual', R(G.render(ST.q, { help: true })));
  swap('hintArea', R('<div class="hintbox">' + G.help(ST.q) + '</div>'));
}
function drawInput() {
  var I = ST.input;
  document.querySelectorAll('#answerArea [data-slot]').forEach(function (el) {
    var k = el.dataset.slot;
    el.textContent = I.vals[k] || '';
    el.classList.toggle('cur', I.f === k && !ST.answered);
  });
}
function slotIndex(k) { var s = ST.input.slots; for (var i = 0; i < s.length; i++) if (s[i].k === k) return i; return 0; }
function inputKey(k) {
  if (ST.answered || !document.querySelector('#answerArea [data-slot]')) return;
  var I = ST.input, f = I.f, idx = slotIndex(f), s = I.slots[idx], cur = I.vals[f] || '';
  if (k === 'del') {
    if (cur === '' && idx > 0) I.f = I.slots[idx - 1].k; else I.vals[f] = cur.slice(0, -1);
  } else if (k === 'go') {
    if (!I.vals.v) { I.f = 'v'; drawInput(); return; }
    var v = parseInt(I.vals.v, 10), r = I.vals.r ? parseInt(I.vals.r, 10) : 0;
    answer({ v: v, r: r }, GUTIL.ansText(v, r), -1);
    return;
  } else if (/^[0-9]$/.test(k)) {
    if (cur.length < s.len) I.vals[f] = cur + k;
    if ((I.vals[f] || '').length >= s.len && idx < I.slots.length - 1) I.f = I.slots[idx + 1].k;
  }
  drawInput();
}
function switchSlot() {
  var I = ST.input;
  if (!I || I.slots.length < 2) return;
  I.f = I.slots[(slotIndex(I.f) + 1) % I.slots.length].k;
  drawInput();
}
function pickChoice(i) {
  if (ST.answered) return;
  var c = ST.ch[i];
  answer(c, c.text, i);
}
function answer(given, pickText, idx) {
  if (ST.answered) return;

      var pg = per.filter(function (p) { return p.n > 0 || p.mastered || p.genre === sn.cur; }).map(function (p) {
        var lvTxt = p.n > 0 && p.n === p.mixN ? 'まぜた問題だけ' : 'レベル' + lvChange(p.start, p.level);
        return esc(gname(p.genre)) + ' ' + lvTxt + '（' + p.n + '問・' + p.correct + '）' + (p.mastered ? '<span class="mst">習得' + (p.newly ? '！' : '') + '</span>' : '') + (p.flag ? '<span class="flag">要確認</span>' : '') + (p.top ? '<br><span class="subc" style="font-size:13px">多いまちがい：' + esc(p.top) + '</span>' : '');
      }).join('<br>');
      var rem = sn.remain != null && !m.done ? (sn.remain >= 60 ? Math.floor(sn.remain / 60) + '分' : '') + (sn.remain % 60) + '秒' : '';
      return '<tr><td><b>' + esc(m.name) + '</b>' + (flag ? '<span class="flag">要確認</span>' : '') + '</td><td>' + st + '</td><td>' + cur + '</td><td style="font-size:14px">' + (pg || '<span class="subc">まだ</span>') + '</td><td>' + (sn.correct || 0) + ' / ' + (sn.count || 0) + '</td><td>' + rem + '</td><td>' + netCell(r, m) + '</td></tr>';

}

/* --- いまの問題（みんなで：大きな画面に映して説明できる） --- */
function renderTQ() {
  var r = T.round, s = r.settings, on = r.status === 'open' && s.mode === 'all' && r.curQ >= 0;
  if (!on) T.qView = false;
  $('qCard').classList.toggle('hidden', !(on && T.qView));
  if (!on || !T.qView) return;
  KANA_ONLY = s.kana === 'hira';
  var keys = (r.sets || []).filter(function (k) { return T.sets[k]; });
  if (!keys.length) return;
  if (!T.sets[T.viewKey]) T.viewKey = keys[0];
  $('tLvTabs').innerHTML = keys.length > 1 ? keys.map(function (k) { return '<button class="gchip' + (k === T.viewKey ? ' on' : '') + '" data-k="' + k + '">レベル' + k + 'の問題</button>'; }).join('') : '';
  $('tLvTabs').querySelectorAll('[data-k]').forEach(function (b) { b.addEventListener('click', function () { T.viewKey = b.dataset.k; T.shownQ = ''; renderTQ(); }); });
  var it = T.sets[T.viewKey][r.curQ];
  if (!it) return;
  var q = it.q, G = GENRES[q.g];
  swap('tqTitle', (r.curQ + 1) + '問目（' + G.name + '・レベル' + q.level + '）');
  swap('tqState', '全部で ' + s.count + '問');
  var sig = r.id + '|' + T.viewKey + '|' + r.curQ + '|' + T.reveal;
  if (T.shownQ !== sig) {
    T.shownQ = sig;
    swap('tqPrompt', R(promptOf(G, q)));
    swap('tqVisual', R(G.render(q, { explain: T.reveal })));
  }
  var cnt = {}, rv = T.reveal;
  T.members.filter(function (m) { return String(m.lv) === String(T.viewKey); }).forEach(function (m) { var a = m.ans && m.ans[r.curQ]; if (a) cnt[a[3] || '－'] = (cnt[a[3] || '－'] || 0) + 1; });
  $('revealBtn').textContent = rv ? '正解と答えの分かれ方をかくす' : '正解と答えの分かれ方を表示する';
  var html;
  if (it.ch) {
    html = '<div class="choices">' + it.ch.map(function (c, i) {
      return '<div class="choice' + (rv && c.correct ? ' is-correct' : '') + (rv && !c.correct ? ' dim' : '') + '"><span class="chip">' + 'ABCD'.charAt(i) + '</span><span class="ctext">' + c.text + '</span><span class="cmark">' + (rv ? (cnt[c.text] || 0) + '人' + (c.correct ? '・正解' : '') : '') + '</span></div>';
    }).join('') + '</div>';
  } else {
    html = '<div class="feedback">数字を入力する問題です。' + (rv ? '正解は <b>' + q.answerText + '</b>。<div>' + (Object.keys(cnt).map(function (k) { return esc(k) + '：' + cnt[k] + '人'; }).join('<br>') || '<span class="subc">まだ答えた人はいません</span>') + '</div>' : '') + '</div>';
  }
  swap('tqChoices', html);
  renderWho(r);
}
// 答えた人：名前で出すか、人数だけ出すか（だれが何を答えたかは出さない）
function whoMode() { try { return localStorage.getItem('minna-keisan-who') === 'count' ? 'count' : 'name'; } catch (e) { return T.who || 'name'; } }
function setWhoMode(v) { T.who = v; try { localStorage.setItem('minna-keisan-who', v); } catch (e) {} }
function renderWho(r) {
  var names = T.members.filter(function (m) { return m.ans && m.ans[r.curQ]; }).map(function (m) { return m.name; });
  var mode = T.who || whoMode(), h;
  if (mode === 'count') h = '<div class="who-lab">答えた人 <span class="who-n">' + names.length + '</span> / ' + T.members.length + '人</div>';
  else h = '<div class="who-lab">答えた人（' + names.length + ' / ' + T.members.length + '人）</div><div class="who-names">' +
    (names.length ? names.map(function (n) { return '<span class="who-name">' + esc(n) + '</span>'; }).join('') : '<span class="subc">まだだれも答えていません</span>') + '</div>';
  swap('tqWho', h);
  $('whoBtn').textContent = mode === 'count' ? '答えた人の名前を出す' : '答えた人を人数だけにする';
}

function renderCfgTabs() {
  $('cfgGenre').innerHTML = GENRE_ORDER.map(function (k) { return '<button data-v="' + k + '" class="' + (k === T.cfg.genre ? 'on' : '') + '">' + gname(k) + '</button>'; }).join('');
  var groups = T.rosters.map(function (r) { return r.title; });

function loadCfg() {
  renderCfgTabs();
  swap('cfgBody', '<tr><td colspan="9" class="subc">読み込んでいます…</td></tr>');
  api('api_tGetConf', T.cfg.genre, T.cfg.group).then(function (rows) { T.cfg.rows = rows || []; drawCfg(); })
    .catch(function (e) { swap('cfgBody', '<tr><td colspan="9">' + esc(errText(e, '読み込めませんでした。もう一度ページを開きなおしてください。')) + '</td></tr>'); });
}
function stepCell(i, k, v) {
  return '<div class="stepper"><button data-i="' + i + '" data-k="' + k + '" data-d="-1">▽</button><input data-i="' + i + '" data-k="' + k + '" inputmode="numeric" value="' + (v === '' || v == null ? '' : v) + '"><button data-i="' + i + '" data-k="' + k + '" data-d="1">△</button></div>';
}
function drawCfg() {
  var rows = T.cfg.rows;
  if (!rows.length) { swap('cfgBody', '<tr><td colspan="9" class="subc">' + (T.cfg.group === '__guest' ? 'まだゲストで参加した端末はありません。' : 'この名簿には名前がありません。') + '</td></tr>'); return; }
  var b = swap('cfgBody', rows.map(function (r, i) {
    return '<tr><td><b>' + esc(r.name) + '</b></td><td><button class="tog' + (r.show ? ' on' : '') + '" data-i="' + i + '" data-t="show">' + (r.show ? '○ 出す' : '× 出さない') + '</button></td>' +
      '<td>' + stepCell(i, 'level', r.level) + '</td><td>' + stepCell(i, 'lower', r.lower) + '</td><td>' + stepCell(i, 'upper', r.upper) + '</td>' +
      '<td><button class="tog' + (r.auto ? ' on' : '') + '" data-i="' + i + '" data-t="auto">' + (r.auto ? 'する' : 'しない') + '</button></td>' +
      '<td>' + (r.last || '<span class="subc">—</span>') + '</td>' +
      '<td>' + (r.mastered && !r.unmaster ? '<span class="mst">習得 ' + esc(r.mdate || '') + '</span> <button class="ghost" style="font-size:14px;min-height:40px" data-i="' + i + '" data-t="unmaster">外す</button>' : (r.unmaster ? '<span class="subc">外します（保存で反映）</span>' : '<span class="subc">—</span>')) + '</td>' +
      '<td>' + (r.best === '' || r.best == null ? '<span class="subc">—</span>' : r.best + '問') + '</td></tr>';
  }).join(''));
  var max = GENRES[T.cfg.genre].levels.length;
  var setV = function (i, k, v) {
    v = (v === '' || isNaN(v) || v < 1) ? '' : Math.min(max, v);
    T.cfg.rows[i][k] = v;
    var inp = b.querySelector('input[data-i="' + i + '"][data-k="' + k + '"]'); if (inp) inp.value = v;
    swap('cfgMsg', '');
  };
  b.querySelectorAll('.stepper button').forEach(function (x) {
    x.addEventListener('click', function () {
      var i = +x.dataset.i, k = x.dataset.k, cur = T.cfg.rows[i][k], d = +x.dataset.d;
      setV(i, k, cur === '' || cur == null ? (d > 0 ? 1 : '') : +cur + d);
    });
  });
  b.querySelectorAll('.stepper input').forEach(function (x) {
    x.addEventListener('change', function () { setV(+x.dataset.i, x.dataset.k, x.value === '' ? '' : parseInt(x.value.replace(/[^0-9]/g, ''), 10)); });
  });
  b.querySelectorAll('[data-t]').forEach(function (x) {
    x.addEventListener('click', function () {
      var r = T.cfg.rows[+x.dataset.i], t = x.dataset.t;
      if (t === 'unmaster') r.unmaster = true; else r[t] = !r[t];
      swap('cfgMsg', ''); drawCfg();
    });
  });
}
function saveCfg() {
  swap('cfgMsg', '保存しています…');
  api('api_tSaveConf', T.cfg.genre, T.cfg.rows).then(function () { swap('cfgMsg', '保存しました（' + fmtHM(Date.now()) + '）'); loadCfg(); })
    .catch(function (e) { swap('cfgMsg', errText(e, '保存できませんでした。もう一度押してください。')); });
}

/* --- レベルごとの内容（問題を作るのと同じ定義から出す） --- */
function renderLevels() {
  $('lvList').innerHTML = GENRE_ORDER.map(function (k) {

    }).join('') + '</ol></div>';
  }).join('');
}

function initTeacher() {
  document.body.classList.add('t-page');
  $('teacherApp').classList.remove('hidden');
  $('setCard').addEventListener('click', function (e) {
    var g = e.target.closest('.gchip[data-g]');
    if (g) {
      var s = T.settings, k = g.dataset.g, i = s.genres.indexOf(k);
      if (s.mode !== 'self') s.genres = [k];
      else if (i >= 0) { if (s.genres.length > 1) s.genres.splice(i, 1); }
      else s.genres = GENRE_ORDER.filter(function (x) { return x === k || s.genres.indexOf(x) >= 0; });
      renderSettingsT(); return;
    }
    var rb = e.target.closest('[data-roster]');
    if (rb) { T.settings.roster = rb.dataset.roster; renderSettingsT(); return; }
    var st = e.target.closest('[data-step]');
    if (st) { setNum(st.dataset.step, T.settings[st.dataset.step] + (+st.dataset.d)); return; }
    var b = e.target.closest('.seg[data-key] button');
    if (b) {
      var key = b.parentElement.dataset.key, v = b.dataset.val;
      T.settings[key] = v === 'true' ? true : v === 'false' ? false : v;
      renderSettingsT();
    }
  });
  Object.keys(LIMITS).forEach(function (k) {
    var el = $('in-' + k);
    if (!el) return;
    el.addEventListener('change', function () { setNum(k, el.value); });
    el.addEventListener('focus', function () { el.select(); });
  });
  $('startBtn').addEventListener('click', startRound);
  $('cfgGenre').addEventListener('click', function (e) { var b = e.target.closest('button[data-v]'); if (b) { T.cfg.genre = b.dataset.v; loadCfg(); } });
  $('cfgGroup').addEventListener('click', function (e) { var b = e.target.closest('button[data-v]'); if (b) { T.cfg.group = b.dataset.v; loadCfg(); } });
  $('cfgSave').addEventListener('click', saveCfg);
  $('advBtn').addEventListener('click', function () {
    var open = $('advBox').classList.contains('hidden');
    $('advBox').classList.toggle('hidden', !open);
    $('advBtn').textContent = open ? 'くわしい設定を閉じる' : 'くわしい設定を開く';
  });
  $('lvOpenBtn').addEventListener('click', function () {
    var open = $('cfgCard').classList.contains('hidden');
    $('cfgCard').classList.toggle('hidden', !open); $('lvCard').classList.toggle('hidden', !open);
    $('lvOpenBtn').textContent = open ? '生徒ごとのレベル設定を閉じる' : '生徒ごとのレベル設定を開く';
    if (open) { loadCfg(); renderLevels(); $('cfgCard').scrollIntoView({ block: 'start', behavior: 'smooth' }); }
  });
  $('lvRefresh').addEventListener('click', renderLevels);
  $('recBtn').addEventListener('click', loadRecords);
  $('whoBtn').addEventListener('click', function () { setWhoMode(whoMode() === 'count' ? 'name' : 'count'); if (T.round) renderWho(T.round); });
  $('qvClose').addEventListener('click', function () { T.qView = false; renderTQ(); });
  $('revealBtn').addEventListener('click', function () { T.reveal = !T.reveal; renderTQ(); });
  $('qrBigBtn').addEventListener('click', function () {
    $('qrBigSvg').innerHTML = qrSvg(studentUrl()) || '<p style="font-size:24px">' + esc(studentUrl()) + '</p>';
    $('qrBigCode').textContent = T.round && T.round.code ? '参加コード ' + T.round.code : '';
    $('qrBig').classList.remove('hidden');
  });
  $('qrBigClose').addEventListener('click', function () { $('qrBig').classList.add('hidden'); });
  $('tHelpBtn').addEventListener('click', teacherGuide);
  renderLevels();
  api('api_tInit').then(function (d) {
    T.settings = Object.assign({}, DEF_T, d.settings || {}); T.rosters = d.rosters || []; T.guests = d.guests || []; T.url = d.url || ''; T.round = d.round;
    setNetT(true);
  }).catch(function (e) {
    T.settings = Object.assign({}, DEF_T);
    swap('startMsg', esc(errText(e, '読み込めませんでした。もう一度ページを開きなおしてください。')));
    setNetT(false);
  }).then(function () {
    renderSettingsT();
    if (T.round) tPoll(); else renderAll();
  });
}
